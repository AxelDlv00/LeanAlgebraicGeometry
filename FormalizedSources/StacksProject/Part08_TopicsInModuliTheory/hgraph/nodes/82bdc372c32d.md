theorem morphismScheme_representable {C : Type u} [Category.{v} C]
    [HasPullbacks C] {X Y : C} (f : X ⟶ Y) :
    RepresentableTransformation C (morphismScheme f) := by
  exact Functor.relativelyRepresentable.map yoneda f
