def FamilyOfCurves.SatisfiesFamilyConditions {GeometricFiber : Type u}
    (f : FamilyOfCurves GeometricFiber) : Prop :=
  f.isFlat ∧ f.isProper ∧ f.isOfFinitePresentation ∧
    f.hasRelativeDimensionAtMostOne
