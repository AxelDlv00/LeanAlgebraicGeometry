theorem locusOn_union {X I : Type*} [TopologicalSpace X]
    (s : NumericalSituation X I) (J K : Set I) :
    s.locusOn (J ∪ K) = s.locusOn J ∩ s.locusOn K := by
  ext x
  constructor
  · intro hx
    constructor
    · intro i hi
      exact hx i (Or.inl hi)
    · intro i hi
      exact hx i (Or.inr hi)
  · rintro ⟨hxJ, hxK⟩ i hi
    rcases hi with hi | hi
    · exact hxJ i hi
    · exact hxK i hi
