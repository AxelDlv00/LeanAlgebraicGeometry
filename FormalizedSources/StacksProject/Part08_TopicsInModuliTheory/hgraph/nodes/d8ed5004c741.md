def pullback {X Y : Type*} [TopologicalSpace X] [TopologicalSpace Y]
    (ν : NumericalInvariant X) (f : Y → X) (hf : Continuous f) :
    NumericalInvariant Y where
  value := ν.value ∘ f
  locallyConstant := ν.locallyConstant.comp_continuous hf

@[simp]
