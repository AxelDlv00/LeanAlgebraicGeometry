theorem Stable.semistable {GeometricFiber : Type u}
    {f : FamilyOfCurves GeometricFiber} (hf : Stable f) : Semistable f := by
  refine ⟨hf.prestable, ?_⟩
  intro s
  exact ⟨le_trans (by decide : 1 ≤ 2) (hf.2 s).1, (hf.2 s).2.1⟩
