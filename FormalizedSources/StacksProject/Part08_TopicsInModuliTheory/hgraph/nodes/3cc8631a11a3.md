def Stable {GeometricFiber : Type u} (f : FamilyOfCurves GeometricFiber) : Prop :=
  Prestable f ∧
    ∀ s, 2 ≤ f.genus s ∧
      ¬ f.hasRationalTail s ∧ ¬ f.hasRationalBridge s
