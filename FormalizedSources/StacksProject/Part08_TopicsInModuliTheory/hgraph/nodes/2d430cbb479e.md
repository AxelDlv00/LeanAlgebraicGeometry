theorem locus_isClosed {X I : Type*} [TopologicalSpace X]
    (s : NumericalSituation X I) : IsClosed s.locus := by
  rw [s.locus_eq_iInter]
  apply isClosed_iInter
  intro i
  exact ((s.invariant i).fiber_isClopen (s.prescribed i)).isClosed
