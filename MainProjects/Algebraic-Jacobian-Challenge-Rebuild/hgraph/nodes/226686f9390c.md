lemma stalkIdeal_le_span_germ_iff_le_ord (d : (relCurve C K).LocalEquations)
    (s : relThetaSections C K π a) {z : relCurve C K}
    (hzg : z ≠ genericPoint (relCurve C K)) :
    Scheme.divisorBound
        (thetaFieldDivisor C K π a - Scheme.presentationDivisor K d.presentation) hzg
      ≤ Scheme.ord (relCurve C K ↘ Spec (CommRingCat.of K)) hzg
          (thetaFieldRead C K π a s)
    ↔ d.stalkIdeal z ≤ Ideal.span
        {((relCurve C K).presheaf.germ (⊤ ⊓ (thetaFieldPointedCover C K π).opens z) z
            ⟨trivial, (thetaFieldPointedCover C K π).mem_opens z⟩).hom
          ((thetaFieldGluedEquiv C K π a s).val z)} := by
  have hsplit : Scheme.divisorBound
      (thetaFieldDivisor C K π a - Scheme.presentationDivisor K d.presentation) hzg
      = Scheme.divisorBound
          (Scheme.presentationDivisor K (thetaFieldPresentation C K π a)) hzg
        * Scheme.divisorBound (-Scheme.presentationDivisor K d.presentation) hzg := by
    rw [sub_eq_add_neg]
    exact Scheme.divisorBound_add _ _ hzg
  have hpos : (0 : WithZero (Multiplicative ℤ))
      < Scheme.divisorBound
          (Scheme.presentationDivisor K (thetaFieldPresentation C K π a)) hzg := by
    rw [show Scheme.divisorBound
        (Scheme.presentationDivisor K (thetaFieldPresentation C K π a)) hzg
        = ((Multiplicative.ofAdd (coeffAt hzg
            (Scheme.presentationDivisor K (thetaFieldPresentation C K π a))) :
          Multiplicative ℤ) : WithZero (Multiplicative ℤ)) from rfl]
    exact zero_lt_iff.mpr WithZero.coe_ne_zero
  have helem : (d.presentation.elem z : (relCurve C K).functionField)
      = algebraMap ((relCurve C K).presheaf.stalk z) (relCurve C K).functionField
          (((relCurve C K).presheaf.germ (d.cover.opens z) z
            (d.cover.mem_opens z)).hom (d.eqn z)) := by
    rw [Scheme.LocalEquations.presentation_elem_val]
    exact Scheme.germ_generic_eq_algebraMap_germ (d.cover.genericPoint_mem_opens z)
      (d.cover.mem_opens z) (d.eqn z)
  have hβ : ((relCurve C K).presheaf.germ
        (⊤ ⊓ (thetaFieldPresentation C K π a).cover.opens z)
        (genericPoint (relCurve C K))
        ⟨trivial, (thetaFieldPresentation C K π a).cover.genericPoint_mem_opens z⟩).hom
        ((thetaFieldGluedEquiv C K π a s).val z)
      = algebraMap ((relCurve C K).presheaf.stalk z) (relCurve C K).functionField
          (((relCurve C K).presheaf.germ
            (⊤ ⊓ (thetaFieldPointedCover C K π).opens z) z
            ⟨trivial, (thetaFieldPointedCover C K π).mem_opens z⟩).hom
            ((thetaFieldGluedEquiv C K π a s).val z)) :=
    Scheme.germ_generic_eq_algebraMap_germ
      (⟨trivial, (thetaFieldPresentation C K π a).cover.genericPoint_mem_opens z⟩ :
        genericPoint (relCurve C K)
          ∈ ⊤ ⊓ (thetaFieldPresentation C K π a).cover.opens z)
      (⟨trivial, (thetaFieldPresentation C K π a).cover.mem_opens z⟩ :
        z ∈ ⊤ ⊓ (thetaFieldPresentation C K π a).cover.opens z)
      ((thetaFieldGluedEquiv C K π a s).val z)
  rw [thetaFieldRead_apply,
    Scheme.MeromorphicPresentation.gluedVal_eq_elem_inv_mul K
      (thetaFieldPresentation C K π a) z (W := ⊤) trivial
      (thetaFieldGluedEquiv C K π a s),
    map_mul, hsplit,
    Scheme.MeromorphicPresentation.ord_elem_inv K (thetaFieldPresentation C K π a) z hzg,
    mul_le_mul_iff_right₀ hpos,
    ← Scheme.LocalEquations.ord_presentation_elem K d hzg,
    hβ, helem, Scheme.LocalEquations.stalkIdeal, Ideal.span_singleton_le_iff_mem,
    Ideal.mem_span_singleton, Scheme.dvd_iff_ord_le K hzg]
