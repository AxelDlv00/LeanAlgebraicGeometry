private lemma final_core {G : Type u} [CommGroup G] {tx ty ty' c γ : G}
    (h₁ : tx = ty * c) (h₂ : ty * γ = ty') : tx⁻¹ * c = γ * ty'⁻¹ := by
  subst h₁
  rw [← h₂, mul_inv, inv_mul_cancel_right, mul_inv, mul_comm ty⁻¹ γ⁻¹,
    mul_inv_cancel_left]
