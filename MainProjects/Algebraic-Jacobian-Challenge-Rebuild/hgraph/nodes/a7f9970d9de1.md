theorem subtype_lTensor_injective_of_flat_quotient
    (J : Ideal B) [Module.Flat R (B ⧸ J)]
    (A : Type u) [AddCommGroup A] [Module R A] :
    Function.Injective ((J.restrictScalars R).subtype.lTensor A) := by
  apply LinearMap.lTensor_injective_of_exact_of_flat
    (Ideal.Quotient.mkₐ R J).toLinearMap
    Ideal.Quotient.mk_surjective
    (J.restrictScalars R).subtype
    (Submodule.injective_subtype _)
  rw [LinearMap.exact_iff, restrictScalars_ker_quotient]
  exact (Submodule.range_subtype _).symm
