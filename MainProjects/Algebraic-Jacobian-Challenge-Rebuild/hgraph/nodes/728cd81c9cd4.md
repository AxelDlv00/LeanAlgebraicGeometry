def Scheme.Hom.IsProjective {X S : Scheme.{u}} (pi : X ⟶ S) : Prop :=
  ∃ (n : Type u) (_ : Finite n) (i : X ⟶ ℙ(n; S)),
    IsClosedImmersion i ∧ i ≫ (ℙ(n; S) ↘ S) = pi
