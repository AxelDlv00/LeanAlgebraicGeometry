lemma map_maximalIdeal_eq_of_surjective [IsLocalRing B] [IsLocalRing A]
    (φ : B →+* A) (hφ : Function.Surjective φ) :
    (IsLocalRing.maximalIdeal B).map φ = IsLocalRing.maximalIdeal A := by
  have hker_le : RingHom.ker φ ≤ IsLocalRing.maximalIdeal B :=
    IsLocalRing.le_maximalIdeal (RingHom.ker_ne_top φ)
  apply le_antisymm
  · rw [Ideal.map_le_iff_le_comap]
    intro b hb
    by_contra hnb
    have hunit : IsUnit (φ b) := by
      by_contra h
      exact hnb ((IsLocalRing.mem_maximalIdeal _).mpr h)
    obtain ⟨c, hc⟩ := hunit.exists_right_inv
    obtain ⟨c', rfl⟩ := hφ c
    have hker : b * c' - 1 ∈ RingHom.ker φ := by
      rw [RingHom.mem_ker, map_sub, map_mul, map_one, hc, sub_self]
    have h1 : (1 : B) ∈ IsLocalRing.maximalIdeal B := by
      have hbc : b * c' ∈ IsLocalRing.maximalIdeal B :=
        Ideal.mul_mem_right _ _ hb
      simpa using sub_mem hbc (hker_le hker)
    exact (IsLocalRing.maximalIdeal.isMaximal B).ne_top
      ((Ideal.eq_top_iff_one _).mpr h1)
  · intro a ha
    obtain ⟨b, rfl⟩ := hφ a
    have hb : b ∈ IsLocalRing.maximalIdeal B := by
      by_contra hnb
      exact (IsLocalRing.mem_maximalIdeal _).mp ha
        ((IsLocalRing.notMem_maximalIdeal.mp hnb).map φ)
    exact Ideal.mem_map_of_mem φ hb
