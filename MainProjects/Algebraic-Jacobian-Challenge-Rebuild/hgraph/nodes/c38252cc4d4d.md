lemma germ_resHom {V W : X.Opens} (h : W ≤ V) (x : X) (hx : x ∈ W)
    (t : Γ(X, V)) :
    (X.presheaf.germ W x hx).hom (X.resHom h t) =
      (X.presheaf.germ V x (h hx)).hom t :=
  X.presheaf.germ_res_apply (homOfLE h) x hx t
