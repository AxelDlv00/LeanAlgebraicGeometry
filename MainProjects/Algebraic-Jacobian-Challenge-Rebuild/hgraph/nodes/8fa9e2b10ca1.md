theorem ker_finiteComponentSum_le_range_finiteKoszulBoundary_of_pencil
    (row : ι → M →ₗ[R] N) (step : ι → L →ₗ[R] M)
    (c : Fin 2 → ι → R)
    (hcomm : ∀ (i j : ι) (z : L),
      row i (step j z) = row j (step i z))
    (hspan : ∀ x : M, ∃ z : Fin 2 → L,
      x = ∑ q, ∑ i, c q i • step i (z q))
    (hpencil : ∀ y : Fin 2 → M,
      (∑ q, ∑ i, c q i • row i (y q)) = 0 →
        ∃ z : L,
          y 0 = ∑ i, c 1 i • step i z ∧
          y 1 = -∑ i, c 0 i • step i z) :
    LinearMap.ker (finiteComponentSum row) ≤
      LinearMap.range (finiteKoszulBoundary step) := by
  classical
  intro x hx
  choose d hd using fun i => hspan (x i)
  let y : Fin 2 → M := fun q => ∑ i, step i (d i q)
  let zMain : ι × ι → L := fun ij => ∑ q, c q ij.2 • d ij.1 q
  have hzMain : ∀ i,
      finiteKoszulBoundary step zMain i = x i - ∑ q, c q i • y q := by
    intro i
    rw [finiteKoszulBoundary_apply]
    simp only [zMain, y, map_sum, map_smul]
    rw [Finset.sum_sub_distrib]
    have hleft : (∑ j, ∑ q, c q j • step j (d i q)) = x i := by
      rw [hd i, Finset.sum_comm]
    rw [hleft]
    congr 1
    rw [Finset.sum_comm]
    apply Finset.sum_congr rfl
    intro q hq
    rw [Finset.smul_sum]
  have hpairZero : (∑ q, ∑ i, c q i • row i (y q)) = 0 := by
    have hx0 : finiteComponentSum row x = 0 := LinearMap.mem_ker.mp hx
    have hboundary0 : finiteComponentSum row (finiteKoszulBoundary step zMain) = 0 :=
      LinearMap.mem_ker.mp
        (range_finiteKoszulBoundary_le_ker_finiteComponentSum row step hcomm
          (LinearMap.mem_range_self (finiteKoszulBoundary step) zMain))
    have hdiff : x - finiteKoszulBoundary step zMain =
        fun i => ∑ q, c q i • y q := by
      funext i
      simp only [Pi.sub_apply, hzMain i, sub_sub_cancel]
    have hzero : finiteComponentSum row
        (fun i => ∑ q, c q i • y q) = 0 := by
      rw [← hdiff, map_sub, hx0, hboundary0, sub_zero]
    simp only [finiteComponentSum, LinearMap.sum_apply, LinearMap.comp_apply,
      LinearMap.proj_apply, map_sum, map_smul] at hzero
    rw [Finset.sum_comm] at hzero
    exact hzero
  obtain ⟨a, hy0, hy1⟩ := hpencil y hpairZero
  let zPair : ι × ι → L := fun ij => (c 0 ij.1 * c 1 ij.2) • a
  have hzPair : ∀ i,
      finiteKoszulBoundary step zPair i = ∑ q, c q i • y q := by
    intro i
    rw [finiteKoszulBoundary_apply, Finset.sum_sub_distrib]
    simp only [zPair, map_smul]
    calc
      (∑ j, (c 0 i * c 1 j) • step j a) -
          ∑ j, (c 0 j * c 1 i) • step j a =
          c 0 i • (∑ j, c 1 j • step j a) -
            c 1 i • (∑ j, c 0 j • step j a) := by
        rw [Finset.smul_sum, Finset.smul_sum]
        apply congrArg₂ (· - ·)
        · apply Finset.sum_congr rfl
          intro j hj
          rw [smul_smul]
        · apply Finset.sum_congr rfl
          intro j hj
          rw [smul_smul, mul_comm (c 1 i)]
      _ = c 0 i • y 0 + c 1 i • y 1 := by
        rw [hy0, hy1, smul_neg]
        abel
      _ = ∑ q, c q i • y q := by rw [Fin.sum_univ_two]
  refine ⟨zMain + zPair, ?_⟩
  funext i
  simp only [map_add, Pi.add_apply, hzMain i, hzPair i, sub_add_cancel]
