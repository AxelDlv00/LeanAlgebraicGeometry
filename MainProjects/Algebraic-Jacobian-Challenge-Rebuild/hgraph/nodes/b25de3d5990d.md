noncomputable def pointDivisorCover : X.PointedCover where
  opens y := if y = x then pointOpen K hx else pointDivisorCompl K hx
  mem_opens y := by
    by_cases h : y = x
    · rw [if_pos h]; subst h; exact mem_pointOpen K hx
    · rw [if_neg h]; exact h

@[simp]
