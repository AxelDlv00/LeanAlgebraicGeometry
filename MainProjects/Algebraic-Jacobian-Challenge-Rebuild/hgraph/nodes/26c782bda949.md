noncomputable def pic0GaloisAction (gamma : L ≃ₐ[k] L) :
    (pic0TwistTestFunctor gamma).op ⋙
        pic0TypeFunctor ((baseChange k L).obj C) ≅
      pic0TypeFunctor ((baseChange k L).obj C) :=
  Functor.isoWhiskerLeft _ (pic0ThetaType k L C) ≪≫
    pic0GaloisActionRestricted C gamma ≪≫
