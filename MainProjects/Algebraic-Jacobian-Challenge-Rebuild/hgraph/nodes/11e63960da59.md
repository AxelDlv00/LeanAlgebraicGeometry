lemma isAffineOpen_chart_inf (i j : D.index) : IsAffineOpen (D.chart i ⊓ D.chart j) := by
  cases i <;> cases j
  · change IsAffineOpen (fiberChart₀ π ⊓ fiberChart₀ π)
    rw [inf_idem]
    exact (fiberTwoCover π).isAffineOpen₀
  · exact (fiberTwoCover π).isAffineOpen_inf
  · change IsAffineOpen (fiberChart₁ π ⊓ fiberChart₀ π)
    rw [inf_comm]
    exact (fiberTwoCover π).isAffineOpen_inf
  · change IsAffineOpen (fiberChart₁ π ⊓ fiberChart₁ π)
    rw [inf_idem]
    exact (fiberTwoCover π).isAffineOpen₁
