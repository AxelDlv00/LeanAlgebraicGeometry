def pic0PullbackNat (g : D ⟶ E) : pic0Functor E ⟶ pic0Functor D where
  app T := CommGrpCat.ofHom (pic0Pullback g T.unop)
  naturality T T' f := by
    ext lam
    exact (pic0Map_pic0Pullback g f.unop lam).symm

/-- The components of `pic0PullbackNat` are `pic0Pullback`. -/
@[simp]
