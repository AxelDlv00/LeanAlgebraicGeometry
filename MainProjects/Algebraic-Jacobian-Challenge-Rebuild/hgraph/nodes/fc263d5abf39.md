theorem classDeg_inv (L : X.CechPic) : classDeg K (L⁻¹ : X.CechPic) = -classDeg K L := by
  have h := classDeg_mul K L (L⁻¹ : X.CechPic)
  rw [mul_inv_cancel, classDeg_one] at h
  omega
