private noncomputable def smulSection {U : X.Opens} (c : Γ(X, U)) :
    Γ(X, U) →ₗ[k] Γ(X, U) where
  toFun s := c * s
  map_add' := mul_add c
  map_smul' r s := by
    simp only [RingHom.id_apply, Scheme.overModule_smul_def, mul_left_comm]
