theorem posPoly_add_negPoly (z : LaurentPolynomial A) :
    Polynomial.toLaurent (posPoly z) + rightChart A (negPoly z) = z := by
  apply Finsupp.ext
  intro j
  have hsum : (Polynomial.toLaurent (posPoly z) + rightChart A (negPoly z)) j
      = (Polynomial.toLaurent (posPoly z)) j + (rightChart A (negPoly z)) j := rfl
  rw [hsum, toLaurent_posPoly_apply, rightChart_negPoly_apply]
  by_cases hj : 0 ≤ j <;> simp [hj]
