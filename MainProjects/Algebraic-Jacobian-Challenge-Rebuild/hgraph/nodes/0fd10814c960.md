noncomputable def pointSectionMap :
    B ⊗[Polynomial k] F →ₐ[Polynomial k] (B ⊗[k] F) ⧸ Ideal.span {pointGen k B F} :=
  Algebra.TensorProduct.lift
    ((Ideal.Quotient.mkₐ (Polynomial k) _).comp Algebra.TensorProduct.includeLeft)
    (pointQRight B F) (fun _ _ => Commute.all _ _)
