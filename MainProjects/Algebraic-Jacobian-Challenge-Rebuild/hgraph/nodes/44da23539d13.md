def res (γ : OneCocycle G U) (f : ∀ i, V i ⟶ U i) : OneCocycle G V where
  toOneCochain := γ.toOneCochain.res f
  ev_trans i j k T a b c := by
    simp only [OneCochain.res_ev]
    exact γ.ev_trans i j k (a ≫ f i) (b ≫ f j) (c ≫ f k)

@[simp]
