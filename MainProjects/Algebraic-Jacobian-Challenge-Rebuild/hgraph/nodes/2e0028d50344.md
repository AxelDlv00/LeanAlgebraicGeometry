theorem matrixLinearForm_mem (M : Matrix σ σ R) (i : σ) :
    matrixLinearForm M i ∈ homogeneousSubmodule σ R 1 :=
  Submodule.sum_mem _ fun j _ =>
    Submodule.smul_mem _ _ (X_mem_homogeneousSubmodule_one R j)
