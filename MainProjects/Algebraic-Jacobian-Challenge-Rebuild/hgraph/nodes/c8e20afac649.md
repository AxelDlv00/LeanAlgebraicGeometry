noncomputable def relSectionMulRight (A B : X.CurveDivisor)
    (a : ↥(divisorSections K A ⊤)) :
    R ⊗[K] ↥(divisorSections K B ⊤) →ₗ[R] R ⊗[K] ↥(divisorSections K (A + B) ⊤) :=
  LinearMap.baseChange R (sectionMulBilin K A B a)

@[simp]
