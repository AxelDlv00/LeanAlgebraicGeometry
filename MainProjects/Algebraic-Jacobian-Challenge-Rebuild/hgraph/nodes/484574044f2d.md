theorem ker_pointEv_map_localization_eq :
    (RingHom.ker (pointEv c)).map (algebraMap (B ⊗[k] F) L)
      = Ideal.span {algebraMap (B ⊗[k] F) L (pointGen k B F)} := by
  have hidemF : IsIdempotentElem (pointBaseChange B F (mapRight c elift)) := by
    rw [pointBaseChange_mapRight]
    exact IsIdempotentElem.map hidem (mapRightTwo c)
  have hdvd : algebraMap (B ⊗[k] F) L (mapRight c elift) ∈
      Ideal.span {algebraMap (B ⊗[k] F) L (pointGen k B F)} := by
    have hunit : IsUnit (algebraMap (B ⊗[k] F) L (1 - mapRight c elift)) :=
      IsLocalization.map_units L
        (⟨1 - mapRight c elift, Submonoid.mem_powers _⟩ : Submonoid.powers _)
    obtain ⟨v, hv⟩ := hunit.exists_right_inv
    have hmem : mapRight c elift * (1 - mapRight c elift)
        ∈ Ideal.span {pointGen k B F} := by
      rw [← ker_pointBaseChange, RingHom.mem_ker, map_mul, map_sub, map_one, mul_sub,
        mul_one, hidemF.eq, sub_self]
    have hfmem : algebraMap (B ⊗[k] F) L (mapRight c elift * (1 - mapRight c elift)) ∈
        Ideal.span {algebraMap (B ⊗[k] F) L (pointGen k B F)} := by
      have hmap := Ideal.mem_map_of_mem (algebraMap (B ⊗[k] F) L) hmem
      rwa [Ideal.map_span, Set.image_singleton] at hmap
    have heq : algebraMap (B ⊗[k] F) L (mapRight c elift)
        = algebraMap (B ⊗[k] F) L (mapRight c elift * (1 - mapRight c elift)) * v := by
      rw [map_mul, mul_assoc, hv, mul_one]
    rw [heq]; exact Ideal.mul_mem_right _ _ hfmem
  apply le_antisymm
  · rw [Ideal.map_le_iff_le_comap]
    refine (ker_pointEv_le_sup c elift hgen).trans ?_
    rw [sup_le_iff]
    refine ⟨?_, ?_⟩
    · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe, Ideal.mem_comap]
      exact Ideal.subset_span rfl
    · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe, Ideal.mem_comap]
      exact hdvd
  · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe]
    exact Ideal.mem_map_of_mem _ (by rw [RingHom.mem_ker]; exact pointEv_pointGen c)
