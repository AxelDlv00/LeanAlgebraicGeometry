noncomputable def frameMinor (k : Type u) [Field k] (d r : ℕ) (S : Type u) [CommRing S]
    [Algebra k S] (X : Matrix (Fin d) (Fin r) S) (I : Finset (Fin r))
    (hI : I.card = d) : Matrix (Fin d) (Fin d) S :=
  X.submatrix id (fun j => (I.orderIsoOfFin hI j : Fin r))
