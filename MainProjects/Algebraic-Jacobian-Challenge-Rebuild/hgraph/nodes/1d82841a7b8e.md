noncomputable def thetaFieldChartIndex (x : relCurve C K) : (thetaChartCover C K π).index :=
  if x ∈ (relCover C K (fiberTwoCover π)).V₀ then Sum.inl PUnit.unit
  else Sum.inr PUnit.unit
