noncomputable def pieceDescentEquiv :
    (S ⊗[A] B) ⊗[S] (S ⊗[A] B) ≃ₐ[S] S ⊗[A] (B ⊗[A] B) :=
  (Algebra.TensorProduct.cancelBaseChange A S S (S ⊗[A] B) B).trans
    (Algebra.TensorProduct.assoc A A S S B B)
