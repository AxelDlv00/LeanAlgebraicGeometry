lemma rightAlgebra_smul_tmul (a : A) (s : S) (b : A) :
    a • (s ⊗ₜ[R] b) = s ⊗ₜ[R] (a * b) := by
  change ((1 : S) ⊗ₜ[R] a) * (s ⊗ₜ[R] b) = s ⊗ₜ[R] (a * b)
  rw [Algebra.TensorProduct.tmul_mul_tmul, one_mul]
