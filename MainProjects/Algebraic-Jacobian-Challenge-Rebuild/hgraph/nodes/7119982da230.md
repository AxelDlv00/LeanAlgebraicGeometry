private lemma glue_telescope {G : Type u} [CommGroup G] {bx by' Lxy Lxs Lys m : G}
    (hb : bx * Lxy = by') (hL : Lxy * Lys = Lxs) :
    bx⁻¹ * Lxs⁻¹ * m⁻¹ = by'⁻¹ * Lys⁻¹ * m⁻¹ := by
  subst hb; rw [← hL]
  simp only [mul_inv, inv_inv]
  simp only [mul_comm, mul_left_comm, mul_assoc]
