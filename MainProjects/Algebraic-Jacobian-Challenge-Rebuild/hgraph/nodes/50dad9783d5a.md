def unitsAppLE (f : X.Hom Y) (V : Y.Opens) (U : X.Opens) (e : U ≤ f ⁻¹ᵁ V) :
    Γ(Y, V)ˣ →* Γ(X, U)ˣ :=
  Units.map (f.appLE V U e).hom
