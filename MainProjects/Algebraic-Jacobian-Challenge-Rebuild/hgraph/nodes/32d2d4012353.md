theorem basicOpen_algHom_ext [IsAffine X] (g : Γ(X, ⊤)) {C : Type u} [CommRing C]
    [Algebra Γ(X, ⊤) C] (φ ψ : Γ(X, X.basicOpen g) →ₐ[Γ(X, ⊤)] C) : φ = ψ :=
  (IsLocalization.algHom_subsingleton (Submonoid.powers g)).elim φ ψ
