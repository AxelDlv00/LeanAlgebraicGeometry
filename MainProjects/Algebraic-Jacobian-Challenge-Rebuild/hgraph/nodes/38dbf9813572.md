theorem sectionIdeal_eq_span_restrict_eqn (A : AffAdaptation D d) (j : D.index)
    {U : (relCurve C R).Opens} (hU : U ≤ D.pieces j) :
    d.sectionIdeal U = Ideal.span {
      ((relCurve C R).presheaf.map (homOfLE hU).op).hom (A.eqn j)} := by
  apply d.sectionIdeal_eq_span_of_generator
  · intro z hz
    rw [TopCat.Presheaf.germ_res_apply]
    exact A.eqn_regular j z (hU hz)
  · intro z hz
    rw [TopCat.Presheaf.germ_res_apply]
    exact A.germ_eqn_span_eq_stalkIdeal j (hU hz)
