theorem h0Map_injective (h : IsShortExact φ ψ) : Function.Injective φ.h0Map := by
  intro z w hzw
  have hval := congrArg Subtype.val hzw
  apply Subtype.ext
  apply Prod.ext
  · exact h.injective₀ (congrArg Prod.fst hval)
  · exact h.injective₁ (congrArg Prod.snd hval)
