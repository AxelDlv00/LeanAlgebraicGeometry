private noncomputable def resOverAlgHom (K : Type u) [CommRing K] (X : Scheme.{u})
    [X.Over (Spec (.of K))] {W U : X.Opens} (h : W ≤ U) :
    Γ(X, U) →ₐ[K] Γ(X, W) :=
  { X.resHom h with
    commutes' := fun r => X.overAlgebraMap_apply_res K (homOfLE h).op r }
