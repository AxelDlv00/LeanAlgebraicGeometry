theorem exists_divisorSections_pencil_decomposition
    (A B : X.CurveDivisor) (w : X.functionFieldˣ)
    (hsup : B ⊔
        (B - Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w) = A)
    (hB : Subsingleton (Sheaf.HModule (X.divisorSheaf K B) 1))
    (hBw : Subsingleton (Sheaf.HModule (X.divisorSheaf K
      (B - Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w)) 1))
    (hinf : Subsingleton (Sheaf.HModule (X.divisorSheaf K
      (B ⊓ (B - Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w))) 1))
    (hsupH : Subsingleton (Sheaf.HModule (X.divisorSheaf K
      (B ⊔ (B - Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w))) 1))
    (x : divisorSections K A ⊤) :
    ∃ z : Fin 2 → divisorSections K B ⊤,
      (x : X.functionField) =
        (z 0 : X.functionField) + (w : X.functionField) * (z 1 : X.functionField) := by
  let D := Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w
  have hmul : Submodule.map (Scheme.mulLinear K (w : X.functionField))
        (divisorSections K B ⊤) = divisorSections K (B - D) ⊤ := by
    rw [map_mulLinear_divisorSections_top K (Units.ne_zero w) B]
    have hmk : Units.mk0 (w : X.functionField) (Units.ne_zero w) = w := Units.ext rfl
    rw [hmk]
  have hsections : divisorSections K B ⊤ ⊔
      Submodule.map (Scheme.mulLinear K (w : X.functionField))
        (divisorSections K B ⊤) = divisorSections K A ⊤ := by
    rw [hmul, divisorSections_sup K B (B - D) hB hBw hinf hsupH, hsup]
  have hx : (x : X.functionField) ∈ divisorSections K B ⊤ ⊔
      Submodule.map (Scheme.mulLinear K (w : X.functionField))
        (divisorSections K B ⊤) := by
    rw [hsections]
    exact x.property
  obtain ⟨a, ha, b, hb, hab⟩ := Submodule.mem_sup.mp hx
  obtain ⟨z, hz, rfl⟩ := hb
  let z0 : divisorSections K B ⊤ := ⟨a, ha⟩
  let z1 : divisorSections K B ⊤ := ⟨z, hz⟩
  let z' : Fin 2 → divisorSections K B ⊤ := fun q => if q = 0 then z0 else z1
  refine ⟨z', ?_⟩
  simpa [z', z0, z1, Scheme.mulLinear_apply] using hab.symm
