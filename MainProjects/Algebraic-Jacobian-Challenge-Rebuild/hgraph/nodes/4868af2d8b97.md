def DivFam.toZar : DivFam C R π n → DivFamZar C R π n :=
  Quotient.lift
    (fun F : CertifiedDivisorFamily C R π n =>
      DivFamZar.mk F.eqns F.isLocallyCertified)
    (fun _ _ h => DivFamZar.mk_eq_mk_iff.mpr h)

@[simp]
