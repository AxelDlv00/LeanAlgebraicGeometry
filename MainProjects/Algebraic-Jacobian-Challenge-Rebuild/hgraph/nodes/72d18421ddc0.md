    PicEtAff C Γ(T.left, W.1)) : Prop where
  res : ∀ (i : ι) (W W' : T.left.affineOpens) (hW : W.1 ≤ O i) (hW' : W'.1 ≤ O i)

      v i W hi = v j W hj
