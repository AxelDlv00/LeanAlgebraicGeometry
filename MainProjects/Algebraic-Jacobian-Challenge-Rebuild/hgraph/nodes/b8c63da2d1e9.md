noncomputable def stalkIdeal (d : X.LocalEquations) (z : X) :
    Ideal (X.presheaf.stalk z) :=
  Ideal.span {(X.presheaf.germ (d.cover.opens z) z (d.cover.mem_opens z)).hom (d.eqn z)}
