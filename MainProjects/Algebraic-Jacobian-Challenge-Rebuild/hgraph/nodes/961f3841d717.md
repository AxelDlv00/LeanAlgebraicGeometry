theorem divisorBound_add (D₁ D₂ : X.CurveDivisor) {x : X} (hx : x ≠ genericPoint X) :
    divisorBound (D₁ + D₂) hx = divisorBound D₁ hx * divisorBound D₂ hx := by
  simp only [divisorBound, ← WithZero.coe_mul, ← ofAdd_add]
  congr 2
