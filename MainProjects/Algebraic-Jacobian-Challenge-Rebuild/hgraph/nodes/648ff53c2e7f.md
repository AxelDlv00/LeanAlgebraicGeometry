theorem sectionIdeal_eq_span_of_generator (d : X.LocalEquations) {U : X.Opens}
    (b : Γ(X, U))
    (hreg : ∀ (z : X) (hz : z ∈ U),
      (X.presheaf.germ U z hz).hom b ∈ nonZeroDivisors (X.presheaf.stalk z))
    (hspan : ∀ (z : X) (hz : z ∈ U),
      Ideal.span {(X.presheaf.germ U z hz).hom b} = d.stalkIdeal z) :
    d.sectionIdeal U = Ideal.span {b} := by
  apply le_antisymm
  · intro s hs
    apply Scheme.mem_span_singleton_of_forall_germ hreg
    intro z hz
    rw [hspan z hz]
    exact hs z hz
  · rw [Ideal.span_singleton_le_iff_mem]
    intro z hz
    rw [← hspan z hz]
    exact Ideal.subset_span (Set.mem_singleton _)
