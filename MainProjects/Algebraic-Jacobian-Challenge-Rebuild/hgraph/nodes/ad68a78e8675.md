lemma range_liftBaseChange_eq_span (f : M →ₗ[R] N) :
    LinearMap.range (LinearMap.liftBaseChange B f) =
      Submodule.span B (Set.range f) := by
  apply le_antisymm
  · rintro y ⟨x, rfl⟩
    induction x using TensorProduct.induction_on with
    | zero => exact Submodule.zero_mem _
    | add x y hx hy =>
        simpa only [map_add] using Submodule.add_mem _ hx hy
    | tmul b m =>
        rw [LinearMap.liftBaseChange_tmul]
        exact Submodule.smul_mem _ b (Submodule.subset_span ⟨m, rfl⟩)
  · rw [Submodule.span_le]
    rintro _ ⟨m, rfl⟩
    exact ⟨(1 : B) ⊗ₜ[R] m, by
      rw [LinearMap.liftBaseChange_tmul, one_smul]⟩
