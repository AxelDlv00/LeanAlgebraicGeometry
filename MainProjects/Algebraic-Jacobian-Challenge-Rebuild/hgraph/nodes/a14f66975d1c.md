theorem incl_toProjInt :
    incl J i S ≫ toProjInt J S =
      pullback.snd (toProjInt J S)
          (Proj.awayι P[J] (X i) (X_i_mem_deg_one J i) Nat.zero_lt_one) ≫
        Proj.awayι P[J] (X i) (X_i_mem_deg_one J i) Nat.zero_lt_one :=
  pullback.condition
