def submoduleQuotientAddEquiv {R : Type v} {M : Type w} [Ring R] [AddCommGroup M]
    [Module R M] (p : Submodule R M) : (M ⧸ p) ≃+ (M ⧸ p.toAddSubgroup) :=
  { Equiv.refl (M ⧸ p) with map_add' := fun _ _ => rfl }
