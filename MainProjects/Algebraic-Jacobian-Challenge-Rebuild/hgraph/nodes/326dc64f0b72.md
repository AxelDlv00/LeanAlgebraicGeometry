lemma comm_tN_inv_pow (f : P.Hom P') (m : ℕ) (n : N) :
    f.homN ((P.tN.inv ^ m) n) = (P'.tN.inv ^ m) (f.homN n) := by
  induction m with
  | zero => simp
  | succ m ih =>
    rw [pow_succ', pow_succ']
    simp only [Module.End.mul_apply]
    rw [f.comm_tN_inv, ih]
