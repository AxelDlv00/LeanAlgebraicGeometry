private theorem baseDivisorAt_sub_baseDivisor_eq_zero
    {L : Type u} [Field L] {X : Scheme.{u}} [X.Over (Spec (CommRingCat.of L))]
    [SmoothOfRelativeDimension 1 (X ↘ Spec (CommRingCat.of L))] [IsIntegral X]
    [LocallyOfFiniteType (X ↘ Spec (CommRingCat.of L))]
    [QuasiCompact (X ↘ Spec (CommRingCat.of L))]
    {T : Submodule L X.functionField} {A : X.CurveDivisor}
    (hTA : T ≤ Scheme.divisorSections L A ⊤) (hT : ∃ f ∈ T, f ≠ 0)
    {x : X} (hx : x ≠ genericPoint X) :
    (Scheme.baseDivisorAt L T (A - Scheme.baseDivisor L T A hT) ⟨x, hx⟩ : ℤ) = 0 := by
  have hTsub := Scheme.le_divisorSections_sub_baseDivisor L hTA hT
  obtain ⟨f, hfT, hf0, hzero⟩ := Scheme.exists_achiever_baseDivisor_sub L hTA hT hx
  have hle := Scheme.baseDivisorAt_le_coeffAt L hTsub hfT hf0 hx
  rw [hzero] at hle
  omega
