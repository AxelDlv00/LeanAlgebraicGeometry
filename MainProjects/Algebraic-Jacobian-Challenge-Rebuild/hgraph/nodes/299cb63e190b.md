            (normalizationCover_le_inr C σ 𝒩 𝒲 y) (ρ ((q₂).base y))

          ((normalizationCover C σ 𝒩 𝒲).opens y)
          (normalizationCover_le_inl C σ 𝒩 𝒲 y) (ρ ((q₁).base y)))
