lemma triple_le₁₂ (i j k : P.ι) :
    X.basicOpen (P.r i * (P.r j * P.r k)) ≤ X.basicOpen (P.r i * P.r j) := by
  have h : P.r i * (P.r j * P.r k) = P.r i * P.r j * P.r k := (mul_assoc _ _ _).symm
  rw [h]
  exact (X.basicOpen_mul _ _).trans_le inf_le_left
