lemma H1.res_res (f : ∀ i, V i ⟶ U i) (g : ∀ i, W i ⟶ V i) (x : H1 G U) :
    H1.res g (H1.res f x) = H1.res (fun i ↦ g i ≫ f i) x := by
  induction x using Quot.ind with
  | _ γ => exact congrArg (Quot.mk _) (γ.res_res f g)

@[simp]
