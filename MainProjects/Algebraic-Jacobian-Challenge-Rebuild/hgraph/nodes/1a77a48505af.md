def descentClasses (E : Algebra.EtaleCover A) :
    Subgroup (relPic C (overSpec k E.Carrier)) :=
  MonoidHom.eqLocus
    (relPicAlgMap (B := E.Carrier ⊗[A] E.Carrier) C (doubleInl E))
    (relPicAlgMap (B := E.Carrier ⊗[A] E.Carrier) C (doubleInr E))
