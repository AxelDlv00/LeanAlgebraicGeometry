lemma fiberCocycle_zero : fiberCocycle π 0 = 1 := by
  refine unitsCocycle_ext fun x y => ?_
  rw [one_unitsEvInf]
  refine Units.ext ?_
  have h := fiberCocycle_ratio π 0 x y

  rw [Units.val_one, ← h]
