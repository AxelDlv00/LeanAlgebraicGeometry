def SwallowedBy (D : AffCoverData C R) (d : (relCurve C R).LocalEquations) : Prop :=
  ∃ j₀ : D.index, d.supportLocus ⊆ (D.pieces j₀ : Set (relCurve C R)) ∧
    ∀ j : D.index, j ≠ j₀ → Disjoint d.supportLocus (D.pieces j : Set (relCurve C R))

end AffCoverData
