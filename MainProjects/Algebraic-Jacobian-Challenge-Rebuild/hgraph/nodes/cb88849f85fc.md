    linearEquiv₀ (R := R) ((mk₀ f).comp x (zero_add 0)) = f ≫ linearEquiv₀ (R := R) x := by
  apply (mk₀_bijective X Z).injective
  rw [mk₀_linearEquiv₀_apply, ← mk₀_comp_mk₀, mk₀_linearEquiv₀_apply]
