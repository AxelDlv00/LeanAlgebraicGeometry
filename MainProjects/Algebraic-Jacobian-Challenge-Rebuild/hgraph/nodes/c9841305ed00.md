lemma Scheme.coheight_add_one_le_of_specializes {Y : Scheme.{u}} {a b : ↥Y}
    (h : a ⤳ b) (hne : a ≠ b) :
    Order.coheight a + 1 ≤ Order.coheight b := by
  refine Order.coheight_add_one_le (lt_of_le_not_ge (show b ≤ a from h) ?_)
  intro hba
  exact hne (h.antisymm hba).eq
