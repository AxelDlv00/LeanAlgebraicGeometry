theorem span_singleton_algebraMap_eq_of_le_sup_map [IsLocalRing L]
    (q : Ideal B) [q.IsPrime] [IsLocalization.AtPrime L q]
    {f : B} (hf : f ∈ nonZeroDivisors B) [Module.Flat R (B ⧸ Ideal.span {f})]
    {s : Ideal R} (hs : ∀ r ∈ s, algebraMap R L r ∈ IsLocalRing.maximalIdeal L)
    {E : Ideal L} (hE : E ≤ Ideal.span {algebraMap B L f})
    (hcov : Ideal.span {algebraMap B L f} ≤ E ⊔ Ideal.map (algebraMap R L) s) :
    Ideal.span {algebraMap B L f} = E := by
  refine le_antisymm ?_ hE
  -- decompose the generator: `f = e + w` with `w ∈ s·L ∩ (f)`, so `w = c·f`
  obtain ⟨e, he, w, hw, hew⟩ := Submodule.mem_sup.mp
    (hcov (Ideal.mem_span_singleton_self _))
  have hwa : w ∈ Ideal.span {algebraMap B L f} := by
    have hsub : w = algebraMap B L f - e := by rw [← hew]; ring
    rw [hsub]
    exact Submodule.sub_mem _ (Ideal.mem_span_singleton_self _) (hE he)
  obtain ⟨c, hc⟩ := Ideal.mem_span_singleton'.mp hwa
  -- the map ideal through the tower
  have hsB : Ideal.map (algebraMap R L) s
      = Ideal.map (algebraMap B L) (Ideal.map (algebraMap R B) s) := by
    rw [Ideal.map_map, ← IsScalarTower.algebraMap_eq]
  -- clear denominators: `f · (c₀·u·v) = t·σ·v ∈ s·B`
  have hcw : c * algebraMap B L f
      ∈ Ideal.map (algebraMap B L) (Ideal.map (algebraMap R B) s) := by
    rw [← hsB, hc]
    exact hw
  obtain ⟨⟨⟨σ, hσ⟩, u⟩, hσu⟩ :=
    (IsLocalization.mem_map_algebraMap_iff q.primeCompl L).mp hcw
  obtain ⟨⟨c₀, t⟩, hct⟩ := IsLocalization.surj q.primeCompl c
  have hkey : algebraMap B L (c₀ * f * ↑u) = algebraMap B L (↑t * σ) := by
    simp only [map_mul]
    rw [← hct, ← hσu]
    ring
  obtain ⟨v, hv⟩ := (IsLocalization.eq_iff_exists q.primeCompl L).mp hkey
  have hfmul : f * (c₀ * ↑u * ↑v) ∈ Ideal.map (algebraMap R B) s := by
    have hre : f * (c₀ * ↑u * ↑v) = ↑v * (c₀ * f * ↑u) := by ring
    rw [hre, hv]
    exact Ideal.mul_mem_left _ _ (Ideal.mul_mem_left _ _ hσ)
  -- cancel `f` through the flat colength
  have hcuv : c₀ * ↑u * ↑v ∈ Ideal.map (algebraMap R B) s := by
    have h1 : f * (c₀ * ↑u * ↑v) ∈ s • (⊤ : Submodule R B) := by
      rw [Ideal.smul_top_eq_map]
      exact hfmul
    have h2 := mul_mem_smul_top_of_flat_coker_of_mem_smul_top hf s h1
    rwa [Ideal.smul_top_eq_map] at h2
  -- descend: `c ∈ s·L`
  have hcmem : c ∈ Ideal.map (algebraMap R L) s := by
    have h3 : c * algebraMap B L (↑t * ↑u * ↑v) = algebraMap B L (c₀ * ↑u * ↑v) := by
      simp only [map_mul]
      rw [← hct]
      ring
    have hunit : IsUnit (algebraMap B L (↑t * ↑u * ↑v : B)) := by
      have hco : (↑t * ↑u * ↑v : B) = ((t * u * v : q.primeCompl) : B) := by
        push_cast; ring
      rw [hco]
      exact IsLocalization.map_units L (t * u * v)
    obtain ⟨ω, hω⟩ := hunit
    have hcval : c = algebraMap B L (c₀ * ↑u * ↑v) * ↑ω⁻¹ := by
      rw [← h3, ← hω, mul_assoc, Units.mul_inv, mul_one]
    rw [hcval, hsB]
    exact Ideal.mul_mem_right _ _ (Ideal.mem_map_of_mem _ hcuv)
  have hcmax : c ∈ IsLocalRing.maximalIdeal L :=
    (Ideal.map_le_iff_le_comap.mpr fun r hr => Ideal.mem_comap.mpr (hs r hr)) hcmem
  -- the unit trick
  have hunit1c : IsUnit (1 - c) :=
    IsLocalRing.isUnit_one_sub_self_of_mem_nonunits c hcmax
  obtain ⟨ω, hω⟩ := hunit1c
  have hae : ↑ω * algebraMap B L f = e := by
    have hfe : algebraMap B L f - c * algebraMap B L f = e := by
      rw [hc, ← hew]
      ring
    rw [hω, sub_mul, one_mul]
    exact hfe
  rw [Ideal.span_singleton_le_iff_mem]
  have hfval : algebraMap B L f = ↑ω⁻¹ * e := by
    rw [← hae, ← mul_assoc, Units.inv_mul, one_mul]
  rw [hfval]
  exact Ideal.mul_mem_left _ _ he

end LocalBridge
