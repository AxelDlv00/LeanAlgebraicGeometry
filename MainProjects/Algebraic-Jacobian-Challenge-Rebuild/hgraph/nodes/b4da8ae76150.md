lemma algebraMap_tmul_sub_mem_span (p : Polynomial k) :
    (algebraMap (Polynomial k) B p) ⊗ₜ[k] (1 : B)
        - (1 : B) ⊗ₜ[k] (algebraMap (Polynomial k) B p)
      ∈ Ideal.span {diagGen (k := k) (B := B)} := by
  rw [Ideal.mem_span_singleton, diagGen, algebraMap_eq_aeval]
  have hL : (aeval (coord k B) p) ⊗ₜ[k] (1 : B)
      = aeval (coord k B ⊗ₜ[k] (1 : B)) p := by
    rw [← Algebra.TensorProduct.includeLeft_apply (S := k),
      ← Polynomial.aeval_algHom_apply, Algebra.TensorProduct.includeLeft_apply]
  have hR : (1 : B) ⊗ₜ[k] (aeval (coord k B) p)
      = aeval ((1 : B) ⊗ₜ[k] coord k B) p := by
    rw [← Algebra.TensorProduct.includeRight_apply,
      ← Polynomial.aeval_algHom_apply, Algebra.TensorProduct.includeRight_apply]
  rw [hL, hR]
  exact aeval_sub_dvd _ _ p
