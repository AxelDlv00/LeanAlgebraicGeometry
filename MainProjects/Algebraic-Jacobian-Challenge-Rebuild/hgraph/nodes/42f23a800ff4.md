lemma lmul'_baseChange (z : B ⊗[k] B) :
    Algebra.TensorProduct.lmul' (R := Polynomial k) (S := B) (baseChange z)
      = Algebra.TensorProduct.lmul' k z := by
  induction z with
  | zero => simp
  | tmul x y => rw [baseChange_tmul, Algebra.TensorProduct.lmul'_apply_tmul,
      Algebra.TensorProduct.lmul'_apply_tmul]
  | add a b ha hb => simp only [map_add, ha, hb]
