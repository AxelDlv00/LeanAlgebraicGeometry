lemma comm_tN_inv (f : P.Hom P') (n : N) :
    f.homN (P.tN.inv n) = P'.tN.inv (f.homN n) := by
  have h : f.homN n = P'.tN.val (f.homN (P.tN.inv n)) := by
    conv_lhs => rw [← P.tN_val_inv_apply n]
    exact f.comm_tN _
  rw [h, P'.tN_inv_val_apply]
