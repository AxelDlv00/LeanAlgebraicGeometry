theorem twoChartCover_pullback (f : X ⟶ Y) (sel : Y → Bool) (hmem : ∀ y, y ∈ V (sel y)) :
    (twoChartCover V sel hmem).pullback f
      = twoChartCover (fun s ↦ f ⁻¹ᵁ V s) (fun x ↦ sel (f.base x))
          (fun x ↦ hmem (f.base x)) :=
  rfl
