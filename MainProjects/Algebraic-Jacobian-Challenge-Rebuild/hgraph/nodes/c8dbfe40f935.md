theorem chi_skyModule [Nontrivial K] (x : X) (M : ModuleCat.{u} K) :
    Sheaf.chi (skyModule x M) = Module.finrank K M := by
  rw [Sheaf.chi_eq_h0 (skyModule_subsingleton_hModule_one x M), h0_skyModule]
