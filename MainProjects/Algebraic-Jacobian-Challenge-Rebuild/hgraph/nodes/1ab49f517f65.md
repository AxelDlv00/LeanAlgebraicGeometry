theorem isDivisorDegree_nat_of_point (p : overSpec k k ⟶ C) (n : ℕ) :
    IsDivisorDegree C (n : ℤ) := by
  obtain ⟨D, hD⟩ := isDivisorDegree_one_of_point C p
  refine ⟨n • D, ?_⟩
  rw [Scheme.CurveDivisor.deg_nsmul' k, hD]
  simp
