noncomputable def projectiveQuotientBaseChangeEquiv (N : Submodule R M)
    [Module.Projective R (M ⧸ N)] :
    S ⊗[R] N ≃ₗ[S] N.baseChange S := by
  have hsplit : quotRetract N ∘ₗ N.subtype = LinearMap.id :=
    LinearMap.ext fun x => quotRetract_apply_of_mem N x.2
  have hinj : Function.Injective (LinearMap.baseChange S N.subtype) := by
    have hleft : LinearMap.baseChange S (quotRetract N) ∘ₗ
        LinearMap.baseChange S N.subtype = LinearMap.id := by
      rw [← LinearMap.baseChange_comp, hsplit, LinearMap.baseChange_id]
    exact Function.LeftInverse.injective
      (g := LinearMap.baseChange S (quotRetract N))
      fun x => by rw [← LinearMap.comp_apply, hleft, LinearMap.id_apply]
  refine LinearEquiv.ofBijective (N.toBaseChange S) ⟨?_, N.toBaseChange_surjective S⟩
  intro x y hxy
  apply hinj
  exact congrArg Subtype.val hxy
