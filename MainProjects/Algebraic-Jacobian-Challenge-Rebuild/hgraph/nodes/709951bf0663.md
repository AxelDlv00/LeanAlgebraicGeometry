theorem exists_h_mem_basicOpen_forall_mul_read_mem_span
    (K : Submodule R (relThetaSections C R π a)) (b : Bool)
    (s : relThetaSections C R π a) [Module.Finite R ↥K]
    {z : relCurve C R} (hz : z ∈ relPinnedChart C R π b)
    (hsupp : (isAffineOpen_relPinnedChart C R π b).primeIdealOf ⟨z, hz⟩ ∉
      Module.support Γ(relCurve C R, relPinnedChart C R π b) ↥(chartColengthModule K b s)) :
    ∃ h : Γ(relCurve C R, relPinnedChart C R π b),
      z ∈ (relCurve C R).basicOpen h ∧
      ∀ ⦃ψ : relThetaSections C R π a⦄, ψ ∈ K →
        h * relThetaResSide a b (le_rfl) ψ
          ∈ Ideal.span {relThetaResSide a b (le_rfl) s} := by
  haveI := chartColengthModule_finite K b s
  obtain ⟨h, hmem, hann⟩ :=
    (isAffineOpen_relPinnedChart C R π b).exists_mem_basicOpen_forall_smul_eq_zero
      (↥(chartColengthModule K b s)) hz hsupp
  exact ⟨h, hmem, forall_mul_read_mem_span_of_forall_smul_eq_zero K b s hann⟩
