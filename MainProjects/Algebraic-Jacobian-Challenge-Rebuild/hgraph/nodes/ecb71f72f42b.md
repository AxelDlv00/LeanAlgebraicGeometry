noncomputable def fiberWeilDivisor : Y.CurveDivisor :=
  Finsupp.mapRange (fun n => max n 0) (max_self 0)
    (Scheme.divOf (Y ↘ Spec (CommRingCat.of K)) (fiberCoordUnit π))
