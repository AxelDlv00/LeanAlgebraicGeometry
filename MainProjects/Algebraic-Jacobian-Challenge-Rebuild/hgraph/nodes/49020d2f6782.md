noncomputable def divisorBound (D : X.CurveDivisor) {x : X} (hx : x ≠ genericPoint X) :
    WithZero (Multiplicative ℤ) :=
  letI D' : {x : X // x ≠ genericPoint X} →₀ ℤ := D
  ((Multiplicative.ofAdd (D' ⟨x, hx⟩) : Multiplicative ℤ) : WithZero (Multiplicative ℤ))
