noncomputable def piBaseChangeDistrib :
    ((∀ i, S i) ⊗[A] N) ≃ₗ[A] ∀ i, S i ⊗[A] N :=
  (TensorProduct.comm A _ _).trans ((TensorProduct.piRight A A N S).trans
    (LinearEquiv.piCongrRight fun i ↦ TensorProduct.comm A N (S i)))

@[simp]
