theorem mul_mem_smul_top_of_flat_coker_of_mem_smul_top
    {f : B} (hf : f ∈ nonZeroDivisors B) [Module.Flat R (B ⧸ Ideal.span {f})]
    (s : Ideal R) {x : B} (h : f * x ∈ s • (⊤ : Submodule R B)) :
    x ∈ s • (⊤ : Submodule R B) := by
  have hreg := Algebra.TensorProduct.includeRight_mem_nonZeroDivisors_of_flat_coker
    (R := R) (R' := R ⧸ s) f hf
  -- the tensor image of `f * x` vanishes
  have hz : ((1 : R ⧸ s) ⊗ₜ[R] (f * x) : (R ⧸ s) ⊗[R] B) = 0 := by
    apply (TensorProduct.quotTensorEquivQuotSMul B s).injective
    rw [map_zero, TensorProduct.quotTensorEquivQuotSMul_mk_one_tmul,
      Submodule.Quotient.mk_eq_zero]
    exact h
  -- split off the regular factor
  have hsplit : ((1 : R ⧸ s) ⊗ₜ[R] (f * x) : (R ⧸ s) ⊗[R] B)
      = Algebra.TensorProduct.includeRight f * ((1 : R ⧸ s) ⊗ₜ[R] x) := by
    rw [Algebra.TensorProduct.includeRight_apply, Algebra.TensorProduct.tmul_mul_tmul,
      one_mul]
  have hx0 : ((1 : R ⧸ s) ⊗ₜ[R] x : (R ⧸ s) ⊗[R] B) = 0 :=
    (mul_left_mem_nonZeroDivisors_eq_zero_iff hreg).mp (by rw [← hsplit]; exact hz)
  have hmk : (TensorProduct.quotTensorEquivQuotSMul B s) ((1 : R ⧸ s) ⊗ₜ[R] x)
      = (TensorProduct.quotTensorEquivQuotSMul B s) 0 := by rw [hx0]
  rwa [TensorProduct.quotTensorEquivQuotSMul_mk_one_tmul, map_zero,
    Submodule.Quotient.mk_eq_zero] at hmk
