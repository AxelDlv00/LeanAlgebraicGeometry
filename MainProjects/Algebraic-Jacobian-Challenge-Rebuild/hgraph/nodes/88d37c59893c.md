theorem isUnit_iff_C_mul_T {f : LaurentPolynomial R} :
    IsUnit f ↔ ∃ (c : R) (n : ℤ), IsUnit c ∧ f = LaurentPolynomial.C c * T n := by

  rintro ⟨c, n, hc, rfl⟩
  exact isUnit_C_mul_T hc n
