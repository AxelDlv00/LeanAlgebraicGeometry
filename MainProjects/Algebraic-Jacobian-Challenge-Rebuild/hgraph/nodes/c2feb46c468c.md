def openSubfunctor (D : Over (Spec (.of k))) (U : D.left.Opens) :
    Subfunctor (yoneda.obj D) :=
  Subfunctor.range (yoneda.map
    (Over.homMk (U := Over.mk (U.ι ≫ D.hom)) U.ι))
