noncomputable def DivFamZar.toAff {n : ℕ} (F : DivFamZar C R π n) : DivFamZarAff C R n :=

    (fun dp => DivFamZarAff.mk dp.1 (isLocallyCertifiedAff_of_isLocallyCertified dp.2))
