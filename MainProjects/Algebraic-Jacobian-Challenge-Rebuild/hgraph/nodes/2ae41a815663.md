def mixedParamFamily {ι : Type u}
    (nn : ι → ℕ)
    (D : ι → Over (Spec (.of k)))
    (rep : ∀ i, (divFunctor C π (nn i)).RepresentableBy (D i))
    (m : ι → ℕ) (Z : ι → (C ⊗ overSpec k k).left.CurveDivisor)
    (hdeg : ∀ i, Scheme.CurveDivisor.deg k (Z i)
      = (m i : ℤ) * classDeg k (thetaCechClass C) - (nn i : ℤ))
    (V : ∀ i, (D i).left.Opens) :
    ∀ i, yoneda.obj ((V i : Scheme.{u})) ⟶ (pic0SigmaSheaf C).1 :=
  fun i => restrictChart (abelSigmaChart C π (nn i) (rep i) (m i) (Z i) (hdeg i)) (V i)
