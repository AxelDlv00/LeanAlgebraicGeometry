noncomputable def relSectionMul (A B : X.CurveDivisor) :
    R ⊗[K] (↥(divisorSections K A ⊤) ⊗[K] ↥(divisorSections K B ⊤)) →ₗ[R]
      R ⊗[K] ↥(divisorSections K (A + B) ⊤) :=
  LinearMap.baseChange R (sectionMul K A B)

@[simp]
