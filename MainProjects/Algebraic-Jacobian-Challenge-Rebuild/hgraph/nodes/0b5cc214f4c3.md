theorem surjective_of_testGeneral {J : Over (Spec (.of k))}
    (rep : (pic0TypeFunctor C).RepresentableBy J)
    (lam : pic0Subgroup C DivO)
    (h : ∀ y : J.left, ∃ (T : Over (Spec (.of k))) (p : T.left) (q : T ⟶ DivO) (t : T ⟶ J),
      t.left.base p = y ∧ pic0Map C q lam = rep.homEquiv t) :
    Function.Surjective (abelOfPic0Class rep lam).left.base := by
  intro y
  obtain ⟨T, p, q, t, hp, hcl⟩ := h y
  have hqt : q ≫ abelOfPic0Class rep lam = t := by
    apply rep.homEquiv.injective
    rw [abelOfPic0Class_comp_class]
    exact hcl
  refine ⟨q.left.base p, ?_⟩
  have := congrArg (fun (m : T ⟶ J) => m.left.base p) hqt
  simpa [hp] using this
