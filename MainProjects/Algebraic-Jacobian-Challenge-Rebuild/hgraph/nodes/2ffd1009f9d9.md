noncomputable def unitIso : Over.mk (𝟙 (Spec (CommRingCat.of k))) ≅ overSpec k k :=
  Over.isoMk (Iso.refl _) (by
    change 𝟙 _ ≫ (overSpec k k).hom = (Over.mk (𝟙 (Spec (CommRingCat.of k)))).hom
    rw [Category.id_comp]
    exact specMap_algebraMap_self_eq_id k)

@[simp]
