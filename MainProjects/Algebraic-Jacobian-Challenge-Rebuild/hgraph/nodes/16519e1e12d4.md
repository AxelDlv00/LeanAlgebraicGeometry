theorem CertifiedDivisorFamily.stalkIdeal_eq_span_windowGerm (g : ℕ)
    (G : CertifiedDivisorFamily C R π g)
    (hwin : ∀ z : relCurve C R, ∃ s : Ideal R,
      (∀ r ∈ s, algebraMap R ((relCurve C R).presheaf.stalk z) r
        ∈ IsLocalRing.maximalIdeal ((relCurve C R).presheaf.stalk z)) ∧
      G.eqns.stalkIdeal z
        ≤ Ideal.span (divFamEpsWindowGermSet hπ g (DivFam.mk G) z)
          ⊔ Ideal.map (algebraMap R ((relCurve C R).presheaf.stalk z)) s)
    (z : relCurve C R) :
    G.eqns.stalkIdeal z
      = Ideal.span (divFamEpsWindowGermSet hπ g (DivFam.mk G) z) := by
  obtain ⟨s, hs, hcov⟩ := hwin z
  exact G.adaptation.stalkIdeal_eq_of_le_sup_map
    (fun j => G.certified.projective_colength j) hs
    (span_divFamEpsWindowGermSet_le hπ g G z) hcov
