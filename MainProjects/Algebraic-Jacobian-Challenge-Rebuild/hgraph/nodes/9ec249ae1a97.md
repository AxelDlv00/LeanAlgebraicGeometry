lemma evInf_ofPairs (γ : OneCocycle G U) :
    ofPairs (fun i j ↦ γ.evInf i j) (fun i j k ↦ γ.evInf_trans i j k) = γ := by
  ext1
  ext i j T a b
  exact (γ.toOneCochain.ev_eq_evInf i j a b).symm
