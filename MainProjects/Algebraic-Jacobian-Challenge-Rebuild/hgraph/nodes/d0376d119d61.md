theorem app_surjective_of_isLocallySurjective {F G : Scheme.{u}ᵒᵖ ⥤ Type v} (φ : F ⟶ G)
    [Presheaf.IsLocallySurjective Scheme.zariskiTopology φ]
    {X : Scheme.{u}} [Subsingleton (X : Type u)] (x : X) :
    Function.Surjective (φ.app (op X)) := by
  intro s
  have hcov : Presheaf.imageSieve φ s ∈ Scheme.zariskiTopology X :=
    Presheaf.imageSieve_mem Scheme.zariskiTopology φ s
  obtain ⟨t, ht⟩ := id_mem_of_mem_zariskiTopology x _ hcov
  refine ⟨t, ?_⟩
  simpa using ht
