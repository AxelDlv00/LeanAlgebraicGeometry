noncomputable def Scheme.finiteMulMap
    (U T : Submodule K X.functionField) (b : Module.Basis ι K U) :
    (ι → T) →ₗ[K] X.functionField := by
  classical
  exact finiteComponentSum fun i =>
    (Scheme.mulLinear K (b i : X.functionField)).comp T.subtype

@[simp]
