def IsLineBundle (M : X.Modules) : Prop :=
  ∀ x : X, ∃ U : X.Opens, x ∈ U ∧
    Nonempty ((restrictFunctor U.ι).obj M ≅ SheafOfModules.unit U.toScheme.ringCatSheaf)
