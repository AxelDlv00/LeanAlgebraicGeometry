lemma comp_rowSnd :
    p ≫ rowSnd q hq = pullback.lift p q (by rw [hp, hq]) := by
  apply pullback.hom_ext
  · rw [Category.assoc, rowSnd_fst, Category.comp_id, pullback.lift_fst]
  · rw [Category.assoc, rowSnd_snd, pullback.lift_snd, ← Category.assoc, hp,
      Category.id_comp]
