    (x : G n) (hx : x ∈ K) :

  have ha : a = ∑ t, (basis.repr a t) • basis t := (basis.sum_repr a).symm

        ∑ t, (basis.repr a t) •

            hpi g r1 r2 b1 b2 i j n (basis t) x := by

        (divUniversalHighWindowShiftMulLinear (C := C) (pi := pi) hpi g n)
    change
      (T a) x =
        ∑ t, (basis.repr a t) •
