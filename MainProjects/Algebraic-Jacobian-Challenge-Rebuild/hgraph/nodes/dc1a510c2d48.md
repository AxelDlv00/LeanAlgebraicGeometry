def Pic0GaloisInvariant (T : Over (Spec (CommRingCat.of K))) : Type u :=
  {x : (pic0TypeFunctor C).obj
      (op ((restrictTest K L).obj (baseTest (k' := L) T))) //
    ∀ gamma : L ≃ₐ[K] L,
      (pic0TypeFunctor C).map (twistTest T gamma).op x = x}
