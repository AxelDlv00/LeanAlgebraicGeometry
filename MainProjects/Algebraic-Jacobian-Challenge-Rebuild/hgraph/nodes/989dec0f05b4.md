lemma exists_mem_ker_notMem_sq [IsLocalRing B] [IsNoetherianRing B]
    [IsLocalRing A] [IsNoetherianRing A] (φ : B →+* A)
    (hφ : Function.Surjective φ)
    (hlt : (IsLocalRing.maximalIdeal A).spanFinrank
      < (IsLocalRing.maximalIdeal B).spanFinrank) :
    ∃ x ∈ RingHom.ker φ, x ∉ (IsLocalRing.maximalIdeal B) ^ 2 := by
  classical
  by_contra hcon
  push Not at hcon
  -- generators of `𝔪_A` and their lifts into `𝔪_B`
  obtain ⟨t, htcard, htspan⟩ :=
    (IsNoetherian.noetherian (IsLocalRing.maximalIdeal A)
      : (IsLocalRing.maximalIdeal A).FG).exists_span_finset_card_eq_spanFinrank
  have hmap := map_maximalIdeal_eq_of_surjective φ hφ
  have hlift : ∀ a : A, a ∈ IsLocalRing.maximalIdeal A →
      ∃ b, b ∈ IsLocalRing.maximalIdeal B ∧ φ b = a := by
    intro a ha
    rw [← hmap] at ha
    exact (Ideal.mem_map_iff_of_surjective φ hφ).mp ha
  choose bfun hbmem hbval using hlift
  have htsub : (↑t : Set A) ⊆ IsLocalRing.maximalIdeal A := by
    rw [← htspan]
    exact Submodule.subset_span
  -- the lifted finset
  set bF : Finset B :=
    t.attach.image (fun a => bfun a.1 (htsub a.2)) with hbFdef
  have hbFsub : (↑bF : Set B) ⊆ IsLocalRing.maximalIdeal B := by
    intro b hb
    simp only [hbFdef, Finset.coe_image, Set.mem_image] at hb
    obtain ⟨a, -, rfl⟩ := hb
    exact hbmem _ _
  -- `𝔪_B ≤ span bF ⊔ 𝔪_B • 𝔪_B`
  have hkey : IsLocalRing.maximalIdeal B
      ≤ Ideal.span ↑bF ⊔ IsLocalRing.maximalIdeal B • IsLocalRing.maximalIdeal B := by
    intro u hu
    have hφu : φ u ∈ Submodule.span A (↑t : Set A) := by
      have hm : φ u ∈ IsLocalRing.maximalIdeal A :=
        hmap ▸ Ideal.mem_map_of_mem φ hu
      rwa [← htspan] at hm
    rw [Submodule.mem_span_finset] at hφu
    obtain ⟨f, -, hf⟩ := hφu
    -- lift the coefficients
    have hcoef : ∀ a : A, ∃ g : B, φ g = f a := fun a => hφ (f a)
    choose gfun hgval using hcoef
    set v : B := ∑ a ∈ t.attach, gfun a.1 * bfun a.1 (htsub a.2) with hvdef
    have hvmem : v ∈ Ideal.span (↑bF : Set B) := by
      refine Submodule.sum_mem _ fun a _ => ?_
      refine Ideal.mul_mem_left _ _ (Submodule.subset_span ?_)
      simp only [hbFdef, Finset.coe_image, Set.mem_image]
      exact ⟨a, by simp, rfl⟩
    have hφv : φ v = φ u := by
      rw [hvdef, map_sum]
      rw [← hf, ← Finset.sum_attach t (fun a => f a • a)]
      refine Finset.sum_congr rfl fun a _ => ?_
      rw [map_mul, hgval, hbval, smul_eq_mul]
    have huv : u - v ∈ RingHom.ker φ := by
      rw [RingHom.mem_ker, map_sub, hφv, sub_self]
    have huvsq := hcon _ huv
    have : u = v + (u - v) := by ring
    rw [this]
    refine Submodule.add_mem _ (Ideal.mem_sup_left hvmem) (Ideal.mem_sup_right ?_)
    rwa [Ideal.smul_eq_mul, ← pow_two]
  -- Nakayama: `𝔪_B = span bF`, contradicting the spanFinrank gap
  have hsup := Submodule.sup_eq_sup_smul_of_le_smul_of_le_jacobson
    (I := IsLocalRing.maximalIdeal B) (J := ⊥) (N := Ideal.span ↑bF)
    (N' := IsLocalRing.maximalIdeal B)
    (IsNoetherian.noetherian _) (IsLocalRing.maximalIdeal_le_jacobson ⊥) hkey
  rw [Submodule.bot_smul, sup_bot_eq] at hsup
  have hMle : IsLocalRing.maximalIdeal B ≤ Ideal.span ↑bF := by
    rw [← hsup]
    exact le_sup_right
  have hMeq : Ideal.span (↑bF : Set B) = IsLocalRing.maximalIdeal B :=
    le_antisymm (Ideal.span_le.mpr hbFsub) hMle
  have hcount : (IsLocalRing.maximalIdeal B).spanFinrank
      ≤ (IsLocalRing.maximalIdeal A).spanFinrank := by
    calc (IsLocalRing.maximalIdeal B).spanFinrank
        = (Ideal.span (↑bF : Set B)).spanFinrank := by rw [hMeq]
      _ ≤ (↑bF : Set B).ncard :=
          Submodule.spanFinrank_span_le_ncard_of_finite bF.finite_toSet
      _ = bF.card := Set.ncard_coe_finset _
      _ ≤ t.attach.card := by rw [hbFdef]; exact Finset.card_image_le
      _ = t.card := Finset.card_attach
      _ = _ := htcard
  exact absurd (hlt.trans_le hcount) (lt_irrefl _)
