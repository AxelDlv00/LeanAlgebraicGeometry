theorem Submodule.le_of_forall_dual_ker {S W : Submodule K V}
    (h : ∀ Λ : Module.Dual K V, S ≤ LinearMap.ker Λ → W ≤ LinearMap.ker Λ) :
    W ≤ S := by
  intro w hw
  by_contra hws
  have hq : S.mkQ w ≠ 0 := by
    rw [Submodule.mkQ_apply, Ne, Submodule.Quotient.mk_eq_zero]
    exact hws
  obtain ⟨φ, hφ⟩ := Module.Projective.exists_dual_ne_zero K hq
  have hker : S ≤ LinearMap.ker (φ.comp S.mkQ) := by
    intro s hs
    rw [LinearMap.mem_ker, LinearMap.comp_apply, Submodule.mkQ_apply,
      (Submodule.Quotient.mk_eq_zero S).mpr hs, map_zero]
  have hwk := h (φ.comp S.mkQ) hker hw
  rw [LinearMap.mem_ker, LinearMap.comp_apply] at hwk
  exact hφ hwk

end Shell
