noncomputable def frameMatrix (X : Matrix (Fin d) (Fin r) S) (I : Finset (Fin r))
    (hI : I.card = d) : Matrix (Fin d) (Fin r) S :=
  (frameMinor k d r S X I hI)⁻¹ * X
