theorem comp_eq_toUnit_comp_of_comp_snd_eq {T X Y Z : C} {p : T ⟶ X ⊗ Y} {y : 𝟙_ C ⟶ Y}
    {f : X ⊗ Y ⟶ Z} {c : 𝟙_ C ⟶ Z} (h : p ≫ snd X Y = toUnit T ≫ y)
    (hf : lift (𝟙 X) (toUnit X ≫ y) ≫ f = toUnit X ≫ c) :
    p ≫ f = toUnit T ≫ c := by
