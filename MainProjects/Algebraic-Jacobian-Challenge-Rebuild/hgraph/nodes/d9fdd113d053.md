theorem pic0SigmaFunctor_isSheafFor_etaleSingleton
    {k : Type u} [Field k]
    (C : Over (Spec (.of k))) [SmoothOfRelativeDimension 1 C.hom]
    [IsProper C.hom] [GeometricallyIrreducible C.hom]
    [GeometricallyReduced C.hom]
    {R S : CommRingCat.{u}} (f : R ⟶ S)
    (hf : Etale (Spec.map f)) (hs : Surjective (Spec.map f)) :
    Presieve.IsSheafFor (pic0SigmaFunctor C)
      (.singleton (Spec.map f)) := by
  algebraize [f.hom]
  haveI : Algebra.Etale R S :=
    RingHom.etale_algebraMap.mp
      ((HasRingHomProperty.Spec_iff (P := @Etale)).mp hf)
  have hcomap :
      Function.Surjective (PrimeSpectrum.comap (algebraMap R S)) := by
    exact (surjective_iff (Spec.map f)).mp hs
  let E : Algebra.EtaleCover R := Algebra.EtaleCover.of S hcomap
  let e : Spec S ≅ Spec (.of E.Carrier) :=
    Scheme.Spec.mapIso
      (Algebra.EtaleCover.ofEquiv S hcomap).toRingEquiv.toCommRingCatIso.op
  have he : e.hom ≫
      Spec.map (CommRingCat.ofHom (algebraMap R E.Carrier)) =
      Spec.map f := by
    change Spec.map (CommRingCat.ofHom
        (Algebra.EtaleCover.ofEquiv S hcomap).toRingEquiv.toRingHom) ≫
      Spec.map (CommRingCat.ofHom
        (algebraMap R (Algebra.EtaleCover.of S hcomap).Carrier)) =
      Spec.map f
    rw [← Spec.map_comp]
    congr 1
    ext r
    exact (Algebra.EtaleCover.ofEquiv S hcomap).commutes r
  exact (Presieve.isSheafFor_singleton_iff_of_iso
    (Spec.map f)
    (Spec.map (CommRingCat.ofHom (algebraMap R E.Carrier))) e he).mpr
      (pic0SigmaFunctor_isSheafFor_etaleCover C E)
