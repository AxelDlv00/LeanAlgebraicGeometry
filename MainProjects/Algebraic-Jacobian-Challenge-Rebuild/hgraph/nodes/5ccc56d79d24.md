theorem divisorClass_add (D D' : X.CurveDivisor) :
    divisorClass K (D + D') = divisorClass K D * divisorClass K D' := by
  simp only [divisorClass]
  exact Finsupp.prod_add_index' (fun p => zpow_zero _) (fun p m n => zpow_add _ m n)
