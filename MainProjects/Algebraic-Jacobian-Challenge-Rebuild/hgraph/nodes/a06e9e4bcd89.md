theorem exists_diagonalLocalization
    [Algebra.FormallyUnramified (Polynomial k) B] [Algebra.EssFiniteType (Polynomial k) B] :
    ∃ elift : B ⊗[k] B, Algebra.TensorProduct.lmul' k elift = 0 ∧
      IsIdempotentElem (baseChange elift) ∧
      RingHom.ker (Algebra.TensorProduct.lmul' (R := Polynomial k) (S := B))
        = Ideal.span {baseChange elift} ∧
      (RingHom.ker (Algebra.TensorProduct.lmul' k (S := B))).map
          (algebraMap (B ⊗[k] B) (Localization.Away (1 - elift)))
        = Ideal.span {algebraMap (B ⊗[k] B) (Localization.Away (1 - elift))
            (diagGen (k := k) (B := B))} := by
  obtain ⟨elift, hidem, hgen⟩ := exists_idempotent_lift (k := k) (B := B)
  exact ⟨elift, lmul'_lift elift hgen, hidem, hgen,
    ker_map_localization_eq elift hidem hgen (Localization.Away (1 - elift))⟩
