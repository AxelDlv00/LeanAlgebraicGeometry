lemma baseChange_surjective : Function.Surjective (baseChange (k := k) (B := B)) := by
  intro z
  induction z with
  | zero => exact ⟨0, map_zero _⟩
  | tmul x y => exact ⟨x ⊗ₜ[k] y, baseChange_tmul x y⟩
  | add a b ha hb =>
      obtain ⟨a', ha'⟩ := ha; obtain ⟨b', hb'⟩ := hb
      exact ⟨a' + b', by rw [map_add, ha', hb']⟩
