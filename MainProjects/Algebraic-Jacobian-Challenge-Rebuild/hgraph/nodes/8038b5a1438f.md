lemma exists_mem_pieces (z : relCurve C R) : ∃ j : D.index, z ∈ D.pieces j := by
  have hz : z ∈ (relCover C R (fiberTwoCover π)).V₀ ⊔ (relCover C R (fiberTwoCover π)).V₁ := by
    rw [relCover_sup]
    trivial
  rcases Opens.mem_sup.mp hz with h | h
  · have hcov := D.cover₀ h
    rw [Opens.mem_iSup] at hcov
    obtain ⟨j, hj⟩ := hcov
    exact ⟨Sum.inl j, hj⟩
  · have hcov := D.cover₁ h
    rw [Opens.mem_iSup] at hcov
    obtain ⟨j, hj⟩ := hcov
    exact ⟨Sum.inr j, hj⟩
