noncomputable def finiteComponentSum (f : ι → M →ₗ[R] N) :
    (ι → M) →ₗ[R] N := by
  classical
  exact ∑ t : ι, (f t).comp (LinearMap.proj t)

/-- A finite component sum evaluated on a vector supported in one coordinate
is that component map. -/
@[simp]
