  charts : Set (pic0_sepClosed_representableBy (C := C)).1.left.affineOpens
  finite_charts : charts.Finite

    (⨆ U ∈ charts,

    (pic0_sepClosed_representableBy (C := C)).1.left.FiniteAffineOverlapPresentation U V
