lemma tmul_sub_tmul_mem_span_mapRightTwo (x : B) :
    x ⊗ₜ[Polynomial k] (1 : F) - (1 : B) ⊗ₜ[Polynomial k] c x
      ∈ Ideal.span {mapRightTwo c e} := by
  have hker : x ⊗ₜ[Polynomial k] (1 : B) - (1 : B) ⊗ₜ[Polynomial k] x
      ∈ RingHom.ker (Algebra.TensorProduct.lmul' (R := Polynomial k) (S := B)) := by
    rw [RingHom.mem_ker, map_sub, Algebra.TensorProduct.lmul'_apply_tmul,
      Algebra.TensorProduct.lmul'_apply_tmul, mul_one, one_mul, sub_self]
  rw [hgen, Ideal.mem_span_singleton] at hker
  obtain ⟨w, hw⟩ := hker
  rw [Ideal.mem_span_singleton]
  refine ⟨mapRightTwo c w, ?_⟩
  have := congrArg (mapRightTwo c) hw
  rw [map_sub, mapRightTwo_tmul, mapRightTwo_tmul, map_one, map_mul] at this
  rw [this]
