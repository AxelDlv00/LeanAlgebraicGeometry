theorem isAffine_of_V₀_eq_bot (h : D.V₀ = ⊥) : IsAffine Y := by
  have hv1 : D.V₁ = ⊤ := by
    have hs := D.sup_eq_top
    rw [h, bot_sup_eq] at hs
    exact hs
  have h2 := D.isAffineOpen₁
  rw [hv1] at h2
  haveI : IsAffine (↑(⊤ : Y.Opens)) := h2
  exact IsAffine.of_isIso Y.topIso.inv
