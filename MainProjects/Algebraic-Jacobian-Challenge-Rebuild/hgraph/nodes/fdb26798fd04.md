theorem quotientChartBaseChange_isPullback
    [FiniteDimensional K L] [IsGalois K L] [HasStableAffineCover K L ρ]
    (i : StableAffineOpen ρ) :

      (quotientChartBaseChangeToGlued ρ i)

    exact (Scheme.Opens.opensRange_ι i.U).symm
