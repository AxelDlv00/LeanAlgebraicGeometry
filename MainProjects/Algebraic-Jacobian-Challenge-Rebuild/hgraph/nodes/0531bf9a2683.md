noncomputable def basePrime {B : Type*} [CommRing B] [IsLocalRing B] (φ : S →+* B) :
    PrimeSpectrum R :=
  ⟨Ideal.comap (φ.comp (algebraMap R S)) (IsLocalRing.maximalIdeal B), inferInstance⟩
