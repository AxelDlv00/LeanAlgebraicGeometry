    (a : U) (z : X.functionField) :
    (∑ i, (b.repr a i) • ((b i : X.functionField) * z)) =
      (a : X.functionField) * z := by
  have hterm : ∀ i,
      (b.repr a i) • ((b i : X.functionField) * z) =

    ring

      symm
