structure Hom (D₁ : DescentDatum A B R) (D₂ : DescentDatum A B S) where
  hom : R →ₐ[B] S
  coaction_comm (x : R) :
    D₂.coaction (hom x) =
      Algebra.TensorProduct.map (AlgHom.id A B) (hom.restrictScalars A)
        (D₁.coaction x)
