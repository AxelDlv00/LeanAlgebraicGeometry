lemma supportLeak_eq_empty_of_forall_fibre (f : X ⟶ S) {U : X.Opens}
    (hfib : ∀ s : S, f.base ⁻¹' {s} ∩ closure (d.supportLocus ∩ (U : Set X))
      ⊆ (U : Set X)) :
    d.supportLeak U = ∅ := by
  rw [Set.eq_empty_iff_forall_notMem]
  rintro x ⟨hxc, hxU⟩
  exact hxU (hfib (f.base x) ⟨rfl, hxc⟩)
