theorem redL_toLaurent (φ : A →+* B) (p : Polynomial A) :
    redL φ (Polynomial.toLaurent p)
      = Polynomial.toLaurent (Polynomial.mapRingHom φ p) := by
  induction p using Polynomial.induction_on' with
  | add p q hp hq => simp [hp, hq]
  | monomial n a =>
      rw [← Polynomial.C_mul_X_pow_eq_monomial, Polynomial.toLaurent_C_mul_X_pow, redL_C_mul_T]
      simp
