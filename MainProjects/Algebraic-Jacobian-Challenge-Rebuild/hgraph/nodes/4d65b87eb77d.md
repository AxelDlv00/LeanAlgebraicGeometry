def Scheme.baseDivisorSet (T : Submodule K X.functionField) (A : X.CurveDivisor)
    (p : {x : X // x ≠ genericPoint X}) : Set ℕ :=
  {n : ℕ | ∃ (f : X.functionField) (_ : f ∈ T) (hf : f ≠ 0),
    (coeffAt p.2
      (A + Scheme.divOf (X ↘ Spec (CommRingCat.of K)) (Units.mk0 f hf))).toNat = n}
