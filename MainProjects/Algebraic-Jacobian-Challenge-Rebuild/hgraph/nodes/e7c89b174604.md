theorem isDivisorDegree_one_of_point (p : overSpec k k ⟶ C) :
    IsDivisorDegree C 1 := by
  obtain ⟨D, hD⟩ :=
    Scheme.CurveDivisor.exists_picClass_eq k (Over.graphPicClass C p)
  refine ⟨D, ?_⟩
  rw [← classDeg_picClass, hD]
  exact classDeg_graphPicClass C p
