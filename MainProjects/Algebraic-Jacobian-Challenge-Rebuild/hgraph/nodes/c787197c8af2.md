theorem truncExpClass_add (b c : Γ(X, U₀ ⊓ U₁)) :
    truncExpClass X U₀ U₁ (b + c)
      = truncExpClass X U₀ U₁ b + truncExpClass X U₀ U₁ c := by
  rw [truncExpClass_eq_engine X U₀ U₁ b, truncExpClass_eq_engine X U₀ U₁ c, ← map_add,
