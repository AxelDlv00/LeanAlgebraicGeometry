noncomputable def glueSection : picEt C T :=
  ⟨fun W => glueValue hcov hv W, by
    intro U V h
    refine (glueValue_eq_of hcov hv ?_).symm
    intro W₀ hW₀ i hi
    rw [← PicEtAff.mapAlg_comp, Over.resAlgHom_comp]
    exact glueValue_spec hcov hv V W₀ (hW₀.trans h) i hi⟩
