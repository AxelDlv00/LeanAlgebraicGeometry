noncomputable abbrev baseChange (k L : Type u) [Field k] [Field L] [Algebra k L] :
    Over (Spec (.of k)) ⥤ Over (Spec (.of L)) :=
  Over.pullback (Spec.map (CommRingCat.ofHom (algebraMap k L)))
