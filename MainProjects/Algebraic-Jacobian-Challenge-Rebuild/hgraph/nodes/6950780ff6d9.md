theorem quasiCompact_of_surjective_from_divScheme (J : Over (Spec (CommRingCat.of k)))
    (abel : DivScheme k A B g r₁ r₂ b₁ b₂ ⟶ J.left)
    (hsurj : Function.Surjective abel.base) :
    QuasiCompact J.hom :=
  quasiCompact_of_surjective abel J.hom hsurj

end Qc
