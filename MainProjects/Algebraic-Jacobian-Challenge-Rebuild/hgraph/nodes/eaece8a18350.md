lemma ker_eq_bot_of_spanFinrank_eq [IsRegularLocalRing B] [IsRegularLocalRing A]
    (φ : B →+* A) (hφ : Function.Surjective φ)
    (h : (IsLocalRing.maximalIdeal A).spanFinrank
      = (IsLocalRing.maximalIdeal B).spanFinrank) :
    RingHom.ker φ = ⊥ := by
  haveI : IsDomain B := isDomain_of_regularLocal B
  haveI : IsDomain A := isDomain_of_regularLocal A
  haveI hDpr : (RingHom.ker φ).IsPrime := RingHom.ker_isPrime φ
  have hdimB : (((IsLocalRing.maximalIdeal B).spanFinrank : ℕ) : WithBot ℕ∞)
      = ringKrullDim B := IsRegularLocalRing.spanFinrank_maximalIdeal
  have hdimA : (((IsLocalRing.maximalIdeal A).spanFinrank : ℕ) : WithBot ℕ∞)
      = ringKrullDim A := IsRegularLocalRing.spanFinrank_maximalIdeal
  have hquot : ringKrullDim (B ⧸ RingHom.ker φ) = ringKrullDim A :=
    ringKrullDim_eq_of_ringEquiv (RingHom.quotientKerEquivOfSurjective hφ)
  have hcoh : (Order.coheight
        (⟨RingHom.ker φ, hDpr⟩ : PrimeSpectrum B) : WithBot ℕ∞)
      = ringKrullDim B := by
    rw [← ringKrullDim_quotient_eq_coheight (RingHom.ker φ), hquot, ← hdimA,
      h, hdimB]
  exact eq_bot_of_mem_minimalPrimes_of_isDomain
    (mem_minimalPrimes_of_coheight_eq_ringKrullDim hcoh)
