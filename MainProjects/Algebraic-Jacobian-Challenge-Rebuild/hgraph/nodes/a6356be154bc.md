lemma coord_tmul_one :
    coord k B ⊗ₜ[Polynomial k] (1 : B) = (1 : B) ⊗ₜ[Polynomial k] coord k B := by
  have h1 : (coord k B : B) = (X : Polynomial k) • (1 : B) := by
    rw [Algebra.smul_def, mul_one]; rfl
  rw [h1, TensorProduct.smul_tmul]
