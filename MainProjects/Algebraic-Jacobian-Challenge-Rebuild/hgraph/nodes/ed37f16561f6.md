theorem mem_smul_top_of_mem_map_of_flat_quotient
    (J : Ideal B) [Module.Flat R (B ⧸ J)] (p : Ideal R) (x : J)
    (hx : (x : B) ∈ Ideal.map (algebraMap R B) p) :
    x ∈ p • (⊤ : Submodule R J) := by
  have hzeroAmbient :
      ((J.restrictScalars R).subtype.lTensor (R ⧸ p))
          ((1 : R ⧸ p) ⊗ₜ[R] x) = 0 := by
    rw [LinearMap.lTensor_tmul]
    apply (TensorProduct.quotTensorEquivQuotSMul B p).injective
    rw [map_zero, TensorProduct.quotTensorEquivQuotSMul_mk_one_tmul,
      Submodule.Quotient.mk_eq_zero, Ideal.smul_top_eq_map]
    exact hx
  have hzero : ((1 : R ⧸ p) ⊗ₜ[R] x) = 0 :=
    (IdealPurity.subtype_lTensor_injective_of_flat_quotient
      (R := R) (B := B) J (R ⧸ p)) hzeroAmbient
  have hmk :
      (Submodule.Quotient.mk x : J ⧸ (p • (⊤ : Submodule R J))) = 0 := by
    rw [← TensorProduct.quotTensorEquivQuotSMul_mk_one_tmul]
    rw [hzero, map_zero]
  rw [Submodule.Quotient.mk_eq_zero] at hmk
  exact hmk
