theorem IsPlusHonest.mul {T : Over (Spec (.of k))} {μ ν : picEt C T}
    (hμ : IsPlusHonest C T μ) (hν : IsPlusHonest C T ν) :
    IsPlusHonest C T (μ * ν) := by
  intro U
  obtain ⟨z, hz⟩ := hμ U
  obtain ⟨w, hw⟩ := hν U
  exact ⟨z * w, by rw [map_mul, hz, hw, map_mul]⟩
