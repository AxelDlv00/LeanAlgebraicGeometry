  X₂ := Sheaf.freeModuleSheaf J R S.X₂ ⊞ Sheaf.freeModuleSheaf J R S.X₃
  X₃ := Sheaf.freeModuleSheaf J R S.X₄

    (-Sheaf.freeModuleSheafMap J R S.f₁₃)

    (Sheaf.freeModuleSheafMap J R S.f₃₄)
