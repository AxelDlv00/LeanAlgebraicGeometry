noncomputable def quotRetract : M →ₗ[R] ↥N :=
  ((LinearMap.id : M →ₗ[R] M) - quotSection N ∘ₗ N.mkQ).codRestrict N fun c => by
    rw [LinearMap.sub_apply, LinearMap.id_apply, LinearMap.comp_apply]
    exact sub_quotSection_mem N c
