theorem slice_iff_bare {D : Over (Spec (.of k))}
    (Y : Scheme.{u}) (s : (pic0SigmaSheaf C).1.obj (op Y)) (W : Y.Opens) :
    Nonempty (Over.mk (W.ι ≫ s.1) ⟶ D) ↔
      ∃ x : (W : Scheme.{u}) ⟶ D.left, x ≫ D.hom = W.ι ≫ s.1 :=
  ⟨fun ⟨g⟩ => ⟨g.left, Over.w g⟩, fun ⟨x, hx⟩ => ⟨Over.homMk x hx⟩⟩
