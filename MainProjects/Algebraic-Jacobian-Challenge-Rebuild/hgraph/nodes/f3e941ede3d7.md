    (hx : x ≠ genericPoint C.left) :

      -coeffAt hx (Scheme.divOf (C.left ↘ Spec (CommRingCat.of k)) v) := by

      Scheme.divOf (C.left ↘ Spec (CommRingCat.of k)) v = 0 := by
    rw [← Scheme.divOf_mul, inv_mul_cancel, Scheme.divOf_one]
