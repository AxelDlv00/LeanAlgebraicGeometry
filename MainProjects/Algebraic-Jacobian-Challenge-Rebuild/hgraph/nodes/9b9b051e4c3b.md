@[reducible] noncomputable def Scheme.stalkOverAlgebra (z : X) :
    Algebra K (X.presheaf.stalk z) :=
  ((X.presheaf.germ ⊤ z trivial).hom.comp (X.overAlgebraMap K ⊤)).toAlgebra
