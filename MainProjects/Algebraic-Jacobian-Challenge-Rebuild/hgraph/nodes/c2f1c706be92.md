instance isIso_transportLeft : IsIso (transportLeft k C) := by
  have h : IsIso (C ◁ (unitIso k).hom) :=
    inferInstanceAs (IsIso (whiskerLeftIso C (unitIso k)).hom)
  exact inferInstanceAs (IsIso ((Over.forget _).map (C ◁ (unitIso k).hom)))
