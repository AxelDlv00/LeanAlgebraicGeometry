theorem ker_finiteComponentSum_eq_range_finiteKoszulBoundary_of_pencil
    (row : ι → M →ₗ[R] N) (step : ι → L →ₗ[R] M)
    (c : Fin 2 → ι → R)
    (hcomm : ∀ (i j : ι) (z : L),
      row i (step j z) = row j (step i z))
    (hspan : ∀ x : M, ∃ z : Fin 2 → L,
      x = ∑ q, ∑ i, c q i • step i (z q))
    (hpencil : ∀ y : Fin 2 → M,
      (∑ q, ∑ i, c q i • row i (y q)) = 0 →
        ∃ z : L,
          y 0 = ∑ i, c 1 i • step i z ∧
          y 1 = -∑ i, c 0 i • step i z) :
    LinearMap.ker (finiteComponentSum row) =
      LinearMap.range (finiteKoszulBoundary step) := by
  apply le_antisymm
  · exact ker_finiteComponentSum_le_range_finiteKoszulBoundary_of_pencil
      row step c hcomm hspan hpencil
  · exact range_finiteKoszulBoundary_le_ker_finiteComponentSum row step hcomm
