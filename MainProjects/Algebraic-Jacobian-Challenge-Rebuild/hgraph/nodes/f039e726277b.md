theorem isDegree_sub {c c' : ℤ} (h : IsDivisorDegree C c) (h' : IsDivisorDegree C c') :
    IsDivisorDegree C (c - c') := by
  obtain ⟨W, hW⟩ := h
  obtain ⟨W', hW'⟩ := h'
  exact ⟨W - W', by rw [Scheme.CurveDivisor.deg_sub' k, hW, hW']⟩
