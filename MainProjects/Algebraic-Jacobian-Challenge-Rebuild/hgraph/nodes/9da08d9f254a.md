def ChartLocusAffineLocal (m : ℕ) (Z : (C ⊗ overSpec k k).left.CurveDivisor)
    (T : Over (Spec (.of k))) (lam : picEt C T) : Prop :=
  ∀ U : T.left.affineOpens,
    IsOpen (chartLocus C m Z (picEtMap C (Over.fromSpecAffine T U) lam))
