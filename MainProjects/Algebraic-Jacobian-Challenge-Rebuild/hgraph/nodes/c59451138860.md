    (Over.pullbackComp f (𝟙 Y)).hom ≫
      whiskerRight Over.pullbackId.hom (Over.pullback f) ≫
        (Over.pullback f).leftUnitor.hom =

    (Category.comp_id f))).hom

    (Over.pullbackCongrIso_eq_eqToIso (Category.comp_id f))]
