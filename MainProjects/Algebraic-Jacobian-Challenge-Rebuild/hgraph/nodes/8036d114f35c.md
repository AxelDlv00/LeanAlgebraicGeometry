theorem awayToPoly_chartCoord (i : J) (j : {j : J // j ≠ i}) :
    awayToPoly R J i (chartCoord R J i j) = X j :=
  (awayToPoly_mk R J i 1 (X j.1)
    (by simpa using X_mem_deg_one R J j.1)).trans
      (dehomogenize_X_ne R J i j)
