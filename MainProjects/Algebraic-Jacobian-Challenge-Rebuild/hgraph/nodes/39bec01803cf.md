structure IsCoverCocycle (γ : ∀ i j, (T i j)ˣ) : Prop where
  diag_eq_one : ∀ i, diag f S T i (γ i i).val = 1
  cocycle : ∀ i j k,
    face₂₃ f T W i j k (γ j k).val * face₁₂ f T W i j k (γ i j).val
      = face₁₃ f T W i j k (γ i k).val
