lemma IsCohomologous.mul {γ₁ γ₂ δ₁ δ₂ : OneCocycle (G ⋙ forget₂ CommGrpCat GrpCat) U}
    (h : γ₁.IsCohomologous γ₂) (h' : δ₁.IsCohomologous δ₂) :
    (γ₁ * δ₁).IsCohomologous (γ₂ * δ₂) := by
  obtain ⟨α, hα⟩ := h
  obtain ⟨β, hβ⟩ := h'
  exact ⟨α * β, hα.mul hβ⟩
