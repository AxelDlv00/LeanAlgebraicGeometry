theorem existsUnique_of_baseChange_subsingleton {X J A : Over (Spec (.of k))}
    (u : X ⟶ J) (f : X ⟶ A)
    (hex : ∃ g : J ⟶ A, f = u ≫ g)
    (hL : Subsingleton
      ((Over.pullback (Spec.map (CommRingCat.ofHom (algebraMap k L)))).obj J ⟶
        (Over.pullback (Spec.map (CommRingCat.ofHom (algebraMap k L)))).obj A)) :
    ∃! g : J ⟶ A, f = u ≫ g := by
