noncomputable def divRepPullAt
    (U : ∀ (i : (glueData k g r1).J) (j : (glueData k g r2).J),
      CertifiedDivisorFamily C (ChartRing i j) pi g)
    {S : Type u} [CommRing S] [Algebra k S]
