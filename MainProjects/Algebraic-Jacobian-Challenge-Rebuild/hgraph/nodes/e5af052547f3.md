lemma thetaFieldGluedEquiv_val (s : relThetaSections C K π a) (x : relCurve C K) :
    (thetaFieldGluedEquiv C K π a s).val x
      = secRes ((relCurve C K).moduleKSheaf K) (inf_le_inf_left ⊤ le_rfl)
          ((twistToGluedApp C K π a ⊤ s).val (thetaFieldChartIndex C K π x)) := by
  have h1 : thetaFieldGluedEquiv C K π a s
      = gluedSubordEquiv K (U := (thetaChartDatum C K π a).pieces)
          (g := (thetaChartDatum C K π a).unit) (σ := thetaFieldChartIndex C K π)
          (fun _ => le_rfl)
          (fun x y => congrArg Units.val
            (gluedSubordCocycle_evInf (thetaChartDatum C K π a).isGluingCocycle
              (thetaFieldPointedCover C K π) (thetaFieldChartIndex C K π)
              (fun _ => le_rfl) x y))
          (thetaChartDatum C K π a).isGluingCocycle
          (fun z => ⟨z, (thetaFieldPointedCover C K π).mem_opens z⟩) ⊤
          ((gluedTwistEquiv C K π a ⊤).symm s) :=
    (gluedTwistEquiv C K π a ⊤).symm.trans_apply _ s
  have h2 : (gluedTwistEquiv C K π a ⊤).symm s = twistToGluedApp C K π a ⊤ s := by
    rw [gluedTwistEquiv]
    exact LinearEquiv.ofLinear_symm_apply _ _ _ _ s
  rw [h1, gluedSubordEquiv_apply_coe, h2]
