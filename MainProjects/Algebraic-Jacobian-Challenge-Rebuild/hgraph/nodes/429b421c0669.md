def Scheme.IsGluingCoboundary {X : Scheme.{u}} {J : Type u} (U : J → X.Opens)
    (g g' : ∀ i j : J, Γ(X, U i ⊓ U j)ˣ) (c : ∀ j : J, Γ(X, U j)ˣ) : Prop :=
  ∀ i j : J,
    X.resHom (inf_le_left : U i ⊓ U j ≤ U i) (c i : Γ(X, U i)) *
        (g i j : Γ(X, U i ⊓ U j)) =
      (g' i j : Γ(X, U i ⊓ U j)) *
        X.resHom (inf_le_right : U i ⊓ U j ≤ U j) (c j : Γ(X, U j))
