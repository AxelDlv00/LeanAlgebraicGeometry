noncomputable def piecesMap : ∀ j : D.index,
    Γ(relCurve C B, D.pieces j) →+*
      Γ(relCurve C B', (D.baseChange B').pieces j)
  | .inl j => pieceSectionsMap B' (fiberChart₀ pi) (D.h₀ j)
  | .inr j => pieceSectionsMap B' (fiberChart₁ pi) (D.h₁ j)
