theorem relCurveMap_eq_whiskerLeft_overDualNumberZero
    (C : Over (Spec (CommRingCat.of k))) :
    relCurveMap C (DualNumber k) k = (C ◁ overDualNumberZero k).left := by
  rw [relCurveMap]
  congr 1
  exact congrArg (fun φ ↦ C ◁ φ) (Over.OverMorphism.ext (by rfl))
