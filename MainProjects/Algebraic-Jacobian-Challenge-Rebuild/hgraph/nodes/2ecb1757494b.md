def testPoint {T : Over (Spec (.of k))} (t : T.left) :
    overSpec k (testPointField t) ⟶ T :=
  Over.homMk (T.left.fromSpecResidueField t)
    (Spec.map_preimage (T.left.fromSpecResidueField t ≫ T.hom)).symm
