theorem comp_isClosedImmersion (h : pi.IsProjective) {Y : Scheme.{u}}
    (j : Y ⟶ X) [IsClosedImmersion j] : (j ≫ pi).IsProjective := by
  obtain ⟨n, hn, i, hi, hcomp⟩ := h
  letI : Finite n := hn
  haveI := hi
  exact ⟨n, hn, j ≫ i, inferInstance, by rw [Category.assoc, hcomp]⟩
