lemma ker_lmul'_le_sup :
    RingHom.ker (Algebra.TensorProduct.lmul' k (S := B))
      ≤ Ideal.span {diagGen (k := k) (B := B)} ⊔ Ideal.span {elift} := by
  intro x hx
  rw [RingHom.mem_ker] at hx
  -- `π x` lies in the `Polynomial k`-diagonal ideal `(π elift)`
  have hπx : baseChange x ∈ RingHom.ker (Algebra.TensorProduct.lmul' (R := Polynomial k)
      (S := B)) := by
    rw [RingHom.mem_ker, lmul'_baseChange x, hx]
  rw [hgen, Ideal.mem_span_singleton] at hπx
  obtain ⟨c, hc⟩ := hπx
  obtain ⟨w, rfl⟩ := baseChange_surjective c
  -- `x - elift * w ∈ ker π = (u⊗1 − 1⊗u)`
  have hkey : x - elift * w ∈ Ideal.span {diagGen (k := k) (B := B)} := by
    rw [← ker_baseChange, RingHom.mem_ker, map_sub, map_mul, hc]
    ring
  have hx' : x = (x - elift * w) + elift * w := by ring
  rw [hx']
  exact add_mem (Ideal.mem_sup_left hkey)
    (Ideal.mem_sup_right (Ideal.mul_mem_right _ _ (Ideal.subset_span rfl)))

  [IsLocalization.Away (1 - elift) L]
