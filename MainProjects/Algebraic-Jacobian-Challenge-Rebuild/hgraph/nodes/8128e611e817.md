private lemma germ_resHom_coherence {X : Scheme.{u}} {W V : X.Opens}
    (h : W ≤ V) (z : X) (hz : z ∈ W) (t : Γ(X, V)) :
    (X.presheaf.germ W z hz).hom (X.resHom h t) =
      (X.presheaf.germ V z (h hz)).hom t :=
  X.presheaf.germ_res_apply (homOfLE h) z hz t
