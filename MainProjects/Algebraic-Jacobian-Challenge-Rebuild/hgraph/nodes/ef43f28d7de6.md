lemma Scheme.map_mulLinear_le_mulSpan {U T : Submodule K X.functionField}
    {f : X.functionField} (hf : f ∈ T) :
    Submodule.map (Scheme.mulLinear K f) U ≤ Scheme.mulSpan K U T := by
  rintro _ ⟨h, hh, rfl⟩
  rw [Scheme.mulLinear_apply, mul_comm]
  exact Scheme.mul_mem_mulSpan K hh hf

end MulSpan
