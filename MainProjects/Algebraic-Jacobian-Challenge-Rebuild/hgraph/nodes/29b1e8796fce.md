    pieceRingEquiv C hU (s ⊗ₜ b)
      = (cg).appLE U ((cg) ⁻¹ᵁ U) le_rfl s *
        ((Scheme.ΓSpecIso (.of B)).inv
          ≫ (pB).appLE ⊤ ((cg) ⁻¹ᵁ U)

    rw [Algebra.TensorProduct.tmul_mul_tmul, mul_one, one_mul]
