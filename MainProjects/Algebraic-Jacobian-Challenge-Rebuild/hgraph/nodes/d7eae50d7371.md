theorem nilpotent_one_add_mem_laurentCoboundaryUnits {z : LaurentPolynomial A}
    (hz : IsNilpotent z) (u : (LaurentPolynomial A)ˣ)
    (hu : (u : LaurentPolynomial A) = 1 + z) : u ∈ laurentCoboundaryUnits A := by
  classical
  -- the ideal generated by z's (finitely many) coefficients
  set I : Ideal A := Ideal.span (Set.range fun i : z.support => z i) with hI
  have hzmem : ∀ i, z i ∈ I := by
    intro i
    by_cases hi : i ∈ z.support
    · exact Ideal.subset_span ⟨⟨i, hi⟩, rfl⟩
    · rw [Finsupp.notMem_support_iff.mp hi]; exact Ideal.zero_mem _
  have hIfg : I.FG := ⟨(Set.finite_range _).toFinset, by rw [hI, Set.Finite.coe_toFinset]⟩
  have hInil : IsNilpotent I := by
    refine (Ideal.FG.isNilpotent_iff_le_nilradical hIfg).mpr ?_
    rw [hI, Ideal.span_le]
    rintro _ ⟨i, rfl⟩
    exact coeff_isNilpotent_of_isNilpotent hz i
  obtain ⟨N, hN⟩ := hInil
  obtain ⟨n, hn⟩ : ∃ n, N ≤ 2 ^ n := ⟨N, Nat.le_of_lt (Nat.lt_two_pow_self)⟩
  have hIpow : I ^ (2 ^ n) = ⊥ := by
    obtain ⟨j, hj⟩ : ∃ j, 2 ^ n = N + j := ⟨2 ^ n - N, by omega⟩
    rw [hj, pow_add, hN, zero_mul]; rfl
  exact laurentCoboundary_one_add_of_coeffs_mem n I hIpow z hzmem u hu
