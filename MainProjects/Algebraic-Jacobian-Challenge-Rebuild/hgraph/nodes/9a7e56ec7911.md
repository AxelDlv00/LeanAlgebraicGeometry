noncomputable def piAssemblyUnit (w : ∀ i j, (S i ⊗[A] S j)ˣ) :
    ((∀ i, S i) ⊗[A] ∀ i, S i)ˣ :=
  Units.map ((piDoubleEquivA A S).symm :
      (∀ p : ι × ι, S p.1 ⊗[A] S p.2) ≃ₐ[A] _).toAlgHom.toRingHom.toMonoidHom
    (Pi.unitOf _ fun p => w p.1 p.2)
