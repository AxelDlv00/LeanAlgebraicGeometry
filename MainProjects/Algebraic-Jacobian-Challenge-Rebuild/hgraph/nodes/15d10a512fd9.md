noncomputable def entriesIdeal (φ : M →ₗ[R] N) : Ideal R :=
  Ideal.span {x : R | ∃ (i : Fin (genCard R M)) (j : Fin (splitRank R N)),
    x = splitSect R N (φ (gen R M i)) j}
