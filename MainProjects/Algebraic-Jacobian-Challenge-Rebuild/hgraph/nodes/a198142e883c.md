theorem CertifiedDivisorFamily.stalkIdeal_le_span_windowGerm_of_field (g : ℕ)
    (G : CertifiedDivisorFamily C K π g)
    (hOk : Sheaf.h0 (C.left.moduleKSheaf k) = 1)
    (hχk : Sheaf.chi (C.left.moduleKSheaf k) = 1 - (g : ℤ))
    (hOK : Sheaf.h0 ((relCurve C K).moduleKSheaf K) = 1)
    (hχK : Sheaf.chi ((relCurve C K).moduleKSheaf K) = 1 - (g : ℤ))
    (z : relCurve C K) :
    G.eqns.stalkIdeal z
      ≤ Ideal.span (divFamEpsWindowGermSet hπ g (DivFam.mk G) z) := by
  -- the shared window data
  have hT : Submodule.map
        (divFamPhi C K π (windowM_choice π hπ g) (relThetaPairH1_windowM C π hπ g))
        ((divFamEps hπ g (DivFam.mk G)).1)
      = Scheme.divisorSections K
          (windowN C K hπ g - divFamDivisor (DivFam.mk G)) ⊤ :=
    map_divFamPhi_eps_fst K hπ g G
  have hdeg : Scheme.CurveDivisor.deg K (divFamDivisor (DivFam.mk G)) = (g : ℤ) :=
    deg_divFamDivisor (DivFam.mk G)
  have hD0 : 0 ≤ divFamDivisor (DivFam.mk G) := zero_le_divFamDivisor _
  have hNnorm : ∀ D' : (relCurve C K).CurveDivisor,
      Scheme.CurveDivisor.deg K D' ≤ 2 * (g : ℤ) →
      Subsingleton (Sheaf.HModule
        ((relCurve C K).divisorSheaf K (windowN C K hπ g - D')) 1) :=
    fun D' hD' => subsingleton_h1_windowN_sub C K hπ g hOk hχk hχK D' hD'
  have hNdeg : 2 * (g : ℤ) ≤ Scheme.CurveDivisor.deg K (windowN C K hπ g) :=
    two_mul_genus_le_deg_windowN C K hπ g hOk hχk
  have hne : ∃ f ∈ Submodule.map
      (divFamPhi C K π (windowM_choice π hπ g) (relThetaPairH1_windowM C π hπ g))
      ((divFamEps hπ g (DivFam.mk G)).1), f ≠ 0 :=
    exists_mem_ne_zero_of_window_normalization g hχK (windowN C K hπ g) hNnorm hNdeg
      (divFamDivisor (DivFam.mk G)) hdeg hT
  -- the window membership of a window element's reading, in ord form
  have hread_le : ∀ (x : K ⊗[k] ↥(Scheme.divisorSections k
        (windowM_choice π hπ g • fiberWeilDivisor π) ⊤))
      (_ : x ∈ (divFamEps hπ g (DivFam.mk G)).1)
      {w : relCurve C K} (hwg : w ≠ genericPoint (relCurve C K)),
      Scheme.ord (relCurve C K ↘ Spec (CommRingCat.of K)) hwg
          (thetaFieldRead C K π (windowM_choice π hπ g)
            (relThetaWindowEquiv C K π (windowM_choice π hπ g)
              (relThetaPairH1_windowM C π hπ g) x))
        ≤ Scheme.divisorBound (thetaFieldDivisor C K π (windowM_choice π hπ g)
            - Scheme.presentationDivisor K G.eqns.presentation) hwg := by
    intro x hx w hwg
    have hvan : relThetaWindowEquiv C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x
        ∈ G.eqns.vanishingSubmodule K (relCover C K (fiberTwoCover π)).V₀
            (relCover C K (fiberTwoCover π)).V₁
            (relThetaCocycle C K π (windowM_choice π hπ g)) := hx
    have hmem := (mem_vanishingSubmodule_iff_thetaFieldRead_mem C K π
      (windowM_choice π hπ g) G.eqns _).mp hvan
    rw [Scheme.mem_divisorSections_of_nonempty K
      (top_opens_nonempty (X := relCurve C K))] at hmem
    exact hmem w hwg trivial
  by_cases hzg : z = genericPoint (relCurve C K)
  · -- the generic point: any nonzero window element generates the field stalk
    subst hzg
    rw [Scheme.LocalEquations.stalkIdeal_genericPoint]
    obtain ⟨f, hfT, hf⟩ := hne
    obtain ⟨x, hx, rfl⟩ := Submodule.mem_map.mp hfT
    have hΦval : divFamPhi C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x
        = ((thetaFieldShiftUnit C K π (windowM_choice π hπ g) :
            (relCurve C K).functionFieldˣ) : (relCurve C K).functionField)
          * thetaFieldRead C K π (windowM_choice π hπ g)
              (relThetaWindowEquiv C K π (windowM_choice π hπ g)
                (relThetaPairH1_windowM C π hπ g) x) := by
      simp only [divFamPhi, LinearMap.coe_comp, Function.comp_apply,
        LinearEquiv.coe_coe, Scheme.mulLinear_apply]
    have hs_ne : relThetaWindowEquiv C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x ≠ 0 := by
      intro h0
      apply hf
      rw [hΦval, h0, map_zero, mul_zero]
    have hcomp : (relThetaWindowEquiv C K π (windowM_choice π hπ g)
          (relThetaPairH1_windowM C π hπ g) x).val.1 ≠ 0
        ∨ (relThetaWindowEquiv C K π (windowM_choice π hπ g)
          (relThetaPairH1_windowM C π hπ g) x).val.2 ≠ 0 := by
      rcases eq_or_ne (relThetaWindowEquiv C K π (windowM_choice π hπ g)
          (relThetaPairH1_windowM C π hπ g) x).val.1 0 with h1 | h1
      · refine Or.inr fun h2 => hs_ne (Subtype.ext (Prod.ext h1 h2))
      · exact Or.inl h1
    have hsmem : relThetaWindowEquiv C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x
        ∈ Submodule.map (relThetaWindowEquiv C K π (windowM_choice π hπ g)
            (relThetaPairH1_windowM C π hπ g)).toLinearMap
          ((divFamEps hπ g (DivFam.mk G)).1) :=
      Submodule.mem_map_of_mem hx
    rcases hcomp with h1 | h2
    · -- chart 0 carries a nonzero component
      have hnonempty : ((⊤ ⊓ (relCover C K (fiberTwoCover π)).V₀ :
          (relCurve C K).Opens) : Set (relCurve C K)).Nonempty := by
        by_contra hemp
        have hbot : (⊤ ⊓ (relCover C K (fiberTwoCover π)).V₀ :
            (relCurve C K).Opens) ≤ ⊥ := fun w hw => absurd ⟨w, hw⟩ hemp
        haveI := (relCurve C K).subsingleton_sections_of_le_bot hbot
        exact h1 (Subsingleton.elim _ _)
      have hη₀ : genericPoint (relCurve C K)
          ∈ ⊤ ⊓ (relCover C K (fiberTwoCover π)).V₀ :=
        Scheme.genericPoint_mem_of_nonempty hnonempty
      have hγne : ((relCurve C K).presheaf.germ
          (⊤ ⊓ (relCover C K (fiberTwoCover π)).V₀) (genericPoint (relCurve C K))
          hη₀).hom (relThetaWindowEquiv C K π (windowM_choice π hπ g)
            (relThetaPairH1_windowM C π hπ g) x).val.1 ≠ 0 := fun h0 =>
        h1 (germ_injective_of_isIntegral (relCurve C K) (genericPoint (relCurve C K))
          hη₀ (by rw [h0, map_zero]))
      have hγmem : ((relCurve C K).presheaf.germ
            (⊤ ⊓ (relCover C K (fiberTwoCover π)).V₀) (genericPoint (relCurve C K))
            hη₀).hom (relThetaWindowEquiv C K π (windowM_choice π hπ g)
              (relThetaPairH1_windowM C π hπ g) x).val.1
          ∈ divFamEpsWindowGermSet hπ g (DivFam.mk G) (genericPoint (relCurve C K)) :=
        Set.mem_union_left _ ⟨_, hsmem, hη₀, rfl⟩
      exact (Ideal.eq_top_of_isUnit_mem _ (Ideal.subset_span hγmem)
        (Units.mk0 (G₀ := (relCurve C K).functionField) _ hγne).isUnit).ge
    · -- chart 1 carries a nonzero component
      have hnonempty : ((⊤ ⊓ (relCover C K (fiberTwoCover π)).V₁ :
          (relCurve C K).Opens) : Set (relCurve C K)).Nonempty := by
        by_contra hemp
        have hbot : (⊤ ⊓ (relCover C K (fiberTwoCover π)).V₁ :
            (relCurve C K).Opens) ≤ ⊥ := fun w hw => absurd ⟨w, hw⟩ hemp
        haveI := (relCurve C K).subsingleton_sections_of_le_bot hbot
        exact h2 (Subsingleton.elim _ _)
      have hη₁ : genericPoint (relCurve C K)
          ∈ ⊤ ⊓ (relCover C K (fiberTwoCover π)).V₁ :=
        Scheme.genericPoint_mem_of_nonempty hnonempty
      have hγne : ((relCurve C K).presheaf.germ
          (⊤ ⊓ (relCover C K (fiberTwoCover π)).V₁) (genericPoint (relCurve C K))
          hη₁).hom (relThetaWindowEquiv C K π (windowM_choice π hπ g)
            (relThetaPairH1_windowM C π hπ g) x).val.2 ≠ 0 := fun h0 =>
        h2 (germ_injective_of_isIntegral (relCurve C K) (genericPoint (relCurve C K))
          hη₁ (by rw [h0, map_zero]))
      have hγmem : ((relCurve C K).presheaf.germ
            (⊤ ⊓ (relCover C K (fiberTwoCover π)).V₁) (genericPoint (relCurve C K))
            hη₁).hom (relThetaWindowEquiv C K π (windowM_choice π hπ g)
              (relThetaPairH1_windowM C π hπ g) x).val.2
          ∈ divFamEpsWindowGermSet hπ g (DivFam.mk G) (genericPoint (relCurve C K)) :=
        Set.mem_union_right _ ⟨_, hsmem, hη₁, rfl⟩
      exact (Ideal.eq_top_of_isUnit_mem _ (Ideal.subset_span hγmem)
        (Units.mk0 (G₀ := (relCurve C K).functionField) _ hγne).isUnit).ge
  · -- a closed point: the achiever generates
    have hTA : Submodule.map
        (divFamPhi C K π (windowM_choice π hπ g) (relThetaPairH1_windowM C π hπ g))
        ((divFamEps hπ g (DivFam.mk G)).1)
        ≤ Scheme.divisorSections K (windowN C K hπ g) ⊤ := by
      have hle : windowN C K hπ g - divFamDivisor (DivFam.mk G) ≤ windowN C K hπ g := by
        refine Scheme.CurveDivisor.le_iff_coeffAt.mpr fun x hx => ?_
        rw [Scheme.CurveDivisor.coeffAt_sub]
        have h0 := Scheme.CurveDivisor.le_iff_coeffAt.mp hD0 x hx
        rw [Scheme.CurveDivisor.coeffAt_zero] at h0
        omega
      rw [hT, Scheme.divisorSections_of_nonempty K
          (top_opens_nonempty (X := relCurve C K)),
        Scheme.divisorSections_of_nonempty K (top_opens_nonempty (X := relCurve C K))]
      exact Scheme.boundedSections_mono K hle ⊤
    obtain ⟨f, hfT, hf, hach0⟩ :=
      Scheme.exists_coeffAt_eq_baseDivisorAt K hTA hne hzg
    have hbd : (Scheme.baseDivisorAt K (Submodule.map
          (divFamPhi C K π (windowM_choice π hπ g)
            (relThetaPairH1_windowM C π hπ g)) ((divFamEps hπ g (DivFam.mk G)).1))
          (windowN C K hπ g) ⟨z, hzg⟩ : ℤ)
        = coeffAt hzg (divFamDivisor (DivFam.mk G)) := by
      rw [hT]
      exact baseDivisorAt_window_normalization g hOK hχK (windowN C K hπ g) hNnorm
        hNdeg (divFamDivisor (DivFam.mk G)) hD0 hdeg hzg
    rw [hbd] at hach0
    obtain ⟨x, hx, rfl⟩ := Submodule.mem_map.mp hfT
    have hΦval : divFamPhi C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x
        = ((thetaFieldShiftUnit C K π (windowM_choice π hπ g) :
            (relCurve C K).functionFieldˣ) : (relCurve C K).functionField)
          * thetaFieldRead C K π (windowM_choice π hπ g)
              (relThetaWindowEquiv C K π (windowM_choice π hπ g)
                (relThetaPairH1_windowM C π hπ g) x) := by
      simp only [divFamPhi, LinearMap.coe_comp, Function.comp_apply,
        LinearEquiv.coe_coe, Scheme.mulLinear_apply]
    have hread_ne : thetaFieldRead C K π (windowM_choice π hπ g)
        (relThetaWindowEquiv C K π (windowM_choice π hπ g)
          (relThetaPairH1_windowM C π hπ g) x) ≠ 0 := by
      intro h0
      apply hf
      rw [hΦval, h0, mul_zero]
    -- the unit decomposition `f = w · read s` and its divisor
    have hfU : Units.mk0 (divFamPhi C K π (windowM_choice π hπ g)
          (relThetaPairH1_windowM C π hπ g) x) hf
        = thetaFieldShiftUnit C K π (windowM_choice π hπ g)
          * Units.mk0 _ hread_ne := by
      refine Units.ext ?_
      rw [Units.val_mul]
      exact hΦval
    have hdiv : coeffAt hzg (Scheme.divOf (relCurve C K ↘ Spec (CommRingCat.of K))
          (Units.mk0 (divFamPhi C K π (windowM_choice π hπ g)
            (relThetaPairH1_windowM C π hπ g) x) hf))
        = coeffAt hzg (Scheme.divOf (relCurve C K ↘ Spec (CommRingCat.of K))
            (thetaFieldShiftUnit C K π (windowM_choice π hπ g)))
          + coeffAt hzg (Scheme.divOf (relCurve C K ↘ Spec (CommRingCat.of K))
            (Units.mk0 _ hread_ne)) := by
      rw [hfU, Scheme.divOf_mul, Scheme.CurveDivisor.coeffAt_add]
    have hshift : coeffAt hzg (thetaFieldDivisor C K π (windowM_choice π hπ g))
        - coeffAt hzg (windowN C K hπ g)
        = coeffAt hzg (Scheme.divOf (relCurve C K ↘ Spec (CommRingCat.of K))
            (thetaFieldShiftUnit C K π (windowM_choice π hπ g))) := by
      have h := congrArg (coeffAt hzg)
        (divOf_thetaFieldShiftUnit C K π (windowM_choice π hπ g))
      rwa [Scheme.CurveDivisor.coeffAt_sub] at h
    have hach1 : coeffAt hzg (windowN C K hπ g)
        + coeffAt hzg (Scheme.divOf (relCurve C K ↘ Spec (CommRingCat.of K))
            (Units.mk0 (divFamPhi C K π (windowM_choice π hπ g)
              (relThetaPairH1_windowM C π hπ g) x) hf))
        = coeffAt hzg (divFamDivisor (DivFam.mk G)) := by
      rw [← Scheme.CurveDivisor.coeffAt_add]
      exact hach0
    -- the generation-side bound
    have hge : Scheme.divisorBound
          (thetaFieldDivisor C K π (windowM_choice π hπ g)
            - Scheme.presentationDivisor K G.eqns.presentation) hzg
        ≤ Scheme.ord (relCurve C K ↘ Spec (CommRingCat.of K)) hzg
            (thetaFieldRead C K π (windowM_choice π hπ g)
              (relThetaWindowEquiv C K π (windowM_choice π hπ g)
                (relThetaPairH1_windowM C π hπ g) x)) := by
      have hval : Scheme.ord (relCurve C K ↘ Spec (CommRingCat.of K)) hzg
          (thetaFieldRead C K π (windowM_choice π hπ g)
            (relThetaWindowEquiv C K π (windowM_choice π hπ g)
              (relThetaPairH1_windowM C π hπ g) x))
          = Scheme.divisorBound
              (-Scheme.divOf (relCurve C K ↘ Spec (CommRingCat.of K))
                (Units.mk0 _ hread_ne)) hzg :=
        Scheme.ord_val_eq K (Units.mk0 _ hread_ne) hzg
      rw [hval, divisorBound_le_divisorBound_iff hzg,
        Scheme.CurveDivisor.coeffAt_sub, Scheme.CurveDivisor.coeffAt_neg]
      have hDpres : coeffAt hzg (divFamDivisor (DivFam.mk G))
          = coeffAt hzg (Scheme.presentationDivisor K G.eqns.presentation) := by
        rw [divFamDivisor_mk]
      omega
    -- the membership-side bound
    have hle := hread_le x hx hzg
    -- assemble: the germ generates and lies in the window germ set
    have hspan := span_germ_thetaFieldGluedEquiv_eq_stalkIdeal C K π
      (windowM_choice π hπ g) G.eqns
      (relThetaWindowEquiv C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x) hzg hle hge
    have hsmem : relThetaWindowEquiv C K π (windowM_choice π hπ g)
        (relThetaPairH1_windowM C π hπ g) x
        ∈ Submodule.map (relThetaWindowEquiv C K π (windowM_choice π hπ g)
            (relThetaPairH1_windowM C π hπ g)).toLinearMap
          ((divFamEps hπ g (DivFam.mk G)).1) :=
      Submodule.mem_map_of_mem hx
    rw [← hspan]
    refine Ideal.span_mono (Set.singleton_subset_iff.mpr ?_)
    by_cases hz₀ : z ∈ (relCover C K (fiberTwoCover π)).V₀
    · refine Set.mem_union_left _ ⟨_, hsmem, ⟨trivial, hz₀⟩, ?_⟩
      exact germ_thetaFieldGluedEquiv_fst C K π (windowM_choice π hπ g) _ hz₀
        ⟨trivial, (thetaFieldPointedCover C K π).mem_opens z⟩ ⟨trivial, hz₀⟩
    · refine Set.mem_union_right _
        ⟨_, hsmem, ⟨trivial, mem_V₁_of_notMem_V₀ C K π hz₀⟩, ?_⟩
      exact germ_thetaFieldGluedEquiv_snd C K π (windowM_choice π hπ g) _ hz₀
        ⟨trivial, (thetaFieldPointedCover C K π).mem_opens z⟩
        ⟨trivial, mem_V₁_of_notMem_V₀ C K π hz₀⟩
