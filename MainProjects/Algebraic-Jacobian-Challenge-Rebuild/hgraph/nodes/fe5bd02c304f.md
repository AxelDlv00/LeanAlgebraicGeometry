noncomputable def ofChartPair
    (eqn : ∀ j : (chartPairCoverData C R π).index,
      Γ(relCurve C R, (chartPairCoverData C R π).pieces j))
    (pt : (chartPairCoverData C R π).index → relCurve C R)
    (piece_le : ∀ j, (chartPairCoverData C R π).pieces j ≤ d.cover.opens (pt j))
    (unit : ∀ j, Γ(relCurve C R, (chartPairCoverData C R π).pieces j)ˣ)
    (eqn_eq : ∀ j, eqn j
      = (unit j : Γ(relCurve C R, (chartPairCoverData C R π).pieces j))
        * ((relCurve C R).presheaf.map (homOfLE (piece_le j)).op).hom (d.eqn (pt j))) :
    DivisorAdaptation C R π d :=
  DivisorAdaptation.ofAnchors (chartPairCoverData C R π) eqn pt piece_le unit eqn_eq
