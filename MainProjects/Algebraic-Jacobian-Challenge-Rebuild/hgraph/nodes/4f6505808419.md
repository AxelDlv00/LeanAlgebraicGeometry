noncomputable def pointQRight :
    F →ₐ[Polynomial k] (B ⊗[k] F) ⧸ Ideal.span {pointGen k B F} where
  toFun y := Ideal.Quotient.mk _ ((1 : B) ⊗ₜ[k] y)
  map_one' := by rw [← Algebra.TensorProduct.one_def]; exact map_one _
  map_mul' a b := by rw [← map_mul, Algebra.TensorProduct.tmul_mul_tmul, one_mul]
  map_zero' := by simp
  map_add' a b := by rw [← map_add, TensorProduct.tmul_add]
  commutes' p := by
    change Ideal.Quotient.mk _ ((1 : B) ⊗ₜ[k] algebraMap (Polynomial k) F p)
      = algebraMap (Polynomial k) ((B ⊗[k] F) ⧸ Ideal.span {pointGen k B F}) p
    have h : algebraMap (Polynomial k) ((B ⊗[k] F) ⧸ Ideal.span {pointGen k B F}) p
        = Ideal.Quotient.mk _ ((algebraMap (Polynomial k) B p) ⊗ₜ[k] (1 : F)) := rfl
    rw [h, Ideal.Quotient.eq, ← neg_sub]
    exact neg_mem (algebraMap_tmul_sub_mem_span_point p)
