theorem mem_range_diagonal_iff {z : (C ⊗ C).left} :
    z ∈ Set.range (diagonal C).left.base ↔
      (diagonal C).left.base ((fst C C).left.base z) = z :=
  ⟨fun ⟨p, hp⟩ => by rw [← hp, fst_base_diagonal_base], fun h => ⟨_, h⟩⟩
