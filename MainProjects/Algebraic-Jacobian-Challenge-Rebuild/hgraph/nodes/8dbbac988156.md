docstring: '**Chart triviality over the dual numbers.** An invertible `A[ε]`-module
  `M` in which

  every element agrees with some multiple of a fixed `m` modulo `(ε) · M` — i.e. whose

  restriction along `ε ↦ 0` is generated by the image of `m`, hence trivial — is free.


  This is the affine-chart form of "an invertible sheaf on `C ×_k Spec k[ε]` trivial
  along

  `ε ↦ 0` is trivial on each base-changed chart". Note what is *not* assumed: `A`
  is an

  arbitrary commutative ring, and the sections are not assumed finitely generated
  over the

  base field — affine charts of a curve do not have finite-dimensional section spaces,
  so any

  version of this requiring that would be useless here.'

theorem free_of_cyclic_mod_eps
    (M : Type v) [AddCommGroup M] [Module (DualNumber A) M]
    [Module.Invertible (DualNumber A) M] (m : M)
    (h : ∀ x : M, ∃ r : DualNumber A,
      x - r • m ∈ Ideal.span {(ε : DualNumber A)} • (⊤ : Submodule (DualNumber A) M)) :
    Module.Free (DualNumber A) M :=
  Module.Invertible.free_of_nilpotent_of_exists_sub_smul_mem (isNilpotent_span_eps A) h
