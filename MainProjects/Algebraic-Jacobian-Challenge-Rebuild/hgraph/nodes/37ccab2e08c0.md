private noncomputable def modelHomChart (t : Module.End R M₀) (a : ι → M₀) :
    (ι → R[X]) →ₗ[R] M₀ :=
  ∑ i : ι, LinearMap.applyₗ (a i) ∘ₗ (Polynomial.aeval t).toLinearMap ∘ₗ LinearMap.proj i
