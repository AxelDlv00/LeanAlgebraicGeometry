theorem probePicEtIdentity (T : Over (Spec (.of k)))
    (lam : pic0Subgroup ((baseChange k k).obj C) T) :
    picEtCrossBaseInv k k C T lam.1 =
      picEtMap C ((probeMId k).hom.app T)
        (picEtPullback ((baseChange.idIso k).app C).inv T lam.1) := by
  apply Subtype.ext
  funext W
  have hW : W.1 ≤ ((probeMId k).hom.app T).left ⁻¹ᵁ W.1 := by
    rw [probeMId_hom_app_left]
    exact le_rfl
  calc
    (picEtCrossBaseInv k k C T lam.1).1 W =
        (sectionShuffle k k C T W.1).symm (lam.1.1 ⟨W.1, W.2⟩) :=
      picEtCrossBaseInv_val k k C T lam.1 W
    _ = PicEtAff.mapAlg C
        (Over.appLEAlgHom ((probeMId k).hom.app T) W.1 W.1 hW)
        (PicEtAff.curveMap Γ(T.left, W.1) ((baseChange.idIso k).app C).inv
          (lam.1.1 ⟨W.1, W.2⟩)) :=
      probeAffineBridge k C T W hW _
    _ = PicEtAff.mapAlg C
        (Over.appLEAlgHom ((probeMId k).hom.app T) W.1 W.1 hW)
        ((picEtPullback ((baseChange.idIso k).app C).inv T lam.1).1 W) :=
      congrArg (PicEtAff.mapAlg C
        (Over.appLEAlgHom ((probeMId k).hom.app T) W.1 W.1 hW))
        (picEtPullback_val ((baseChange.idIso k).app C).inv T lam.1 W).symm
    _ = picEtMapVal C ((probeMId k).hom.app T)
        (picEtPullback ((baseChange.idIso k).app C).inv T lam.1) W :=
      (picEtMapVal_eq_mapAlg C ((probeMId k).hom.app T)
        (picEtPullback ((baseChange.idIso k).app C).inv T lam.1)
        (W := W) (V := W) hW).symm
    _ = (picEtMap C ((probeMId k).hom.app T)
        (picEtPullback ((baseChange.idIso k).app C).inv T lam.1)).1 W :=
      (picEtMap_val C ((probeMId k).hom.app T)
        (picEtPullback ((baseChange.idIso k).app C).inv T lam.1) W).symm
