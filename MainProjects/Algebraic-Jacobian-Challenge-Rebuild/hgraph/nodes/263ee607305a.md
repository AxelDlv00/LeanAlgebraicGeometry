def h1Map (f : P.Hom P') : P.H1 →ₗ[R] P'.H1 :=
  Submodule.mapQ P.imageLattice P'.imageLattice f.homN <| by
    refine sup_le ?_ ?_
    · rintro n ⟨x, rfl⟩
      exact Submodule.mem_sup_left ⟨f.hom₀ x, (f.comm_ι₀ x).symm⟩
    · rintro n ⟨y, rfl⟩
      exact Submodule.mem_sup_right ⟨f.hom₁ y, (f.comm_ι₁ y).symm⟩

@[simp]
