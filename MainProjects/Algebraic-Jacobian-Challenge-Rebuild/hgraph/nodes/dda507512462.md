theorem ker_baseChange :
    RingHom.ker (baseChange (k := k) (B := B)) = Ideal.span {diagGen (k := k) (B := B)} := by
  apply le_antisymm
  · intro x hx
    rw [RingHom.mem_ker] at hx
    have hmk : Ideal.Quotient.mk (Ideal.span {diagGen (k := k) (B := B)}) x = 0 := by
      rw [← sectionMap_baseChange x, hx, map_zero]
    exact (Ideal.Quotient.eq_zero_iff_mem).mp hmk
  · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe, RingHom.mem_ker]
    exact baseChange_diagGen
