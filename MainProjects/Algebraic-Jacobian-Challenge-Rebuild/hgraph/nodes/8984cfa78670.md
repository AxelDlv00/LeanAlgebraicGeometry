private lemma telescope_eq_one {G : Type u} [CommGroup G] {T₂₃ T₁₂ T₁₃ a b c : G}
    (h₂₃ : T₂₃ * c = b) (h₁₂ : T₁₂ * b = a) (h₁₃ : T₁₃ * c = a) :
    T₂₃ * T₁₂ / T₁₃ = 1 := by
  have h : T₁₃ * c = T₁₂ * (T₂₃ * c) := by rw [h₂₃, h₁₂]; exact h₁₃
  rw [← mul_assoc] at h
  rw [div_eq_one, mul_comm]
  exact mul_right_cancel h.symm
