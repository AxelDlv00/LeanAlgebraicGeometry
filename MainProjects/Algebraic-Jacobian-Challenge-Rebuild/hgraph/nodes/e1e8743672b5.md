docstring: '**Freeness of the base change gives cyclicity of the reduction.** For
  an invertible

  `R`-module `M` and an ideal `I`, if `(R ⧸ I) ⊗[R] M` is free over `R ⧸ I` then `M
  ⧸ I·M` is cyclic

  as an `R`-module.


  This is the step that turns what the geometry supplies — *the sheaf is trivial after
  restriction*,

  i.e. the base change is **free** — into the hypothesis

  `Submodule.exists_sub_smul_mem_of_quotient_cyclic` wants.


  `TensorProduct.quotTensorEquivQuotSMul` identifies the base change with `M ⧸ I·M`,
  and

  `cyclic_of_free` supplies the generator on the tensor side; the rest is transporting
  the generator

  across that equivalence and reconciling the `R ⧸ I`-action with the `R`-action

  (`Ideal.Quotient.algebraMap_eq` + `algebraMap_smul`).


  The raised heartbeat budget is for the `TensorProduct`/`Submodule.Quotient` instance
  towers on

  `(R ⧸ I) ⊗[R] M`, which are deep; the proof itself is six steps.'

theorem quotient_smul_cyclic_of_free_baseChange [Module.Invertible R M] (I : Ideal R)
    (hfree : Module.Free (R ⧸ I) ((R ⧸ I) ⊗[R] M)) :
    ∃ y : M ⧸ (I • (⊤ : Submodule R M)), ∀ z, ∃ r : R, z = r • y := by
  obtain ⟨y, hy⟩ := Module.Invertible.cyclic_of_free (R ⧸ I) ((R ⧸ I) ⊗[R] M)
  set E := TensorProduct.quotTensorEquivQuotSMul M I with hE
  refine ⟨E y, fun z => ?_⟩
  obtain ⟨r, hr⟩ := hy (E.symm z)
  obtain ⟨r', rfl⟩ := Ideal.Quotient.mk_surjective r
  refine ⟨r', ?_⟩
  have key : E (E.symm z) = E (Ideal.Quotient.mk I r' • y) := congrArg E hr
  rw [E.apply_symm_apply] at key
  rw [key]
  have hsm : (Ideal.Quotient.mk I r') • y = r' • y := by
    rw [← Ideal.Quotient.algebraMap_eq, algebraMap_smul]
  rw [hsm, E.map_smul]

end Module.Invertible
