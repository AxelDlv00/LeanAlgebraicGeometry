lemma div_mul_div_eq_of_mul_div {G : Type u} [CommGroup G]
    {t₂₃ t₁₂ t₁₃ c₂₃ c₁₂ c₁₃ : G} (W : t₂₃ * t₁₂ / t₁₃ = c₂₃ * c₁₂ / c₁₃) :
    t₂₃ / c₂₃ * (t₁₂ / c₁₂) = t₁₃ / c₁₃ := by
  rw [div_mul_div_comm, div_eq_div_iff_mul_eq_mul]
  exact (div_eq_div_iff_mul_eq_mul.mp W).trans (mul_comm _ _)
