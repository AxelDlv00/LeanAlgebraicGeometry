lemma germGenericUnits_unitsRestrict {U W : X.Opens} (h : W ≤ U)
    (hηW : genericPoint X ∈ W) (u : Γ(X, U)ˣ) :
    germGenericUnits hηW (X.unitsRestrict h u) = germGenericUnits (h hηW) u := by
