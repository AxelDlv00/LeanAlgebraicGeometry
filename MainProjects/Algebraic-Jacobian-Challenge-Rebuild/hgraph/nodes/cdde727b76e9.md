    {k A : Type u} [Field k] [CommRing A] [Algebra k A]
    (E : Algebra.EtaleCover A) :
    let d₀ : overSpec k (E.Carrier ⊗[A] E.Carrier) ⟶ overSpec k E.Carrier :=
      Over.overSpecMap (doubleInl (k := k) E)
    let d₁ : overSpec k (E.Carrier ⊗[A] E.Carrier) ⟶ overSpec k E.Carrier :=
      Over.overSpecMap (doubleInr (k := k) E)
    let g : Spec (.of E.Carrier) ⟶ Spec (.of A) :=
      Spec.map (CommRingCat.ofHom (algebraMap A E.Carrier))
    d₀.left ≫ g = d₁.left ≫ g := by
  dsimp only
  let f : A →ₐ[k] E.Carrier :=
    (Algebra.ofId A E.Carrier).restrictScalars k
  let d : overSpec k E.Carrier ⟶ overSpec k A := Over.overSpecMap f
  let d₀ : overSpec k (E.Carrier ⊗[A] E.Carrier) ⟶ overSpec k E.Carrier :=
    Over.overSpecMap (doubleInl (k := k) E)
  let d₁ : overSpec k (E.Carrier ⊗[A] E.Carrier) ⟶ overSpec k E.Carrier :=
    Over.overSpecMap (doubleInr (k := k) E)
  have hcomp :
      (doubleInl (k := k) E).comp f =
        (doubleInr (k := k) E).comp f := by
    ext r
    simp [f, doubleInl, doubleInr]
  have hover : d₀ ≫ d = d₁ ≫ d := by
    dsimp only [d₀, d₁, d]
    rw [← Over.overSpecMap_comp, ← Over.overSpecMap_comp, hcomp]
  have hleft := congrArg Over.Hom.left hover
  have hf : f.toRingHom = algebraMap A E.Carrier := rfl
  simpa only [Over.comp_left, d₀, d₁, d, Over.overSpecMap_left, hf,
    overSpec_left] using hleft
