theorem classDeg_fiberTwist (n : ℕ) :
    classDeg K (fiberTwist π n) = n • classDeg K (fiberTwist π 1) := by
  induction n with
  | zero => rw [fiberTwist_zero, classDeg_one, zero_smul]
  | succ n ih => rw [fiberTwist_succ, classDeg_mul, ih, succ_nsmul]
