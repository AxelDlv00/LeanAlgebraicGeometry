    {F : Type u} [Field F] [Algebra F k] [Algebra.IsAlgebraic F k] :
    ∃ L : DatG0.FinSubext F k,
      ∀ U : { U // U ∈ (pic0FiniteStageAtlas C).charts },

    finitePresentation_pic0FiniteStageChartRing C U
