theorem testGeneral_of_surjective {J : Over (Spec (.of k))}
    (rep : (pic0TypeFunctor C).RepresentableBy J)
    (lam : pic0Subgroup C DivO)
    (hsurj : Function.Surjective (abelOfPic0Class rep lam).left.base) :
    ∀ y : J.left, ∃ (T : Over (Spec (.of k))) (p : T.left) (q : T ⟶ DivO) (t : T ⟶ J),
      t.left.base p = y ∧ pic0Map C q lam = rep.homEquiv t := by
  intro y
  obtain ⟨x, hx⟩ := hsurj y
  refine ⟨DivO, x, 𝟙 _, abelOfPic0Class rep lam, hx, ?_⟩
  have h := abelOfPic0Class_comp_class rep lam (T := DivO) (𝟙 _)
  rw [Category.id_comp] at h
  simpa using h.symm
