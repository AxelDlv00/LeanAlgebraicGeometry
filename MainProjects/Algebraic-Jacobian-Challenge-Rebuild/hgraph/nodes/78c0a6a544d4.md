theorem exists_matrixPresentation_of_isLocalizedModule
    [Module.FinitePresentation R M] (p : Ideal R) [p.IsPrime] :
    ∃ g : R, g ∉ p ∧
      ∀ (R' : Type u') [CommRing R'] [Algebra R R'] [IsLocalization.Away g R']
        (N : Type v') [AddCommGroup N] [Module R N] [Module R' N]
        [IsScalarTower R R' N] (l : M →ₗ[R] N)
        [IsLocalizedModule (Submonoid.powers g) l],
      ∃ mm : ℕ, Nonempty (MatrixPresentation R' N (p.fiberRank M) mm) := by
  classical
  -- a basis of the fiber, rescaled to consist of pure tensors `1 ⊗ mᵢ`
  let b : Basis (Fin (p.fiberRank M)) p.ResidueField (p.Fiber M) :=
    Module.finBasis p.ResidueField (p.Fiber M)
  have hb : ∀ i, ∃ r ∉ p, ∃ s : M, r • b i = (1 : p.ResidueField) ⊗ₜ[R] s := fun i ↦
    exists_smul_eq_one_tmul p (b i)
  choose r hr mvec hm using hb
  -- the images `1 ⊗ mᵢ` span the fiber
  have hunit : ∀ i, IsUnit (algebraMap R p.ResidueField (r i)) := by
    intro i
    rw [isUnit_iff_ne_zero, Ne, Ideal.algebraMap_residueField_eq_zero]
    exact hr i
  have hspan : Submodule.span p.ResidueField
      (Set.range fun i ↦ (1 : p.ResidueField) ⊗ₜ[R] mvec i) = ⊤ := by
    have hsm : Set.range (fun i ↦ (1 : p.ResidueField) ⊗ₜ[R] mvec i) =
        Set.range ⇑(b.unitsSMul fun i ↦ (hunit i).unit) := by
      apply congrArg
      funext i
      rw [← hm i, ← algebraMap_smul p.ResidueField (r i) (b i),
        Basis.unitsSMul_apply, Units.smul_def, IsUnit.unit_spec]
    rw [hsm]
    exact (b.unitsSMul fun i ↦ (hunit i).unit).span_eq
  -- the `mᵢ` generate `M` modulo Nakayama: the quotient has vanishing fiber
  have hQfiber : Subsingleton (p.ResidueField ⊗[R]
      (M ⧸ LinearMap.range (Fintype.linearCombination R mvec))) := by
    have hsurj : Function.Surjective
        ((LinearMap.range (Fintype.linearCombination R mvec)).mkQ.baseChange
          p.ResidueField) := by
      rw [LinearMap.baseChange_eq_ltensor]
      exact LinearMap.lTensor_surjective _ (Submodule.mkQ_surjective _)
    have hcomp : ((LinearMap.range (Fintype.linearCombination R mvec)).mkQ.baseChange
        p.ResidueField) ∘ₗ
        ((Fintype.linearCombination R mvec).baseChange p.ResidueField) = 0 := by
      rw [← LinearMap.baseChange_comp]
      have h0 : (LinearMap.range (Fintype.linearCombination R mvec)).mkQ ∘ₗ
          Fintype.linearCombination R mvec = 0 := by
        ext x
        simp only [LinearMap.comp_apply, Submodule.mkQ_apply, LinearMap.zero_apply,
          Submodule.Quotient.mk_eq_zero]
        exact ⟨_, rfl⟩
      rw [h0, LinearMap.baseChange_zero]
    have hrange : LinearMap.range
        ((Fintype.linearCombination R mvec).baseChange p.ResidueField) = ⊤ := by
      rw [← top_le_iff, ← hspan, Submodule.span_le]
      rintro _ ⟨i, rfl⟩
      exact ⟨(1 : p.ResidueField) ⊗ₜ Pi.single i 1, by
        simp [LinearMap.baseChange_tmul, Fintype.linearCombination_apply_single]⟩
    apply subsingleton_of_forall_eq 0
    intro x
    obtain ⟨y, rfl⟩ := hsurj x
    have hy : y ∈ LinearMap.range
        ((Fintype.linearCombination R mvec).baseChange p.ResidueField) :=
      hrange ▸ Submodule.mem_top
    obtain ⟨a, rfl⟩ := hy
    exact congr($hcomp a)
  -- Nakayama at `p`: the finite quotient localizes to zero at `p`
  haveI : Module.Finite R (M ⧸ LinearMap.range (Fintype.linearCombination R mvec)) :=
    Module.Finite.of_surjective _ (Submodule.mkQ_surjective _)
  have e2 : Localization.AtPrime p ⊗[R]
      (M ⧸ LinearMap.range (Fintype.linearCombination R mvec))
      ≃ₗ[Localization.AtPrime p] LocalizedModule p.primeCompl
        (M ⧸ LinearMap.range (Fintype.linearCombination R mvec)) :=
    (IsLocalizedModule.isBaseChange p.primeCompl (Localization.AtPrime p)
      (LocalizedModule.mkLinearMap p.primeCompl _)).equiv
  haveI : Module.Finite (Localization.AtPrime p) (LocalizedModule p.primeCompl
      (M ⧸ LinearMap.range (Fintype.linearCombination R mvec))) :=
    Module.Finite.equiv e2
  have hQpsub : Subsingleton (LocalizedModule p.primeCompl
      (M ⧸ LinearMap.range (Fintype.linearCombination R mvec))) := by
    haveI := hQfiber
    haveI : Subsingleton (p.ResidueField ⊗[Localization.AtPrime p]
        (LocalizedModule p.primeCompl
          (M ⧸ LinearMap.range (Fintype.linearCombination R mvec)))) :=
      Equiv.subsingleton (((TensorProduct.congr
        (LinearEquiv.refl (Localization.AtPrime p) p.ResidueField) e2.symm).toEquiv).trans
        (AlgebraTensorModule.cancelBaseChange R (Localization.AtPrime p) p.ResidueField
          p.ResidueField _).toEquiv)
    exact (IsLocalRing.subsingleton_tensorProduct (R := Localization.AtPrime p)).mp ‹_›
  -- extract `g ∉ p` annihilating the quotient
  obtain ⟨g, hgann, hgp⟩ : ∃ g ∈ Module.annihilator R
      (M ⧸ LinearMap.range (Fintype.linearCombination R mvec)), g ∉ p := by
    have hnmem : (⟨p, ‹_›⟩ : PrimeSpectrum R) ∉ Module.support R
        (M ⧸ LinearMap.range (Fintype.linearCombination R mvec)) :=
      Module.notMem_support_iff.mpr hQpsub
    rw [Module.mem_support_iff_of_finite] at hnmem
    exact SetLike.not_le_iff_exists.mp hnmem
  have hg : ∀ x : M, g • x ∈ LinearMap.range (Fintype.linearCombination R mvec) := by
    intro x
    have hx := Module.mem_annihilator.mp hgann (Submodule.Quotient.mk
      (p := LinearMap.range (Fintype.linearCombination R mvec)) x)
    rwa [← Submodule.Quotient.mk_smul, Submodule.Quotient.mk_eq_zero] at hx
  refine ⟨g, hgp, ?_⟩
  intro R' _ _ _ N _ _ _ _ l _
  -- `N` is finitely presented over `R'`
  haveI : Module.FinitePresentation R' N :=
    _root_.FinitePresentation.of_isBaseChange l
      (IsLocalizedModule.isBaseChange (Submonoid.powers g) R' l)
  -- the images `l mᵢ` generate `N` over `R'`
  have hgen : Function.Surjective (Fintype.linearCombination R' fun i ↦ l (mvec i)) := by
    rw [← LinearMap.range_eq_top, Fintype.range_linearCombination, ← top_le_iff]
    intro n _
    obtain ⟨⟨x, s⟩, hxs⟩ := IsLocalizedModule.surj (Submonoid.powers g) l n
    have hs : IsUnit (algebraMap R R' (s : R)) := IsLocalization.map_units R' s
    have hgu : IsUnit (algebraMap R R' g) :=
      IsLocalization.map_units R' ⟨g, Submonoid.mem_powers g⟩
    -- `l (g • x)` lies in the `R'`-span of the `l mᵢ`
    have hmem : l (g • x) ∈ Submodule.span R' (Set.range fun i ↦ l (mvec i)) := by
      obtain ⟨c, hc⟩ := hg x
      have hsum : l (g • x) = ∑ i, c i • l (mvec i) := by
        rw [← hc]
        simp [Fintype.linearCombination_apply, map_sum]
      rw [hsum]
      exact Submodule.sum_mem _ fun i _ ↦ Submodule.smul_of_tower_mem _ (c i)
        (Submodule.subset_span ⟨i, rfl⟩)
    -- clear the units `s` and `g`
    have hxs' : (s : R) • n = l x := hxs
    have hn : algebraMap R R' ((s : R) * g) • n = l (g • x) := by
      rw [map_mul, mul_smul, algebraMap_smul, algebraMap_smul, smul_comm, hxs',
        ← map_smul]
    have hu : IsUnit (algebraMap R R' ((s : R) * g)) := by
      rw [map_mul]
      exact hs.mul hgu
    have hnn : n = hu.unit⁻¹ • l (g • x) := by
      rw [eq_inv_smul_iff, Units.smul_def, IsUnit.unit_spec]
      exact hn
    rw [hnn]
    exact Submodule.smul_mem _ _ hmem
  -- the kernel is finitely generated; its generators are the relation matrix
  obtain ⟨T, hT⟩ := Module.FinitePresentation.fg_ker
    (Fintype.linearCombination R' fun i ↦ l (mvec i)) hgen
  refine ⟨T.card, ⟨{
    relMatrix := Matrix.of fun i j ↦ (T.equivFin.symm j : Fin (p.fiberRank M) → R') i
    proj := Fintype.linearCombination R' fun i ↦ l (mvec i)
    surjective_proj := hgen
    exact_mulVecLin_proj := ?_ }⟩⟩
  rw [LinearMap.exact_iff, ← hT]
  apply le_antisymm
  · -- each generator is a column of the matrix
    rw [Submodule.span_le]
    intro t ht
    obtain ⟨j, hj⟩ := T.equivFin.symm.surjective ⟨t, ht⟩
    refine ⟨Pi.single j 1, ?_⟩
    funext i
    simp only [Matrix.mulVecLin_apply, Matrix.mulVec, Matrix.of_apply]
    rw [dotProduct_single, mul_one, hj]
  · -- every matrix-vector product is a combination of the columns
    rintro _ ⟨x, rfl⟩
    have hcol : (Matrix.of fun i j ↦
          (T.equivFin.symm j : Fin (p.fiberRank M) → R') i).mulVecLin x =
        ∑ j, x j • ((T.equivFin.symm j : Fin (p.fiberRank M) → R')) := by
      funext i
      simp [Matrix.mulVec, dotProduct, Finset.sum_apply, mul_comm]
    rw [hcol]
    exact Submodule.sum_mem _ fun j _ ↦ Submodule.smul_mem _ _
      (Submodule.subset_span (T.equivFin.symm j).2)
