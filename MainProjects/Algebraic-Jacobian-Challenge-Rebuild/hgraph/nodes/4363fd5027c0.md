      (relCover C R D).V₀)) :

        (relCocycleBaseChange C R R' D g) (le_refl (relCover C R' D).V₀)

            (le_refl (relCover C R D).V₀) p) := by

    LinearEquiv.trans_apply]

    (relCocycleBaseChange C R R' D g)

    (relTermBaseChange C R R' D.V₀ D.isAffineOpen₀.isCompact
