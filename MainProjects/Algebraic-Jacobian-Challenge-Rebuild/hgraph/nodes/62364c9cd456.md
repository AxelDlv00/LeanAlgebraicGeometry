theorem ker_fstRingHom :
    RingHom.ker (fstRingHom (R := A)) = Ideal.span {(ε : DualNumber A)} := by
  apply le_antisymm
  · intro x hx
    rw [RingHom.mem_ker, fstRingHom_apply] at hx
    rw [Ideal.mem_span_singleton]
    refine ⟨inl x.snd, ?_⟩
    apply TrivSqZeroExt.ext
    · rw [hx, TrivSqZeroExt.fst_mul, DualNumber.fst_eps, zero_mul]
    · rw [TrivSqZeroExt.snd_mul, DualNumber.fst_eps, DualNumber.snd_eps,
        TrivSqZeroExt.fst_inl, TrivSqZeroExt.snd_inl]
      simp
  · rw [Ideal.span_le, Set.singleton_subset_iff]
    show fstRingHom (ε : DualNumber A) = 0
    rw [fstRingHom_apply, DualNumber.fst_eps]
