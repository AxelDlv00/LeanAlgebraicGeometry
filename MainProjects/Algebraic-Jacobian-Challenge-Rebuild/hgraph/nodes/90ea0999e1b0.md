lemma comm_t₀_pow (f : P.Hom P') (m : ℕ) (x : M₀) :
    f.hom₀ ((P.t₀ ^ m) x) = (P'.t₀ ^ m) (f.hom₀ x) := by
  induction m with
  | zero => simp
  | succ m ih =>
    rw [pow_succ', pow_succ']
    simp only [Module.End.mul_apply]
    rw [f.comm_t₀, ih]
