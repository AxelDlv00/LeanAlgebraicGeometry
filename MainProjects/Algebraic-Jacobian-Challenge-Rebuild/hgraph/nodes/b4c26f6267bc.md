theorem thetaToTriple_from_third (A : AffAdaptation D d) (a : ℕ)

    A.thetaOverlapToTriple (π := π) a i l i j l
        (A.thetaTripleOpen_le_pair13 i j l)
        (A.thetaToOverlapRight (π := π) a i l x) =
