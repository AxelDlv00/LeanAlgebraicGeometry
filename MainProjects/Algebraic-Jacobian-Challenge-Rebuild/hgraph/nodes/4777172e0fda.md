theorem probeAlgebraEqOfTower {B : Type u} [CommRing B]
    (iT iD : Algebra k B)
    (tw : @IsScalarTower k k B (Algebra.id k).toSMul iT.toSMul iD.toSMul) :
    iD = iT := by
  have hiMap := @IsScalarTower.algebraMap_eq k k B _ _ _
    (Algebra.id k) iT iD tw
  exact Algebra.algebra_ext iD iT fun r =>
    (DFunLike.congr_fun hiMap r).trans (by rfl)
