lemma unitsRestrict_top_self {Y : Scheme.{u}} (u : Γ(Y, ⊤)ˣ) :
    Y.unitsRestrict le_top u = u := by
  apply Units.ext
  change (Y.presheaf.map (homOfLE _).op) u.val = u.val
  rw [show (homOfLE (le_top : (⊤ : Y.Opens) ≤ ⊤)).op = 𝟙 (op ⊤) from rfl,
    Y.presheaf.map_id]
