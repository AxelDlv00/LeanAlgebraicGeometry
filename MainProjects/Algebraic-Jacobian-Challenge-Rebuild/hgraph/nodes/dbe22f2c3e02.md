private theorem pow_smul_mem {S : Type*} [Monoid S] [MulAction S N] {Q : Submodule k N}
    {r : S} (hr : ∀ x ∈ Q, r • x ∈ Q) {x : N} (hx : x ∈ Q) : ∀ m : ℕ, r ^ m • x ∈ Q
  | 0 => by simpa using hx
  | m + 1 => by
    rw [pow_succ', mul_smul]
    exact hr _ (pow_smul_mem hr hx m)
