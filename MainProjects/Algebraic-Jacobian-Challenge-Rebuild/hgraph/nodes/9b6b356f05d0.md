def divFam (T : Over (Spec (.of k))) : Type u :=
  {s : Π U : T.left.affineOpens, DivFam C Γ(T.left, U.1) π n //
    ∀ (U V : T.left.affineOpens) (h : U.1 ≤ V.1),
      DivFam.mapAlgHom (Over.resAlgHom T h) (s V) = s U}
