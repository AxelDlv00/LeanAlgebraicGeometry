noncomputable def divFamDivisor (F : DivFam C K π n) : (relCurve C K).CurveDivisor :=
  Quotient.liftOn F (fun G => Scheme.presentationDivisor K G.eqns.presentation)
    (fun _ _ h => Scheme.presentationDivisor_eq_of_divEq K h)
