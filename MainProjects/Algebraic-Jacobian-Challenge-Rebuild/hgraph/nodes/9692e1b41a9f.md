private theorem span_range_comp_surjective
    {R B M N : Type*} [CommRing R] [CommRing B] [Algebra R B]
    [AddCommGroup M] [Module R M] [AddCommGroup N] [Module R N]
    (f : M →ₗ[R] B) (e : N →ₗ[R] M) (he : Function.Surjective e) :
    Ideal.span (Set.range (f.comp e)) = Ideal.span (Set.range f) := by
  apply le_antisymm
  · refine Ideal.span_le.mpr ?_
    rintro _ ⟨x, rfl⟩
    exact Ideal.subset_span ⟨e x, rfl⟩
  · refine Ideal.span_le.mpr ?_
    rintro _ ⟨x, rfl⟩
    obtain ⟨y, rfl⟩ := he x
    exact Ideal.subset_span ⟨y, rfl⟩
