theorem forall_finite_colength_of_pieces_eq_chart
    (hpieces : ∀ j : A.index, A.pieces j = (relCover C R (fiberTwoCover π)).V₀
      ∨ A.pieces j = (relCover C R (fiberTwoCover π)).V₁)
    (hsub : d.supportLocus ⊆ (((relCover C R (fiberTwoCover π)).V₀
      ⊓ (relCover C R (fiberTwoCover π)).V₁ : (relCurve C R).Opens) : Set (relCurve C R))) :
    ∀ j : A.index, Module.Finite R (A.colength j) :=
  A.forall_finite_colength_of_forall_subset_or_disjoint
    (A.forall_subset_or_disjoint_of_pieces_eq_chart hpieces hsub)
