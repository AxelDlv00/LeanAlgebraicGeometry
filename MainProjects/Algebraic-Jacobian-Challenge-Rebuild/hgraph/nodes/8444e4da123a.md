theorem exists_divisorSections_pencil_relation
    (A B : X.CurveDivisor) (w : X.functionFieldˣ)
    (hinf : A ⊓
        (A + Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w) = B)
    (y : Fin 2 → divisorSections K A ⊤)
    (hy : (y 0 : X.functionField) +
      (w : X.functionField) * (y 1 : X.functionField) = 0) :
    ∃ z : divisorSections K B ⊤,
      (y 0 : X.functionField) = (w : X.functionField) * (z : X.functionField) ∧
      (y 1 : X.functionField) = -(z : X.functionField) := by
  let D := Scheme.divOf (X ↘ Spec (CommRingCat.of K)) w
  have hwyA : (w : X.functionField) * (y 1 : X.functionField) ∈
      divisorSections K A ⊤ := by
    have heq : (w : X.functionField) * (y 1 : X.functionField) =
        -(y 0 : X.functionField) := by
      linear_combination hy
    rw [heq]
    exact (divisorSections K A ⊤).neg_mem (y 0).property
  have hmap : Submodule.map (Scheme.mulLinear K (w : X.functionField))
      (divisorSections K (A + D) ⊤) = divisorSections K A ⊤ := by
    rw [map_mulLinear_divisorSections_top K (Units.ne_zero w)]
    have hmk : Units.mk0 (w : X.functionField) (Units.ne_zero w) = w := Units.ext rfl
    rw [hmk]
    change divisorSections K (A + D - D) ⊤ = divisorSections K A ⊤
    congr 2
    abel
  have hwyMap : (w : X.functionField) * (y 1 : X.functionField) ∈
      Submodule.map (Scheme.mulLinear K (w : X.functionField))
        (divisorSections K (A + D) ⊤) := by
    rw [hmap]
    exact hwyA
  obtain ⟨t, ht, htw⟩ := hwyMap
  have hty : t = (y 1 : X.functionField) := by
    change (w : X.functionField) * t =
      (w : X.functionField) * (y 1 : X.functionField) at htw
    exact mul_left_cancel₀ (Units.ne_zero w) htw
  have hyAD : (y 1 : X.functionField) ∈ divisorSections K (A + D) ⊤ := by
    rwa [hty] at ht
  have hyB : (y 1 : X.functionField) ∈ divisorSections K B ⊤ := by
    have hmem : (y 1 : X.functionField) ∈
        divisorSections K A ⊤ ⊓ divisorSections K (A + D) ⊤ :=
      Submodule.mem_inf.mpr ⟨(y 1).property, hyAD⟩
    rw [divisorSections_inf K A (A + D), hinf] at hmem
    exact hmem
  let z : divisorSections K B ⊤ := ⟨-(y 1 : X.functionField),
    (divisorSections K B ⊤).neg_mem hyB⟩
  refine ⟨z, ?_, ?_⟩
  · change (y 0 : X.functionField) = (w : X.functionField) * -(y 1 : X.functionField)
    calc
      (y 0 : X.functionField) =
          -((w : X.functionField) * (y 1 : X.functionField)) :=
        by linear_combination hy
      _ = (w : X.functionField) * -(y 1 : X.functionField) := by ring
  · simp only [z, neg_neg]
