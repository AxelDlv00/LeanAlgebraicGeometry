noncomputable def awayLiftAlgHom (x : A) (a : A →ₐ[k] B) (hx : IsUnit (a x)) :
    Localization.Away x →ₐ[k] B :=
  IsLocalization.liftAlgHom (M := Submonoid.powers x) (f := a) fun y => by
    obtain ⟨n, hn⟩ := (Submonoid.mem_powers_iff _ _).mp y.2
    rw [← hn, map_pow]
    exact hx.pow n
