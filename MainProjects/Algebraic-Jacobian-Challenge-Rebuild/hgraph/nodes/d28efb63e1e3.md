noncomputable def directedQuotientMap (N : ι → Submodule R M) (hN : Monotone N)
    (i j : ι) (hij : i ≤ j) : (M ⧸ N i) →ₗ[R] (M ⧸ N j) :=
  (N i).mapQ (N j) LinearMap.id (by simpa using hN hij)
