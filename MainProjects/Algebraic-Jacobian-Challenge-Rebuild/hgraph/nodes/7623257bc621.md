theorem chartIndex_of_isDegree {c : ℤ} (h : IsDivisorDegree C c) :
    ∃ (m : ℕ) (Z : (C ⊗ overSpec k k).left.CurveDivisor),
      Scheme.CurveDivisor.deg k Z = (m : ℤ) * classDeg k (thetaCechClass C) - c := by
  obtain ⟨W, hW⟩ := h
  refine ⟨0, -W, ?_⟩
  rw [Scheme.CurveDivisor.deg_neg, hW]
