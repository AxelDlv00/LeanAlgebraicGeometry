theorem mem_zariskiTopology_iSup_of_pointwise {ι : Type u} {T : Scheme.{u}}
    (Sv : ι → Sieve T) (W : ↥T → T.Opens) (hW : ∀ t, t ∈ W t) (idx : ↥T → ι)
    (hmem : ∀ t, (Sv (idx t)).arrows (W t).ι) :
    (⨆ i, Sv i) ∈ Scheme.zariskiTopology T := by
  refine Scheme.mem_grothendieckTopology_iff.mpr
    ⟨Scheme.Cover.mkOfCovers (↥T) (fun t => (W t : Scheme.{u})) (fun t => (W t).ι)
      (fun t => ⟨t, ⟨t, hW t⟩, rfl⟩) (fun t => inferInstance), ?_⟩
  rintro V g ⟨t⟩
  exact le_iSup Sv (idx t) _ (hmem t)
