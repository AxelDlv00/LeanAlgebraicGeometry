noncomputable def pic0OverlapIso :
    (Over.map (picRepOverlapSpecInl k L)).op ⋙
        pic0TypeFunctor ((baseChange k L).obj C) ≅
      (Over.map (picRepOverlapSpecInr k L)).op ⋙
        pic0TypeFunctor ((baseChange k L).obj C) :=
  Functor.isoWhiskerLeft (Over.map (picRepOverlapSpecInl k L)).op
      (pic0ThetaType k L C) ≪≫
    (Functor.associator _ _ _).symm ≪≫
    Functor.isoWhiskerRight (picRepOverlapOpPushIso k L) (pic0TypeFunctor C) ≪≫
    Functor.associator _ _ _ ≪≫
    (Functor.isoWhiskerLeft (Over.map (picRepOverlapSpecInr k L)).op
      (pic0ThetaType k L C)).symm
