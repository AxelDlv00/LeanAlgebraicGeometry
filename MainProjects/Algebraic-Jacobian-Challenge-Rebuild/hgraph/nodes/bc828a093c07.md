theorem T_neg_natCast_smul_mem {Q : Submodule k N} (hQ : ∀ x ∈ Q, (T (-1) : k[T;T⁻¹]) • x ∈ Q)
    {x : N} (hx : x ∈ Q) (m : ℕ) : (T (-(m : ℤ)) : k[T;T⁻¹]) • x ∈ Q := by
  have h : (T (-(m : ℤ)) : k[T;T⁻¹]) = T (-1) ^ m := by rw [T_pow, mul_neg_one]
  rw [h]
  exact pow_smul_mem hQ hx m
