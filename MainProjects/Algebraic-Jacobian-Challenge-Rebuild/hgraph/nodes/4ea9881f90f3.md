theorem cartierIdealData_eq_cartierIdeal [IsProper C.hom]
    (A : AffAdaptation D d) : d.cartierIdealData = A.cartierIdeal := by
  apply Scheme.IdealSheafData.ext_of_iSup_eq_top
    (fun j : D.index ↦ ⟨D.pieces j, D.isAffineOpen j⟩) D.cover
  intro j
  exact (A.cartierIdeal_ideal_eq_sectionIdeal j).symm
