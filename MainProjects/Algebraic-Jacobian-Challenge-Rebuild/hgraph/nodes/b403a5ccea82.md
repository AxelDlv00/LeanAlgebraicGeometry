private lemma app_naturality {P Q : Cᵒᵖ ⥤ ModuleCat.{u} R} (η : P ⟶ Q) {W W' : Cᵒᵖ}
    (i : W ⟶ W') (x : P.obj W) :
    (η.app W').hom ((P.map i).hom x) = (Q.map i).hom ((η.app W).hom x) :=
  DFunLike.congr_fun (congrArg ModuleCat.Hom.hom (η.naturality i)) x
