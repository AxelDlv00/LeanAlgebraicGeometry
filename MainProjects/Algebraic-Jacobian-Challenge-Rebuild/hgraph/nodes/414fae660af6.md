noncomputable def cechPicMapEquivOfIso (f : X ⟶ Y) [IsIso f] : Y.CechPic ≃* X.CechPic :=
  MulEquiv.ofBijective (CechPic.map f)
    ⟨cechPicMap_injective_of_isIso f, cechPicMap_surjective_of_isIso f⟩

@[simp]
