def relPicHom : relPic E (overSpec kE B) →* relPic D (overSpec kD B) :=
  QuotientGroup.map (picFromBase E (overSpec kE B)) (picFromBase D (overSpec kD B))
    (CechPic.map (T.hom B)) (T.picFromBase_le_comap_hom B)
