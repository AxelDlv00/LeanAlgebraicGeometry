def sigmaExtensionMap {T' T : C} (g : T' ⟶ T) :
    (Σ a : T ⟶ S, F.obj (op (Over.mk a))) → Σ a : T' ⟶ S, F.obj (op (Over.mk a)) :=
  fun x =>
    ⟨g ≫ x.1, F.map (Over.homMk g rfl : Over.mk (g ≫ x.1) ⟶ Over.mk x.1).op x.2⟩
