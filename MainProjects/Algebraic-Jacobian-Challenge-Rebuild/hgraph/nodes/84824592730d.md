theorem forall_subset_or_disjoint_of_pieces_eq_chart
    (hpieces : ∀ j : A.index, A.pieces j = (relCover C R (fiberTwoCover π)).V₀
      ∨ A.pieces j = (relCover C R (fiberTwoCover π)).V₁)
    (hsub : d.supportLocus ⊆ (((relCover C R (fiberTwoCover π)).V₀
      ⊓ (relCover C R (fiberTwoCover π)).V₁ : (relCurve C R).Opens) : Set (relCurve C R))) :
    ∀ j : A.index, d.supportLocus ⊆ (A.pieces j : Set (relCurve C R))
      ∨ Disjoint d.supportLocus (A.pieces j : Set (relCurve C R)) := by
  intro j
  refine Or.inl fun x hx => ?_
  have hx' := hsub hx
  rcases hpieces j with h | h <;> rw [h]
  · exact hx'.1
  · exact hx'.2
