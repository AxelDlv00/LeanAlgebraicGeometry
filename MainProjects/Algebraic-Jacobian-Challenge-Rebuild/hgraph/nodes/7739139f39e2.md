theorem isSMulRegular_self_of_mem_nonZeroDivisors {t : A}
    (ht : t ∈ nonZeroDivisors A) : IsSMulRegular A t := by
  intro u v huv
  have h0 : (u - v) * t = 0 := by
    have hm : t * u = t * v := huv
    rw [sub_mul, mul_comm u, mul_comm v, hm, sub_self]
  exact sub_eq_zero.mp ((mem_nonZeroDivisors_iff.mp ht).2 (u - v) h0)
