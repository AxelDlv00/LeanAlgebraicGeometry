    (S : Type u) [CommRing S] [Algebra k S]
    [Algebra (PairChartRing k g r₁ g r₂ i j) S]
    [IsScalarTower k (PairChartRing k g r₁ g r₂ i j) S] :
    carveIdeal k g r₁ r₂ μ i j
        ≤ RingHom.ker (algebraMap (PairChartRing k g r₁ g r₂ i j) S) ↔
