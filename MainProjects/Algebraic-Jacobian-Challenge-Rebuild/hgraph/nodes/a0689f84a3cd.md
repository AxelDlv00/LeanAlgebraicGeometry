    (w : PairChartRing k d₁ r₁ d₂ r₂ i j →ₐ[k] B)

  have h1 := congrArg (· ≫ grPairSnd k d₁ r₁ d₂ r₂) h

    Spec.map_comp, Spec.map_comp, Category.assoc, Category.assoc]
