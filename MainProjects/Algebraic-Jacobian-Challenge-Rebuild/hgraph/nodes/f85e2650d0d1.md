theorem sectionIdeal_map_ideal_basicOpen [IsProper C.hom]
    (d : (relCurve C R).LocalEquations) (U : (relCurve C R).affineOpens)
    (f : Γ(relCurve C R, U.1)) :
    Ideal.map ((relCurve C R).presheaf.map
      (homOfLE ((relCurve C R).basicOpen_le f)).op).hom (d.sectionIdeal U.1) =
      d.sectionIdeal ((relCurve C R).affineBasicOpen f).1 := by
  obtain ⟨D, ⟨A⟩⟩ := exists_affAdaptation_of_isProper C R d
  rw [← A.cartierIdeal_ideal_eq_sectionIdeal_affine U,
    ← A.cartierIdeal_ideal_eq_sectionIdeal_affine
      ((relCurve C R).affineBasicOpen f)]
  exact A.cartierIdeal.map_ideal_basicOpen U f
