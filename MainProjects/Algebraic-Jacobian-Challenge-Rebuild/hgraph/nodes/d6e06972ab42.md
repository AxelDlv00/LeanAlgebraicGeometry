noncomputable def diagonalChart (C : Over (Spec (.of k))) {U : C.left.Opens}
    (hU : IsAffineOpen U) (elift : Γ(C.left, U) ⊗[k] Γ(C.left, U)) :
    (C ⊗ C).left.Opens :=
  (C ⊗ C).left.basicOpen (productChartSections C C hU hU (1 - elift))
