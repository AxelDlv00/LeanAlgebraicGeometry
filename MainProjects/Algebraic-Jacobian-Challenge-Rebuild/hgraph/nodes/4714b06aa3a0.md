noncomputable def diff : (Γ(X, U₀) × Γ(X, U₁)) →ₗ[k] Γ(X, U₀ ⊓ U₁) :=
  ((X.moduleKSheaf k).obj.map (homOfLE (inf_le_left : U₀ ⊓ U₁ ≤ U₀)).op).hom.comp
      (LinearMap.fst k _ _) -
    ((X.moduleKSheaf k).obj.map (homOfLE (inf_le_right : U₀ ⊓ U₁ ≤ U₁)).op).hom.comp
