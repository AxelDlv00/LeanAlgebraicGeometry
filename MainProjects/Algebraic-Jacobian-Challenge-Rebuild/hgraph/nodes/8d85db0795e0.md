noncomputable def twoChartCandidate (sel : X → Bool) (hmem : ∀ x, x ∈ V (sel x))
    (γ : X.unitsCocycle (twoChartCover V sel hmem))
    {x₀ x₁ : X} (h₀ : sel x₀ = false) (h₁ : sel x₁ = true) : Γ(X, V false ⊓ V true)ˣ :=
  mixedValue h₀ h₁ (Scheme.unitsEvInf γ x₀ x₁)
