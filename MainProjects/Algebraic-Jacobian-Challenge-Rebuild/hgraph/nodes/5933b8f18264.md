noncomputable def unitCoaction : B →ₗ[B] B ⊗[A] B where
  toFun x := u.val * x ⊗ₜ 1
  map_add' x y := by rw [add_tmul, mul_add]
  map_smul' b x := by
    rw [RingHom.id_apply, ← smul_tmul', mul_smul_comm]

@[simp]
