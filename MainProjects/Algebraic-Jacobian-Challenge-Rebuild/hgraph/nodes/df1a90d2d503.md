def IsEquivariant {X' : Scheme.{u}}
    {f' : X' ⟶ Spec (CommRingCat.of L)}
    (rho : SemilinearGalAction K L X f)
    (rho' : SemilinearGalAction K L X' f') (h : X ⟶ X') : Prop :=
  ∀ gamma : L ≃ₐ[K] L,
    (rho.act gamma).hom ≫ h = h ≫ (rho'.act gamma).hom
