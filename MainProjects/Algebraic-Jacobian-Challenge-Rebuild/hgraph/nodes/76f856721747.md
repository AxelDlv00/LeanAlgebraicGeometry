lemma span_germ_thetaFieldGluedEquiv_eq_stalkIdeal (d : (relCurve C K).LocalEquations)
    (s : relThetaSections C K π a) {z : relCurve C K}
    (hzg : z ≠ genericPoint (relCurve C K))
    (hle : Scheme.ord (relCurve C K ↘ Spec (CommRingCat.of K)) hzg
        (thetaFieldRead C K π a s)
      ≤ Scheme.divisorBound
          (thetaFieldDivisor C K π a - Scheme.presentationDivisor K d.presentation) hzg)
    (hge : Scheme.divisorBound
          (thetaFieldDivisor C K π a - Scheme.presentationDivisor K d.presentation) hzg
      ≤ Scheme.ord (relCurve C K ↘ Spec (CommRingCat.of K)) hzg
          (thetaFieldRead C K π a s)) :
    Ideal.span
        {((relCurve C K).presheaf.germ (⊤ ⊓ (thetaFieldPointedCover C K π).opens z) z
            ⟨trivial, (thetaFieldPointedCover C K π).mem_opens z⟩).hom
          ((thetaFieldGluedEquiv C K π a s).val z)}
      = d.stalkIdeal z := by
  refine le_antisymm ?_
    ((stalkIdeal_le_span_germ_iff_le_ord C K π a d s hzg).mp hge)
  rw [Ideal.span_singleton_le_iff_mem]
  exact (ord_thetaFieldRead_le_iff C K π a d s hzg).mp hle

end ExactOrder
