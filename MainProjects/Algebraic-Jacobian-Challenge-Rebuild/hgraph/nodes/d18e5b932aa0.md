theorem isUnit_germ_genericPoint_of_windowCompare_ne_zero
    (b : Bool)
    (hη : genericPoint (relCurve C K) ∈ relPinnedChart C K π b)
    {x : R ⊗[k] ↥(Scheme.divisorSections k (a • fiberWeilDivisor π) ⊤)}
    (hx : windowCompare R K x ≠ 0) :
    IsUnit (((relCurve C K).presheaf.germ (relPinnedChart C K π b)
      (genericPoint (relCurve C K)) hη).hom
      (relPinnedSectionsMap C R K π b
        (relThetaResSide a b le_rfl (relThetaWindowEquiv C R π a hH1 x)))) := by
  have hb : relPinnedChart C K π b ≠ ⊥ := by
    intro hbot
    rw [hbot] at hη
    exact hη
  have hs := relPinnedSectionsMap_relThetaResSide_ne_zero C R K π a hH1 b hx hb
  have hg : ((relCurve C K).presheaf.germ (relPinnedChart C K π b)
      (genericPoint (relCurve C K)) hη).hom
      (relPinnedSectionsMap C R K π b
        (relThetaResSide a b le_rfl (relThetaWindowEquiv C R π a hH1 x))) ≠ 0 := by
    intro h0
    apply hs
    apply germ_injective_of_isIntegral (relCurve C K) (genericPoint (relCurve C K)) hη
    rw [h0, map_zero]
  exact isUnit_iff_ne_zero.mpr hg
