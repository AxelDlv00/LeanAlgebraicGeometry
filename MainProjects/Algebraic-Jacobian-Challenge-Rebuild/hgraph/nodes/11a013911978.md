docstring: '**The Nakayama iteration, with no finiteness hypothesis.** If `N` together
  with `I • M`

  spans `M`, then `N` together with `Iⁿ • M` spans `M`, for every `n`.


  Substituting `M = N + I·M` into itself `n` times: `Iⁿ·M = Iⁿ·N + Iⁿ·(I·M)`, the
  first

  summand lies in `N` and the second is `Iⁿ⁺¹·M`.


  Mathlib''s Nakayama lemmas (`Submodule.le_of_le_smul_of_le_jacobson_bot` and friends,

  `Mathlib/RingTheory/Nakayama.lean`) reach a stronger conclusion but require `N''.FG`
  and

  `I ≤ jacobson ⊥`. For a *nilpotent* `I` the iteration below terminates on its own,
  so

  neither hypothesis is needed — which matters here because the modules in the tangent-space

  application are sections over an affine chart, where finite generation is exactly
  the

  hypothesis one does not want to have to supply.'

theorem top_le_span_sup_pow_smul_top {I : Ideal R} {N : Submodule R M}
    (hN : N ⊔ I • (⊤ : Submodule R M) = ⊤) (n : ℕ) :
    (⊤ : Submodule R M) ≤ N ⊔ (I ^ n) • (⊤ : Submodule R M) := by
  induction n with
  | zero => simp
  | succ n ih =>
    have h₁ : (I ^ n) • (⊤ : Submodule R M)
        = (I ^ n) • N ⊔ (I ^ n) • (I • (⊤ : Submodule R M)) := by
      rw [← Submodule.smul_sup, hN]
    have h₂ : (I ^ n) • (I • (⊤ : Submodule R M)) = (I ^ (n + 1)) • (⊤ : Submodule R M) := by
      rw [← Submodule.smul_assoc, Ideal.smul_eq_mul, ← pow_succ]
    calc (⊤ : Submodule R M) ≤ N ⊔ (I ^ n) • (⊤ : Submodule R M) := ih
      _ = N ⊔ ((I ^ n) • N ⊔ (I ^ (n + 1)) • (⊤ : Submodule R M)) := by rw [h₁, h₂]
      _ ≤ N ⊔ (I ^ (n + 1)) • (⊤ : Submodule R M) :=
          sup_le le_sup_left
            (sup_le (le_trans Submodule.smul_le_right le_sup_left) le_sup_right)

end Submodule
