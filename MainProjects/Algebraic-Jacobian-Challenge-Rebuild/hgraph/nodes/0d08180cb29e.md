private lemma depth_eq_sSup_of_ne_top {R : Type u} [CommRing R] {I : Ideal R}
    {M : Type v} [AddCommGroup M] [Module R M]
    (h : I • (⊤ : Submodule R M) ≠ ⊤) :
    Module.depth I M = sSup { n : ℕ∞ | ∃ rs : List R, (rs.length : ℕ∞) = n ∧
      (∀ r ∈ rs, r ∈ I) ∧ RingTheory.Sequence.IsRegular M rs } := by
  unfold Module.depth
  rw [if_neg h]
