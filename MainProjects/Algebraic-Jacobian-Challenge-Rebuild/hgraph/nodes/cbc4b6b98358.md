noncomputable def pieceTensorIso {U : (XA).Opens} (hU : IsAffineOpen U) :
    CommRingCat.of (Γ(XA, U) ⊗[A] B) ≅ Γ(XB, (cg) ⁻¹ᵁ U) :=
  (CommRingCat.isPushout_tensorProduct A Γ(XA, U) B).isoIsPushout _ _
    (isPushout_cgAlgebraMap C hU)
