theorem mem_thetaPieceVanishing_iff_forall_germ (A : AffAdaptation D d) (a : ℕ)
    (j : D.index) (s : A.ThetaPieceSections (π := π) a j) :
    s ∈ A.thetaPieceVanishing (π := π) a j ↔
      ∀ (q : (thetaChartDatum C R π a).index) (z : relCurve C R)
        (hz : z ∈ D.pieces j ⊓ (thetaChartDatum C R π a).pieces q),
        ((relCurve C R).presheaf.germ
          (D.pieces j ⊓ (thetaChartDatum C R π a).pieces q) z hz).hom (s.val q) ∈
            d.stalkIdeal z := by
  let Mj := A.thetaPieceSectionsModel (π := π) a j
  letI : Scheme.QcohOn (thetaChartDatum C R π a).sheaf (D.pieces j) := Mj.qcoh
  letI : Module Γ(relCurve C R, D.pieces j)
      (A.ThetaPieceSections (π := π) a j) :=
    A.thetaPieceSectionsModule (π := π) a j
  constructor
  · intro hs q z hz
    rw [thetaPieceVanishing, Submodule.ideal_span_singleton_smul] at hs
    obtain ⟨t, -, ht⟩ :=
      (Submodule.mem_smul_pointwise_iff_exists s (A.eqn j) ⊤).mp hs
    have hq := congrArg
      (fun y : A.ThetaPieceSections (π := π) a j => y.val q) ht
    change (Scheme.QcohOn.qsmul (F := (thetaChartDatum C R π a).sheaf)
      (le_refl (D.pieces j)) (A.eqn j) t).val q = s.val q at hq
    rw [Mj.qsmul_eq, gluedQsmul_coe] at hq
    have hgerm := congrArg ((relCurve C R).presheaf.germ
      (D.pieces j ⊓ (thetaChartDatum C R π a).pieces q) z hz).hom hq
    rw [map_mul] at hgerm
    have hswap :
        ((relCurve C R).presheaf.germ
          (D.pieces j ⊓ (thetaChartDatum C R π a).pieces q) z hz).hom
            ((relCurve C R).resHom inf_le_left (A.eqn j)) =
          ((relCurve C R).presheaf.germ (D.pieces j) z hz.1).hom (A.eqn j) :=
      TopCat.Presheaf.germ_res_apply _ _ _ _ _
    rw [hswap] at hgerm
    rw [← A.germ_eqn_span_eq_stalkIdeal j hz.1, ← hgerm]
    exact Ideal.mul_mem_right _ _ (Ideal.subset_span rfl)
  · intro hs
    let c : ∀ q : (thetaChartDatum C R π a).index,
        Γ(relCurve C R, D.pieces j ⊓ (thetaChartDatum C R π a).pieces q) :=
      fun q => A.eqnDiv j inf_le_left (s.val q) (hs q)
    have hc (q : (thetaChartDatum C R π a).index) :
        (relCurve C R).resHom
            (inf_le_left : D.pieces j ⊓ (thetaChartDatum C R π a).pieces q ≤ D.pieces j)
            (A.eqn j) * c q = s.val q := by
      simpa only [c] using A.eqn_mul_eqnDiv j inf_le_left (s.val q) (hs q)
    have hc_mem : c ∈ AlgebraicGeometry.gluedSubmodule R
        (thetaChartDatum C R π a).pieces
        (thetaChartDatum C R π a).unit (D.pieces j) := by
      intro q r
      refine A.eqn_res_cancel j (inf_le_left.trans inf_le_left) ?_
      have hL := congrArg ((relCurve C R).resHom
        (inf_le_left : D.pieces j ⊓ (thetaChartDatum C R π a).pieces q ⊓
          (thetaChartDatum C R π a).pieces r ≤
            D.pieces j ⊓ (thetaChartDatum C R π a).pieces q)) (hc q)
      have hR := congrArg ((relCurve C R).resHom
        (gluedInclSnd (thetaChartDatum C R π a).pieces (D.pieces j) q r)) (hc r)
      rw [map_mul] at hL hR
      simp only [Scheme.resHom_resHom] at hL hR
      calc
        _ = (relCurve C R).resHom inf_le_left (s.val q) := hL
        _ = (relCurve C R).resHom
              (gluedInclCoc (thetaChartDatum C R π a).pieces (D.pieces j) q r)
              ((thetaChartDatum C R π a).unit q r :
                Γ(relCurve C R, (thetaChartDatum C R π a).pieces q ⊓
                  (thetaChartDatum C R π a).pieces r)) *
              (relCurve C R).resHom
                (gluedInclSnd (thetaChartDatum C R π a).pieces (D.pieces j) q r)
                (s.val r) := s.property q r
        _ = (relCurve C R).resHom
              (gluedInclCoc (thetaChartDatum C R π a).pieces (D.pieces j) q r)
              ((thetaChartDatum C R π a).unit q r :
                Γ(relCurve C R, (thetaChartDatum C R π a).pieces q ⊓
                  (thetaChartDatum C R π a).pieces r)) *
              ((relCurve C R).resHom (inf_le_left.trans inf_le_left) (A.eqn j) *
                (relCurve C R).resHom
                  (gluedInclSnd (thetaChartDatum C R π a).pieces (D.pieces j) q r)
                  (c r)) := by rw [hR]
        _ = (relCurve C R).resHom (inf_le_left.trans inf_le_left) (A.eqn j) *
              ((relCurve C R).resHom
                (gluedInclCoc (thetaChartDatum C R π a).pieces (D.pieces j) q r)
                ((thetaChartDatum C R π a).unit q r :
                  Γ(relCurve C R, (thetaChartDatum C R π a).pieces q ⊓
                    (thetaChartDatum C R π a).pieces r)) *
                (relCurve C R).resHom
                  (gluedInclSnd (thetaChartDatum C R π a).pieces (D.pieces j) q r)
                  (c r)) := by ring
    let t : A.ThetaPieceSections (π := π) a j := ⟨c, hc_mem⟩
    rw [thetaPieceVanishing, Submodule.ideal_span_singleton_smul]
    apply (Submodule.mem_smul_pointwise_iff_exists s (A.eqn j) ⊤).mpr
    refine ⟨t, Submodule.mem_top, ?_⟩
    apply Subtype.ext
    funext q
    change (Scheme.QcohOn.qsmul (F := (thetaChartDatum C R π a).sheaf)
      (le_refl (D.pieces j)) (A.eqn j) t).val q = s.val q
    rw [Mj.qsmul_eq, gluedQsmul_coe]
    exact hc q
