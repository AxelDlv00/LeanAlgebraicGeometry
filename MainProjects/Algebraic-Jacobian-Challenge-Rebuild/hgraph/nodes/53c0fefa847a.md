theorem probeWrongDirection {X : Scheme.{u}} (V : X.Opens)
    (r : X ⟶ (V : Scheme.{u})) (hr : V.ι ≫ r = 𝟙 _) :
    Function.Surjective (r.base) := by
  intro v
  exact ⟨(V.ι).base v, by
    have := congrArg (fun m => Scheme.Hom.base m) hr
    exact congrFun (congrArg (fun g => g) (by rw [← Scheme.comp_base]; exact
      congrArg Scheme.Hom.base hr)) v⟩
