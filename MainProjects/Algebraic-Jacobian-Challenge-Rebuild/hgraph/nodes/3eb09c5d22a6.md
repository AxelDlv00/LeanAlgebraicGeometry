def chartLocus (m : ℕ) (Z : (C ⊗ overSpec k k).left.CurveDivisor)
    {T : Over (Spec (.of k))} (lam : picEt C T) : Set T.left :=
  {t : T.left | IsSplitWitness C
    (picEtMap C (Over.testPoint t) (chartTwist C m Z T lam))}
