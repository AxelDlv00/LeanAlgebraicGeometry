lemma Scheme.CurveDivisor.le_iff_coeffAt {D D' : X.CurveDivisor} :
    D ≤ D' ↔ ∀ (x : X) (hx : x ≠ genericPoint X), coeffAt hx D ≤ coeffAt hx D' := by
  constructor
  · intro h x hx
    exact Finsupp.le_def.mp h ⟨x, hx⟩
  · intro h
    exact Finsupp.le_def.mpr (fun p => h p.1 p.2)
