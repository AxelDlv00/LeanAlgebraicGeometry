theorem toLaurent_posPoly_apply (z : LaurentPolynomial A) (j : ℤ) :
    (Polynomial.toLaurent (posPoly z)) j = if 0 ≤ j then z j else 0 := by
  classical
  unfold posPoly
  rw [map_sum]
  have hterm : ∀ i ∈ z.support.filter (0 ≤ ·),
      Polynomial.toLaurent (Polynomial.monomial i.toNat (z i))
        = (Finsupp.single i (z i) : LaurentPolynomial A) := by
    intro i hi
    rw [Finset.mem_filter] at hi
    rw [Polynomial.toLaurent_C_mul_T, ← LaurentPolynomial.single_eq_C_mul_T,
      Int.toNat_of_nonneg hi.2]
  rw [Finset.sum_congr rfl hterm]
  change (Finset.sum (z.support.filter (0 ≤ ·))
    (fun i => (Finsupp.single i (z i) : ℤ →₀ A)) : ℤ →₀ A) j = _
  rw [Finset.sum_apply']
  simp only [Finsupp.single_apply]
  by_cases hj : 0 ≤ j
  · rw [if_pos hj]
    by_cases hmem : j ∈ z.support.filter (0 ≤ ·)
    · rw [Finset.sum_eq_single j]
      · rw [if_pos rfl]
      · intro b _ hbj; rw [if_neg hbj]
      · intro h; exact absurd hmem h
    · rw [Finset.sum_eq_zero]
      · rw [Finset.mem_filter, not_and] at hmem
        by_cases hjs : j ∈ z.support
        · exact absurd hj (hmem hjs)
        · rw [Finsupp.notMem_support_iff] at hjs; rw [hjs]
      · intro b hb; rw [if_neg]; rintro rfl; exact hmem hb
  · rw [if_neg hj, Finset.sum_eq_zero]
    intro b hb
    rw [Finset.mem_filter] at hb
    rw [if_neg]; rintro rfl; exact hj hb.2
