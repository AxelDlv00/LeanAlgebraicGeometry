theorem subsingleton_chartIdealColengthModule_tensor_residueField_of_ideal_span
    (K : Submodule R (relThetaSections C R π a)) (b : Bool)
    (s : relThetaSections C R π a) (hsK : s ∈ K)
    (p : PrimeSpectrum Γ(relCurve C R, relPinnedChart C R π b))
    (hspan : ∀ v : p.asIdeal.ResidueField ⊗[
        Γ(relCurve C R, relPinnedChart C R π b)]
        (chartReadIdeal K b), ∃ c : p.asIdeal.ResidueField,
          c • ((1 : p.asIdeal.ResidueField) ⊗ₜ[
            Γ(relCurve C R, relPinnedChart C R π b)]
            (⟨relThetaResSide a b (le_rfl) s,
              by
                exact Ideal.subset_span ⟨⟨s, hsK⟩, rfl⟩⟩ :
              chartReadIdeal K b)) = v) :
    Subsingleton ((chartIdealColengthModule K b s) ⊗[
      Γ(relCurve C R, relPinnedChart C R π b)] p.asIdeal.ResidueField) := by
  let B : Type u := Γ(relCurve C R, relPinnedChart C R π b)
  let I : Ideal B := Ideal.span {relThetaResSide a b le_rfl s}
  let P : Submodule B B := chartReadIdeal K b
  let f : B →ₗ[B] (B ⧸ I) := (Ideal.Quotient.mkₐ B I).toLinearMap
  have hsP : relThetaResSide a b le_rfl s ∈ P := by
    exact Ideal.subset_span ⟨⟨s, hsK⟩, rfl⟩
  have hfs : f (relThetaResSide a b le_rfl s) = 0 := by
    change Ideal.Quotient.mk I (relThetaResSide a b le_rfl s) = 0
    exact Ideal.Quotient.eq_zero_iff_mem.mpr (Ideal.subset_span (by simp))
  have hzero := RankOneFibre.subsingleton_map_tensor_of_span_singleton
    P f (relThetaResSide a b le_rfl s) hsP (by
      intro v
      exact hspan v) hfs
  change Subsingleton ((P.map f) ⊗[B] p.asIdeal.ResidueField) at hzero
  exact hzero
