noncomputable def pointCover (d : PointUniformizerData K hx) : X.PointedCover where
  opens z := if z = x then d.opens else pointCompl K hx
  mem_opens z := by
    by_cases h : z = x
    · rw [if_pos h]; subst h; exact d.mem
    · rw [if_neg h]; exact h

@[simp]
