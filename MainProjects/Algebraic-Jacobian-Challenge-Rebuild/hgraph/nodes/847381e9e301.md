def openSubfunctorMap (D : Over (Spec (.of k))) (U : D.left.Opens) :
    yoneda.obj (Over.mk (U.ι ≫ D.hom)) ⟶ yoneda.obj D :=
  yoneda.map (Over.homMk (U := Over.mk (U.ι ≫ D.hom)) U.ι)
