theorem probeIsIso {X : Scheme.{u}} [IsReduced X] [X.IsSeparated] (V : X.Opens)
    [IsDominant (V.ι)]
    (r : X ⟶ (V : Scheme.{u})) (hr : V.ι ≫ r = 𝟙 _) :
    IsIso (V.ι) := by
  refine ⟨r, hr, ?_⟩
  -- r >> V.i = id_X : the two agree after precomposing with the dominant V.i
  refine ext_of_isDominant (X := X) (Y := X) (W := (V : Scheme.{u})) (V.ι) ?_
  rw [← Category.assoc, hr, Category.id_comp, Category.comp_id]
