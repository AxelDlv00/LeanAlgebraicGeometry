theorem cartierIdeal_ideal_eq_sectionIdeal_affine [IsProper C.hom]
    (A : AffAdaptation D d) (U : (relCurve C R).affineOpens) :
    A.cartierIdeal.ideal U = d.sectionIdeal U.1 := by
  apply le_antisymm
  · intro s hs z hz
    obtain ⟨j, hzj⟩ := D.exists_mem_pieces z
    let W : (relCurve C R).affineOpens :=
      ⟨U.1 ⊓ D.pieces j, Over.isAffineOpen_inf C U.2 (D.isAffineOpen j)⟩
    have hsW :
        ((relCurve C R).presheaf.map
          (homOfLE (inf_le_left : W.1 ≤ U.1)).op).hom s ∈
          A.cartierIdeal.ideal W := by
      rw [← A.cartierIdeal.map_ideal (U := W) (V := U)
        (show W.1 ≤ U.1 from inf_le_left)]
      exact Ideal.mem_map_of_mem _ hs
    rw [A.cartierIdeal_ideal_eq_sectionIdeal_of_le j W inf_le_right] at hsW
    have hzW : z ∈ W.1 := ⟨hz, hzj⟩
    have := hsW z hzW
    rw [TopCat.Presheaf.germ_res_apply] at this
    exact this
  · have hlocal : ∀ x : U.1, ∃ (j : D.index) (r : Γ(relCurve C R, U.1))
        (rj : Γ(relCurve C R, D.pieces j)),
        (relCurve C R).basicOpen r = (relCurve C R).basicOpen rj ∧
          x.1 ∈ (relCurve C R).basicOpen r := by
      intro ⟨x, hxU⟩
      obtain ⟨j, hxj⟩ := D.exists_mem_pieces x
      exact ⟨j, exists_basicOpen_le_affine_inter U.2 (D.isAffineOpen j) x ⟨hxU, hxj⟩⟩
    choose j r rj hr hx using hlocal
    have hspan : Ideal.span (Set.range r) = ⊤ := by
      rw [← U.2.self_le_iSup_basicOpen_iff]
      exact fun x hxU ↦
        Opens.mem_iSup.mpr ⟨⟨_, ⟨x, hxU⟩, rfl⟩, hx ⟨x, hxU⟩⟩
    have inst := U.2.isLocalization_basicOpen
    refine Submodule.le_of_isLocalized_span _ hspan
      (fun q ↦ Γ(relCurve C R, (relCurve C R).basicOpen q.1))
      (fun q ↦ Algebra.linearMap Γ(relCurve C R, U.1)
        Γ(relCurve C R, (relCurve C R).basicOpen q.1)) ?_
    rintro ⟨_, x, rfl⟩
    simp +instances only [← Submodule.restrictScalars_localized'
      Γ(relCurve C R, (relCurve C R).basicOpen (r x)),
      Ideal.localized'_eq_map, RingHom.algebraMap_toAlgebra]
    erw [A.cartierIdeal.map_ideal
      (U := ⟨_, U.2.basicOpen (r x)⟩) ((relCurve C R).basicOpen_le (r x))]
    have hpiece : (relCurve C R).basicOpen (r x) ≤ D.pieces (j x) := by
      rw [hr x]
      exact (relCurve C R).basicOpen_le (rj x)
    rw [A.cartierIdeal_ideal_eq_sectionIdeal_of_le (j x)
      ⟨_, U.2.basicOpen (r x)⟩ hpiece]
    exact d.map_sectionIdeal_le ((relCurve C R).basicOpen_le (r x))
