theorem degAt_inv (lam : picEt C T) (t : overSpec k K ⟶ T) :
    degAt lam⁻¹ t = -degAt lam t := by
  change PicEtAff.degAff K (picEtAffineEquiv C K (picEtMap C t lam⁻¹)) = -degAt lam t
  rw [map_inv, map_inv, PicEtAff.degAff_inv]
  rfl
