theorem Over.isPullback_left (X T : Over S) :
    IsPullback (fst X T).left (snd X T).left X.hom T.hom := by
  simp only [Over.fst_left, Over.snd_left]
  exact IsPullback.of_hasPullback X.hom T.hom

end Bridge
