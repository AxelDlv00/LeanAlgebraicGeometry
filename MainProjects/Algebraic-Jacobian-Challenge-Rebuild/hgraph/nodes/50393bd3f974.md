private noncomputable def modelHomOverlap (c : ι → N) : (ι → R[T;T⁻¹]) →ₗ[R] N :=
  ∑ i : ι, LinearMap.applyₗ (c i) ∘ₗ P.laurentToEnd.toLinearMap ∘ₗ LinearMap.proj i
