structure SemilinearGalAction (X : Scheme.{u})
    (f : X ⟶ Spec (CommRingCat.of L)) where
  act : (L ≃ₐ[K] L) →* Aut X
  compat (gamma : L ≃ₐ[K] L) :
    (act gamma).hom ≫ f =
      f ≫ (toSpecAut (L ≃ₐ[K] L) L gamma).hom
