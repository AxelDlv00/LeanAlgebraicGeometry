lemma classDeg_pow (L : X.CechPic) (n : ℕ) :
    classDeg K (L ^ n) = n * classDeg K L := by
  induction n with
  | zero => rw [pow_zero, classDeg_one, Nat.cast_zero, zero_mul]
  | succ n ih => rw [pow_succ, classDeg_mul, ih, Nat.cast_succ]; ring

end ClassArith
