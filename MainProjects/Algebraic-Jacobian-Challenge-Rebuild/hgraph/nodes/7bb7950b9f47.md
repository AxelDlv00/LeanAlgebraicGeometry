noncomputable def Scheme.annKernel (U V : Submodule K X.functionField)
    (Λ : Module.Dual K X.functionField) : Submodule K X.functionField where
  carrier := {h | h ∈ U ∧ ∀ f ∈ V, Λ (h * f) = 0}
  zero_mem' := ⟨U.zero_mem, fun f _ => by rw [zero_mul, map_zero]⟩
  add_mem' := fun {a b} ha hb =>
    ⟨U.add_mem ha.1 hb.1, fun f hf => by
      rw [add_mul, map_add, ha.2 f hf, hb.2 f hf, add_zero]⟩
  smul_mem' := fun r a ha =>
    ⟨U.smul_mem r ha.1, fun f hf => by
      rw [Scheme.functionFieldOverModule_smul_def, mul_assoc,
        ← Scheme.functionFieldOverModule_smul_def, map_smul, ha.2 f hf, smul_zero]⟩
