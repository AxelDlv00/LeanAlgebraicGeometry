lemma sigmaExtension_map_mk {T' T : C} (g : T' ⟶ T) (a : T ⟶ S)
    (ξ : F.obj (op (Over.mk a))) :
    (sigmaExtension S F).map g.op ⟨a, ξ⟩
      = ⟨g ≫ a, F.map (Over.homMk g rfl : Over.mk (g ≫ a) ⟶ Over.mk a).op ξ⟩ :=
  rfl
