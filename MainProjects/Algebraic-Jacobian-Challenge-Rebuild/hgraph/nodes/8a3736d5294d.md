lemma Scheme.mul_mem_mulSpan {U T : Submodule K X.functionField}
    {h f : X.functionField} (hh : h ∈ U) (hf : f ∈ T) :
    h * f ∈ Scheme.mulSpan K U T :=
  Submodule.subset_span ⟨h, hh, f, hf, rfl⟩
