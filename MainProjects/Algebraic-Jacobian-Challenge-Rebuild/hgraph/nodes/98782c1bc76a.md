theorem pic0SigmaFunctor_isSheafFor_etaleCover
    {k A : Type u} [Field k] [CommRing A]
    (C : Over (Spec (.of k))) [SmoothOfRelativeDimension 1 C.hom]
    [IsProper C.hom] [GeometricallyIrreducible C.hom]
    [GeometricallyReduced C.hom]
    (E : Algebra.EtaleCover A) :
    Presieve.IsSheafFor (pic0SigmaFunctor C)
      (.singleton (Spec.map
        (CommRingCat.ofHom (algebraMap A E.Carrier)))) := by
  rw [Presieve.isSheafFor_singleton]
  rintro ⟨b, x⟩ hx
  let g : Spec (.of E.Carrier) ⟶ Spec (.of A) :=
    Spec.map (CommRingCat.ofHom (algebraMap A E.Carrier))
  letI : Surjective g :=
    ⟨E.comap_surjective⟩
  haveI : Etale g := by
    dsimp [g]
    rw [HasRingHomProperty.Spec_iff (P := @Etale)]
    exact RingHom.etale_algebraMap.mpr inferInstance
  have hbase : ∀ {Z : Scheme.{u}} (p₁ p₂ : Z ⟶ Spec (.of E.Carrier)),
      p₁ ≫ g = p₂ ≫ g → p₁ ≫ b = p₂ ≫ b := by
    intro Z p₁ p₂ hp
    exact congrArg Sigma.fst (hx p₁ p₂ hp)
  let a : Spec (.of A) ⟶ Spec (.of k) :=
    EffectiveEpi.desc g b hbase
  have ha : g ≫ a = b := EffectiveEpi.fac g b hbase
  obtain ⟨φ, hφ⟩ := Spec.map_surjective a
  algebraize [φ.hom]
  have haφ : g ≫ Spec.map φ = b := by
    rw [hφ]
    exact ha
  have hE : (overSpec k E.Carrier).hom = b := by
    change Spec.map (CommRingCat.ofHom (algebraMap k E.Carrier)) = b
    rw [IsScalarTower.algebraMap_eq k A E.Carrier]
    change Spec.map (φ ≫ CommRingCat.ofHom (algebraMap A E.Carrier)) = b
    rw [Spec.map_comp]
    exact haφ
  let eIso : overSpec k E.Carrier ≅ Over.mk b :=
    Over.isoMk (Iso.refl _) hE.symm
  let e : overSpec k E.Carrier ⟶ Over.mk b := eIso.hom
  let xE : pic0Subgroup C (overSpec k E.Carrier) := pic0Map C e x
  let d₀ : overSpec k (E.Carrier ⊗[A] E.Carrier) ⟶ overSpec k E.Carrier :=
    Over.overSpecMap (doubleInl (k := k) E)
  let d₁ : overSpec k (E.Carrier ⊗[A] E.Carrier) ⟶ overSpec k E.Carrier :=
    Over.overSpecMap (doubleInr (k := k) E)
  have hlegs : d₀.left ≫ g = d₁.left ≫ g := by
    exact etaleCover_doubleProjection_comp_base E
  have hd₀e : (d₀ ≫ e).left = d₀.left := by
    simp only [Over.comp_left, e, eIso, Over.isoMk_hom_left, Iso.refl_hom]
    exact Category.comp_id _
  have hd₁e : (d₁ ≫ e).left = d₁.left := by
    simp only [Over.comp_left, e, eIso, Over.isoMk_hom_left, Iso.refl_hom]
    exact Category.comp_id _
  have hSigma :
      (pic0SigmaFunctor C).map (d₀ ≫ e).left.op ⟨b, x⟩ =
        (pic0SigmaFunctor C).map (d₁ ≫ e).left.op ⟨b, x⟩ := by
    rw [hd₀e, hd₁e]
    exact hx d₀.left d₁.left hlegs
  have hFib := Over.map_eq_of_sigmaExtension_map_eq
    (F := pic0TypeFunctor C) (d₀ ≫ e) (d₁ ≫ e) hSigma
  have hdouble :
      pic0Map C d₀ xE = pic0Map C d₁ xE := by
    change (pic0TypeFunctor C).map d₀.op
        ((pic0TypeFunctor C).map e.op x) =
      (pic0TypeFunctor C).map d₁.op
        ((pic0TypeFunctor C).map e.op x)
    rw [← FunctorToTypes.map_comp_apply, ← FunctorToTypes.map_comp_apply]
    exact hFib
  obtain ⟨z, hz, hzu⟩ :=
    pic0Subgroup_existsUnique_of_etaleCover C E xE hdouble
  let d : overSpec k E.Carrier ⟶ overSpec k A :=
    Over.overSpecMap ((Algebra.ofId A E.Carrier).restrictScalars k)
  let q : Over.mk b ⟶ overSpec k A := Over.homMk g haφ
  have heq : e ≫ q = d := by
    apply Over.OverMorphism.ext
    rfl
  have hrestrict (t : pic0Subgroup C (overSpec k A)) :
      pic0Map C e (pic0Map C q t) = pic0Map C d t := by
    change (pic0TypeFunctor C).map e.op
        ((pic0TypeFunctor C).map q.op t) =
      (pic0TypeFunctor C).map d.op t
    rw [← FunctorToTypes.map_comp_apply, ← op_comp, heq]
  have hq : pic0Map C q z = x := by
    apply (ConcreteCategory.bijective_of_isIso
      ((pic0TypeFunctor C).map e.op)).1
    change (pic0TypeFunctor C).map e.op
        ((pic0TypeFunctor C).map q.op z) =
      (pic0TypeFunctor C).map e.op x
    rw [← FunctorToTypes.map_comp_apply, ← op_comp, heq]
    change pic0Map C d z = xE
    exact hz
  refine ⟨⟨Spec.map φ, z⟩, ?_, ?_⟩
  · refine (Over.sigmaExtension_map_mk_eq_iff
      (F := pic0TypeFunctor C) g haφ z x).mpr ?_
    rw [pic0TypeFunctor_map_apply]
    exact hq
  · rintro ⟨a', z'⟩ hz'
    have ha' : a' = Spec.map φ := by
      exact (EffectiveEpi.uniq g b hbase a'
        (congrArg Sigma.fst hz')).trans hφ.symm
    subst a'
    change pic0Subgroup C (overSpec k A) at z'
    refine congrArg (Sigma.mk (Spec.map φ)) (hzu z' ?_)
    have hq' : pic0Map C q z' = x := by
      have hmem := (Over.sigmaExtension_map_mk_eq_iff
        (F := pic0TypeFunctor C) g haφ z' x).mp hz'
      rw [pic0TypeFunctor_map_apply] at hmem
      exact hmem
    calc
      pic0Map C d z' = pic0Map C e (pic0Map C q z') := (hrestrict z').symm
      _ = pic0Map C e x := congrArg (pic0Map C e) hq'
      _ = xE := rfl
