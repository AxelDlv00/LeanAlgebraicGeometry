noncomputable def uniqueUpToIsoOfIso
    {C : Type u} [Category.{v, u} C]
    {F F' : Cᵒᵖ ⥤ Type v} {Y Y' : C}
    (e : F.RepresentableBy Y) (e' : F'.RepresentableBy Y') (η : F ≅ F') : Y ≅ Y' :=
  (e.ofIso η).uniqueUpToIso e'
