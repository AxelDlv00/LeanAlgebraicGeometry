noncomputable def comparison (D : DescentDatum A B R) :
    B ⊗[A] D.descended →ₐ[B] R :=
  Algebra.TensorProduct.lift (Algebra.ofId B R) D.descended.val
    (fun _ _ => Commute.all _ _)
