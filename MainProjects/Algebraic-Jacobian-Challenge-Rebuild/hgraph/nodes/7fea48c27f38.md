theorem substGradedHom_mul (M N : Matrix σ σ R) :
    substGradedHom (M * N) = (substGradedHom N).comp (substGradedHom M) :=
  GradedRingHom.ext fun p => DFunLike.congr_fun (substAlgHom_mul M N) p
