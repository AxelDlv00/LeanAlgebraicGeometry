    ((Algebra.TensorProduct.lmul' A (S := B)).restrictScalars k).comp
        (tensorInl (A := A) (B := B))
      = AlgHom.id k B := by

  change Algebra.TensorProduct.lmul' A (b ⊗ₜ[A] 1) = b
