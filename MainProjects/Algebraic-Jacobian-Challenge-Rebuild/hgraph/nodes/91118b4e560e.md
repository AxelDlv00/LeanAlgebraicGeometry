    Γ(Y, extOpens w Ω' 𝒰₀ y ⊓ extOpens w Ω' 𝒰₀ y')ˣ :=
  if hy : y ∈ w.opensRange then

        (inf_le_left.trans (extOpens_le_opensRange w Ω' 𝒰₀ hy))).symm

              (le_of_eq (preimage_extOpens_of_mem w Ω' 𝒰₀ hy'))))
