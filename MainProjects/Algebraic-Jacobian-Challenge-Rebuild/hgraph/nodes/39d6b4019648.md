noncomputable def quotientMapOfLE {L P : Submodule R M} (hLP : L ≤ P) :
    (M ⧸ L) →ₗ[R] (M ⧸ P) :=
  L.liftQ P.mkQ fun x hx => by
    rw [LinearMap.mem_ker, Submodule.mkQ_apply, Submodule.Quotient.mk_eq_zero]
    exact hLP hx
