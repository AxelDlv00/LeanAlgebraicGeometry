lemma algebraMap_eq_aeval (p : Polynomial k) :
    algebraMap (Polynomial k) B p = aeval (coord k B) p := by
  have h : (IsScalarTower.toAlgHom k (Polynomial k) B) = aeval (coord k B) := by
    apply Polynomial.algHom_ext; simp [coord]
  simpa [IsScalarTower.toAlgHom_apply] using DFunLike.congr_fun h p
