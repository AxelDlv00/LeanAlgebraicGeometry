lemma Hom.le_preimage_inf (f : X.Hom Y) {U : X.Opens} {V₁ V₂ : Y.Opens}
    (h₁ : U ≤ f ⁻¹ᵁ V₁) (h₂ : U ≤ f ⁻¹ᵁ V₂) : U ≤ f ⁻¹ᵁ (V₁ ⊓ V₂) :=
  fun _ hx ↦ ⟨h₁ hx, h₂ hx⟩
