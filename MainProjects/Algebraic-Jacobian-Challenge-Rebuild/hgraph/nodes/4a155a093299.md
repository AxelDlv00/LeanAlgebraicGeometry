lemma exists_idempotent_lift :
    ∃ elift : B ⊗[k] B, IsIdempotentElem (baseChange elift) ∧
      RingHom.ker (Algebra.TensorProduct.lmul' (R := Polynomial k) (S := B))
        = Ideal.span {baseChange elift} := by
  have hfg : (KaehlerDifferential.ideal (Polynomial k) B).FG := KaehlerDifferential.ideal_fg _ _
  have hidem : IsIdempotentElem (KaehlerDifferential.ideal (Polynomial k) B) :=
    (Ideal.cotangent_subsingleton_iff _).mp <|
      inferInstanceAs <| Subsingleton Ω[B⁄Polynomial k]
  obtain ⟨e, he, hsp⟩ := (Ideal.isIdempotentElem_iff_of_fg _ hfg).mp hidem
  obtain ⟨elift, rfl⟩ := baseChange_surjective e
  refine ⟨elift, he, ?_⟩
  simpa [Ideal.span] using hsp

end Unramified
