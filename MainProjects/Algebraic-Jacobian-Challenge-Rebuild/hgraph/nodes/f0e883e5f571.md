noncomputable def thetaChartIndex (z : relCurve C k) : (thetaChartCover C k π).index :=
  if ((fst C (overSpec k k)).left).base z ∈ fiberChart₀ π then Sum.inl PUnit.unit
  else Sum.inr PUnit.unit
