noncomputable def directedQuotientMapOfCompatible (i j : ι) (hij : i ≤ j) :
    (G i ⧸ K i) →ₗ[R] (G j ⧸ K j) :=
  (K i).mapQ (K j) (f i j hij) (by
    intro x hx
    exact hK i j hij ⟨x, hx, rfl⟩)
