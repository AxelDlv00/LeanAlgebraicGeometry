theorem ker_fstHom_eq_span_eps :
    RingHom.ker (TrivSqZeroExt.fstHom A A A) = Ideal.span {(ε : DualNumber A)} := by
  refine le_antisymm (fun x hx => ?_) (Ideal.span_le.mpr ?_)
  · have hx' : x.fst = 0 := hx
    refine Ideal.mem_span_singleton.mpr ⟨TrivSqZeroExt.inl x.snd, ?_⟩
    refine TrivSqZeroExt.ext ?_ ?_
    · simp [TrivSqZeroExt.fst_mul, hx']
    · simp [TrivSqZeroExt.snd_mul]
  · intro y hy
    rw [Set.mem_singleton_iff] at hy
    subst hy
    change (ε : DualNumber A).fst = 0
    simp
