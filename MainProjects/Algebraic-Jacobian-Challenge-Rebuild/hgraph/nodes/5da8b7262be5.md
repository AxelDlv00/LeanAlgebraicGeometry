noncomputable def kerBaseChangeEquivPulledGlued
    (R' : Type u) [CommRing R'] [Algebra k R'] [Algebra R R'] [IsScalarTower k R R'] :
    LinearMap.ker ((A.deltaLeft - A.deltaRight).baseChange R') ≃ₗ[R']
      ↥(A.pulledGluedSubmodule R') := by
  let e := A.chartProdBaseChange R'
  refine LinearEquiv.ofSubmodules e _ _ ?_
  have hmap : (A.deltaLeft - A.deltaRight).baseChange R' =
      AlgebraTensorModule.lTensor R' R' (A.deltaLeft - A.deltaRight) := by
    ext x
    exact congrFun (LinearMap.baseChange_eq_ltensor (A.deltaLeft - A.deltaRight)) x
  rw [hmap]
  exact A.map_ker_lTensor_delta R'
