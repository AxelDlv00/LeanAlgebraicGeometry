lemma Scheme.CurveDivisor.sub_sup (C D₁ D₂ : X.CurveDivisor) :
    C - (D₁ ⊔ D₂) = (C - D₁) ⊓ (C - D₂) := by
  refine CurveDivisor.ext_coeffAt (fun x hx => ?_)

    Scheme.CurveDivisor.coeffAt_inf, CurveDivisor.coeffAt_sub, CurveDivisor.coeffAt_sub]
  omega
