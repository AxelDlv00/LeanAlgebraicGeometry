theorem descentEquiv_naturality (f : Hom D₁ D₂) [Module.Flat A B]
    (x : B ⊗[A] D₁.descended) :
    D₂.descentEquiv
        (Algebra.TensorProduct.map (AlgHom.id A B) f.descendedMap x) =
      f.hom (D₁.descentEquiv x) :=
  f.comparison_naturality x
