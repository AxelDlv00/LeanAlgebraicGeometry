class HasStableAffineCover {X : Scheme.{u}}
    {f : X ⟶ Spec (CommRingCat.of L)}
    (rho : SemilinearGalAction K L X f) : Prop where
  exists_stable_affineOpen (x : X) :
    ∃ U : X.Opens, IsAffineOpen U ∧ x ∈ U ∧ rho.IsStableOpen U
