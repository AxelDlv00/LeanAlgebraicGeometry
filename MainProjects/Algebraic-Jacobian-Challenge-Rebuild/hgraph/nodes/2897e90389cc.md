theorem isUnit_germ_at_eq_of_isUnit {X : Scheme.{u}} {U : X.Opens}
    {y z : X} (hy : y ∈ U) (hz : z ∈ U) (h : y = z) {s : Γ(X, U)}
    (hu : IsUnit ((X.presheaf.germ U y hy).hom s)) :
    IsUnit ((X.presheaf.germ U z hz).hom s) := by
  subst z
  exact hu
