theorem map_twoChartClassHom (f : X ⟶ Y) (sel : Y → Bool) (hmem : ∀ y, y ∈ V (sel y))
    (u : Γ(Y, V false ⊓ V true)ˣ) :

      = twoChartClassHom (fun s ↦ f ⁻¹ᵁ V s) (fun x ↦ sel (f.base x))
