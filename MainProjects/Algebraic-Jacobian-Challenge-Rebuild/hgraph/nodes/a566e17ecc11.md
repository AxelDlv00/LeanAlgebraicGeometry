theorem exists_refines_prod (E E' : EtaleCover A) :
    ∃ E'' : EtaleCover A, E''.Refines E ∧ E''.Refines E' :=
  ⟨E.prod E', E.prod_refines_left E', E.prod_refines_right E'⟩
