noncomputable def rowFst (pt : Spec (.of kbar) ⟶ X.left) (hpt : pt ≫ X.hom = 𝟙 _) :
    X.left ⟶ pullback X.hom X.hom :=
  pullback.lift (X.hom ≫ pt) (𝟙 _)
    (by rw [Category.id_comp, Category.assoc, hpt, Category.comp_id])
