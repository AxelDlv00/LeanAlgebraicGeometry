lemma comm_t₁_pow (f : P.Hom P') (m : ℕ) (y : M₁) :
    f.hom₁ ((P.t₁ ^ m) y) = (P'.t₁ ^ m) (f.hom₁ y) := by
  induction m with
  | zero => simp
  | succ m ih =>
    rw [pow_succ', pow_succ']
    simp only [Module.End.mul_apply]
    rw [f.comm_t₁, ih]
