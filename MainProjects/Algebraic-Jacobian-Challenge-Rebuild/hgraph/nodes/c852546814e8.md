theorem span_range_read_eq_of_surjective_mul_of_unit
    (m : S →ₗ[R] B) (r : K →ₗ[R] B) (r' : K' →ₗ[R] B)
    (μ : S ⊗[R] K →ₗ[R] K')
    (hsurj : Function.Surjective μ)
    (hread : ∀ x, r' (μ x) = productReadMap (R := R) m r x)
    (s₁ : S) (hmunit : IsUnit (m s₁))
    (ι : K → K')
    (hι : ∀ k, μ (s₁ ⊗ₜ[R] k) = ι k) :
    Ideal.span (Set.range r') = Ideal.span (Set.range r) := by
  apply le_antisymm
  · exact span_range_read_le_of_surjective_mul (R := R) m r r' μ hsurj hread
  · refine Ideal.span_le.mpr ?_
    rintro _ ⟨k, rfl⟩
    have hprod : r' (ι k) = m s₁ * r k := by
      rw [← hι k, hread, productReadMap_tmul]
    have hmem : m s₁ * r k ∈ Ideal.span (Set.range r') := by
      rw [← hprod]
      exact Ideal.subset_span ⟨ι k, rfl⟩
    obtain ⟨u, hu⟩ := hmunit
    rw [← hu] at hmem
    have hinv := Ideal.mul_mem_left (Ideal.span (Set.range r')) ((u⁻¹ : Bˣ) : B) hmem
    change r k ∈ Ideal.span (Set.range r')
    simpa only [← mul_assoc, Units.inv_mul, Units.val_one, one_mul] using hinv
