lemma telescope_mul_div {G : Type u} [CommGroup G] (T₂₃ T₁₂ T₁₃ M₁ M₂ M₃ : G) :
    T₂₃ * (M₃ / M₂) * (T₁₂ * (M₂ / M₁)) / (T₁₃ * (M₃ / M₁)) = T₂₃ * T₁₂ / T₁₃ := by
  rw [mul_mul_mul_comm, mul_div_mul_comm,
    show M₃ / M₂ * (M₂ / M₁) / (M₃ / M₁) = 1 by rw [div_mul_div_cancel, div_self'],
    mul_one]
