theorem exists_notMem_mul_mem_span_singleton_of_fibre_of_flat_quotient_of_isLocalization
    (q : PrimeSpectrum B) [IsLocalization.AtPrime L q.asIdeal]
    (J : Ideal B) (hJfg : J.FG) [Module.Flat R (B ⧸ J)]
    (e : B) (he : e ∈ J)
    (hfibre : ∀ x : J, ∃ r : B, r ∉ q.asIdeal ∧
      r * (x : B) ∈
        Ideal.span ({e} : Set B) ⊔
          Ideal.map (algebraMap R B)
            (Ideal.comap (algebraMap R B) q.asIdeal)) :
    ∀ x : J, ∃ r : B, r ∉ q.asIdeal ∧
      r * (x : B) ∈ Ideal.span ({e} : Set B) := by
  letI : IsLocalRing L :=
    IsLocalization.AtPrime.isLocalRing L q.asIdeal
  let p : Ideal R := Ideal.comap (algebraMap R B) q.asIdeal
  let JL : Ideal L := Ideal.map (algebraMap B L) J
  let E : Ideal L := Ideal.span ({algebraMap B L e} : Set L)
  let I : Ideal L := Ideal.map (algebraMap R L) p
  have hImax : I ≤ IsLocalRing.maximalIdeal L := by
    change Ideal.map (algebraMap R L) p ≤ IsLocalRing.maximalIdeal L
    rw [Ideal.map_le_iff_le_comap]
    intro r hr
    change algebraMap R L r ∈ IsLocalRing.maximalIdeal L
    rw [IsScalarTower.algebraMap_apply R B L]
    exact (IsLocalization.AtPrime.to_map_mem_maximal_iff
      L q.asIdeal (algebraMap R B r)).2 hr
  have hIjac : I ≤ (⊥ : Ideal L).jacobson :=
    hImax.trans (IsLocalRing.maximalIdeal_le_jacobson ⊥)
  have hlocal : JL ≤ E ⊔ I * JL := by
    change Ideal.map (algebraMap B L) J ≤ E ⊔ I * JL
    rw [Ideal.map_le_iff_le_comap]
    intro x hx
    obtain ⟨r, hr, hrx⟩ := hfibre ⟨x, hx⟩
    obtain ⟨a, ha, b, hb, hab⟩ := Submodule.mem_sup.mp hrx
    have haJ : a ∈ J :=
      (Ideal.span_le.mpr (Set.singleton_subset_iff.mpr he)) ha
    have hbJ : b ∈ J := by
      have hbeq : b = r * x - a := by rw [← hab]; ring
      rw [hbeq]
      exact J.sub_mem (J.mul_mem_left r hx) haJ
    have hbMul : b ∈ Ideal.map (algebraMap R B) p * J :=
      mem_map_mul_of_mem_map_of_flat_quotient
        (R := R) (B := B) J p ⟨b, hbJ⟩ hb
    have hma : algebraMap B L a ∈ E := by
      have := Ideal.mem_map_of_mem (algebraMap B L) ha
      simpa [E, Ideal.map_span] using this
    have hmb : algebraMap B L b ∈ I * JL := by
      have := Ideal.mem_map_of_mem (algebraMap B L) hbMul
      simpa [I, JL, p, Ideal.map_mul, Ideal.map_map,
        IsScalarTower.algebraMap_eq R B L] using this
    have hrxLocal :
        algebraMap B L r * algebraMap B L x ∈ E ⊔ I * JL := by
      have habLocal := congrArg (algebraMap B L) hab
      rw [map_add, map_mul] at habLocal
      rw [← habLocal]
      exact Submodule.add_mem _
        (Submodule.mem_sup_left hma) (Submodule.mem_sup_right hmb)
    have hrUnit : IsUnit (algebraMap B L r) :=
      IsLocalization.map_units L (⟨r, hr⟩ : q.asIdeal.primeCompl)
    exact (Ideal.unit_mul_mem_iff_mem (E ⊔ I * JL) hrUnit).mp hrxLocal
  have hNak : (JL : Submodule L L) ≤ (E : Submodule L L) := by
    apply Submodule.le_of_le_smul_of_le_jacobson_bot
      (hJfg.map (algebraMap B L)) hIjac
    simpa only [Ideal.smul_eq_mul] using hlocal
  intro x
  have hxLocal :
      algebraMap B L (x : B) ∈
        Ideal.map (algebraMap B L) (Ideal.span ({e} : Set B)) := by
    have hxJL : algebraMap B L (x : B) ∈ JL := by
      exact Ideal.mem_map_of_mem (algebraMap B L) x.2
    have hxE : algebraMap B L (x : B) ∈ E := hNak hxJL
    simpa [E, Ideal.map_span] using hxE
  obtain ⟨r, hr, hrx⟩ :=
    (IsLocalization.algebraMap_mem_map_algebraMap_iff
      q.asIdeal.primeCompl L (Ideal.span ({e} : Set B)) (x : B)).mp hxLocal
  exact ⟨r, hr, hrx⟩
