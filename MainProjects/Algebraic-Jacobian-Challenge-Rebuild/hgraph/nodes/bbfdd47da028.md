theorem exists_ofList_eq_ker_length_add_ringKrullDim
    [IsRegularLocalRing B] [IsRegularLocalRing A]
    (φ : B →+* A) (hφ : Function.Surjective φ) :
    ∃ xs : List B, Ideal.ofList xs = RingHom.ker φ ∧
      (xs.length : WithBot ℕ∞) + ringKrullDim A = ringKrullDim B := by
  obtain ⟨xs, hspan, hlen⟩ := exists_ofList_eq_ker_of_surjective φ hφ
  refine ⟨xs, hspan, ?_⟩
  have hdimB : (((IsLocalRing.maximalIdeal B).spanFinrank : ℕ) : WithBot ℕ∞)
      = ringKrullDim B := IsRegularLocalRing.spanFinrank_maximalIdeal
  have hdimA : (((IsLocalRing.maximalIdeal A).spanFinrank : ℕ) : WithBot ℕ∞)
      = ringKrullDim A := IsRegularLocalRing.spanFinrank_maximalIdeal
  rw [← hdimA, ← hdimB, ← hlen]
  push_cast
  ring
