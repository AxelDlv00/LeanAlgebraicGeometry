theorem existsUnique_isPullbackValue (f : T' ⟶ T) (s : picEt C T)
    (W : T'.left.affineOpens) : ∃! v, IsPullbackValue C f s W v := by
  obtain ⟨v, hv⟩ := exists_isPullbackValue C f s W
  exact ⟨v, hv, fun v' hv' => pullbackValue_unique C hv' hv⟩

end picEt
