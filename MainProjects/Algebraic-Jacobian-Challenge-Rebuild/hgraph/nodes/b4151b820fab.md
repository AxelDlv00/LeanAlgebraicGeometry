theorem cyclic_of_free [Module.Invertible S N] [Module.Free S N] :
    ∃ y : N, ∀ z : N, ∃ r : S, z = r • y := by
  obtain ⟨e⟩ := (Module.Invertible.free_iff_linearEquiv (R := S) (M := N)).mp ‹_›
  refine ⟨e.symm 1, fun z => ⟨e z, ?_⟩⟩
  calc z = e.symm (e z) := (e.symm_apply_apply z).symm
    _ = e.symm (e z • (1 : S)) := by rw [smul_eq_mul, mul_one]
    _ = e z • e.symm 1 := map_smul _ _ _

end Module.Invertible
