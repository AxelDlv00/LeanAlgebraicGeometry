noncomputable def Scheme.residueOverAlgebraMap (x : X) : K →+* X.residueField x :=
  (X.residue x).hom.comp
    (((X.presheaf.germ ⊤ x trivial).hom).comp (X.overAlgebraMap K ⊤))
