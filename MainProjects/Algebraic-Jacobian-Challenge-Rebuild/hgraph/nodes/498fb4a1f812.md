    (hfin : Module.Finite R A.Glued) (hproj' : Module.Projective R A.Glued)

  haveI := hproj'
  rw [← Module.rankAtStalk_eq_of_equiv (A.gluedBaseChange R' hproj hinf),
    Module.rankAtStalk_baseChange]
