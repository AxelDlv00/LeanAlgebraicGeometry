theorem poly_nilpotent_of_coeffs_mem {I : Ideal A} (hI : IsNilpotent I) {p : Polynomial A}
    (hp : ∀ m, p.coeff m ∈ I) : IsNilpotent p := by
  rw [Polynomial.isNilpotent_iff]
  intro m
  obtain ⟨N, hN⟩ := hI
  exact ⟨N, by have := Ideal.pow_mem_pow (hp m) N; rw [hN] at this; simpa using this⟩
