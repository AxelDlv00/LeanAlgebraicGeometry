  refine AlgEquiv.ofBijective (Algebra.ofId k (𝒜 0)) ⟨fun a b hab => ?_, fun c => ?_⟩
  · have h : (MvPolynomial.C a : MvPolynomial (Fin 2) k) = MvPolynomial.C b := by
      have h' := congrArg Subtype.val hab
