theorem surjective_congr (h : ∀ a : A, e₂ (δ a) = δ' (e₁ a))
    (hδ : Function.Surjective δ) : Function.Surjective δ' := by
  intro b
  obtain ⟨a, ha⟩ := hδ (e₂.symm b)
  refine ⟨e₁ a, ?_⟩
  rw [← h a, ha, e₂.apply_symm_apply]

end AlgebraicJacobian.RigidEngine
