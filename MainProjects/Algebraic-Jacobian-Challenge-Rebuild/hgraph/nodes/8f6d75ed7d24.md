def scalarExtensionMap {M K A B : Type u}
    [CommRing M] [CommRing K] [CommRing A] [CommRing B]
    [Algebra M K] [Algebra M A] [Algebra M B] [Algebra A B]
    [IsScalarTower M A B] :
    (K ⊗[M] A) →ₐ[K] (K ⊗[M] B) :=
  Algebra.TensorProduct.map (AlgHom.id K K) (IsScalarTower.toAlgHom M A B)
