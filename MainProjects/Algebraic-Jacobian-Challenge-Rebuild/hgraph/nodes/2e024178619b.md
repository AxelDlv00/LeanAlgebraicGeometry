noncomputable def reindex (A : AffAdaptation D d) {m' : ℕ} (e : Fin m' ≃ D.index) :
    AffAdaptation (D.reindex e) d where
  eqn := fun j => A.eqn (e j)
  eqn_rel := fun j y => A.eqn_rel (e j) y
