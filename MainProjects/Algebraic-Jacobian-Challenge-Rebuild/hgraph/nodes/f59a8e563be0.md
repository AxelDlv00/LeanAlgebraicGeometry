def cechCoboundaryUnits (ρ₁ : A₁ →+* B) (ρ₂ : A₂ →+* B) : Subgroup Bˣ :=
  (Units.map ρ₁.toMonoidHom).range ⊔ (Units.map ρ₂.toMonoidHom).range
