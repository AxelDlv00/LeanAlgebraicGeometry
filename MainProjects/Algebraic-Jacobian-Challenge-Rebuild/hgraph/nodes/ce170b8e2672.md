theorem surjective_selector_of_not_isAffine (h : ¬ IsAffine Y) :
    Function.Surjective D.selector :=
  (D.surjective_selector_iff).mpr
    ⟨fun hbot => h (D.isAffine_of_V₀_eq_bot hbot), fun htop => h (D.isAffine_of_V₀_eq_top htop)⟩
