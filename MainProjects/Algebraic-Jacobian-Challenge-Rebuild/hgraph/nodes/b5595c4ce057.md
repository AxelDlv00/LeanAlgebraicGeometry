noncomputable def transitionOfLE (i j : Nat) (h : i ≤ j) : G i →ₗ[R] G j :=
  (reindex (R := R) G (i + (j - i)) j (Nat.add_sub_of_le h)).toLinearMap.comp
    (iterateSuccessor G step i (j - i))
