theorem probe_containment_monotone {ι : Type u} {X : ι → Scheme.{u}}
    (f : ∀ i, yoneda.obj (X i) ⟶ (pic0SigmaSheaf C).1) (Vc Vf : ∀ i, (X i).Opens)
    (hle : ∀ i, Vc i ≤ Vf i)
    (h : ∀ (T : Scheme.{u}) (s : (pic0SigmaSheaf C).1.obj (op T)) (t : ↥T),
      ∃ (W : T.Opens) (_ : t ∈ W) (i : ι) (x : (W : Scheme.{u}) ⟶ X i),
        (f i).app (op (W : Scheme.{u})) x = (pic0SigmaSheaf C).1.map (W.ι).op s ∧
          Set.range (x.base) ⊆ Set.range ((Vc i).ι.base)) :
    ∀ (T : Scheme.{u}) (s : (pic0SigmaSheaf C).1.obj (op T)) (t : ↥T),
      ∃ (W : T.Opens) (_ : t ∈ W) (i : ι) (x : (W : Scheme.{u}) ⟶ X i),
        (f i).app (op (W : Scheme.{u})) x = (pic0SigmaSheaf C).1.map (W.ι).op s ∧
          Set.range (x.base) ⊆ Set.range ((Vf i).ι.base) := by
  intro T s t
  obtain ⟨W, htW, i, x, hx, hrange⟩ := h T s t
  refine ⟨W, htW, i, x, hx, hrange.trans ?_⟩
  have : (Vc i).ι = (X i).homOfLE (hle i) ≫ (Vf i).ι := (Scheme.homOfLE_ι _ _).symm
  rw [this]
  intro z hz
  obtain ⟨w, rfl⟩ := hz
  exact ⟨((X i).homOfLE (hle i)).base w, rfl⟩
