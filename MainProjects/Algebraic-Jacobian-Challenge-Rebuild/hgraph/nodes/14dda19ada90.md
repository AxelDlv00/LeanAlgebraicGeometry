theorem surjective_of_extensionTolerant {J : Over (Spec (.of k))}
    (rep : (pic0TypeFunctor C).RepresentableBy J)
    (lam : pic0Subgroup C DivO)
    (h : ∀ y : J.left, ∃ (T : Over (Spec (.of k))) (_ : Nonempty T.left)
      (e : T ⟶ overSpec k (Over.testPointField y)) (q : T ⟶ DivO),
      pic0Map C q lam = rep.homEquiv (e ≫ Over.testPoint y)) :
    Function.Surjective (abelOfPic0Class rep lam).left.base := by
  refine surjective_of_testGeneral rep lam fun y => ?_
  obtain ⟨T, ⟨p⟩, e, q, hcl⟩ := h y
  exact ⟨T, p, q, e ≫ Over.testPoint y, Over.testPoint_base y _, hcl⟩
