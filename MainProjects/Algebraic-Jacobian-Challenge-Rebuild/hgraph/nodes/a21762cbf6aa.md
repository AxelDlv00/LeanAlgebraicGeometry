lemma mulVecLin_map_algebraMap (ψ : Matrix (Fin e) (Fin m) R) :
    (ψ.map (algebraMap R A)).mulVecLin =
      (TensorProduct.piScalarRight R A A (Fin e)).toLinearMap
        ∘ₗ ψ.mulVecLin.baseChange A
        ∘ₗ (TensorProduct.piScalarRight R A A (Fin m)).symm.toLinearMap := by
  refine LinearMap.pi_ext' fun j ↦ LinearMap.ext_ring ?_
  ext i
  simp only [LinearMap.comp_apply, LinearMap.coe_single, LinearEquiv.coe_toLinearMap,
    TensorProduct.piScalarRight_symm_single, LinearMap.baseChange_tmul,
    TensorProduct.piScalarRight_apply, TensorProduct.piScalarRightHom_tmul,
    Matrix.mulVecLin_apply, Matrix.mulVec, Matrix.map_apply]
  rw [dotProduct_single, dotProduct_single]
  simp [Algebra.smul_def]
