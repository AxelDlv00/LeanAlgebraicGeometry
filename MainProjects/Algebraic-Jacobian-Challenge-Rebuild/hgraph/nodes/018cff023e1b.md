private theorem span_range_read_map_submodule_map
    {R A A' B : Type*} [CommRing R] [CommRing B] [Algebra R B]
    [AddCommGroup A] [Module R A] [AddCommGroup A'] [Module R A']
    (f : A' →ₗ[R] B) (e : A →ₗ[R] A') (N : Submodule R A) :
    Ideal.span (Set.range (f.comp (Submodule.map e N).subtype)) =
      Ideal.span (Set.range ((f.comp e).comp N.subtype)) := by
  let em := e.submoduleMap N
  calc
    Ideal.span (Set.range (f.comp (Submodule.map e N).subtype)) =

      symm

        (f := f.comp (Submodule.map e N).subtype) (e := em)
          (LinearMap.submoduleMap_surjective e N)
    _ = Ideal.span (Set.range ((f.comp e).comp N.subtype)) := by
      have hmaps :
          (f.comp (Submodule.map e N).subtype).comp em =
            (f.comp e).comp N.subtype := by
        apply LinearMap.ext
        intro x
        rfl
      rw [hmaps]
