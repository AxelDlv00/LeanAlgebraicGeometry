theorem baseChange_universalMulMap_eq_sum
    (K : Type u) [CommRing K] [Algebra RZ K] [Algebra k K]
      [IsScalarTower k RZ K] :
    LinearMap.baseChange K (universalMulMap (hπ := hπ) g r₁ r₂ b₁ b₂ i j)
      = universalMulComponentSum (hπ := hπ) g r₁ r₂ b₁ b₂ i j K ∘ₗ
          universalMulSourceBaseChangeHom (hπ := hπ) g r₁ r₂ b₁ b₂ i j K := by
  apply LinearMap.ext
  intro x
  induction x using TensorProduct.induction_on with
  | zero => simp
  | add x y hx hy => simp [hx, hy]
  | tmul a v =>
      simp [universalMulMap, universalMulComponent,
        universalMulComponentSum,
        universalMulSourceBaseChangeHom, LinearMap.comp_apply,
        LinearMap.baseChange_tmul]
      rw [TensorProduct.tmul_sum]
