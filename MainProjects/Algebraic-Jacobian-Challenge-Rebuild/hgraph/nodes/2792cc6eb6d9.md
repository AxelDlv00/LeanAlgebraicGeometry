theorem ker_pointBaseChange :
    RingHom.ker (pointBaseChange (k := k) B F) = Ideal.span {pointGen k B F} := by
  apply le_antisymm
  · intro x hx
    rw [RingHom.mem_ker] at hx
    have hmk : Ideal.Quotient.mk (Ideal.span {pointGen k B F}) x = 0 := by
      rw [← pointSectionMap_pointBaseChange x, hx, map_zero]
    exact (Ideal.Quotient.eq_zero_iff_mem).mp hmk
  · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe, RingHom.mem_ker]
    exact pointBaseChange_pointGen
