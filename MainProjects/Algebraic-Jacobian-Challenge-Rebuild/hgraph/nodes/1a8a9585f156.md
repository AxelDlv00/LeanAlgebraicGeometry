def degAt {T : Over (Spec (.of k))} (lam : picEt C T) {K : Type u} [Field K]
    [Algebra k K] (t : overSpec k K ⟶ T) : ℤ :=
  PicEtAff.degAff K (picEtAffineEquiv C K (picEtMap C t lam))
