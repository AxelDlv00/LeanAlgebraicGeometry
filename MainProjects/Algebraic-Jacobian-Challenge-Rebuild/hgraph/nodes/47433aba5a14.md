theorem finite_app_overlap [IsFinite π] :
    RingHom.Finite (π.app (Proj.basicOpen (homogeneousSubmodule (Fin 2) k)
      (X 0 * X 1))).hom :=
  π.finite_app _ (P1.isAffineOpen_overlap k)
