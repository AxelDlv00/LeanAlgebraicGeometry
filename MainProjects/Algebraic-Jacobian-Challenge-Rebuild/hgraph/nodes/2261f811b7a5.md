    (hE : Scheme.CechPic.map (cg)
        (Scheme.CechPic.mk ℰ (CategoryTheory.PresheafOfGroups.OneCocycle.class lam))

    Scheme.CechPic.mk_eq_one_iff] at hE

  have h := hα v v'

  exact h
