noncomputable def chartFrameMap (X : Matrix (Fin d) (Fin r) S) (I : Finset (Fin r))
    (hI : I.card = d) : ChartRing k d r I →ₐ[k] S :=
  MvPolynomial.aeval (fun e => frameMatrix k d r S X I hI e.1 e.2.1)
