noncomputable def relPicMap (g : T' ⟶ T) : relPic C T →* relPic C T' :=
  QuotientGroup.map (picFromBase C T) (picFromBase C T')
    (CechPic.map (C ◁ g).left) (picFromBase_le_comap C g)
