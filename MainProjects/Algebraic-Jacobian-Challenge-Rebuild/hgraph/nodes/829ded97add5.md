theorem isProper_left [IsProper D.hom] [IsSeparated E.hom] (h : D ⟶ E) :
    IsProper h.left := by
  haveI : IsProper (h.left ≫ E.hom) := by rw [Over.w h]; infer_instance
  exact IsProper.of_comp h.left E.hom
