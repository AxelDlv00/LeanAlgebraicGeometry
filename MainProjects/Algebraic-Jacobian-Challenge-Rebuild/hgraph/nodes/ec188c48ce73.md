def divFamZarToAffVehicle {T : Over (Spec (.of k))} (s : divFamZar C π n T) :
    divFamZarAff C n T :=
  ⟨fun U => (s.1 U).toAff,
    fun U V h => by
      beta_reduce
      rw [← s.compat U V h, DivFamZar.toAff_mapAlgHom]⟩
