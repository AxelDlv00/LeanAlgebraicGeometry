noncomputable def ovlProdBaseChange
    (hinf : ∀ i j : D.index, IsAffineOpen (D.pieces i ⊓ D.pieces j)) :
    R' ⊗[R] A.ovlProd ≃ₗ[R'] (A.pullback R' hproj).ovlProd :=

      (fun p : D.index × D.index => A.ovlColength p.1 p.2)).trans
