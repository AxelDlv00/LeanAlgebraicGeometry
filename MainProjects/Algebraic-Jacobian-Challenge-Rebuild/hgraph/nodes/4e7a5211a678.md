theorem cartierIdeal_ideal_eq_sectionIdeal_of_le [IsProper C.hom]
    (A : AffAdaptation D d) (j : D.index) (U : (relCurve C R).affineOpens)
    (hU : U.1 ≤ D.pieces j) :
    A.cartierIdeal.ideal U = d.sectionIdeal U.1 := by
  rw [← A.cartierIdeal.map_ideal (U := U)
    (V := ⟨D.pieces j, D.isAffineOpen j⟩) hU]
  rw [A.cartierIdeal_ideal_eq_span_eqn j, Ideal.map_span, Set.image_singleton]
  exact (A.sectionIdeal_eq_span_restrict_eqn j hU).symm
