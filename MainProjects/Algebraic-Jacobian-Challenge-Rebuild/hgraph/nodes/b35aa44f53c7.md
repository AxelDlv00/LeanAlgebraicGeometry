theorem Module.Finite.span_range_of_finite_isScalarTower
    {R S M N : Type*} [CommRing R] [CommRing S] [Algebra R S]
    [AddCommGroup M] [Module R M] [Module.Finite R M]
    [AddCommGroup N] [Module R N] [Module S N] [IsScalarTower R S N]
    (f : M →ₗ[R] N) :
    Module.Finite S (Submodule.span S (Set.range f)) := by
  obtain ⟨T, hT⟩ := (Module.Finite.fg_top (R := R) (M := M))
  have hspan : Submodule.span S (Set.range f) = Submodule.span S (f '' T) := by
    apply le_antisymm
    · rw [Submodule.span_le]
      rintro _ ⟨m, rfl⟩
      have hm : m ∈ Submodule.span R (T : Set M) := by rw [hT]; trivial
      have hfm : f m ∈ Submodule.span R (f '' T) := by
        rw [← Submodule.map_span]
        exact Submodule.mem_map_of_mem hm
      exact (Submodule.span_le_restrictScalars R S (f '' T)) hfm
    · exact Submodule.span_mono (Set.image_subset_range f T)
  rw [hspan]
  exact Module.Finite.iff_fg.mpr (Submodule.fg_span (T.finite_toSet.image f))
