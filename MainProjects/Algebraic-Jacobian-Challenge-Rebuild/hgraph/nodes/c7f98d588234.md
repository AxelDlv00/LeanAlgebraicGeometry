theorem map_id (s : divFamZarAff C n T) : map C n (𝟙 T) s = s := by
  refine ext fun W => ?_
  rw [map_val]
  refine mapVal_eq_of C n (𝟙 T) s ?_

  have hV' : W₀.1 ≤ V.1 := by
    intro p hp
