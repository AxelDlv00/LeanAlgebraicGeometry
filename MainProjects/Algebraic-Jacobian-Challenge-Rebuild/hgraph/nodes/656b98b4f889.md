theorem degAt_pow (θ : picEt C T) (m : ℕ) (t : overSpec k K ⟶ T) :
    degAt (θ ^ m) t = (m : ℤ) * degAt θ t := by
  induction m with
  | zero => rw [pow_zero, degAt_one, Nat.cast_zero, zero_mul]
  | succ n ih => rw [pow_succ, degAt_mul, ih]; push_cast; ring
