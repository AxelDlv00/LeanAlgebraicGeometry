noncomputable def Scheme.Hom.functionFieldMap (f : X ⟶ Y)
    (h : f.base (genericPoint X) = genericPoint Y) :
    Y.functionField ⟶ X.functionField :=
  (Y.presheaf.stalkCongr (Inseparable.of_eq h.symm)).hom ≫ f.stalkMap (genericPoint X)
