noncomputable def polyBaseChange (k A : Type u) [CommRing k] [CommRing A] [Algebra k A] :
    (Polynomial k) ⊗[k] A ≃ₐ[k] Polynomial A :=
  (Algebra.TensorProduct.comm k (Polynomial k) A).trans (polyEquivTensor k A).symm
