def divFamZarAff (T : Over (Spec (.of k))) : Type u :=
  {s : Π U : T.left.affineOpens, DivFamZarAff C Γ(T.left, U.1) n //
    ∀ (U V : T.left.affineOpens) (h : U.1 ≤ V.1),
      DivFamZarAff.mapAlgHom (Over.resAlgHom T h) (s V) = s U}
