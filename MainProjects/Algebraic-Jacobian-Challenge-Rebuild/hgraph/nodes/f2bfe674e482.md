theorem ker_map_localization_eq :
    (RingHom.ker (Algebra.TensorProduct.lmul' k (S := B))).map (algebraMap (B ⊗[k] B) L)
      = Ideal.span {algebraMap (B ⊗[k] B) L (diagGen (k := k) (B := B))} := by
  -- `elift` becomes a multiple of `u⊗1 − 1⊗u` after inverting `1 - elift`
  have hdvd : algebraMap (B ⊗[k] B) L elift ∈
      Ideal.span {algebraMap (B ⊗[k] B) L (diagGen (k := k) (B := B))} := by
    have hunit : IsUnit (algebraMap (B ⊗[k] B) L (1 - elift)) :=
      IsLocalization.map_units L (⟨1 - elift, Submonoid.mem_powers _⟩ : Submonoid.powers _)
    obtain ⟨v, hv⟩ := hunit.exists_right_inv
    have hmem : elift * (1 - elift) ∈ Ideal.span {diagGen (k := k) (B := B)} := by
      rw [← ker_baseChange, RingHom.mem_ker, map_mul, map_sub, map_one, mul_sub, mul_one,
        hidem.eq, sub_self]
    have hfmem : algebraMap (B ⊗[k] B) L (elift * (1 - elift)) ∈
        Ideal.span {algebraMap (B ⊗[k] B) L (diagGen (k := k) (B := B))} := by
      have hmap := Ideal.mem_map_of_mem (algebraMap (B ⊗[k] B) L) hmem
      rwa [Ideal.map_span, Set.image_singleton] at hmap
    have heq : algebraMap (B ⊗[k] B) L elift
        = algebraMap (B ⊗[k] B) L (elift * (1 - elift)) * v := by
      rw [map_mul, mul_assoc, hv, mul_one]
    rw [heq]; exact Ideal.mul_mem_right _ _ hfmem
  apply le_antisymm
  · rw [Ideal.map_le_iff_le_comap]
    refine (ker_lmul'_le_sup elift hgen).trans ?_
    rw [sup_le_iff]
    refine ⟨?_, ?_⟩
    · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe, Ideal.mem_comap]
      exact Ideal.subset_span rfl
    · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe, Ideal.mem_comap]
      exact hdvd
  · rw [Ideal.span_le, Set.singleton_subset_iff, SetLike.mem_coe]
    exact Ideal.mem_map_of_mem _ (by rw [RingHom.mem_ker]; exact lmul'_diagGen)

end Localization
