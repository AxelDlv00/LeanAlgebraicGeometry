private lemma div_defect_mul {G : Type u} [CommGroup G] (a b c : G) :
    a / (a * (c / b)) * c = b := by
  rw [← mul_div_assoc a c b, div_div_eq_mul_div, div_mul_eq_mul_div,
    mul_right_comm a b c]
  exact mul_div_cancel_left _ b
