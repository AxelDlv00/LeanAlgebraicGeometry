lemma comp_rowFst :
    q ≫ rowFst p hp = pullback.lift p q (by rw [hp, hq]) := by
  apply pullback.hom_ext
  · rw [Category.assoc, rowFst_fst, pullback.lift_fst, ← Category.assoc, hq,
      Category.id_comp]
  · rw [Category.assoc, rowFst_snd, Category.comp_id, pullback.lift_snd]
