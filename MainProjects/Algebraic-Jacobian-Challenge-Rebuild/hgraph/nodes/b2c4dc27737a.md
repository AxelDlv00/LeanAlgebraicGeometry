lemma h1Map_surjective (f : P.Hom P') (hN : Function.Surjective f.homN) :
    Function.Surjective f.h1Map := by
  intro c
  obtain ⟨n', rfl⟩ := Submodule.Quotient.mk_surjective _ c
  obtain ⟨n, rfl⟩ := hN n'
  exact ⟨Submodule.Quotient.mk n, f.h1Map_mk n⟩
