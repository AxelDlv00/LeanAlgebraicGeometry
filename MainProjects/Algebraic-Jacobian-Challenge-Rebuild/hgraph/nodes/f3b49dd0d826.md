theorem probeDominantFromIrred {X : Scheme.{u}} [IrreducibleSpace X] (V : X.Opens)
    (hne : (V : Set X).Nonempty) :
    IsDominant (V.ι) := by
  rw [isDominant_iff]
  simpa [DenseRange, Scheme.Opens.range_ι] using V.2.dense hne
