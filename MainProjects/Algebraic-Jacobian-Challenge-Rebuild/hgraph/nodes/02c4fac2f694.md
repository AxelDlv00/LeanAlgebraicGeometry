def divFam.toZarVehicle {T : Over (Spec (.of k))} (s : divFam C π n T) :
    divFamZar C π n T :=
  ⟨fun U => (s.1 U).toZar,
    fun U V h => by
      beta_reduce
      rw [← s.compat U V h, DivFam.toZar_mapAlgHom]⟩
