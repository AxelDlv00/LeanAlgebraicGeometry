    {X : Scheme.{u}} (f : yoneda.obj X ⟶ (pic0SigmaSheaf C).1) (V : X.Opens)
    (hV : V ≠ ⊤) (hinj : ∀ T : Scheme.{u}ᵒᵖ, Function.Injective (f.app T)) :
