theorem SmoothOfRelativeDimension.geometricallyReduced (n : ℕ) (f : X ⟶ Y)
    [SmoothOfRelativeDimension n f] : GeometricallyReduced f :=
  have : Smooth f := SmoothOfRelativeDimension.smooth n f
  inferInstance
