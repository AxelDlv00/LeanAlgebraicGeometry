def DivEq (d₁ d₂ : X.LocalEquations) : Prop :=
  ∃ (𝒲 : X.PointedCover) (h₁ : 𝒲 ≤ d₁.cover) (h₂ : 𝒲 ≤ d₂.cover),
    ∀ x : X, ∃ u : Γ(X, 𝒲.opens x)ˣ,
      (X.presheaf.map (homOfLE (h₁ x)).op).hom (d₁.eqn x)
        = (u : Γ(X, 𝒲.opens x))
          * (X.presheaf.map (homOfLE (h₂ x)).op).hom (d₂.eqn x)
