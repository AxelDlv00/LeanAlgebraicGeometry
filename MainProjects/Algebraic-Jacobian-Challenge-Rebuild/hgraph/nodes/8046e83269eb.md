private theorem degree_eq (i : J) {e : J → ℕ} {a : ℕ}
    (hae : ∑ l, e l • (1 : ℕ) = a • (1 : ℕ)) :
    e i + ∑ j : {j : J // j ≠ i}, e j = a := by
  rw [← (Equiv.optionSubtypeNe i).sum_comp, Fintype.sum_option] at hae
  simpa using hae
