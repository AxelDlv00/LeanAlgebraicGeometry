theorem comparison_naturality (f : Hom D₁ D₂)
    (x : B ⊗[A] D₁.descended) :
    D₂.comparison
        (Algebra.TensorProduct.map (AlgHom.id A B) f.descendedMap x) =
      f.hom (D₁.comparison x) := by
  induction x with
  | zero => simp
  | tmul b x => simp [Algebra.smul_def]
  | add x y hx hy => simp [hx, hy]
