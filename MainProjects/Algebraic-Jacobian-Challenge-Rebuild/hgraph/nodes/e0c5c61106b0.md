    (E.prod E').Carrier ≃ₐ[A] E.Carrier ⊗[A] E'.Carrier :=

    Algebra.Etale.comp A E.Carrier (E.Carrier ⊗[A] E'.Carrier)

    Module.FaithfullyFlat.trans A E.Carrier (E.Carrier ⊗[A] E'.Carrier)
