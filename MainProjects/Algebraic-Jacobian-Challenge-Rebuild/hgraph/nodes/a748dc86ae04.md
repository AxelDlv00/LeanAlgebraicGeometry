def ProjectiveSpace (n : Type u) (S : Scheme.{u}) : Scheme.{u} :=
  pullback (terminal.from S)
    (terminal.from (Proj (homogeneousSubmodule n (ULift.{u} ℤ))))

/-- `ℙ(n; S)` is the projective `n`-space over `S`. -/
