lemma isUnit_algHom_powers {k R S : Type u} [Field k] [CommRing R] [CommRing S]
    [Algebra k R] [Algebra k S] (f : R →ₐ[k] S) {x : R} (hx : IsUnit (f x)) :
    ∀ y : Submonoid.powers x, IsUnit (f y) := fun y => by
  obtain ⟨n, hn⟩ := y.2
  simpa [← hn, map_pow] using hx.pow n
