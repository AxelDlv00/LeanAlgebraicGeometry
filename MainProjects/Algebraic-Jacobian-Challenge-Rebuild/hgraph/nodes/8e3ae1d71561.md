theorem exists_ofList_eq_ker_of_surjective
    [IsRegularLocalRing B] [IsRegularLocalRing A]
    (φ : B →+* A) (hφ : Function.Surjective φ) :
    ∃ xs : List B, Ideal.ofList xs = RingHom.ker φ ∧
      xs.length + (IsLocalRing.maximalIdeal A).spanFinrank
        = (IsLocalRing.maximalIdeal B).spanFinrank := by
  suffices haux : ∀ (n : ℕ) (B : Type u) [CommRing B] [IsRegularLocalRing B]
      (φ : B →+* A), Function.Surjective φ →
      (IsLocalRing.maximalIdeal B).spanFinrank = n →
      ∃ xs : List B, Ideal.ofList xs = RingHom.ker φ ∧
        xs.length + (IsLocalRing.maximalIdeal A).spanFinrank = n by
    obtain ⟨xs, h1, h2⟩ :=
      haux (IsLocalRing.maximalIdeal B).spanFinrank B φ hφ rfl
    exact ⟨xs, h1, h2⟩
  intro n
  induction n with
  | zero =>
    intro B _ _ φ hφ hn
    -- `spanFinrank 𝔪_A ≤ 0`, so both span-ranks agree and the kernel is
    -- trivial.
    have hle := spanFinrank_maximalIdeal_le_of_surjective φ hφ
    rw [hn, Nat.le_zero] at hle
    refine ⟨[], ?_, by simp [hle]⟩
    rw [Ideal.ofList_nil]
    exact (ker_eq_bot_of_spanFinrank_eq φ hφ (by rw [hn, hle])).symm
  | succ k ih =>
    intro B _ _ φ hφ hn
    rcases eq_or_lt_of_le (spanFinrank_maximalIdeal_le_of_surjective φ hφ)
      with heq | hlt
    · -- equal spanFinranks: kernel is `⊥`
      refine ⟨[], ?_, by rw [heq, hn]; simp⟩
      rw [Ideal.ofList_nil]
      exact (ker_eq_bot_of_spanFinrank_eq φ hφ heq).symm
    · -- strict drop: peel one kernel element off `𝔪_B ∖ 𝔪_B²`
      rw [hn] at hlt
      obtain ⟨x, hxker, hxsq⟩ := exists_mem_ker_notMem_sq φ hφ (by rwa [hn])
      have hx𝔪 : x ∈ IsLocalRing.maximalIdeal B :=
        IsLocalRing.le_maximalIdeal (RingHom.ker_ne_top φ) hxker
      obtain ⟨hNT, hLR, hReg, hspan₁⟩ :=
        regularLocal_quotient_isRegularLocal_of_notMemSq hn x hx𝔪 hxsq
      haveI := hNT
      haveI := hLR
      haveI := hReg
      -- descend `φ` to `B/(x)`
      have hxle : Ideal.span ({x} : Set B) ≤ RingHom.ker φ := by
        rwa [Ideal.span_singleton_le_iff_mem]
      set φ₁ : B ⧸ Ideal.span ({x} : Set B) →+* A :=
        Ideal.Quotient.lift _ φ (fun y hy => hxle hy) with hφ₁def
      have hφ₁surj : Function.Surjective φ₁ := by
        intro a
        obtain ⟨b, rfl⟩ := hφ a
        exact ⟨Ideal.Quotient.mk _ b, by simp [hφ₁def]⟩
      obtain ⟨ys', hys'span, hys'len⟩ := ih _ φ₁ hφ₁surj hspan₁
      -- lift the generators
      obtain ⟨ys, hysmap⟩ :=
        exists_list_map_eq (Ideal.Quotient.mk (Ideal.span ({x} : Set B)))
          Ideal.Quotient.mk_surjective ys'
      refine ⟨x :: ys, ?_, ?_⟩
      · -- span equality through `map`/`comap` along the quotient
        have hmapof : Ideal.map (Ideal.Quotient.mk (Ideal.span ({x} : Set B)))
            (Ideal.ofList (x :: ys)) = RingHom.ker φ₁ := by
          rw [Ideal.ofList_cons, Ideal.map_sup, Ideal.map_ofList, hysmap,
            hys'span]
          have hx0 : Ideal.map (Ideal.Quotient.mk (Ideal.span ({x} : Set B)))
              (Ideal.span ({x} : Set B)) = ⊥ := by
            rw [Ideal.map_span]
            simp
          rw [hx0, bot_sup_eq]
        have hkerφ₁ : RingHom.ker φ₁
            = Ideal.map (Ideal.Quotient.mk (Ideal.span ({x} : Set B)))
              (RingHom.ker φ) := Ideal.ker_quotient_lift φ _
        have hcomap := congrArg
          (Ideal.comap (Ideal.Quotient.mk (Ideal.span ({x} : Set B))))
          (hmapof.trans hkerφ₁)
        rw [Ideal.comap_map_of_surjective _ Ideal.Quotient.mk_surjective,
          Ideal.comap_map_of_surjective _ Ideal.Quotient.mk_surjective,
          ← RingHom.ker_eq_comap_bot, Ideal.mk_ker] at hcomap
        have hxle' : Ideal.span ({x} : Set B) ≤ Ideal.ofList (x :: ys) := by
          rw [Ideal.ofList_cons]
          exact le_sup_left
        rwa [sup_eq_left.mpr hxle', sup_eq_left.mpr hxle] at hcomap
      · -- length bookkeeping
        have hlen : ys.length = ys'.length := by
          rw [← hysmap, List.length_map]
        simp only [List.length_cons, hlen]
        omega
