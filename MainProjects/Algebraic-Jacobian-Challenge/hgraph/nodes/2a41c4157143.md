lemma fiberRank_congr (p : Ideal R) [p.IsPrime] (σ : M ≃ₗ[R] N) :
    p.fiberRank M = p.fiberRank N :=
  LinearEquiv.finrank_eq (LinearEquiv.baseChange R p.ResidueField M N σ)
