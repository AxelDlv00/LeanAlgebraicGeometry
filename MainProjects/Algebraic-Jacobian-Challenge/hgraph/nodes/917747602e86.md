theorem comp_powSum {C A W : K} [MonObj A] [IsCommMonObj A] (n : ℕ) (φ : C ⟶ A)
    (u : W ⟶ (∏ᶜ (fun _ : Fin n => C))) :
    u ≫ powSum n φ = ∏ i, (u ≫ Pi.π (fun _ : Fin n => C) i ≫ φ) := by
  classical
  exact map_prod (preHom u) _ Finset.univ
