def IsEquivariantAlgHom (f : A →ₐ[L] L ⊗[K] B) : Prop :=
  ∀ (γ : L ≃ₐ[K] L) (x : A),
    f (γ • x) = Algebra.TensorProduct.map (γ : L →ₐ[K] L) (AlgHom.id K B) (f x)
