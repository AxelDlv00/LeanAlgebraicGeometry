def h1Map (h : d ∘ₗ a0 = a1 ∘ₗ k) : (K1 ⧸ range k) →ₗ[R] (M1 ⧸ range d) :=
  Submodule.mapQ _ _ a1 <| by
    rintro x ⟨y, rfl⟩
    have hy := LinearMap.congr_fun h y
    simp only [LinearMap.comp_apply] at hy
    exact ⟨a0 y, hy⟩
