noncomputable def degreeTotal {k : Type u} [Field k]
    (C : Over (Spec (.of k)))
    [SmoothOfRelativeDimension 1 C.hom] [IsProper C.hom]
    [GeometricallyIntegral C.hom] [Scheme.HasPicScheme C] [Scheme.PicScheme.ClassDegreePinned C] :
    (Spec (.of k) ⟶ (Scheme.PicScheme C).left) → ℤ :=
  fun lambda =>
    if h : lambda ≫ (Scheme.PicScheme C).hom = 𝟙 (Spec (.of k)) then
      Scheme.PicScheme.degreeOfSectionPinned C lambda h
    else 0
