theorem Scheme.Hom.IsProjectiveWith.fiber {π : X ⟶ S} {L : X.Modules}
    (h : π.IsProjectiveWith L) (s : S) :
    (π.fiberToSpecResidueField s).IsProjectiveWith (π.fiberModule s L) :=
  h.baseChange (S.fromSpecResidueField s)
