theorem free_of_nilpotent_of_exists_sub_smul_mem [Module.Invertible R M] {I : Ideal R}
    (hI : IsNilpotent I) {m : M}
    (h : ∀ x : M, ∃ r : R, x - r • m ∈ I • (⊤ : Submodule R M)) :
    Module.Free R M := by
  refine free_of_nilpotent_of_span_sup_smul_eq_top hI (m := m) (le_antisymm le_top ?_)
  intro x _
  obtain ⟨r, hr⟩ := h x
  have hx : x = r • m + (x - r • m) := by abel
  rw [hx]
  exact Submodule.add_mem _
    (Submodule.mem_sup_left (Submodule.mem_span_singleton.mpr ⟨r, rfl⟩))
    (Submodule.mem_sup_right hr)

end Module.Invertible
