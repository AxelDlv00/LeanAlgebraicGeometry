theorem HasProperSupport.of_iso {X S : Scheme.{u}} (f : X ⟶ S)
    {A C : X.Modules} (e : A ≅ C) (hA : HasProperSupport f A) :
    HasProperSupport f C := by
  change IsProper ((annihilator C).subschemeι ≫ f)
  rw [← annihilator_eq_of_iso e]
  exact hA
