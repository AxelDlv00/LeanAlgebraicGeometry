noncomputable def quotBaseMap (π : X ⟶ S) {T T' : Over S} (ψ : T' ⟶ T) :
    (Limits.pullback π T'.hom : Scheme.{u}) ⟶ Limits.pullback π T.hom :=
  Limits.pullback.map π T'.hom π T.hom (𝟙 X) ψ.left (𝟙 S) (by simp)
    (by simpa using (Over.w ψ).symm)
