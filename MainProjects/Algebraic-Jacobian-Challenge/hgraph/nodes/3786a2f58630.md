theorem awayToPoly_chartCoord (j : n) :
    awayToPoly R n (chartCoord R n j) = X j :=
  (awayToPoly_mk R n 1 (X (some j))
    (by simpa using X_mem_deg_one R n (some j))).trans
      (dehomogenize_X_some R n j)
