theorem invariantMorphism_kernelPair
    {T Z : Scheme.{u}} (t : T ⟶ Spec (CommRingCat.of K))
    (a : pullback t (fieldBaseMap K L) ⟶ Z)
    (ha : ∀ gamma : L ≃ₐ[K] L, pullbackGalMap K L t gamma ≫ a = a) :
    pullback.fst (pullback.fst t (fieldBaseMap K L))
          (pullback.fst t (fieldBaseMap K L)) ≫ a =
      pullback.snd (pullback.fst t (fieldBaseMap K L))
          (pullback.fst t (fieldBaseMap K L)) ≫ a := by
  let T' : Over (Spec (CommRingCat.of K)) := Over.mk t
  apply AlgebraicGeometry.Scheme.PicScheme.coverSelfSection_hom_ext
    (k := K) (k' := L) T'
  intro gamma
  change (AlgebraicGeometry.Scheme.PicScheme.coverSelfSection T' gamma).left ≫
      pullback.fst (pullback.fst t (fieldBaseMap K L))
          (pullback.fst t (fieldBaseMap K L)) ≫ a =
    (AlgebraicGeometry.Scheme.PicScheme.coverSelfSection T' gamma).left ≫
      pullback.snd (pullback.fst t (fieldBaseMap K L))
          (pullback.fst t (fieldBaseMap K L)) ≫ a
  rw [← Category.assoc, ← Category.assoc]
  change 𝟙 _ ≫ a =
    AlgebraicGeometry.Scheme.PicScheme.twistLeft T' gamma ≫ a
  rw [Category.id_comp]
  exact (ha gamma⁻¹).symm
