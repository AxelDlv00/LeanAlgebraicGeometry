noncomputable def ppTel {Z₁ Z₂ : Scheme.{u}} (a : Z₂ ⟶ Z₁) (p₁ : Z₁ ⟶ X) (p₂ : Z₂ ⟶ X)
    (w : a ≫ p₁ = p₂) :
    Scheme.Modules.pullback p₁ ⋙ Scheme.Modules.pullback a ≅ Scheme.Modules.pullback p₂ :=
  Scheme.Modules.pullbackComp a p₁ ≪≫ Scheme.Modules.pullbackCongr w
