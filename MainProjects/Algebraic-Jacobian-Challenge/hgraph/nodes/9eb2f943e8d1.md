theorem probe_isClosed_section_image {k : Type} [Field k] {X : Scheme}
    (f : X ⟶ Spec (.of k)) (p : Spec (.of k) ⟶ X) (hp : p ≫ f = 𝟙 _) :
    IsClosed ({p.base (IsLocalRing.closedPoint k)} : Set X) := by

  have h := p.isClosedEmbedding.isClosed_range
  rwa [Set.range_eq_singleton (fun x => congrArg p.base (Subsingleton.elim _ _))] at h
