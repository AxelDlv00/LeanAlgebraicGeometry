noncomputable def chartRes {Z : Scheme.{0}} (k : κ) (ρ : Z ⟶ (W k).toScheme)
    {u : Z ⟶ T.left} (h : ρ ≫ (W k).ι = u) :
    (Over.mk (u ≫ T.hom) : Over S) ⟶ Scheme.overRes T (W k) :=
  Over.homMk ρ (by subst h; exact (Category.assoc ρ (W k).ι T.hom).symm)
