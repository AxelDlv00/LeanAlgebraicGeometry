lemma actRes_mul (γ τ : L ≃ₐ[K] L) :
    ρ.actRes hU (γ * τ) = ρ.actRes hU τ ≫ ρ.actRes hU γ := by
  rw [actRes, Scheme.Hom.resLE_congr_hom (ρ.act_mul_hom γ τ)]
  exact (Scheme.Hom.resLE_comp_resLE _ _ _ _).symm
