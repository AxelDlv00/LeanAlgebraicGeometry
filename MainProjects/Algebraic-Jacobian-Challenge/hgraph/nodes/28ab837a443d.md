lemma pow_smul_mem {N : Submodule A M} {t : C} (h : ∀ a ∈ N, t • a ∈ N)
    (n : ℕ) {a : M} (ha : a ∈ N) : t ^ n • a ∈ N := by
  induction n with
  | zero => simpa using ha
  | succ n ih =>
    have hrw : t ^ (n + 1) • a = t • (t ^ n • a) := by
      rw [smul_smul, pow_succ, mul_comm]
    rw [hrw]
    exact h _ ih
