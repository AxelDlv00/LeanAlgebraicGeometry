    Multiplicative B →* (cechUnitsReduction ρ₁ ρ₂).ker where

    ⟨QuotientGroup.mk (truncExpUnit b.toAdd),

    change (QuotientGroup.mk (truncExpUnit ((1 : Multiplicative B).toAdd)) : _) = 1
