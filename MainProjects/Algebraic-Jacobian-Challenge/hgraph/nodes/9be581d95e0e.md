lemma presheaf_res_res {A B C : X.Opens} (j : B ⟶ A) (i : C ⟶ B)
    (k : C ⟶ A) (x : Γ(X, A)) :
    X.presheaf.map i.op (X.presheaf.map j.op x) = X.presheaf.map k.op x := by
  rw [← CommRingCat.comp_apply, ← Functor.map_comp, ← op_comp,
    Subsingleton.elim (i ≫ j) k]
