private lemma res_hom_eq_res (N : Y.Modules) {A B : Y.Opens} (i : B ⟶ A) (x : Γ(N, A)) :
    N.presheaf.map i.op x = N.presheaf.map (homOfLE (leOfHom i)).op x :=
  congrArg (fun (k : B ⟶ A) => N.presheaf.map k.op x) (Subsingleton.elim _ _)
