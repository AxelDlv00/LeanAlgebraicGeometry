theorem span_chart0 {A : Type u} [Semiring A] [Module A R0]
    (d : ℕ) (x : R0) (aa : I → R0)
    (hspan : Submodule.span A (Set.range aa) = ⊤) :
    Submodule.span A (Set.range (chart0 d x aa)) = ⊤ := by
  apply top_unique
  rw [← hspan]
  apply Submodule.span_mono
