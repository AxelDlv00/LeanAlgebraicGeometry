theorem powers_le_comap (b : A) (hb : ∀ g : G, g • b = b) (g : G) :
    Submonoid.powers b ≤
      Submonoid.comap (MulSemiringAction.toRingHom G A g) (Submonoid.powers b) := by
  intro x hx
  obtain ⟨n, rfl⟩ := hx
  refine ⟨n, ?_⟩
  change b ^ n = g • (b ^ n)
  rw [smul_pow', hb g]
