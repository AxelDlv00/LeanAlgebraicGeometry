docstring: '**An invertible module generated by a single element is free.**


  The point is that for an invertible module surjectivity upgrades to bijectivity:

  `Module.Invertible.bijective_of_surjective` (`Mathlib/RingTheory/PicardGroup.lean`)
  says a

  surjection *out of* an invertible module onto an invertible module is injective
  as well, and

  `R` is invertible over itself. So the tautological map `r ↦ r • m`, which is surjective

  exactly when `m` generates, is an isomorphism `R ≃ₗ M`.


  This is emphatically not true for general modules — `ℤ/2` is generated by one element
  over

  `ℤ` and is not free — and it is the step that makes the whole nilpotent-thickening
  argument

  short.'

theorem free_of_span_singleton_eq_top [Module.Invertible R M] {m : M}
    (h : Submodule.span R ({m} : Set M) = ⊤) : Module.Free R M := by
  have hsurj : Function.Surjective (LinearMap.toSpanSingleton R M m) := by
    intro x
    have hx : x ∈ Submodule.span R ({m} : Set M) := h ▸ Submodule.mem_top
    rw [Submodule.mem_span_singleton] at hx
    obtain ⟨r, hr⟩ := hx
    exact ⟨r, hr⟩
  exact Module.Free.of_equiv
    (LinearEquiv.ofBijective _ (Module.Invertible.bijective_of_surjective hsurj))

end Module.Invertible
