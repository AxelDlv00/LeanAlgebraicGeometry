private lemma iso_symm_fold_cancel {C : Type*} [Category C] {W X Y Z : C}
    (a : W ≅ X) (D : W ≅ Y) (g : X ≅ Y) (hD : D = a ≪≫ g) (h : Y ≅ Z) :
    a.symm ≪≫ D ≪≫ h = g ≪≫ h := by
  subst hD; rw [Iso.trans_assoc, Iso.symm_self_id_assoc]
