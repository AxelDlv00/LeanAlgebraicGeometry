theorem fieldGraph_jointlySurjective :
    ∀ x : pullback (fieldBaseMap K L) (fieldBaseMap K L),
      ∃ (gamma : L ≃ₐ[K] L) (y : Spec (CommRingCat.of L)),
        fieldGraph K L gamma y = x := by
  intro x
  let e := AlgebraicGeometry.Scheme.PicScheme.selfTensorSpecCoproduct K L ≪≫
    (pullbackSpecIso K L L).symm
  obtain ⟨gamma, y, hy⟩ := (Scheme.sigmaOpenCover
    (fun _ : L ≃ₐ[K] L ↦ Spec (CommRingCat.of L))).exists_eq (e.inv x)
  refine ⟨gamma, y, ?_⟩
  change (Sigma.ι (fun _ : L ≃ₐ[K] L ↦ Spec (CommRingCat.of L)) gamma ≫
    e.hom) y = x
  rw [← Scheme.Hom.comp_apply, hy]
  change e.hom (e.inv x) = x
  rw [← Scheme.Hom.comp_apply]
  simp
