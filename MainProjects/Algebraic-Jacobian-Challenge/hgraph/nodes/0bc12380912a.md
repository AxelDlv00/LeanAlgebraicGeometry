theorem generic_generate_singleton_over {Y Z : Over X} (f : Z ⟶ Y) :
    Sieve.generate (Presieve.singleton f) =
      (Sieve.overEquiv Y).symm (Sieve.generate (Presieve.singleton f.left)) := by
  ext W g
  constructor
  · rintro ⟨V, a, b, hb, hfac⟩
    cases hb
    rw [Sieve.overEquiv_symm_iff]
    refine ⟨_, a.left, _, Presieve.singleton.mk, ?_⟩
    rw [← hfac]
    rfl
  · intro hg
    rw [Sieve.overEquiv_symm_iff] at hg
    obtain ⟨V, a, b, hb, hfac⟩ := hg
    cases hb
    refine ⟨_, Over.homMk a (by
      rw [← Over.w g, ← hfac, Category.assoc, Over.w f]), f,
      Presieve.singleton.mk, ?_⟩
    apply Over.OverMorphism.ext
    change a ≫ f.left = Over.Hom.left g
    exact hfac

/-- And then the landed lemma is that generic fact, instantiated. -/
example {k : Type u} [Field k] {k' : Type u} [Field k'] [Algebra k k']

        (Sieve.generate (Presieve.singleton
          (pullback.fst T.hom (AlgebraicGeometry.Scheme.PicScheme.specMapAlgebra k k')))) :=
  generic_generate_singleton_over _
