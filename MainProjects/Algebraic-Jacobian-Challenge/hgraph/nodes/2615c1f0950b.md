def dehomogenize (i : J) :
    MvPolynomial J R →ₐ[R] MvPolynomial {j : J // j ≠ i} R := by
  classical
  exact MvPolynomial.aeval fun j ↦ if h : j = i then 1 else X ⟨j, h⟩

@[simp]
