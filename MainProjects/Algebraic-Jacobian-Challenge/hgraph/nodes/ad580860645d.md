private lemma sum_univ_comp {C : Type*} [Category C] [Preadditive C]
    {P Q R : C} {J : Type*} [Fintype J] (f : J → (P ⟶ Q)) (g : Q ⟶ R) :
    (∑ j, f j) ≫ g = ∑ j, f j ≫ g := by
  simpa using Preadditive.sum_comp Finset.univ f g
