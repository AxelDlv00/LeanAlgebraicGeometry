theorem smooth_tensorObj_self (A : Over (Spec (.of kbar))) [Smooth A.hom] :
    Smooth (A ⊗ A).hom := by
  rw [Over.tensorObj_hom]
  exact inferInstanceAs (Smooth (pullback.fst A.hom A.hom ≫ A.hom))
