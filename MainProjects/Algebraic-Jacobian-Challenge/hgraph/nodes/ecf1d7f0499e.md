        (coverMap (k := k) (k' := k') T)).left,

          (coverMap (k := k) (k' := k') T)).op x
      = (picEt C).map (Limits.pullback.snd (coverMap (k := k) (k' := k') T)
          (coverMap (k := k) (k' := k') T)).op x :=
