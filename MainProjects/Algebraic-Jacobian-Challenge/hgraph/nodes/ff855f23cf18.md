theorem polyToAway_surjective (i : J) :
    Function.Surjective (polyToAway R J i) := by
  intro z
  have hz : z ∈
      (⊤ : Submodule (homogeneousSubmodule J R 0)
        (Away (homogeneousSubmodule J R) (X i))) := trivial
  rw [← Away.span_mk_prod_pow_eq_top (X_mem_deg_one R J i) X
    (adjoin_gradeZero_range_X R J) (fun _ ↦ 1)
    (fun l ↦ X_mem_deg_one R J l)] at hz
  induction hz using Submodule.span_induction with
  | mem w hw =>
      obtain ⟨a, e, hae, rfl⟩ := hw
      refine ⟨∏ j : {j : J // j ≠ i}, X j ^ e j, ?_⟩
      rw [map_prod]
      simp_rw [map_pow, polyToAway_X]
      exact (away_mk_prod_eq R J i a e (degree_eq J i hae) _).symm
  | zero => exact ⟨0, map_zero _⟩
  | add p q _ _ hp hq =>
      obtain ⟨p, rfl⟩ := hp
      obtain ⟨q, rfl⟩ := hq
      exact ⟨p + q, map_add _ _ _⟩
  | smul c p _ hp =>
      obtain ⟨p, rfl⟩ := hp
      obtain ⟨r, rfl⟩ := exists_algebraMap_eq_gradeZero R J c
      exact ⟨r • p, by rw [map_smul, IsScalarTower.algebraMap_smul]⟩

omit [Finite J] in
