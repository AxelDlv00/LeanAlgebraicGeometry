lemma horseshoeτ_cocycle (n : ℕ) :
    I_C.cocomplex.d n (n + 1) ≫ horseshoeτ hses I_A I_C (n + 1) =
      -(horseshoeτ hses I_A I_C n ≫ I_A.cocomplex.d (n + 1) (n + 2)) :=
  (twistPair hses I_A I_C n).2.2
