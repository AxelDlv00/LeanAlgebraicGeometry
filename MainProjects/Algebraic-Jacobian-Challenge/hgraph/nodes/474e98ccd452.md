theorem sub_pointDivisor_apply_of_ne (D : X.WeilDivisor) {P Q : X.PrimeDivisor}
    (h : Q ≠ P) :
    (show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) Q =
      (show X.PrimeDivisor →₀ ℤ from D) Q := by
  rw [show (show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) Q =
        (show X.PrimeDivisor →₀ ℤ from D) Q -
          (show X.PrimeDivisor →₀ ℤ from pointDivisor P) Q from Finsupp.sub_apply _ _ _,
    pointDivisor, Finsupp.single_eq_of_ne h, sub_zero]
