lemma isZero_biproduct {C : Type*} [Category C] [HasZeroMorphisms C] {J : Type*}
    (Z : J → C) [HasBiproduct Z] (h : ∀ j, IsZero (Z j)) : IsZero (⨁ Z) := by
  rw [IsZero.iff_id_eq_zero]
  apply biproduct.hom_ext
  intro j
  exact (h j).eq_of_tgt _ _
