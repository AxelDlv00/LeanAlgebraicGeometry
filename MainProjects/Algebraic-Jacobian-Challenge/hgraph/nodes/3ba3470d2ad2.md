lemma scalarEnd_zero : scalarEnd (0 : Γ(X, ⊤)) = 0 := by

  ext Y

          (0 : SheafOfModules.unit X.ringCatSheaf ⟶ SheafOfModules.unit X.ringCatSheaf)).app Y
