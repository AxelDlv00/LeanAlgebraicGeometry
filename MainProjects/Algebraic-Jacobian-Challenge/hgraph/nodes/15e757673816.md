private theorem away_mk_prod_eq (i : J) (a : ℕ) (e : J → ℕ)
    (hae : e i + ∑ j : {j : J // j ≠ i}, e j = a)
    (H : (∏ l, (X l : MvPolynomial J R) ^ e l) ∈
      homogeneousSubmodule J R (a • 1)) :
    Away.mk (homogeneousSubmodule J R) (X_mem_deg_one R J i) a
        (∏ l, X l ^ e l) H =
      ∏ j : {j : J // j ≠ i}, chartCoord R J i j ^ e j := by
  apply (show Function.Injective (algebraMap
      (Away (homogeneousSubmodule J R) (X i))
      (Localization.Away (X i : MvPolynomial J R))) from
    HomogeneousLocalization.val_injective _)
  simp only [map_prod, map_pow, HomogeneousLocalization.algebraMap_apply,
    Away.val_mk, chartCoord, Localization.mk_pow, Localization.mk_prod]
  rw [← (Equiv.optionSubtypeNe i).prod_comp, Fintype.prod_option]
  simp only [Equiv.optionSubtypeNe_none, Equiv.optionSubtypeNe_some]
  rw [Localization.mk_eq_mk_iff, Localization.r_iff_exists]
  refine ⟨1, ?_⟩
  push_cast
  rw [Finset.prod_pow_eq_pow_sum, ← hae, pow_add]
  ring
