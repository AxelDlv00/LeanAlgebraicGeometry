noncomputable def galTwistMor (T : Over (Spec (CommRingCat.of k))) (γ : k' ≃ₐ[k] k') :
    (twistTestFunctor (k := k) γ).obj (baseTest (k' := k') T) ⟶ baseTest (k' := k') T :=
  Over.homMk (pullbackGalMap k k' T.hom γ) (pullbackGalMap_snd k k' T.hom γ)
