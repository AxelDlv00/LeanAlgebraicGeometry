noncomputable def symCocone {T : K} (h : (∏ᶜ (fun _ : Fin n => C)) ⟶ T)
    (hsym : ∀ σ : Equiv.Perm (Fin n), MonObj.permAut C σ ≫ h = h) :
    Cocone (permDiagram C n) where
  pt := T
  ι :=
    { app := fun _ => h
      naturality := by
        intro X Y f
        change MonObj.permAut C (f : Equiv.Perm (Fin n))⁻¹ ≫ h = h ≫ 𝟙 T
        rw [Category.comp_id]
        exact hsym _ }
