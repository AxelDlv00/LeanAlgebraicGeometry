lemma comp_succAbove_zero {α : Type*} (σ : Fin 2 → α) :
    σ ∘ (0 : Fin 2).succAbove = fun _ : Fin 1 => σ 1 := by
  funext k
  have hk : k = 0 := Subsingleton.elim k 0
  subst hk
  rfl
