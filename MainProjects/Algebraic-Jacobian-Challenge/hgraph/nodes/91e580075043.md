theorem finrank_pi_aut : finrank L ((L ≃ₐ[K] L) → L) = finrank K L := by
  haveI : Fintype (L ≃ₐ[K] L) := AlgEquiv.fintype K L
  rw [Module.finrank_fintype_fun_eq_card, ← Nat.card_eq_fintype_card,
    IsGalois.card_aut_eq_finrank K L]
