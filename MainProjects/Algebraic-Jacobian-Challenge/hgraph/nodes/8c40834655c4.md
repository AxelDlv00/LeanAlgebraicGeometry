theorem exists_fg_mem (a : A) : ∃ S : Subalgebra k A, S.FG ∧ a ∈ S :=
  ⟨Algebra.adjoin k {a}, by simpa using Subalgebra.fg_adjoin_finset {a},
    Algebra.self_mem_adjoin_singleton k a⟩
