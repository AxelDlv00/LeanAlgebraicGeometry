def serreTwist (m : ℕ) :
    (Proj (homogeneousSubmodule n₀ (ULift.{0} ℤ))).Modules :=
  (Scheme.Modules.pullback (inv (basicOpenCover n₀).fromGlued)).obj
    (serreTwistGlued n₀ m)
