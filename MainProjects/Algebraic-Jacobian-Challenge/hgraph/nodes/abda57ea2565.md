theorem relMatrix_eq_zero_of_flat (P : MatrixPresentation R M e m)
    [Module.Flat R M]
    (h : ∀ p : PrimeSpectrum R, Module.rankAtStalk M p = e) :
    P.relMatrix = 0 := by
  haveI := P.finitePresentation
  haveI : Module.Projective R M := Module.Flat.projective_of_finitePresentation
  -- split the projection
  obtain ⟨s, hs⟩ := Module.projective_lifting_property P.proj LinearMap.id
    P.surjective_proj
  have hs' : ∀ x : M, P.proj (s x) = x := fun x ↦ congr($hs x)
  -- `R^e ≃ M × ker` via the splitting
  let E : (Fin e → R) ≃ₗ[R] M × LinearMap.ker P.proj :=
    { toFun := fun x ↦ (P.proj x, ⟨x - s (P.proj x), by simp [hs']⟩)
      map_add' := fun x y ↦ by
        ext <;> simp [add_sub_add_comm]
      map_smul' := fun r x ↦ by
        ext <;> simp [smul_sub]
      invFun := fun nk ↦ s nk.1 + nk.2.1
      left_inv := fun x ↦ by simp
      right_inv := fun nk ↦ by
        have hk : P.proj nk.2.1 = 0 := nk.2.2
        ext <;> simp [hs', hk] }
  -- the kernel is finite projective (a direct summand of `R^e`)
  haveI : Module.Projective R (LinearMap.ker P.proj) := by
    refine Module.Projective.of_split (LinearMap.ker P.proj).subtype
      ((LinearMap.snd R M (LinearMap.ker P.proj)) ∘ₗ E.toLinearMap) ?_
    refine LinearMap.ext fun ⟨x, hx⟩ ↦ Subtype.ext ?_
    have hx0 : P.proj x = 0 := hx
    simp [E, hx0]
  haveI : Module.Finite R (LinearMap.ker P.proj) := by
    have hsurj : Function.Surjective
        ((LinearMap.snd R M (LinearMap.ker P.proj)) ∘ₗ E.toLinearMap) := fun k ↦
      ⟨E.symm (0, k), by simp⟩
    exact Module.Finite.of_surjective _ hsurj
  -- rank bookkeeping: `e = e + rankAtStalk (ker) p` at every prime
  have hrank : ∀ p : PrimeSpectrum R,
      Module.rankAtStalk (R := R) (LinearMap.ker P.proj) p = 0 := by
    intro p
    haveI : Nontrivial R := nontrivial_of_primeSpectrum p
    have h1 : Module.rankAtStalk (R := R) (Fin e → R) p = e := by
      simp
    have h2 : Module.rankAtStalk (R := R) (Fin e → R) p
        = Module.rankAtStalk M p + Module.rankAtStalk (LinearMap.ker P.proj) p := by
      rw [congrFun (Module.rankAtStalk_eq_of_equiv E) p,
        congrFun (Module.rankAtStalk_prod (R := R) (M := M) (LinearMap.ker P.proj)) p,
        Pi.add_apply]
    have h3 := h p
    omega
  -- everywhere-rank-zero finite flat module is zero
  haveI : Subsingleton (LinearMap.ker P.proj) :=
    Module.rankAtStalk_eq_zero_iff_subsingleton.mp (funext hrank)
  -- hence the kernel is trivial and the relation matrix vanishes
  have hker : LinearMap.ker P.proj = ⊥ := by
    rw [Submodule.eq_bot_iff]
    intro x hx
    simpa using congrArg Subtype.val (Subsingleton.elim
      (⟨x, hx⟩ : LinearMap.ker P.proj) ⟨0, (LinearMap.ker P.proj).zero_mem⟩)
  have hrange : LinearMap.range P.relMatrix.mulVecLin = ⊥ := by
    rw [← P.ker_proj]
    exact hker
  rw [LinearMap.range_eq_bot] at hrange
  ext i j
  have := congr($hrange (Pi.single j 1) i)
  simpa [Matrix.mulVecLin_apply, Matrix.mulVec_single_one] using this

end HardDirection
