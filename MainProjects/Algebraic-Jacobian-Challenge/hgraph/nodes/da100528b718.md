theorem evalMap_surjective
    (hledger : ∀ D : X.WeilDivisor, chi k U₀ U₁ D = chi k U₀ U₁ 0 + degK k D)
    (D : X.WeilDivisor) (P : X.PrimeDivisor)
    (hD : Subsingleton (H1Mod k U₀ U₁ D))
    (hDP : Subsingleton (H1Mod k U₀ U₁ (D - pointDivisor P)))
    [Module.Finite k (sectionSub k ⊤ D)]
    [Module.Finite k (localStepTgt k P 1)] :
    Function.Surjective (evalMap k D P) := by
  -- the domain is finite-dimensional (a quotient of a finite-dimensional space)
  haveI : Module.Finite k (localStepDom k ⊤ (D - pointDivisor P) D) := inferInstance
  haveI : Module.Finite k
      (localStepTgt k P (-(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P)) := by
    have hrank := finrank_localStepTgt k P
      (-(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P)
    obtain ⟨t, ht, htord⟩ :=
      exists_order_eq P (1 - -(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P)
    exact Module.Finite.equiv
      (LinearEquiv.ofBijective (residueShift k t ht htord)
        ⟨residueShift_injective k t ht htord,
          residueShift_surjective k t ht htord⟩).symm
  -- the two dimensions agree
  have hdim : Module.finrank k (localStepDom k ⊤ (D - pointDivisor P) D) =
      Module.finrank k
        (localStepTgt k P (-(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P)) := by
    rw [finrank_localStepTgt]
    have hrn := finrank_localStepDom k ⊤
      (sectionSub_mono k ⊤ (sub_pointDivisor_le D P))
    have hdrop := ell_sub_ell_sub_pointDivisor_eq k U₀ U₁ hledger D P hD hDP
    simp only [ell] at hdrop
    omega
  exact (LinearMap.injective_iff_surjective_of_finrank_eq_finrank hdim).mp
    (evalMap_injective k D P)

end EvalSurjective
