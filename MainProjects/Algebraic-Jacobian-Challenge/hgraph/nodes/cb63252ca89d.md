theorem Rel.hasFiberDeg_iff {T : Over S} {x y : DivFamily π T} (h : x.Rel y)
    (d : ℕ) : x.HasFiberDeg d ↔ y.HasFiberDeg d :=
  forall_congr' fun t => by rw [h.fiberDeg_eq t]
