noncomputable def glueResHom {Z' Z : Scheme.{0}} (p : Z' ⟶ Z) (u : Z ⟶ T.left) :
    (Over.mk ((p ≫ u) ≫ T.hom) : Over S) ⟶ Over.mk (u ≫ T.hom) :=
  Over.homMk p (Category.assoc p u T.hom).symm
