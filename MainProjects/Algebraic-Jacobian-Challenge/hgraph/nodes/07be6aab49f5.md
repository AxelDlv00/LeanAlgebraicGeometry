def Hom.FiberH1Vanishing (q : X ⟶ S) (L : X.Modules) (t : S) : Prop :=
  ∃ V : (q.fiber t).AffineCoverMVSquare,
    Function.Surjective ⇑(V.moduleSectionDiff (q.fiberModule t L))
