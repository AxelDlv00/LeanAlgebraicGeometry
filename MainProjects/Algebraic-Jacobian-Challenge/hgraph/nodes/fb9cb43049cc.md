    letI : Module Γ(S, V) Γ(F, W₂) := Module.compHom _ (f.appLE V W₂ p₂).hom

  letI : Module Γ(S, V) Γ(F, W₂) := Module.compHom _ (f.appLE V W₂ p₂).hom
  letI : Module Γ(S, V) Γ(F, W₁) := Module.compHom _ (f.appLE V W₁ p₁).hom
