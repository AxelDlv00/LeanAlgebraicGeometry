def smulALinear (A : Type*) [CommRing A] {C : Type*} [CommRing C]
    {M : Type*} [AddCommGroup M] [Module A M] [Module C M] [SMulCommClass A C M]
    (c : C) : M →ₗ[A] M where
  toFun := fun m => c • m
  map_add' := smul_add c
  map_smul' := fun a m => (smul_comm a c m).symm
