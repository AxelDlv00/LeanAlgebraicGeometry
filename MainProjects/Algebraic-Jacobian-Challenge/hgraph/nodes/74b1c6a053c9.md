def h0Map (h : d ∘ₗ a0 = a1 ∘ₗ k) : ker k →ₗ[R] ker d :=
  a0.restrict fun x hx => by
    have hx' := LinearMap.congr_fun h x
    simp only [LinearMap.comp_apply] at hx'
    simp only [mem_ker] at hx ⊢
    rw [hx', hx, map_zero]
