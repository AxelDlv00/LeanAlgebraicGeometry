noncomputable def coprodLift (f : (i : ι) → (A i →ₐ[R] S)) :
    (⨂[R] i, A i) →ₐ[R] S :=
  liftAlgHom (coprodMultilinear f)
    (by simp)
    (by
      intro x y
      simp only [coprodMultilinear_apply, Pi.mul_apply, map_mul, Finset.prod_mul_distrib])
