lemma entry_chain {C : Type*} [Category C] {Y₀ Y₁ Y₂ Y₃ T : C}
    (a : Y₀ ⟶ Y₁) (b : Y₁ ⟶ Y₂) (ci : Y₂ ⟶ Y₃) (pr : Y₃ ⟶ T) (s : Y₀ ⟶ Y₂)
    (h : a ≫ b = s) : (a ≫ b ≫ ci) ≫ pr = s ≫ (ci ≫ pr) := by
