theorem over_hom_ext_of_mono {C : Type*} [Category C] {T : C} {A B : Over T}
    [Mono B.hom] (a b : A ⟶ B) : a = b := by
  ext
  exact (cancel_mono B.hom).mp (by rw [Over.w, Over.w])
