theorem isZero_of_isEmpty {M : Y.Modules} (hY : IsEmpty (Y : Type u)) : IsZero M :=
  isZero_of_forall_subsingleton_sections fun V => by
    haveI := hY
    exact Module.subsingleton Γ(Y, V) Γ(M, V)
