noncomputable def polyModule {r : ℕ} (t : Fin r → Module.End κ M)
    (hcomm : ∀ i j, Commute (t i) (t j)) : Module (MvPolynomial (Fin r) κ) M :=
  Module.compHom M (polyEndHom t hcomm)
