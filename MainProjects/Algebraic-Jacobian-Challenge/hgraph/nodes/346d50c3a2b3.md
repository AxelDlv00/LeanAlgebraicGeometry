theorem iSup_projectiveOpen : (⨆ b, G.projectiveOpen b) = ⊤ := by
  have hpull : (pi.left ⁻¹ᵁ D.V₀) ⊔ (pi.left ⁻¹ᵁ D.V₁) = ⊤ := by
