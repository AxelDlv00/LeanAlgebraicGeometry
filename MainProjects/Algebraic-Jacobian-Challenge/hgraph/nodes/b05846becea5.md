    def strip_comments(t):                    # a `%` comment can hold \end{theorem}
        out = []
        for line in t.split('\n'):
            i = 0
            while i < len(line):
                if line[i] == '\\': i += 2; continue
                if line[i] == '%':  line = line[:i]; break
                i += 1
            out.append(line)
        return '\n'.join(out)
