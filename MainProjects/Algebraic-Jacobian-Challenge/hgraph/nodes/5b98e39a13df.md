noncomputable def permAut (C : K) {n : ℕ} (σ : Equiv.Perm (Fin n)) :
    (∏ᶜ (fun _ : Fin n => C)) ⟶ (∏ᶜ (fun _ : Fin n => C)) :=
  Pi.lift (fun i => Pi.π (fun _ : Fin n => C) (σ i))
