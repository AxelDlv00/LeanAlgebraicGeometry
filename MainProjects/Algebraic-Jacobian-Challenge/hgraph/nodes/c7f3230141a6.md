    [HasFiniteCoproducts C] (p : ℕ) (F : (Fin (p + 2) → ι) → C) :
    (∐ fun i : ι => ∐ fun τ : Fin (p + 1) → ι => F (Fin.cons i τ))
      ≅ ∐ fun σ : Fin (p + 2) → ι => F σ :=
