theorem ell_eq_of_bound
    (hledger : ∀ D : X.WeilDivisor, chi k U₀ U₁ D = chi k U₀ U₁ 0 + degK k D)
    {D : X.WeilDivisor} (hvan : Subsingleton (H1Mod k U₀ U₁ D)) :

  have hh1 := h1dim_eq_zero_of_subsingleton k U₀ U₁ hvan
  have := hledger D
  rw [chi, hh1] at this
