private lemma adj_unit_map_counit {C D : Type*} [Category C] [Category D] {L : C ⥤ D} {R : D ⥤ C}
    (adj : L ⊣ R) (P : C) (M : D) (k : L.obj P ⟶ M) :
    L.map (adj.unit.app P ≫ R.map k) ≫ adj.counit.app M = k :=
  (adj.homEquiv P M).left_inv k
