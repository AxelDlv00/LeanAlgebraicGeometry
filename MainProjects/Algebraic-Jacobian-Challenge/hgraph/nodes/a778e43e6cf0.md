    ∃ b : ℤ, ∀ {x : Y} (hx : x ≠ genericPoint Y) (D : Y.CurveDivisor),

      (Sheaf.h0 (Y.divisorSheaf K D) : ℤ) =
        Sheaf.h0 (Y.divisorSheaf K (D - CurveDivisor.single hx 1)) + Y.residueDeg K x := by
