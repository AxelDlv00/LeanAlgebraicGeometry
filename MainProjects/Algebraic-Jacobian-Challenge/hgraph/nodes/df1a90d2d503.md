def IsEquivariant {X' : Scheme.{u}} {f' : X' ⟶ Spec (CommRingCat.of L)}
    (ρ : SemilinearGalAction K L X f) (ρ' : SemilinearGalAction K L X' f')
    (h : X ⟶ X') : Prop :=
  ∀ γ : L ≃ₐ[K] L, (ρ.act γ).hom ≫ h = h ≫ (ρ'.act γ).hom
