private lemma comp4_solve_last {𝒞 : Type*} [Category 𝒞] {a b c d e : 𝒞}

    (hE : A.hom ≫ B.hom ≫ Cc.hom ≫ D = E) :
    D = Cc.inv ≫ B.inv ≫ A.inv ≫ E := by
