theorem compactSpace_of_universallyClosed {R : Type u} [CommRing R] {X : Scheme.{u}}
    (f : X ⟶ Spec (.of R)) [UniversallyClosed f] : CompactSpace X :=
  QuasiCompact.compactSpace_of_compactSpace f
