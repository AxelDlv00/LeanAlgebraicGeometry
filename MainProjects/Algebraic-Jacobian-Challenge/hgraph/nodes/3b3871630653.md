noncomputable def fiberLattice (D : Y.CurveDivisor) (n : ℕ) : Submodule K Y.functionField :=
  divisorSections K (D + n • fiberWeilDivisor π) (fiberChart₀ π)
    ⊔ divisorSections K (D + n • fiberWeilDivisor π) (fiberChart₁ π)
