noncomputable def representingScheme {V : S.Modules} {r d : ℕ}
    (hV : SheafOfModules.IsLocallyFreeOfRank V r)
    (hd : 1 ≤ d) (hdr : d ≤ r) : Over S :=
  Classical.choose (representable hV hd hdr)
