private lemma comp_p1Eval (r : B →+* B') (b₀ b₁ : B) :
    r.comp (p1Eval b₀ b₁) = p1Eval (r b₀) (r b₁) := by
  apply MvPolynomial.ringHom_ext
  · intro z
    simp [p1Eval]
  · intro i
    rcases i with ⟨i⟩
    fin_cases i <;> simp [p1Eval]
