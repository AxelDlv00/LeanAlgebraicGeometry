noncomputable def AffineCoverMVSquare.moduleSectionRes (V : X.AffineCoverMVSquare)
    (M : X.Modules) :
    Γ(M, (⊤ : X.Opens)) →+ Γ(M, V.U₁) × Γ(M, V.U₂) :=
  ((M.presheaf.map (homOfLE (le_top : V.U₁ ≤ ⊤)).op).hom).prod
    ((M.presheaf.map (homOfLE (le_top : V.U₂ ≤ ⊤)).op).hom)
