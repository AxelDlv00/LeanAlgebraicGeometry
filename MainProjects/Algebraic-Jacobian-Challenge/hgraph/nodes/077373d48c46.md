theorem symTensorPowSubalgebra_toSubring :
    letI := permMulSemiringAction R (ι := ι) A
    (symTensorPowSubalgebra R A).toSubring
      = FixedPoints.subring (⨂[R] _ : ι, A) (Equiv.Perm ι) := by
  letI := permMulSemiringAction R (ι := ι) A
  ext x
  constructor
  · intro h g
    exact h g⁻¹
  · intro h σ
    have hx : permAlgHom R A σ⁻¹⁻¹ x = x := h σ⁻¹
    rwa [inv_inv] at hx
