def GlueCompat : Prop :=
  ∀ k l : κ, ∃ α :
    (Scheme.Modules.pullback ((covGD W hW).f k l)).obj (y k).F ≅
    (Scheme.Modules.pullback ((covGD W hW).t k l ≫
      (covGD W hW).f l k)).obj (y l).F,
    IsGlueIso W y k l rfl (glueSnd_ι W hW k l) α
