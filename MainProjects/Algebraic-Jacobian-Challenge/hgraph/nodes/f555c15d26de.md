private lemma iso_cancel_left {C : Type*} [Category C] {W X Y Z : C}
    (a : W ≅ X) (f : X ≅ Z) (g : X ≅ Y) (h : Y ≅ Z) (H : f = g ≪≫ h) :
    a ≪≫ f = (a ≪≫ g) ≪≫ h := by rw [H, Iso.trans_assoc]
