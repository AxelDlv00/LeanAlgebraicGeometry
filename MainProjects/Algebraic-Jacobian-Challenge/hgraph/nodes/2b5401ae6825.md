theorem galoisSelfTensorHom_surjective :
    Function.Surjective (galoisSelfTensorHom K L) := by
  classical
  set ι := Fin (finrank K L) with hι
  let b : Basis ι K L := Module.finBasis K L
  -- the range of the `L`-linear map underlying `galoisSelfTensorHom`
  set S : Submodule L ((L ≃ₐ[K] L) → L) :=
    LinearMap.range (galoisSelfTensorHom K L).toLinearMap with hS
  have hspan : Submodule.span L
      (Set.range (fun i : ι => (fun γ => γ (b i) : (L ≃ₐ[K] L) → L))) = ⊤ :=
    galoisCol_span b
  have hsub : Submodule.span L
      (Set.range (fun i : ι => (fun γ => γ (b i) : (L ≃ₐ[K] L) → L))) ≤ S := by
    rw [Submodule.span_le]
    rintro x ⟨i, rfl⟩
    exact ⟨(1 : L) ⊗ₜ[K] b i, by ext γ; simp⟩
  have htop : S = ⊤ := top_le_iff.mp (hspan ▸ hsub)
  intro y
  have : y ∈ S := htop ▸ Submodule.mem_top
  obtain ⟨x, hx⟩ := this
  exact ⟨x, hx⟩
