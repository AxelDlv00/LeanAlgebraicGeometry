noncomputable def tensorPowerCofan : Cofan (fun _ : Fin n => CommRingCat.mkUnder k A) :=
  Cofan.mk (CommRingCat.mkUnder k (⨂[k] _ : Fin n, A))
    (fun i => (singleAlgHom (R := k) (A := fun _ : Fin n => A) i).toUnder)
