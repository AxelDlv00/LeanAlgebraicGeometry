def Scheme.Hom.IsHQuasiProjectiveWith {X S : Scheme.{0}} (π : X ⟶ S)
    (L : X.Modules) : Prop :=
  ∃ (n : Type) (_ : Finite n) (i : X ⟶ ℙ(n; S)),
    IsImmersion i ∧ QuasiCompact i ∧ i ≫ (ℙ(n; S) ↘ S) = π ∧
      Nonempty (L ≅ (Scheme.Modules.pullback i).obj
        (ProjectiveSpace.twistingSheaf n S 1))
