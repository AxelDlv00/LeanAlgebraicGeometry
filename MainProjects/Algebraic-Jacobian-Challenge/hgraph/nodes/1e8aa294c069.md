theorem specGal_comp (γ : k' ≃ₐ[k] k') :
    specGal γ ≫ specMapAlgebra k k' = specMapAlgebra k k' := by
  rw [specMapAlgebra, ← Spec.map_comp]
  congr 1
  ext x
  exact γ.commutes x
