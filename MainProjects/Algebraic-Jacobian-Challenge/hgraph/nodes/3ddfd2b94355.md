noncomputable def universalQuotient {V : S.Modules} {r d : ℕ}
    (hV : SheafOfModules.IsLocallyFreeOfRank V r)
    (hd : 1 ≤ d) (hdr : d ≤ r) :
    LocallyFreeQuotient V d (representingScheme hV hd hdr) :=
  Quotient.out (universalClass hV hd hdr)
