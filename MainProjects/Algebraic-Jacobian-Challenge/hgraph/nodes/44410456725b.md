theorem isoAffineSpace_hom_over :
    (isoAffineSpace J i S).hom ≫ (𝔸({j : J // j ≠ i}; S) ↘ S) =
      affineChartAt J i S ↘ S := by
  change ((flattenIso J i S ≪≫ asIso (changeChartHom J i S)).hom) ≫
      (𝔸({j : J // j ≠ i}; S) ↘ S) = affineChartAt J i S ↘ S
  rw [Iso.trans_hom, Category.assoc, asIso_hom, changeChartHom_over,
    flattenIso_hom_over]
  rfl
