noncomputable def representableByTerminal_of_unique (F : Cᵒᵖ ⥤ Type w) {Y : C}
    (hY : Limits.IsTerminal Y) (h : ∀ T : Cᵒᵖ, Unique (F.obj T)) :
    F.RepresentableBy Y :=
  representableByTerminal F hY (fun T => ⟨(h T).default⟩) (fun T => (h T).instSubsingleton)
