class HasDedekindChart (X : Scheme.{u}) [IsIntegral X] : Prop where
  /-- Every nonempty affine chart is a Dedekind domain. -/
  isDedekindDomain : ∀ (U : X.Opens), IsAffineOpen U → [Nonempty U] →
    IsDedekindDomain Γ(X, U)
