def transitionMap {S T : Subalgebra k A} (h : S ≤ T) :
    R ⊗[k] S →ₐ[k] R ⊗[k] T :=
  Algebra.TensorProduct.map (AlgHom.id k R) (Subalgebra.inclusion h)

@[simp]
