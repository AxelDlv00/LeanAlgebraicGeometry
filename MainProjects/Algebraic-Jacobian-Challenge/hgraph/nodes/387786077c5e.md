theorem Hom.fiberEulerIndex_congr (f : X ⟶ S) (s : S)
    {M N : X.Modules} (e : M ≅ N) :
    f.fiberEulerIndex s M = f.fiberEulerIndex s N :=
  chi_toModuleKSheafOfModules_congr
    (Over.mk (f.fiberToSpecResidueField s))
    ((Modules.pullback (f.fiberι s)).mapIso e)
