lemma gamma_pi_ext {x y : Γ(∏ᶜ N, ⊤)}
    (h : ∀ i, (Pi.π N i).app ⊤ x = (Pi.π N i).app ⊤ y) : x = y := by
  have hinj : Function.Injective (gammaPiHom N) :=
    (ModuleCat.mono_iff_injective _).mp inferInstance
  apply hinj
  funext i
  rw [gammaPiHom_apply, gammaPiHom_apply]
  exact h i
