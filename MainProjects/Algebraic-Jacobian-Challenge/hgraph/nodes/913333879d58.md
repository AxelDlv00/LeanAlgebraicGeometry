lemma GluedPoint.ext' {T : Scheme.{0}} {p q : GluedPoint F Y R T}
    (ha : p.a = q.a)
    (hγ : ∀ i, p.γ i = eqToHom (by rw [ha]) ≫ q.γ i) : p = q := by
  obtain ⟨a, γ, c⟩ := p
  obtain ⟨a', γ', c'⟩ := q
  dsimp only at ha
  subst ha
  simp only [eqToHom_refl, Category.id_comp] at hγ
  obtain rfl : γ = γ' := funext hγ
  rfl
