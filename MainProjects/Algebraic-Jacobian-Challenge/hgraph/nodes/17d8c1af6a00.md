theorem algebraMap_mem_fixedAway (b : A) (hb : ∀ g : G, g • b = b) {a : A}
    (ha : ∀ g : G, g • a = a) :
    algebraMap A (Localization.Away b) a ∈ fixedAway b hb := by
  intro g
  rw [awayMap_algebraMap b hb, ha g]
