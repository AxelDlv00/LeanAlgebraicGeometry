private lemma coeffAt_divOf_pow (v : Y.functionFieldˣ) (n : ℕ) {x : Y}
    (hx : x ≠ genericPoint Y) :
    coeffAt hx (Scheme.divOf (Y ↘ Spec (CommRingCat.of K)) (v ^ n)) =
      (n : ℤ) * coeffAt hx (Scheme.divOf (Y ↘ Spec (CommRingCat.of K)) v) := by
  induction n with
