noncomputable def gluedQuotientProjection
    [FiniteDimensional K L] [IsGalois K L] [HasStableAffineCover K L ρ] :
    X ⟶ gluedQuotient ρ :=
  (sourceOpenCover ρ).glueMorphisms (quotientChartProjection ρ)
    (quotientChartProjection_compatible ρ)
