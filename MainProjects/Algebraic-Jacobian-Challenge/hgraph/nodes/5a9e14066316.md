def Peel (D D' : X.WeilDivisor) : Prop :=
  ∀ x ∈ sectionSub k (U₀ ⊓ U₁) D', ∃ y ∈ sectionSub k (U₀ ⊓ U₁) D,
    x - y ∈ coboundarySub k U₀ U₁ D'
