theorem galoisSelfTensorHom_bijective :
    Function.Bijective (galoisSelfTensorHom K L) := by
  haveI : Fintype (L ≃ₐ[K] L) := AlgEquiv.fintype K L
  have hdim : finrank L (L ⊗[K] L) = finrank L ((L ≃ₐ[K] L) → L) := by
    rw [Module.finrank_baseChange, finrank_pi_aut]
  refine ⟨?_, galoisSelfTensorHom_surjective K L⟩
  exact (LinearMap.injective_iff_surjective_of_finrank_eq_finrank hdim).mpr
    (galoisSelfTensorHom_surjective K L)
