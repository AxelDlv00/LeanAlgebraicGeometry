noncomputable def rationalPointSection {S C T : Scheme.{u}} {πC : C ⟶ S} (πT : T ⟶ S)
    (x₀ : S ⟶ C) (hx₀ : x₀ ≫ πC = 𝟙 S) :
    T ⟶ Limits.pullback πC πT :=
  Limits.pullback.lift (πT ≫ x₀) (𝟙 T)
    (by rw [Category.assoc, hx₀, Category.comp_id, Category.id_comp])
