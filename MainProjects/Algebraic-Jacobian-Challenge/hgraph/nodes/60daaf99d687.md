lemma interLegHomφ_interProj (𝒰 : X.OpenCover) {p q : ℕ}
    (φ : SimplexCategory.mk p ⟶ SimplexCategory.mk q) (σ' : Fin (q + 1) → 𝒰.I₀)
    (l : Fin (p + 1)) :
    interLegHomφ 𝒰 φ σ' ≫ interProj 𝒰 (σ' ∘ φ.toOrderHom) l
      = interProj 𝒰 σ' (φ.toOrderHom l) := by
  have hfac : X.homOfLE (le_iInf (fun l' => iInf_le (fun j => coverOpen 𝒰 (σ' j))
        (φ.toOrderHom l'))) ≫ coverInterToMember 𝒰 (σ' ∘ φ.toOrderHom) l
      = coverInterToMember 𝒰 σ' (φ.toOrderHom l) := by
    haveI : IsOpenImmersion (𝒰.f (σ' (φ.toOrderHom l))) := inferInstance
    refine IsOpenImmersion.lift_uniq (𝒰.f (σ' (φ.toOrderHom l)))
      (Scheme.Opens.ι (coverInterOpen 𝒰 σ')) ?_ _ ?_
    refine (Category.assoc _ _ _).trans ?_
    refine (congrArg (fun w => X.homOfLE (le_iInf (fun l' => iInf_le
        (fun j => coverOpen 𝒰 (σ' j)) (φ.toOrderHom l'))) ≫ w)
      (coverInterToMember_fac 𝒰 (σ' ∘ φ.toOrderHom) l)).trans ?_
    exact Scheme.homOfLE_ι _ _
  apply Over.OverMorphism.ext
  exact (Category.assoc _ _ _).symm.trans
    (congrArg (fun w => w ≫ Sigma.ι 𝒰.X (σ' (φ.toOrderHom l))) hfac)
