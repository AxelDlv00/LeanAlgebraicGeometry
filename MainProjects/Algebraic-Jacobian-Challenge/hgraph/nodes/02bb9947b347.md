    ModuleCat.of k
        (F.obj.obj (op (S.coverFamily ⟨0⟩)) × F.obj.obj (op (S.coverFamily ⟨1⟩)))
      ⟶ cechTerm S.coverFamily F 1 j :=

    ModuleCat.ofHom (LinearMap.snd k _ _)
