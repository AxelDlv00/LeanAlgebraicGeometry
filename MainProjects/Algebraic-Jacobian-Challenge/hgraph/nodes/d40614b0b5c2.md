def Scheme.Hom.IsProjectiveWith {X S : Scheme.{0}} (π : X ⟶ S) (L : X.Modules) :
    Prop :=
  ∃ (d : ℕ) (i : X ⟶ ℙ(Fin (d + 1); S)),
    IsClosedImmersion i ∧ i ≫ (ℙ(Fin (d + 1); S) ↘ S) = π ∧
      Nonempty (L ≅ (Scheme.Modules.pullback i).obj
        (ProjectiveSpace.twistingSheaf (Fin (d + 1)) S 1))
