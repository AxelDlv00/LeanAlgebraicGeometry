def Scheme.Hom.IsHQuasiProjective {X S : Scheme.{u}} (pi : X ⟶ S) : Prop :=
  ∃ (n : Type u) (_ : Finite n) (i : X ⟶ ℙ(n; S)),
    IsImmersion i ∧ QuasiCompact i ∧ i ≫ (ℙ(n; S) ↘ S) = pi
