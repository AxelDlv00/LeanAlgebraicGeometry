private lemma exists_list_map_eq {B₁ : Type u} (f : B → B₁)
    (hf : Function.Surjective f) (l' : List B₁) :
    ∃ l : List B, l.map f = l' := by
  induction l' with
  | nil => exact ⟨[], rfl⟩
  | cons b' tl' ih =>
    obtain ⟨tl, htl⟩ := ih
    obtain ⟨b, hb⟩ := hf b'
    exact ⟨b :: tl, by simp [hb, htl]⟩
