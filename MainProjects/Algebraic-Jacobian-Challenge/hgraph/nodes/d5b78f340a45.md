theorem Scheme.Hom.fiberModule_isFinitePresentation (π : X ⟶ S) (s : S)
    {F : X.Modules} (hF : F.IsFinitePresentation) :
    (π.fiberModule s F).IsFinitePresentation :=
  Scheme.Modules.pullback_isFinitePresentation (π.fiberι s) F hF
