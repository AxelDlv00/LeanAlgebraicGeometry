private lemma comp4_solve_mid {𝒞 : Type*} [Category 𝒞] {a b c d e : 𝒞}
    (A : a ≅ b) {B : b ⟶ c} {Cc : c ⟶ d} (D : d ≅ e) {E : a ⟶ e}
    (hE : A.hom ≫ B ≫ Cc ≫ D.hom = E) :
    B ≫ Cc = A.inv ≫ E ≫ D.inv := by
  rw [← hE]; simp
