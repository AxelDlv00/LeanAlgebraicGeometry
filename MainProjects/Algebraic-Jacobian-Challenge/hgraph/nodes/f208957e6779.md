theorem finiteInAffine_of_isProjective {X S : Scheme.{u}} [IsAffine S] {π : X ⟶ S}
    (h : π.IsProjective) : FiniteInAffine X := by
  obtain ⟨n, -, i, hi, -⟩ := h
  haveI := hi
  exact finiteInAffine_of_isAffineHom i (finiteInAffine_projectiveSpace n S)
