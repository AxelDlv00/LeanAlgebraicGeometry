    Modules.flat_sections_of_coherentSheafFlat_id
      x.coherentSheafFlat_id_pushforward (isAffineOpen_top (Spec R))
  letI : Module.Finite R Γ(M, (⊤ : (Spec R).Opens)) := hfin
