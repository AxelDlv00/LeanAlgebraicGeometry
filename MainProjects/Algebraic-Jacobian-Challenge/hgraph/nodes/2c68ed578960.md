def GeneratedAt (D : X.WeilDivisor) (P : X.PrimeDivisor) : Prop :=
  ∃ f : X.functionField, f ∈ sectionSub k ⊤ D ∧ f ≠ 0 ∧
    Scheme.RationalMap.order P f = -(show X.PrimeDivisor →₀ ℤ from D) P
