noncomputable def coordinateSectionsEquivDom (n : ℕ) :
    (Γ(Y, Q.V₀) × Γ(Y, Q.V₁)) ≃ₗ[K]
      (divisorSections K (n • Q.coordinateWeilDivisor (K := K)) Q.V₀ ×
        divisorSections K (n • Q.coordinateWeilDivisor (K := K)) Q.V₁) :=
  (Q.coordinateSectionsEquivV0 (K := K) n).prodCongr
    (Q.coordinateSectionsEquivV1 (K := K) n)
