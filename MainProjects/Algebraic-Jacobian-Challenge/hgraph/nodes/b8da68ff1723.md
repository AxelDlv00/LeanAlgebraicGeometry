theorem degK_divisorOfList (L : List X.PrimeDivisor) :
    degK k (divisorOfList L : X.WeilDivisor) = ((L.map (residueDeg k)).sum : ℤ) := by
  induction L with
  | nil => simp [divisorOfList]
  | cons P L ih =>
    rw [divisorOfList, degK_add, degK_pointDivisor, ih, List.map_cons, List.sum_cons]
    push_cast
    ring
