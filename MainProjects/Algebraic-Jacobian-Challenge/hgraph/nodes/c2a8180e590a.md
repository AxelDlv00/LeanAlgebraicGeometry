private theorem comp_update (f : (i : ι) → (A i →ₐ[R] S)) (x : ∀ i, A i) (i : ι) (v : A i) :
    (fun j => f j (Function.update x i v j))
      = Function.update (fun j => f j (x j)) i (f i v) := by
  funext j
  by_cases hj : j = i
  · subst hj; simp
  · simp [Function.update_of_ne hj]
