lemma _root_.Finsupp.sum_max_zero_eq_sum_filter_pos {α : Type*}
    (D : α →₀ ℤ) :
    D.sum (fun _ n => max n 0) =
      ∑ a ∈ D.support.filter (fun a => 0 < D a), D a := by
  classical
  rw [Finsupp.sum, Finset.sum_filter]
  apply Finset.sum_congr rfl
  intro a _
  by_cases hpos : 0 < D a
  · simp [hpos]
    omega
  · simp [hpos]
    omega
