lemma maximalIdeal_smul_residueField {x : R} (hx : x ∈ maximalIdeal R)
    (c : ResidueField R) : x • c = 0 := by
  rw [Algebra.smul_def]
  have hx0 : algebraMap R (ResidueField R) x = 0 :=
    Ideal.Quotient.eq_zero_iff_mem.mpr hx
  rw [hx0, zero_mul]

end MaximalIdealAction
