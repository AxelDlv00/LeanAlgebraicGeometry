private lemma single₀_hom_ext {X : 𝒜} {D : CochainComplex 𝒜 ℕ}
    {g h : (CochainComplex.single₀ 𝒜).obj X ⟶ D} (h0 : g.f 0 = h.f 0) : g = h := by
  apply (CochainComplex.fromSingle₀Equiv D X).injective
  ext
  exact h0
