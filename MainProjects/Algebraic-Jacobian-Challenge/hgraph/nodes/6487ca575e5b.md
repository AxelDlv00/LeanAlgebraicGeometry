noncomputable def algHomOfMkUnderHom {B : Under k} (f : CommRingCat.mkUnder k A ⟶ B) :
    A →ₐ[k] B where
  __ := f.right.hom
  commutes' r :=
    congrArg (fun (g : k ⟶ B.right) => g.hom r) (f.w : (CommRingCat.mkUnder k A).hom ≫ _ = B.hom)

@[simp]
