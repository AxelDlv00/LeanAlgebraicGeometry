theorem t1Space_of_orbit_of_isClosed {X : Type*} [TopologicalSpace X] (z₀ : X)
    (hz₀ : IsClosed ({z₀} : Set X)) (horb : ∀ z : X, ∃ e : X ≃ₜ X, e z₀ = z) :
    T1Space X := by
  refine ⟨fun z => ?_⟩
  obtain ⟨e, he⟩ := horb z
  rw [← he]
  exact isClosed_singleton_homeomorph e hz₀
