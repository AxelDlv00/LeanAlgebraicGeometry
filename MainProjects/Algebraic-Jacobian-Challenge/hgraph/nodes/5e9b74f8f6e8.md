docstring: "**The two-lattice lemma**, localization form. Let `N` be a finite module\
  \ over the Laurent\npolynomial ring `k[T;T⁻¹]` and let `Q₀ Q₁ : Submodule k N` be\
  \ such that\n\n* `Q₀` is stable under `T` and every element of `N` lands in `Q₀`\
  \ after multiplication by a\n  sufficiently large power of `T`;\n* `Q₁` is stable\
  \ under `T⁻¹` and every element of `N` lands in `Q₁` after multiplication by a\n\
  \  sufficiently negative power of `T`.\n\nThen `N ⧸ (Q₀ ⊔ Q₁)` is a finite `k`-module.\n\
  \nThe hypotheses on `Q₀` and `Q₁` are exactly the \"surjectivity up to powers\"\
  \ statements\nfurnished by `IsLocalization.Away` when `N` is the sections of a scheme\
  \ on an intersection of\ntwo affine opens and `Q₀`, `Q₁` are the images of the restriction\
  \ maps; see also\n`moduleFinite_quotient_sup_of_exists_pow_smul_mem` for the variant\

theorem moduleFinite_quotient_sup_of_exists_smul_mem [Module.Finite k[T;T⁻¹] N]
    {Q₀ Q₁ : Submodule k N}
    (hQ₀ : ∀ x ∈ Q₀, (T 1 : k[T;T⁻¹]) • x ∈ Q₀)
    (hQ₁ : ∀ x ∈ Q₁, (T (-1) : k[T;T⁻¹]) • x ∈ Q₁)
    (hQ₀loc : ∀ n : N, ∃ m : ℕ, (T (m : ℤ) : k[T;T⁻¹]) • n ∈ Q₀)
    (hQ₁loc : ∀ n : N, ∃ m : ℕ, (T (-(m : ℤ)) : k[T;T⁻¹]) • n ∈ Q₁) :
    Module.Finite k (N ⧸ (Q₀ ⊔ Q₁)) := by
  classical
  -- Push a finite `k[T;T⁻¹]`-generating set of `N` into `Q₀` by clearing denominators.
  obtain ⟨t, ht⟩ := Module.Finite.fg_top (R := k[T;T⁻¹]) (M := N)
  choose μ hμ using hQ₀loc
  refine moduleFinite_quotient_sup_of_span_eq_top
    (s := t.image fun n => (T (μ n : ℤ) : k[T;T⁻¹]) • n) ?_ ?_ hQ₀ hQ₁ hQ₁loc
  · intro x hx
    obtain ⟨n, -, rfl⟩ := Finset.mem_image.mp hx
    exact hμ n
  · rw [eq_top_iff, ← ht]
    refine span_le.mpr fun n hn => ?_
    have hmem : (T (μ n : ℤ) : k[T;T⁻¹]) • n ∈
        span k[T;T⁻¹] ((t.image fun n => (T (μ n : ℤ) : k[T;T⁻¹]) • n : Finset N) : Set N) :=
      subset_span <| Finset.mem_coe.mpr <| Finset.mem_image_of_mem _ (Finset.mem_coe.mp hn)
    have hcancel : (T (-(μ n : ℤ)) : k[T;T⁻¹]) • ((T (μ n : ℤ) : k[T;T⁻¹]) • n) = n := by
      rw [← mul_smul, ← T_add, neg_add_cancel, T_zero, one_smul]
    exact hcancel ▸ Submodule.smul_mem _ (T (-(μ n : ℤ))) hmem
