noncomputable def permAlgEquiv (e : Equiv.Perm ι) :
    (⨂[R] _ : ι, A) ≃ₐ[R] (⨂[R] _ : ι, A) :=
  AlgEquiv.ofAlgHom (permAlgHom R A e) (permAlgHom R A e⁻¹)
    (by rw [permAlgHom_comp]; simp)
    (by rw [permAlgHom_comp]; simp)
