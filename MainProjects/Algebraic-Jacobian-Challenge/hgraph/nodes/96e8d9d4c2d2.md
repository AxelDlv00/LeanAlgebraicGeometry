theorem generic_exists_unique_via_mathlib (F : Cᵒᵖ ⥤ Type w) {X Y : C} (f : X ⟶ Y)
    [HasPullback f f]
    (hsh : Presieve.IsSheafFor F (Presieve.singleton f))
    (x : F.obj (op X))
    (h : F.map (pullback.fst f f).op x = F.map (pullback.snd f f).op x) :
    ∃! y, F.map f.op y = x := by
  rw [Equalizer.Presieve.isSheafFor_singleton_iff_of_hasPullback,
    ← Types.type_equalizer_iff_unique] at hsh
  exact hsh x h
