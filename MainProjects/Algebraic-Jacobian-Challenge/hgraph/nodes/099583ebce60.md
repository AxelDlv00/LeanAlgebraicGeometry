theorem stalkMapₗ_injective {M N : ModuleCat R} (f : M ⟶ N) (hf : Function.Injective f.hom)
    (x : PrimeSpectrum.Top R) : Function.Injective (stalkMapₗ f x) := by
  rw [stalkMapₗ_eq]
  exact tilde_toStalk_map_injective f hf x
