private lemma iso_symm_trans_cancel {C : Type*} [Category C] {W X Y Z : C}
    (a : W ≅ X) (g : X ≅ Y) (h : Y ≅ Z) :
    a.symm ≪≫ ((a ≪≫ g) ≪≫ h) = g ≪≫ h := by
  rw [Iso.trans_assoc, Iso.symm_self_id_assoc]
