private lemma map_comp_postcomp_of_eq {C D : Type*} [Category C] [Category D]
    (E : C ⥤ D) {A B C' : C} {T : D} (f : A ⟶ B) (g : B ⟶ C') (h : A ⟶ C')
    (e : f ≫ g = h) (k : E.obj C' ⟶ T) :
    E.map f ≫ E.map g ≫ k = E.map h ≫ k := by
  rw [← Category.assoc, ← E.map_comp, e]
