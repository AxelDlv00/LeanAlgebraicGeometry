theorem finiteInAffine_coprod {X Y : Scheme.{u}} (hX : FiniteInAffine X)
    (hY : FiniteInAffine Y) : FiniteInAffine (X ⨿ Y) := by
  refine finiteInAffine_of_iso (coprodIsoSigma X Y).symm ?_
  apply finiteInAffine_sigma
  rintro ⟨i⟩
  cases i
  · simpa using hX
  · simpa using hY
