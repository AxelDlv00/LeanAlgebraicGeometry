def MVCoboundaryRel (B : Type v) [CommRing B] [Algebra k B]
    (u v : R₀₁ ⊗[k] B) : Prop :=
  ∃ a : R₀ ⊗[k] B, ∃ b : R₁ ⊗[k] B, IsUnit a ∧ IsUnit b ∧
    u * rTensorAlgHom ρ₁ B b = rTensorAlgHom ρ₀ B a * v
