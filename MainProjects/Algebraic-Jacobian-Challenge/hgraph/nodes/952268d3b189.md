theorem degK_eq_of_linearEquivalence
    (hledger : ∀ D : X.WeilDivisor, chi k U₀ U₁ D = chi k U₀ U₁ 0 + degK k D)
    {D D' : X.WeilDivisor} (h : Scheme.WeilDivisor.LinearEquivalence D D') :
    degK k D = degK k D' := by
  have hchi := chi_eq_of_linearEquivalence k U₀ U₁ h
  rw [hledger D, hledger D'] at hchi
  omega

end LedgerNoetherian
