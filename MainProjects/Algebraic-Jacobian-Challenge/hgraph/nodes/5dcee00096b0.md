    (hproj : ∀ σ : Equiv.Perm (Fin n), MonObj.permAut C σ ≫ D.proj = D.proj) :
    Cocone (permDiagram C n) where

        intro X Y f
        change MonObj.permAut C (f : Equiv.Perm (Fin n))⁻¹ ≫ D.proj = D.proj ≫ 𝟙 _
