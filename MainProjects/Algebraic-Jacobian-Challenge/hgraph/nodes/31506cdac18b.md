lemma Scheme.id_appLE {X : Scheme.{u}} {U V : X.Opens}
    (e : V ≤ (𝟙 X : X ⟶ X) ⁻¹ᵁ U) :
    (𝟙 X : X ⟶ X).appLE U V e =
      X.presheaf.map (homOfLE (show V ≤ U from e)).op := by
  simp only [Scheme.Hom.appLE, Scheme.Hom.id_app]
  exact Category.id_comp _
