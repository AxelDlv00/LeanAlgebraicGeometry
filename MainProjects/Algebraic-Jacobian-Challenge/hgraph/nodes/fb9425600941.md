      Subsingleton (Sheaf.HModule (C.left.moduleKSheaf k) 1) := by
  obtain ⟨S⟩ := nonempty_affineCoverMVSquare_of_curve C
  exact Scheme.subsingleton_hModule_one_moduleKSheaf_baseChangeField_iff κ S
