noncomputable def tensorFromPi [Fintype ι] {W : Type*} [AddCommGroup W] [Module K W]
    (b : Basis ι K L) : (ι → W) →ₗ[K] L ⊗[K] W :=
  ∑ i, (TensorProduct.mk K L W (b i)).comp (LinearMap.proj i)
