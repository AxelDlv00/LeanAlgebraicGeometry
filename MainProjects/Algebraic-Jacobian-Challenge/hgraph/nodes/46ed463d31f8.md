theorem span_chart1 {A : Type u} [Semiring A] [Module A R1]
    (d : ℕ) (y : R1) (bb : I → R1)
    (hspan : Submodule.span A (Set.range bb) = ⊤) :
    Submodule.span A (Set.range (chart1 d y bb)) = ⊤ := by
  apply top_unique
  rw [← hspan]
  apply Submodule.span_mono
  rintro z ⟨i, rfl⟩
  exact ⟨Sum.inr i, rfl⟩
