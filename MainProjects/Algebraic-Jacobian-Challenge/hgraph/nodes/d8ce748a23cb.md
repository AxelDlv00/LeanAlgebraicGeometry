theorem orderGe_antitone {P : X.PrimeDivisor} {m m' : ℤ} (h : m ≤ m') :
    orderGe P m' ≤ orderGe P m := by
  intro f hf
  rcases hf with rfl | hf
  · exact Or.inl rfl
  · exact Or.inr (h.trans hf)
