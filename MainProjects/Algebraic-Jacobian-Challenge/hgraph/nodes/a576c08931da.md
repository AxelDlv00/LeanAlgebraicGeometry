private lemma natIso_inv_app_naturality {C D : Type*} [Category C] [Category D]
    {F G : C ⥤ D} (e : F ≅ G) {A B : C} (g : A ⟶ B) :
    G.map g ≫ (e.app B).inv = (e.app A).inv ≫ F.map g :=
  e.inv.naturality g
