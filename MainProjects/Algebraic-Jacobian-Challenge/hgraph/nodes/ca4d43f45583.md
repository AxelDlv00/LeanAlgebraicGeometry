theorem truncExp_injective : Function.Injective (truncExp (R := R)) := by
  intro b c h
  have : (truncExpUnit b.toAdd : R[ε]).snd = (truncExpUnit c.toAdd : R[ε]).snd := by
    rw [truncExp_apply, truncExp_apply] at h
    rw [h]
