noncomputable def basePointShift (P0 : 𝟙_ K ⟶ C) {n : ℕ} (i₀ : Fin n) :
    C ⟶ (∏ᶜ (fun _ : Fin n => C)) :=
  Pi.lift (fun i => if i = i₀ then 𝟙 C else toUnit C ≫ P0)
