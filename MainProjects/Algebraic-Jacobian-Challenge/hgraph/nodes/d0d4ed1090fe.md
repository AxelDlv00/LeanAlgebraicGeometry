theorem sectionOfDivisor_antitone_open {U U' : X.Opens} (h : U ≤ U')
    (D : X.WeilDivisor) : sectionOfDivisor U' D ≤ sectionOfDivisor U D := by
  intro f hf
  rcases hf with rfl | hf
  · exact Or.inl rfl
  · exact Or.inr fun P hP => hf P (h hP)

variable (U₀ U₁ : X.Opens)
