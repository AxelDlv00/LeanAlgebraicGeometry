noncomputable def chartTransition (d r : ℕ) (I J : Finset (Fin r)) (hI : I.card = d)
    (hJ : J.card = d) : chartOverlap d r I J hI hJ ⟶ chartOverlap d r J I hJ hI :=
  Spec.map (CommRingCat.ofHom (transitionMap d r I J hI hJ))
