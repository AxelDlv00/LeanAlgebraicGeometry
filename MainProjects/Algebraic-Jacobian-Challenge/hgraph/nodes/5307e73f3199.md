def IsStableOpen (ρ : SemilinearGalAction K L X f) (U : X.Opens) : Prop :=
  ∀ γ : L ≃ₐ[K] L, (ρ.act γ).hom ⁻¹ᵁ U = U

end SemilinearGalAction
