def pullbackIsoPushforwardInvOfIsIso {X Y : Scheme.{0}} (φ : X ⟶ Y) [IsIso φ] :
    Scheme.Modules.pullback φ ≅ Scheme.Modules.pushforward (inv φ) :=
  Adjunction.leftAdjointUniq (Scheme.Modules.pullbackPushforwardAdjunction φ)
