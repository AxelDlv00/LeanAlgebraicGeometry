theorem finiteInAffine_over_sigma
    {σ : Type v} [Small.{u, v} σ] {S : Scheme.{u}}
    (X : σ → Over S) (hX : ∀ i, FiniteInAffine (X i).left) :
    FiniteInAffine (∐ X).left :=
  finiteInAffine_of_iso (PreservesCoproduct.iso (Over.forget S) X).symm
