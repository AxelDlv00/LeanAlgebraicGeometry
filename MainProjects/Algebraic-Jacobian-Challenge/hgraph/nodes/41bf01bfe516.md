def truncExpUnit (b : R) : (R[ε])ˣ :=
  Units.mkOfMulEqOne (1 + inr b) (1 - inr b) <| by
    calc (1 + inr b) * (1 - inr b)
        = 1 - (inr b : R[ε]) * inr b := by ring
      _ = 1 := by rw [inr_mul_inr_eq_zero, sub_zero]

@[simp]
