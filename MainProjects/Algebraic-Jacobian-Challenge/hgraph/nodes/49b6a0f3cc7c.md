lemma scalarEnd_eq_one_iff (a : Γ(X, ⊤)) :
    scalarEnd a = 𝟙 (SheafOfModules.unit X.ringCatSheaf) ↔ a = 1 := by
  rw [← scalarEnd_one (X := X)]
  exact ⟨fun h => scalarEnd_injective h, fun h => congrArg scalarEnd h⟩
