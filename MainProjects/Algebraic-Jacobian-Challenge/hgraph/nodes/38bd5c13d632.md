noncomputable def preHom {A X W : K} [MonObj A] (u : W ⟶ X) : (X ⟶ A) →* (W ⟶ A) where
  toFun f := u ≫ f
  map_one' := by
    change u ≫ (toUnit X ≫ η[A]) = toUnit W ≫ η[A]
    rw [← Category.assoc]
    congr 1
    exact Subsingleton.elim _ _
  map_mul' f h := by
    change u ≫ (lift f h ≫ μ[A]) = lift (u ≫ f) (u ≫ h) ≫ μ[A]
    rw [← Category.assoc, comp_lift]

@[simp]
