theorem isProjective_of_finiteMap
    (D : LaurentChartData Y) (pi : C ⟶ Y)
    [IsFinite pi.left] [IsProper C.hom] : C.hom.IsProjective := by
  obtain ⟨G⟩ := D.nonempty_finiteMapGenerators pi
  exact G.isProjective
