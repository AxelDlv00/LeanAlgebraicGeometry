docstring: "**Graded prime avoidance at a single relevant homogeneous prime.** If\
  \ a\nhomogeneous ideal `I` is not contained in a homogeneous prime `p`, and `p`\
  \ does not\ncontain the irrelevant ideal, then `I` has a homogeneous element of\
  \ **strictly\npositive** degree outside `p`.\n\nTwo separate uses of the grading,\
  \ and it is worth naming both because the second is\nthe one an ungraded intuition\
  \ drops:\n\n* `I` homogeneous and `¬ I ≤ p` gives a homogeneous element of `I` outside\
  \ `p` —\n  decompose a witness and note that if every component were in `p` so would\
  \ the\n  witness be. Its degree is **not controlled** and may be `0`.\n* Relevance\
  \ of `p` (`¬ \U0001D49C₊ ≤ p`) gives a homogeneous `g ∉ p` of positive degree, by\n\
  \  the same decomposition applied to a witness in the irrelevant ideal, whose\n\
  \  degree-`0` component is zero by definition.\n\nThen `f₀ * g` is homogeneous of\

  \ what makes this possible: over a `p` containing\n`\U0001D49C₊` — i.e. at a point\

theorem exists_homogeneous_pos_mem_notMem
    {I p : Ideal A} [hp : p.IsPrime] (hI : I.IsHomogeneous 𝒜)
    (hrel : ¬ (HomogeneousIdeal.irrelevant 𝒜).toIdeal ≤ p) (hIp : ¬ I ≤ p) :
    ∃ (m : ℕ) (f : A), 0 < m ∧ f ∈ 𝒜 m ∧ f ∈ I ∧ f ∉ p := by
  classical
  obtain ⟨n, f₀, hf₀deg, hf₀I, hf₀p⟩ : ∃ (n : ℕ) (f : A), f ∈ 𝒜 n ∧ f ∈ I ∧ f ∉ p := by
    obtain ⟨x, hxI, hxp⟩ := SetLike.not_le_iff_exists.mp hIp
    by_contra hcon
    push Not at hcon
    exact hxp (by
      rw [← DirectSum.sum_support_decompose 𝒜 x]
      refine Ideal.sum_mem _ fun i _ => ?_
      by_contra h
      exact h (hcon i _ (SetLike.coe_mem _) ((hI.mem_iff).mp hxI i)))
  obtain ⟨m, g, hm, hgdeg, hgp⟩ : ∃ (m : ℕ) (g : A), 0 < m ∧ g ∈ 𝒜 m ∧ g ∉ p := by
    obtain ⟨x, hxirr, hxp⟩ := SetLike.not_le_iff_exists.mp hrel
    by_contra hcon
    push Not at hcon
    exact hxp (by
      rw [← DirectSum.sum_support_decompose 𝒜 x]
      refine Ideal.sum_mem _ fun i _ => ?_
      rcases Nat.eq_zero_or_pos i with rfl | hi
      · have hz : (DirectSum.decompose 𝒜 x 0 : A) = 0 := by
          have := (HomogeneousIdeal.mem_irrelevant_iff 𝒜 x).mp hxirr
          simpa [GradedRing.proj_apply] using this
        simp [hz]
      · exact hcon i _ hi (SetLike.coe_mem _))
  exact ⟨n + m, f₀ * g, by omega, SetLike.mul_mem_graded hf₀deg hgdeg,
    I.mul_mem_right _ hf₀I, fun h => (hp.mem_or_mem h).elim hf₀p hgp⟩
