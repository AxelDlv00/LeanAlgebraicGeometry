theorem exists_algebraMap_eq_gradeZero
    (c : homogeneousSubmodule J R 0) :
    ∃ r : R, algebraMap R (homogeneousSubmodule J R 0) r = c := by
  have hc : (c : MvPolynomial J R) ∈
      (1 : Submodule R (MvPolynomial J R)) := by
    rw [← homogeneousSubmodule_zero (σ := J) (R := R)]
    exact c.2
  obtain ⟨r, hr⟩ := Submodule.mem_one.mp hc
  exact ⟨r, Subtype.ext (by simpa using hr)⟩
