def degLocus {T : Over S} (x : DivFamily π T)
    (hlc : IsLocallyConstant x.fiberDeg) (d : ℕ) : (T.left : Scheme.{u}).Opens :=
  ⟨{t | x.fiberDeg t = d}, hlc.isOpen_fiber d⟩
