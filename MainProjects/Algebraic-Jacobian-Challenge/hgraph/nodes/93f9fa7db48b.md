theorem triple_stable (i j k : StableAffineOpen ρ) :
    ρ.IsStableOpen ((i.U ⊓ j.U) ⊓ (i.U ⊓ k.U)) := by
  intro γ
  simp only [Scheme.Hom.preimage_inf, i.stable γ, j.stable γ, k.stable γ]
