lemma isIso_restrict_map {U V : TopologicalSpace.Opens X.toTopCat} (h : U ≤ V) (h' : V ≤ U) :
    IsIso (F.obj.map (homOfLE h).op) := by
  have heq : U = V := le_antisymm h h'
  subst heq
  have h0 : (homOfLE h).op = 𝟙 (Opposite.op U) := Subsingleton.elim _ _
  rw [h0, F.obj.map_id]
  infer_instance
