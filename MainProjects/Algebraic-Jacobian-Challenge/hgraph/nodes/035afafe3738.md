noncomputable def universalKernelEvaluation (L : X.Modules) {r d : ℕ}
    (hV : SheafOfModules.IsLocallyFreeOfRank
      ((Modules.pushforward π).obj L) r)
    (hd : 1 ≤ d) (hdr : d ≤ r) :=
  LocallyFreeQuotient.kernelEvaluation L
    (universalQuotient hV hd hdr)
