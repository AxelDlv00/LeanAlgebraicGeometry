theorem generic_exists_unique (F : Cᵒᵖ ⥤ Type w) {X Y : C} (f : X ⟶ Y) [HasPullback f f]
    (hsh : Presieve.IsSheafFor F (Presieve.singleton f))
    (x : F.obj (op X))
    (h : F.map (pullback.fst f f).op x = F.map (pullback.snd f f).op x) :
    ∃! y, F.map f.op y = x := by
  rw [Presieve.isSheafFor_singleton] at hsh
  exact hsh x (fun p₁ p₂ hp => generic_compat F f x h p₁ p₂ hp)
