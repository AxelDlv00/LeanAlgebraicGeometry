theorem generatedAt_of_evalMap_surjective (D : X.WeilDivisor) (P : X.PrimeDivisor)
    (hsurj : Function.Surjective (evalMap k D P))
    [Module.Finite k (localStepTgt k P 1)] :
    GeneratedAt k D P := by
  -- the target is nontrivial: its dimension is `[κ(P):k] ≥ 1`
  haveI : Nontrivial
      (localStepTgt k P (-(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P)) := by
    refine Module.nontrivial_of_finrank_pos (R := k) ?_
    rw [finrank_localStepTgt]
    exact one_le_residueDeg k P
  obtain ⟨z, hz⟩ := exists_ne (0 :
    localStepTgt k P (-(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P))
  obtain ⟨q, hq⟩ := hsurj z
  obtain ⟨f, rfl⟩ := Submodule.Quotient.mk_surjective _ q
  refine ⟨(f : X.functionField), f.2, ?_, ?_⟩
  · -- a section whose class is nonzero cannot be `0`
    intro hf0
    apply hz
    rw [← hq, evalMap, localStepMapₖ, Submodule.mapQ_apply]
    rw [Submodule.Quotient.mk_eq_zero, Submodule.mem_comap, Submodule.subtype_apply,
      Submodule.coe_inclusion, mem_orderGeSub, hf0]
    exact (orderGe _ _).zero_mem
  · -- the two inequalities
    have hlow : -(show X.PrimeDivisor →₀ ℤ from D) P ≤
        Scheme.RationalMap.order P (f : X.functionField) := by
      rcases eq_or_ne (f : X.functionField) 0 with hf0 | hf0
      · exact absurd hf0 (by
          intro h
          apply hz
          rw [← hq, evalMap, localStepMapₖ, Submodule.mapQ_apply,
            Submodule.Quotient.mk_eq_zero, Submodule.mem_comap, Submodule.subtype_apply,
            Submodule.coe_inclusion, mem_orderGeSub, h]
          exact (orderGe _ _).zero_mem)
      exact (mem_sectionOfDivisor_of_ne_zero hf0).mp f.2 P
        (TopologicalSpace.Opens.mem_top P.point)
    have hhigh : Scheme.RationalMap.order P (f : X.functionField) <
        -(show X.PrimeDivisor →₀ ℤ from D) P + 1 := by
      by_contra hge
      apply hz
      rw [← hq, evalMap, localStepMapₖ, Submodule.mapQ_apply,
        Submodule.Quotient.mk_eq_zero, Submodule.mem_comap, Submodule.subtype_apply,
        Submodule.coe_inclusion, mem_orderGeSub]
      refine Or.inr ?_
      rw [sub_pointDivisor_apply_self]
      omega
    omega
