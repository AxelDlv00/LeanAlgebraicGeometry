private noncomputable def cochainContractibleOfComponents
    (K : CochainComplex Ab.{u} ℕ)
    (h : ∀ n, K.X (n + 1) ⟶ K.X n)
    (hzero : 𝟙 (K.X 0) = K.d 0 1 ≫ h 0)
    (hsucc : ∀ n, 𝟙 (K.X (n + 1)) =
      h n ≫ K.d n (n + 1) + K.d (n + 1) (n + 2) ≫ h (n + 1)) :
    Homotopy (𝟙 K) 0 := by
  let h' : ∀ i j, (ComplexShape.up ℕ).Rel j i → (K.X i ⟶ K.X j) :=
    fun i j hij => by
      change j + 1 = i at hij
      subst i
      exact h j
  have hid : 𝟙 K = Homotopy.nullHomotopicMap' h' := by
    apply HomologicalComplex.hom_ext
    intro n
    rcases n with (_ | n)
    · rw [Homotopy.nullHomotopicMap'_f_of_not_rel_right
        (ComplexShape.up_mk 0 1 rfl)
        (fun l hl => Nat.succ_ne_zero l hl)]
      exact (HomologicalComplex.id_f K 0).trans hzero
    · rw [Homotopy.nullHomotopicMap'_f
        (ComplexShape.up_mk n (n + 1) rfl)
        (ComplexShape.up_mk (n + 1) (n + 2) rfl)]
      exact (HomologicalComplex.id_f K (n + 1)).trans
        ((hsucc n).trans (add_comm _ _))
  exact (Homotopy.ofEq hid).trans (Homotopy.nullHomotopy' h')
