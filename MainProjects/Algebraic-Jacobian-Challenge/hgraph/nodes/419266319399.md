theorem Opens.eq_top_of_subsingleton {T : Scheme.{u}} [Subsingleton T]
    (U : T.Opens) {x : T} (hx : x ∈ U) : U = ⊤ := by
  ext y
  refine iff_of_true ?_ trivial
  exact Subsingleton.elim y x ▸ hx
