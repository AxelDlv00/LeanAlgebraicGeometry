          (p1Chart k ⟨0⟩ ⊓ p1Chart k ⟨1⟩)
          (p1Chart_inf_eq_preimage_mul k).le).hom
        ((Proj.awayToSection (homogeneousSubmodule (ULift.{u} (Fin 2)) (ULift.{u} ℤ))
            p1XY).hom

            (ProjTwist.X_mem_deg_one (ULift.{u} (Fin 2)) ⟨0⟩)
            p1XY_eq_comm
            (p1CoordAway (ULift.{u} (Fin 2)) ⟨1⟩ ⟨0⟩))) := by
  have My :
      (Proj.awayToSection (homogeneousSubmodule (ULift.{u} (Fin 2)) (ULift.{u} ℤ))
          (X (⟨1⟩ : ULift.{u} (Fin 2))))

            (p1Chart k ⟨0⟩ ⊓ p1Chart k ⟨1⟩)
