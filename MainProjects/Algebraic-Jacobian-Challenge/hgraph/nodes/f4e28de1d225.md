noncomputable def Hom.fiberEulerIndex (f : X ⟶ S) (s : S)
    (M : X.Modules) : ℤ :=
  CategoryTheory.Sheaf.chi
    (toModuleKSheafOfModules
      (Over.mk (f.fiberToSpecResidueField s))
      (f.fiberModule s M))
