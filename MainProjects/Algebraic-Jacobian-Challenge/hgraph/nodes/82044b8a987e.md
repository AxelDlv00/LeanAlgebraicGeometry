def galoisEvalHom : L →ₐ[K] ((L ≃ₐ[K] L) → L) where
  toFun b := fun γ => γ b
  map_one' := by ext γ; simp
  map_mul' x y := by ext γ; simp
  map_zero' := by ext γ; simp
  map_add' x y := by ext γ; simp
  commutes' r := by ext γ; simp
