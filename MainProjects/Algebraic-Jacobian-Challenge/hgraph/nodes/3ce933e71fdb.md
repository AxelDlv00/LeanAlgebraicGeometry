theorem stableAffineQuotientMap_preimage_quotientOpen [FiniteDimensional K L]
    (hUa : IsAffineOpen U) {V : X.Opens} (hVU : V ≤ U)
    (hV : ρ.IsStableOpen V) :
    letI := ρ.sectionsMulSemiringAction hU
    letI := sectionsAlgebra f U
    letI := sectionsAlgebraK (K := K) f U
    letI := sections_isScalarTower (K := K) f U
    letI := ρ.isSemilinear_sections hU
    (stableAffineQuotientMap ρ hU hUa) ⁻¹ᵁ quotientOpenOfStableSubopen ρ hU V =
      U.ι ⁻¹ᵁ V := by
  letI := ρ.sectionsMulSemiringAction hU
  letI := sectionsAlgebra f U
  letI := sectionsAlgebraK (K := K) f U
  letI := sections_isScalarTower (K := K) f U
  letI := ρ.isSemilinear_sections hU
  rw [stableAffineQuotientMap, Scheme.Hom.comp_preimage,
    quotientOpenOfStableSubopen, Scheme.Hom.preimage_iSup,
    Scheme.Hom.preimage_iSup]
  simp only [SemilinearAction.affineQuotientMap]
  rw [hUa.isoSpec_hom]
  calc
    (⨆ i : InvariantBasicOpenIndex ρ hU V,
        U.toSpecΓ ⁻¹ᵁ
          (PrimeSpectrum.basicOpen (i.1 : Γ(X, U)) :
            (Spec (CommRingCat.of Γ(X, U))).Opens))
        = ⨆ i : InvariantBasicOpenIndex ρ hU V,
            U.ι ⁻¹ᵁ X.basicOpen i.1 := by
              congr 1
              funext i
              exact U.toSpecΓ_preimage_basicOpen i.1
    _ = U.ι ⁻¹ᵁ (⨆ i : InvariantBasicOpenIndex ρ hU V, X.basicOpen i.1) :=
      (Scheme.Hom.preimage_iSup U.ι _).symm
    _ = U.ι ⁻¹ᵁ V := by rw [ρ.iSup_invariantBasicOpen_eq hU hUa hVU hV]
