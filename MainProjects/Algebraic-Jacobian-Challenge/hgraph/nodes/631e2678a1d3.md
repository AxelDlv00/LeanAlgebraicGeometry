private noncomputable def relNegVia {S C T : Scheme.{u}} {πC : C ⟶ S} {πT : T ⟶ S} :
    Quotient (relPicSetoid πC πT) → Quotient (relPicSetoid πC πT) :=
  Quotient.lift
    (fun L => Quotient.mk _ (relNegOnProduct L))
    (by
      rintro L M h
      exact Quotient.sound (relPicRel_neg h))
