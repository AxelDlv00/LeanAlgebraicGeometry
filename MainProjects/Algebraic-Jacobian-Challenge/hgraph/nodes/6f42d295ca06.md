lemma coordinateDiff_apply (n : ℕ) (a : Γ(Y, Q.V₀)) (b : Γ(Y, Q.V₁)) :
    Q.coordinateDiff (K := K) n (a, b) =
      Q.overlapInversePower n *
          (Y.presheaf.map
            (homOfLE (inf_le_left : Q.V₀ ⊓ Q.V₁ ≤ Q.V₀)).op).hom a -
        (Y.presheaf.map
          (homOfLE (inf_le_right : Q.V₀ ⊓ Q.V₁ ≤ Q.V₁)).op).hom b := rfl
