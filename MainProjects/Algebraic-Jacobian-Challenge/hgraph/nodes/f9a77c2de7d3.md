theorem exists_stageMap_eq_of_le {S T : Subalgebra k A} (h : S ≤ T)
    {x : R ⊗[k] A} (hx : ∃ y : R ⊗[k] S, stageMap R S y = x) :
    ∃ y : R ⊗[k] T, stageMap R T y = x := by
  obtain ⟨y, rfl⟩ := hx
  exact ⟨transitionMap R h y, stageMap_transitionMap_apply R h y⟩
