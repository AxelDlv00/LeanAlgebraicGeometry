noncomputable def glueChartQuot (k : κ) :
    (Scheme.Modules.pullback ((covGD W hW).ι k)).obj
      ((Scheme.Modules.pullback ((opensCover T.left W hW).fromGlued ≫ T.hom)).obj V)
      ⟶ (y k).F :=
  (pullbackTriangleIso (glueChart_w W hW k) V).hom ≫ (y k).q
