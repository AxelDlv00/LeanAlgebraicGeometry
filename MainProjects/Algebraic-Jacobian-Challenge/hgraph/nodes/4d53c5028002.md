theorem generic_compat (F : Cᵒᵖ ⥤ Type w) {X Y : C} (f : X ⟶ Y) [HasPullback f f]
    (x : F.obj (op X))
    (h : F.map (pullback.fst f f).op x = F.map (pullback.snd f f).op x)
    {Z : C} (p₁ p₂ : Z ⟶ X) (hp : p₁ ≫ f = p₂ ≫ f) :
    F.map p₁.op x = F.map p₂.op x := by
  set l := pullback.lift (f := f) (g := f) p₁ p₂ hp with hl
  have h1 : l ≫ pullback.fst f f = p₁ := pullback.lift_fst _ _ _
  have h2 : l ≫ pullback.snd f f = p₂ := pullback.lift_snd _ _ _
  calc F.map p₁.op x
      = F.map (l ≫ pullback.fst f f).op x := by rw [h1]
    _ = F.map l.op (F.map (pullback.fst f f).op x) := by
          simp only [op_comp, Functor.map_comp, CategoryTheory.comp_apply]
    _ = F.map l.op (F.map (pullback.snd f f).op x) := by rw [h]
    _ = F.map (l ≫ pullback.snd f f).op x := by
          simp only [op_comp, Functor.map_comp, CategoryTheory.comp_apply]
    _ = F.map p₂.op x := by rw [h2]
