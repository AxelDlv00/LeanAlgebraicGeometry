    SmoothOfRelativeDimension (genus C) (Pic0SchemeEt C).hom ↔
      ∀ x : ((Pic0SchemeEt C).left : Scheme.{u}),
        ∃ (U : (Spec (CommRingCat.of k)).Opens) (_ : IsAffineOpen U)

            (letI := ((Pic0SchemeEt C).hom.appLE U V e).hom.toAlgebra;
