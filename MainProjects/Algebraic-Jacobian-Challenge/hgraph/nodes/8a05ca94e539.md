lemma pullbackCongr_hom_app_eqToHom {X Y : Scheme.{u}} {φ ψ : X ⟶ Y} (h : φ = ψ)
    (N : Y.Modules) :
    (Scheme.Modules.pullbackCongr h).hom.app N
      = eqToHom (congrArg (fun α => (Scheme.Modules.pullback α).obj N) h) := by
  subst h
