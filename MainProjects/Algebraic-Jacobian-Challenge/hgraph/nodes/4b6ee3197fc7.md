def mapRingHom (f : R →+* S) : R[ε] →+* S[ε] where
  toFun x := inl (f x.fst) + inr (f x.snd)
  map_one' := by
    refine TrivSqZeroExt.ext ?_ ?_ <;> simp
  map_mul' x y := by
    refine TrivSqZeroExt.ext ?_ ?_
    · simp [fst_mul]
    · simp [smul_eq_mul, mul_comm]
  map_zero' := by
    refine TrivSqZeroExt.ext ?_ ?_ <;> simp
  map_add' x y := by
    refine TrivSqZeroExt.ext ?_ ?_ <;> simp
