lemma T_mem_nonpos_of_nonpos {n : ℤ} (hn : n ≤ 0) :
    (T n : LaurentPolynomial R) ∈ nonposLaurentSubmodule R := by
  obtain ⟨m, rfl⟩ : ∃ m : ℕ, n = -(m : ℤ) := by
    refine ⟨(-n).toNat, ?_⟩
    have := Int.toNat_of_nonneg (neg_nonneg.mpr hn)
    omega
  exact Submodule.subset_span ⟨m, rfl⟩
