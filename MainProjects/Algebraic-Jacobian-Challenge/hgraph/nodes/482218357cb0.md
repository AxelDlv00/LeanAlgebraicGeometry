noncomputable def toAffineCoverMVSquare (D : FiberCoordinateData Y) :
    Y.AffineCoverMVSquare where
  U₁ := D.V₀
  U₂ := D.V₁
  isAffineOpen_U₁ := D.isAffineOpen_V₀
  isAffineOpen_U₂ := D.isAffineOpen_V₁
  isAffineOpen_inf := by
    rw [D.inf_eq_basicOpen_x]
    exact D.isAffineOpen_V₀.basicOpen D.x
  cover := D.cover
