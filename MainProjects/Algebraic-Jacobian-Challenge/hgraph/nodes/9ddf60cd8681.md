theorem ell_le_ell_add_residueDeg {D D' : X.WeilDivisor} {P : X.PrimeDivisor}
    (hstep : (show X.PrimeDivisor →₀ ℤ from D') P =
      (show X.PrimeDivisor →₀ ℤ from D) P + 1)
    (hle : ∀ Q : X.PrimeDivisor, (show X.PrimeDivisor →₀ ℤ from D) Q ≤
      (show X.PrimeDivisor →₀ ℤ from D') Q)
    (hoff : ∀ Q : X.PrimeDivisor, Q ≠ P →
      (show X.PrimeDivisor →₀ ℤ from D) Q = (show X.PrimeDivisor →₀ ℤ from D') Q)
    [Module.Finite k (sectionSub k ⊤ D')]
    [Module.Finite k (localStepTgt k P 1)] :
    ell k D ≤ ell k D' ∧ (ell k D' : ℤ) ≤ ell k D + residueDeg k P := by
  refine ⟨ell_mono k hle, ?_⟩
  have hrn := finrank_localStepDom k ⊤ (sectionSub_mono k ⊤ hle)
  have hN14 := localStep_finrank_le k (TopologicalSpace.Opens.mem_top P.point)
    hstep hle hoff
  have hcast : (Module.finrank k (localStepDom k ⊤ D D') : ℤ) ≤ (residueDeg k P : ℤ) := by
    exact_mod_cast hN14
  simp only [ell] at *
  omega
