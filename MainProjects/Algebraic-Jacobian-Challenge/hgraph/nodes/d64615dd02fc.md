noncomputable def LaurentChartData.chartSquare (D : LaurentChartData Y) :
    Y.left.AffineCoverMVSquare where
  U₁ := D.V₀
  U₂ := D.V₁
  isAffineOpen_U₁ := D.isAffineOpen_V₀
  isAffineOpen_U₂ := D.isAffineOpen_V₁

    rw [D.inf_eq_basicOpen_x]
    exact D.isAffineOpen_V₀.basicOpen D.x
  cover := D.cover
