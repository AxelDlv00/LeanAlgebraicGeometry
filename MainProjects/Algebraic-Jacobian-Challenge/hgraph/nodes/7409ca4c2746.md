noncomputable def abelMapWitness {k : Type u} [Field k] (C : Over (Spec (.of k)))
    [SmoothOfRelativeDimension 1 C.hom] [IsProper C.hom] :
    divFunctor C ⟶ picSharp C :=
  abelKernelNatTrans C ≫ picNeg C
