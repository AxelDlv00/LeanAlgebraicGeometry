theorem ell_eq_zero_of_degK_neg
    (hledger : ∀ D : X.WeilDivisor, chi k U₀ U₁ D = chi k U₀ U₁ 0 + degK k D)
    {D : X.WeilDivisor} (hdeg : degK k D < 0)
    [Module.Finite k (sectionSub k ⊤ D)] :
    ell k D = 0 := by
  by_contra hne
  -- a nonzero section exists
  obtain ⟨f, hfmem, hfne⟩ : ∃ f : X.functionField, f ∈ sectionSub k ⊤ D ∧ f ≠ 0 := by
    by_contra hno
    have hbot : sectionSub k ⊤ D = ⊥ := by
      apply le_antisymm _ bot_le
      intro x hx
      rcases eq_or_ne x 0 with rfl | hx0
      · exact Submodule.zero_mem _
      · exact absurd ⟨x, hx, hx0⟩ hno
    rw [ell, hbot] at hne
    simp at hne
  obtain ⟨E, hEnonneg, hEclass⟩ :=
    exists_effective_linearEquiv_of_ne_zero_mem hfne hfmem
  have hchi := chi_eq_of_linearEquivalence k U₀ U₁ hEclass
  rw [hledger E, hledger D] at hchi
  have : degK k E = degK k D := by omega
  have hEpos := degK_nonneg_of_nonneg k hEnonneg
  omega
