private lemma modules_restr_preimage_top_eq_id {X Y : Scheme.{u}} (g : Y ⟶ X)
    (N : Y.Modules) (e : (⊤ : Y.Opens) ≤ g ⁻¹ᵁ ⊤) :
    N.presheaf.map (homOfLE e).op = 𝟙 _ :=
  (congrArg N.presheaf.map
    (show (homOfLE e).op = 𝟙 (Opposite.op (⊤ : Y.Opens)) from rfl)).trans
    (N.presheaf.map_id _)
