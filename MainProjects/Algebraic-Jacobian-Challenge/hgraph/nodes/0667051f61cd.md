lemma comm_iso_unique {C : Type*} [Category C] {A F F' : C}
    (q : A ⟶ F) [Epi q] {q' : A ⟶ F'} {f g : F ≅ F'}
    (hf : q ≫ f.hom = q') (hg : q ≫ g.hom = q') : f = g :=
  Iso.ext ((cancel_epi q).mp (hf.trans hg.symm))
