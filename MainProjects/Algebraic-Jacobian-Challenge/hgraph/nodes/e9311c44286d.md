theorem probe_preservesLimit_comp_cancel {J C D E : Type*} [Category J] [Category C]
    [Category D] [Category E] (K : J ⥤ C) (F : C ⥤ D) (G : D ⥤ E)
    [HasLimit K] [PreservesLimit K F] [PreservesLimit K (F ⋙ G)] :
