private lemma mapIso_naturality {C D : Type*} [Category C] [Category D] {F G : C ⥤ D}
    (r : F ≅ G) {A B : C} (e : A ≅ B) :
    F.mapIso e ≪≫ r.app B = r.app A ≪≫ G.mapIso e := by
  apply Iso.ext
  simp only [Iso.trans_hom, Functor.mapIso_hom, Iso.app_hom]
  exact r.hom.naturality e.hom
