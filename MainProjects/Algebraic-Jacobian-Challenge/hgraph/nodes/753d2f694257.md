theorem nontrivial_aut_F4 : Nontrivial (F4 ≃ₐ[F2] F4) := by
  have h := IsGalois.card_aut_eq_finrank F2 F4
  rw [GaloisField.finrank _ two_ne_zero] at h
  exact Finite.one_lt_card_iff_nontrivial.mp (by rw [h]; omega)
