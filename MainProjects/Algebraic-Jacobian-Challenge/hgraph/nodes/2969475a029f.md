structure AffineCoverMVSquare (X : Scheme.{u}) where
  /-- First affine open of the cover. -/
  U₁ : X.Opens
  /-- Second affine open of the cover. -/
  U₂ : X.Opens
  /-- `U₁` is affine. -/
  isAffineOpen_U₁ : IsAffineOpen U₁
  /-- `U₂` is affine. -/
  isAffineOpen_U₂ : IsAffineOpen U₂
  /-- The intersection `U₁ ⊓ U₂` is affine. -/
  isAffineOpen_inf : IsAffineOpen (U₁ ⊓ U₂)
  /-- Total cover: `U₁ ⊔ U₂ = ⊤`. -/
  cover : U₁ ⊔ U₂ = ⊤
