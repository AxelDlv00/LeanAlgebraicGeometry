lemma mem_finset_inf {α : Type*} [TopologicalSpace α] {ι : Type*} {s : Finset ι}
    {F : ι → Opens α} {x : α} :
    x ∈ s.inf F ↔ ∀ i ∈ s, x ∈ F i := by
  classical
  induction s using Finset.cons_induction with
  | empty => simp
  | cons a t ha ih => simp [Opens.mem_inf, ih]
