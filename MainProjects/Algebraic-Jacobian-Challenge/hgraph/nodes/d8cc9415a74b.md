theorem exists_bound_generatedAt
    (hledger : ∀ D : X.WeilDivisor, chi k U₀ U₁ D = chi k U₀ U₁ 0 + degK k D)
    (D₀ : X.WeilDivisor) (hbase : Subsingleton (H1Mod k U₀ U₁ D₀))
    (hpeel : ∀ D' : X.WeilDivisor,
      (∀ P : X.PrimeDivisor, (show X.PrimeDivisor →₀ ℤ from D₀) P ≤
        (show X.PrimeDivisor →₀ ℤ from D') P) →
      Peel k U₀ U₁ D₀ D')
    [∀ D : X.WeilDivisor, Module.Finite k (sectionSub k ⊤ D)]
    [∀ P : X.PrimeDivisor, Module.Finite k (localStepTgt k P 1)] :
    ∃ b : ℤ, ∀ (D : X.WeilDivisor) (P : X.PrimeDivisor),
      b + residueDeg k P ≤ degK k D → GeneratedAt k D P := by
  obtain ⟨b, hb⟩ :=
    exists_bound_subsingleton_h1Mod k U₀ U₁ hledger D₀ hbase hpeel
  refine ⟨b, fun D P hD => ?_⟩
  have hdegDP : degK k (D - pointDivisor P) = degK k D - residueDeg k P := by
    rw [degK_sub, degK_pointDivisor]
  refine generatedAt_of_vanishing k U₀ U₁ hledger D P (hb D (by omega)) ?_
  exact hb (D - pointDivisor P) (by omega)
