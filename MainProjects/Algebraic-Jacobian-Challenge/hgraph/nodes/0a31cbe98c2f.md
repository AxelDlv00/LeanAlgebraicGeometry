noncomputable def evalMap (D : X.WeilDivisor) (P : X.PrimeDivisor) :
    localStepDom k ⊤ (D - pointDivisor P) D →ₗ[k]
      localStepTgt k P (-(show X.PrimeDivisor →₀ ℤ from D - pointDivisor P) P) :=
  localStepMapₖ k (TopologicalSpace.Opens.mem_top P.point)
    (sub_pointDivisor_step D P)
