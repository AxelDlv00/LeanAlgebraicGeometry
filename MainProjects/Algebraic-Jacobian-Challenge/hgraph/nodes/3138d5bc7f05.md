def mapAlgHom : DualNumber k →ₐ[k] DualNumber A :=
  { mapRingHom (algebraMap k A) with
    commutes' := fun c => by
      refine TrivSqZeroExt.ext ?_ ?_ <;>
        simp [TrivSqZeroExt.algebraMap_eq_inl' k A, TrivSqZeroExt.algebraMap_eq_inl] }

@[simp]
