noncomputable def twMor (γ : k' ≃ₐ[k] k') : (Tw (k := k) γ).obj X' ⟶ X' :=
  rep.homEquiv.symm ((actOnPicEt C γ).inv.app (Opposite.op X') (rep.homEquiv (𝟙 X')))

/-- ITS COMPAT SQUARE -- the semilinearity condition, for free. -/
example (γ : k' ≃ₐ[k] k') :
    (twMor C rep γ).left ≫ X'.hom = X'.hom ≫ specGal (k := k) γ :=
  Over.w (twMor C rep γ)
