theorem qc_of_classes (J : Over (Spec (.of k)))
    (rep : (pic0TypeFunctor C).RepresentableBy J)
    (lam : pic0Subgroup C DivO)
    (hcl : ∀ y : J.left, ∃ q : overSpec k (Over.testPointField y) ⟶ DivO,
      pic0Map C q lam = rep.homEquiv (Over.testPoint y)) :
    QuasiCompact J.hom := by
  refine qc_of_surj J rep lam ?_
  intro y
  obtain ⟨q, hq⟩ := hcl y
  -- the two Over-morphisms agree because homEquiv is injective
  have hsq : q ≫ rep.homEquiv.symm lam = Over.testPoint y := by
    apply rep.homEquiv.injective
    rw [rep.homEquiv_comp (f := q), Equiv.apply_symm_apply]
    exact hq
  -- so the point y is hit
  refine ⟨q.left.base (Nonempty.some inferInstance), ?_⟩
  have : (q ≫ rep.homEquiv.symm lam).left.base (Nonempty.some inferInstance) = y := by
    rw [hsq]; exact Over.testPoint_base y _
  exact this
