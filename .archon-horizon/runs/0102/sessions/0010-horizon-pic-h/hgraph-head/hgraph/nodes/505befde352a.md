theorem divUniversalSeedFibreDivisor_eq_baseDivisor
    (p : PrimeSpectrum RZ)
    (hne : ∃ f ∈ divUniversalFibreKM C hπ g r₁ r₂ b₁ i j p.asIdeal.ResidueField,
      f ≠ 0) :
    divUniversalSeedFibreDivisor C hπ g r₁ r₂ b₁ b₂ i j hO hχ p =
      Scheme.baseDivisor p.asIdeal.ResidueField
        (divUniversalFibreKM C hπ g r₁ r₂ b₁ i j p.asIdeal.ResidueField)
        (windowN C p.asIdeal.ResidueField hπ g) hne := by
  let K := p.asIdeal.ResidueField
  change divUniversalSeedFibreDivisor C hπ g r₁ r₂ b₁ b₂ i j hO hχ p =
    Scheme.baseDivisor K (divUniversalFibreKM C hπ g r₁ r₂ b₁ i j K)
      (windowN C K hπ g) hne
  have hOK : Sheaf.h0 ((relCurve C K).moduleKSheaf K) = 1 :=
    h0_relCurve_baseField C K
  have hχK : Sheaf.chi ((relCurve C K).moduleKSheaf K) = 1 - (g : ℤ) :=
    chi_relCurve_baseField C K g hχ
  have hNnorm : ∀ D' : (relCurve C K).CurveDivisor,
      CurveDivisor.deg K D' ≤ 2 * (g : ℤ) →
        Subsingleton (Sheaf.HModule
          ((relCurve C K).divisorSheaf K (windowN C K hπ g - D')) 1) :=
    fun D' hD' => subsingleton_h1_windowN_sub C K hπ g hO hχ hχK D' hD'
  have hNdeg : 2 * (g : ℤ) ≤ CurveDivisor.deg K (windowN C K hπ g) :=
    two_mul_genus_le_deg_windowN C K hπ g hO hχ
  have hspec := divUniversalSeedFibreDivisor_spec
    C hπ g r₁ r₂ b₁ b₂ i j hO hχ p
  refine CurveDivisor.ext_coeffAt fun x hx => ?_
  rw [Scheme.coeffAt_baseDivisor K hne hx, hspec.2.2.1]
  exact (baseDivisorAt_window_normalization g hOK hχK
    (windowN C K hπ g) hNnorm hNdeg _ hspec.1 hspec.2.1 hx).symm
