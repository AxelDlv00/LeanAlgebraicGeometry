theorem relPicDeg_mul (K : Type u) [Field K] [Algebra k K]
    (x y : relPic C (overSpec k K)) :
    relPicDeg K (x * y) = relPicDeg K x + relPicDeg K y :=
  (relPicDeg K).map_add x y

/-- The relative degree of the trivial class is zero. -/
@[simp]
