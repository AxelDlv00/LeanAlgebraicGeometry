noncomputable def pointLiftResidue
    (z : relCurve C R) : relCurve C (relCurveBasePoint C R z).asIdeal.ResidueField := by
  let f : relCurve C R ⟶ Spec (CommRingCat.of R) :=
    (snd C (overSpec k R)).left
  let p : Spec (CommRingCat.of R) := relCurveBasePoint C R z
  let K := p.asIdeal.ResidueField
  let e := Scheme.Spec.residueFieldIso (CommRingCat.of R) p
  let q : Spec ((relCurve C R).residueField z) ⟶
      Spec (CommRingCat.of K) :=
    Spec.map (e.inv ≫ (f.residueFieldMap z))
  have hq :
      (Scheme.fromSpecResidueField (relCurve C R) z) ≫ f =
        q ≫ Spec.map (CommRingCat.ofHom (algebraMap R K)) := by
    dsimp [q, e, K, p, f]
    rw [← Scheme.Hom.SpecMap_residueFieldMap_fromSpecResidueField]
    simp only [Category.assoc, ← Spec.map_comp]
    rw [← Category.assoc, Scheme.Spec.map_residueFieldIso_inv_eq_fromSpecResidueField]
  let hpb := Over.isPullback_whiskerLeft_left C
    (overSpecMap (k := k) R K)
  let l : Spec ((relCurve C R).residueField z) ⟶ relCurve C K :=
    hpb.lift (Scheme.fromSpecResidueField (relCurve C R) z) q hq
  exact l (default : Spec ((relCurve C R).residueField z))
