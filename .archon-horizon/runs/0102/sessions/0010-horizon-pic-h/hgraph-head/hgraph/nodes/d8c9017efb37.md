lemma boundedSections_mono {D D' : X.CurveDivisor} (h : D ≤ D') (U : X.Opens) :
    boundedSections K D U ≤ boundedSections K D' U :=
  fun _ hg x hx hxU => le_trans (hg x hx hxU) (divisorBound_mono h hx)
