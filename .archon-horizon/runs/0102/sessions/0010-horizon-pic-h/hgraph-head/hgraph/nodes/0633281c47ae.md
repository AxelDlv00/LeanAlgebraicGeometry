theorem sectionIdeal_eq_span_restrict (d : X.LocalEquations) (x : X) {U : X.Opens}
    (hU : U ≤ d.cover.opens x) :
    d.sectionIdeal U = Ideal.span {
      (X.presheaf.map (homOfLE hU).op).hom (d.eqn x)} := by
  apply d.sectionIdeal_eq_span_of_generator
  · intro z hz
    rw [TopCat.Presheaf.germ_res_apply]
    exact d.regular x z (hU hz)
  · intro z hz
    rw [TopCat.Presheaf.germ_res_apply]
    exact d.germ_eqn_span_eq x z (hU hz)

end Scheme.LocalEquations
