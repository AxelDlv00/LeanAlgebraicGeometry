theorem hcl_false_of_not_compactSpace {J : Over (Spec (.of k))}
    (rep : (pic0TypeFunctor C).RepresentableBy J)
    (lam : pic0Subgroup C DivO)
    (hnc : ¬ CompactSpace J.left) :
    ¬ (∀ y : J.left, ∃ q : overSpec k (Over.testPointField y) ⟶ DivO,
      pic0Map C q lam = rep.homEquiv (Over.testPoint y)) :=
  fun hcl => hnc (hcl_implies_compactSpace rep lam hcl)
