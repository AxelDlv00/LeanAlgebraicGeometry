private lemma id_appLE_apply {X : Scheme.{u}} (e : (⊤ : X.Opens) ≤ 𝟙 X ⁻¹ᵁ ⊤)
    (t : Γ(X, ⊤)) : Scheme.Hom.appLE (𝟙 X) ⊤ ⊤ e t = t := by
  have h : Scheme.Hom.appLE (𝟙 X) ⊤ ⊤ e =
      Scheme.Hom.app (𝟙 X) ⊤ ≫ X.presheaf.map (homOfLE e).op := rfl
  have h2 : (homOfLE e).op = 𝟙 (Opposite.op (⊤ : X.Opens)) := rfl
  rw [h, Scheme.Hom.id_app, Category.id_comp, h2, X.presheaf.map_id,
    CommRingCat.id_apply]
