  of the Laurent

  polynomial `z` lies in `I`, then `1 + z` is a two-chart coboundary.


  The deformation is along the whole ideal `I` (not a single generator), so the induction
  halves the

  *ideal power*.  Split `z = P + Q` into its non-negative and negative degree parts;
  `1 + P` and

  `1 + Q` are chart units (`P, Q` nilpotent, coefficients in the nilpotent `I`), their
  product is a

  coboundary `c = 1 + z + P·Q`, and the cross term `P·Q` has coefficients in `I²`

  (`coeff_mul_mem_mul`).  The residual `c⁻¹·(1 + z) = 1 + w` has `w`''s coefficients
  in `I²`, and

  `(I²) ^ (2 ^ n) = I ^ (2 ^ (n+1)) = ⊥`, so the induction hypothesis applies and
  coboundaries are a

  subgroup.'

theorem laurentCoboundary_one_add_of_coeffs_mem (n : ℕ) :
    ∀ (I : Ideal A), I ^ (2 ^ n) = ⊥ → ∀ (z : LaurentPolynomial A), (∀ i, z i ∈ I) →
      ∀ (u : (LaurentPolynomial A)ˣ), (u : LaurentPolynomial A) = 1 + z →
        u ∈ laurentCoboundaryUnits A := by
  induction n with
  | zero =>
    intro I hI z hz u hu
    have hIbot : I = ⊥ := by simpa using hI
    have hz0 : z = 0 := by
      apply Finsupp.ext; intro i
      have hzi : z i ∈ (⊥ : Ideal A) := hIbot ▸ hz i
      rw [Ideal.mem_bot] at hzi
      rw [hzi]; rfl
    have h1 : (u : LaurentPolynomial A) = 1 := by rw [hu, hz0, add_zero]
    rw [show u = 1 from Units.ext (by rw [h1, Units.val_one])]
    exact Subgroup.one_mem _
  | succ n ih =>
    intro I hI z hz u hu
    have hInil : IsNilpotent I := ⟨2 ^ (n + 1), hI⟩
    set P : Polynomial A := posPoly z with hP
    set Q : Polynomial A := negPoly z with hQ
    have hPu : IsUnit (1 + P) :=
      (poly_nilpotent_of_coeffs_mem hInil (posPoly_coeff_mem hz)).isUnit_one_add
    have hQu : IsUnit (1 + Q) :=
      (poly_nilpotent_of_coeffs_mem hInil (negPoly_coeff_mem hz)).isUnit_one_add
    set c : (LaurentPolynomial A)ˣ :=
      (hPu.unit.map (Polynomial.toLaurent).toMonoidHom)
        * (hQu.unit.map (rightChart A).toMonoidHom) with hc
    have hcmem : c ∈ laurentCoboundaryUnits A :=
      TruncExpCech.mem_cechCoboundaryUnits.mpr ⟨hPu.unit, hQu.unit, rfl⟩
    have hcval : (c : LaurentPolynomial A)
        = 1 + z + Polynomial.toLaurent P * rightChart A Q := by
      rw [hc, Units.val_mul, Units.coe_map, Units.coe_map, IsUnit.unit_spec, IsUnit.unit_spec]
      change Polynomial.toLaurent (1 + P) * rightChart A (1 + Q)
        = 1 + z + Polynomial.toLaurent P * rightChart A Q
      rw [map_add, map_add, map_one, map_one]
      rw [show (1 + Polynomial.toLaurent P) * (1 + rightChart A Q)
          = 1 + (Polynomial.toLaurent P + rightChart A Q)
            + Polynomial.toLaurent P * rightChart A Q from by ring, posPoly_add_negPoly z]
    set w : LaurentPolynomial A :=
      -((↑c⁻¹ : LaurentPolynomial A) * (Polynomial.toLaurent P * rightChart A Q)) with hw
    have hwmem : ∀ m, w m ∈ I ^ 2 := by
      intro m
      have hxy : ∀ j, (Polynomial.toLaurent P * rightChart A Q) j ∈ I * I :=
        coeff_mul_mem_mul (toLaurent_posPoly_coeff_mem hz) (rightChart_negPoly_coeff_mem hz)
      have h2 : ((↑c⁻¹ : LaurentPolynomial A) * (Polynomial.toLaurent P * rightChart A Q)) m
          ∈ (⊤ : Ideal A) * (I * I) := coeff_mul_mem_mul (fun _ => Submodule.mem_top) hxy m
      rw [Ideal.top_mul] at h2
      have hwm : w m = -(((↑c⁻¹ : LaurentPolynomial A)
          * (Polynomial.toLaurent P * rightChart A Q)) m) := rfl
      rw [hwm, sq]
      exact Submodule.neg_mem _ h2
    have hcu : ((c⁻¹ * u : (LaurentPolynomial A)ˣ) : LaurentPolynomial A) = 1 + w := by
      have hinv : (↑c⁻¹ : LaurentPolynomial A) * (c : LaurentPolynomial A) = 1 := by
        rw [← Units.val_mul, inv_mul_cancel, Units.val_one]
      rw [Units.val_mul, hu, hw]
      rw [show (1 : LaurentPolynomial A) + z
          = (1 + z + Polynomial.toLaurent P * rightChart A Q)
            - Polynomial.toLaurent P * rightChart A Q from by ring, ← hcval]
      rw [mul_sub, hinv]; ring
    have hI2 : (I ^ 2) ^ (2 ^ n) = ⊥ := by rw [← pow_mul, ← pow_succ']; exact hI
    have hr : c⁻¹ * u ∈ laurentCoboundaryUnits A := ih (I ^ 2) hI2 w hwmem (c⁻¹ * u) hcu
    rw [show u = c * (c⁻¹ * u) from by group]
    exact Subgroup.mul_mem _ hcmem hr
