private lemma T_neg_one_pow {R : Type u} [CommRing R] (n : ℕ) :
    (LaurentPolynomial.T (-1 : ℤ) : LaurentPolynomial R) ^ n
      = LaurentPolynomial.T (-(n : ℤ)) := by
  have h := LaurentPolynomial.T_pow (R := R) (-1 : ℤ) n
  rw [show ((n : ℤ) * (-1)) = -(n : ℤ) from by ring] at h
  exact h
