theorem negPoly_coeff_mem {I : Ideal A} {z : LaurentPolynomial A} (hz : ∀ i, z i ∈ I) (m : ℕ) :
    (negPoly z).coeff m ∈ I := by
  classical
  unfold negPoly
  rw [Polynomial.finsetSum_coeff]
  refine Ideal.sum_mem _ (fun i _ => ?_)
  rw [Polynomial.coeff_monomial]
  split
  · exact hz i
  · exact Ideal.zero_mem _
