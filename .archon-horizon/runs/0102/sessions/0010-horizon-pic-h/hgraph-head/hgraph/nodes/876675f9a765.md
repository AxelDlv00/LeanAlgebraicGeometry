theorem ker_rTensor_le_range_subtype_of_flat_cokernels
    (δ : M →ₗ[R] N)
    [Module.Flat R (M ⧸ LinearMap.ker δ)]
    [Module.Flat R (N ⧸ LinearMap.range δ)]
    (A : Type u) [AddCommGroup A] [Module R A] :
    LinearMap.ker (δ.rTensor A) ≤
      LinearMap.range ((LinearMap.ker δ).subtype.rTensor A) := by
  intro x hx
  have hcomm : TensorProduct.comm R M A x ∈ LinearMap.ker (δ.lTensor A) := by
    rw [LinearMap.mem_ker, LinearMap.lTensor_comm]
    rw [hx, map_zero]
  obtain ⟨z, hz⟩ :=
    (LinearMap.ker_lTensor_eq_of_flat_coker δ A).mp hcomm
  refine ⟨(TensorProduct.comm R A (LinearMap.ker δ)).symm z, ?_⟩
  apply (TensorProduct.comm R M A).injective
  have hsub := LinearMap.lTensor_comm (LinearMap.ker δ).subtype
    (TensorProduct.comm R A (LinearMap.ker δ)).symm z
  rw [TensorProduct.comm_apply_symm_apply] at hsub
  rw [← hsub, hz]
  exact TensorProduct.comm_apply_symm_apply _ _

end KernelSpan
