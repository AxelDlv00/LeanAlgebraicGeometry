private lemma ratio_telescope {G : Type u} [CommGroup G] {b₁ b₂ L M₁ M₂ P : G}
    (hb : P = b₁ / b₂) (hm : M₂ = P * M₁) :
    b₁⁻¹ * L⁻¹ * M₁⁻¹ = b₂⁻¹ * L⁻¹ * M₂⁻¹ := by
  subst hm; subst hb
  rw [mul_inv (b₁ / b₂) M₁, inv_div b₁ b₂, div_eq_mul_inv b₂ b₁,
    mul_assoc b₂ b₁⁻¹ M₁⁻¹,
    mul_assoc b₂⁻¹ L⁻¹ (b₂ * (b₁⁻¹ * M₁⁻¹)),
    mul_left_comm L⁻¹ b₂ (b₁⁻¹ * M₁⁻¹),
    ← mul_assoc b₂⁻¹ b₂ (L⁻¹ * (b₁⁻¹ * M₁⁻¹)),
    inv_mul_cancel b₂, one_mul, mul_left_comm L⁻¹ b₁⁻¹ M₁⁻¹, ← mul_assoc]
