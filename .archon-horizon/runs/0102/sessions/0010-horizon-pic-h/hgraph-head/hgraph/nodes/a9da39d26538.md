theorem split_of_shared {ι : Type u} {X : ι → Scheme.{u}}
    (f : ∀ i, yoneda.obj (X i) ⟶ (pic0SigmaSheaf C).1) (V : ∀ i, (X i).Opens)
    (huniv : ∀ i, IsOpenImmersion.presheaf (restrictChart (f i) (V i)))
    (hcov : Confined C f V) :
    ∃ Vf Vc : ∀ i, (X i).Opens, (∀ i, Vc i ≤ Vf i) ∧
      (∀ i, IsOpenImmersion.presheaf (restrictChart (f i) (Vf i))) ∧ Confined C f Vc :=
  ⟨V, V, fun _ => le_refl _, huniv, hcov⟩
