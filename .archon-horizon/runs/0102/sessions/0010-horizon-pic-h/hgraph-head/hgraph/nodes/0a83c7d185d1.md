private theorem relCurve_residueField_leg
    (z : relCurve C R) :
    (relCurve C R).fromSpecResidueField z ≫
        (snd C (overSpec k R)).left =
      Spec.map (Scheme.Spec.residueFieldIso (CommRingCat.of R)
          (relCurveBasePoint C R z)).inv ≫
        Spec.map (CommRingCat.ofHom
          (algebraMap R (relCurveBasePoint C R z).asIdeal.ResidueField)) ≫
        (snd C (overSpec k R)).left := by
  -- This is just the residue-field naturality square, with the `Spec R` residue-field
  -- isomorphism expanded on the right.
  rw [← Scheme.Hom.SpecMap_residueFieldMap_fromSpecResidueField]
  rw [← Scheme.Spec.map_residueFieldIso_inv_eq_fromSpecResidueField]
  simp only [Category.assoc, ← Spec.map_comp]
  rfl
