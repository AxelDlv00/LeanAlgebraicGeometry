theorem forall_mul_read_mem_span_of_forall_smul_eq_zero
    (K : Submodule R (relThetaSections C R π a)) (b : Bool)
    (s : relThetaSections C R π a)
    {h : Γ(relCurve C R, relPinnedChart C R π b)}
    (hann : ∀ n : ↥(chartColengthModule K b s), h • n = 0) :
    ∀ ⦃ψ : relThetaSections C R π a⦄, ψ ∈ K →
      h * relThetaResSide a b (le_rfl) ψ
        ∈ Ideal.span {relThetaResSide a b (le_rfl) s} := by
  intro ψ hψ
  set I := Ideal.span {relThetaResSide a b (le_rfl) s} with hI
  have hmem := mk_relThetaResSide_mem_chartColengthModule K b s hψ
  have hz := hann ⟨_, hmem⟩
  have hz' : h • Ideal.Quotient.mk I (relThetaResSide a b (le_rfl) ψ) = 0 := by
    have h2 := congrArg (Subtype.val) hz
    rw [SetLike.val_smul, ZeroMemClass.coe_zero] at h2
    exact h2
  rw [show h • Ideal.Quotient.mk I (relThetaResSide a b (le_rfl) ψ)
        = Ideal.Quotient.mk I (h * relThetaResSide a b (le_rfl) ψ) by
      rw [Algebra.smul_def, Ideal.Quotient.algebraMap_eq, ← map_mul]] at hz'
  exact (Ideal.Quotient.eq_zero_iff_mem).mp hz'
