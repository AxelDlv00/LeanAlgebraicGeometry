theorem baseDivisor_divUniversalFibreKM_eq_seedFibreDivisor
    (p : PrimeSpectrum RZ) :
    Scheme.baseDivisor p.asIdeal.ResidueField
        (divUniversalFibreKM C hpi g r1 r2 b1 i j p.asIdeal.ResidueField)
        (windowN C p.asIdeal.ResidueField hpi g)
        (exists_mem_ne_zero_divUniversalFibreKM_seedPrime
          C hpi g r1 r2 b1 b2 i j hO hchi p) =
      divUniversalSeedFibreDivisor
        C hpi g r1 r2 b1 b2 i j hO hchi p := by
  let K := p.asIdeal.ResidueField
  let D := divUniversalSeedFibreDivisor C hpi g r1 r2 b1 b2 i j hO hchi p
  have hspec := divUniversalSeedFibreDivisor_spec
    C hpi g r1 r2 b1 b2 i j hO hchi p
  have hOK : Sheaf.h0 ((relCurve C K).moduleKSheaf K) = 1 :=
    h0_relCurve_baseField C K
  have hchiK : Sheaf.chi ((relCurve C K).moduleKSheaf K) = 1 - (g : Int) :=
    chi_relCurve_baseField C K g hchi
  have hNnorm : forall D' : (relCurve C K).CurveDivisor,
      CurveDivisor.deg K D' ≤ 2 * (g : Int) →
        Subsingleton
          (Sheaf.HModule ((relCurve C K).divisorSheaf K
            (windowN C K hpi g - D')) 1) :=
    fun D' hD' =>
      subsingleton_h1_windowN_sub C K hpi g hO hchi hchiK D' hD'
  have hNdeg : 2 * (g : Int) ≤ CurveDivisor.deg K (windowN C K hpi g) :=
    two_mul_genus_le_deg_windowN C K hpi g hO hchi
  apply CurveDivisor.ext_coeffAt
  intro x hx
  rw [Scheme.coeffAt_baseDivisor]
  change (Scheme.baseDivisorAt K
      (divUniversalFibreKM C hpi g r1 r2 b1 i j K)
      (windowN C K hpi g) ⟨x, hx⟩ : Int) = coeffAt hx D
  rw [show divUniversalFibreKM C hpi g r1 r2 b1 i j K =
      Scheme.divisorSections K (windowN C K hpi g - D) ⊤ by
    exact hspec.2.2.1]
  exact baseDivisorAt_window_normalization g hOK hchiK
    (windowN C K hpi g) hNnorm hNdeg D hspec.1 hspec.2.1 hx
