lemma algebraMap_tmul_sub_mem_span_point (p : Polynomial k) :
    (algebraMap (Polynomial k) B p) ⊗ₜ[k] (1 : F)
        - (1 : B) ⊗ₜ[k] (algebraMap (Polynomial k) F p)
      ∈ Ideal.span {pointGen k B F} := by
  rw [Ideal.mem_span_singleton, pointGen, algebraMap_eq_aeval,
    algebraMap_eq_aeval (B := F)]
  have hL : (aeval (coord k B) p) ⊗ₜ[k] (1 : F)
      = aeval (coord k B ⊗ₜ[k] (1 : F)) p := by
    rw [← Algebra.TensorProduct.includeLeft_apply (S := k),
      ← Polynomial.aeval_algHom_apply, Algebra.TensorProduct.includeLeft_apply]
  have hR : (1 : B) ⊗ₜ[k] (aeval (coord k F) p)
      = aeval ((1 : B) ⊗ₜ[k] coord k F) p := by
    rw [← Algebra.TensorProduct.includeRight_apply,
      ← Polynomial.aeval_algHom_apply, Algebra.TensorProduct.includeRight_apply]
  rw [hL, hR]
  exact aeval_sub_dvd_point _ _ p
