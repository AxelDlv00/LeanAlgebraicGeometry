@[simp] lemma notMem_basePrime_iff {B : Type*} [CommRing B] [IsLocalRing B] (φ : S →+* B)
    {f : R} : f ∉ (basePrime (R := R) φ).asIdeal ↔ IsUnit (φ (algebraMap R S f)) := by
  rw [← IsLocalRing.notMem_maximalIdeal]
  exact not_congr (by rw [basePrime]; rfl)
