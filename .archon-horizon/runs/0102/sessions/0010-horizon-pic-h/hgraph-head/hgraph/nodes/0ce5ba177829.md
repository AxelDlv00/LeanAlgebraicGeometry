theorem baseDivisorAt_pointwiseFibrePoleDivisor_eq_zero
    (z : relCurve C RZ)
    (hzg : relCurveResiduePoint C RZ z ≠
      genericPoint (relCurve C (relCurveBasePoint C RZ z).asIdeal.ResidueField)) :
    (Scheme.baseDivisorAt (relCurveBasePoint C RZ z).asIdeal.ResidueField
      (divUniversalFibreKM C hπ g r₁ r₂ b₁ i j
        (relCurveBasePoint C RZ z).asIdeal.ResidueField)
      (pointwiseFibrePoleDivisor C hπ g r₁ r₂ b₁ b₂ i j hO hχ z)
      ⟨relCurveResiduePoint C RZ z, hzg⟩ : ℤ) = 0 := by
  let p := relCurveBasePoint C RZ z
  let K := p.asIdeal.ResidueField
  have hne : ∃ f ∈ divUniversalFibreKM C hπ g r₁ r₂ b₁ i j K, f ≠ 0 :=
    exists_mem_ne_zero_divUniversalFibreKM_seedPrime
      C hπ g r₁ r₂ b₁ b₂ i j hO hχ p
  have hKMle := divUniversalFibreKM_le C hπ g r₁ r₂ b₁ i j K
  have hDbase := divUniversalSeedFibreDivisor_eq_baseDivisor
    C hπ g r₁ r₂ b₁ b₂ i j hO hχ p hne
  unfold pointwiseFibrePoleDivisor
  rw [hDbase]
  exact baseDivisorAt_sub_baseDivisor_eq_zero hKMle hne hzg
