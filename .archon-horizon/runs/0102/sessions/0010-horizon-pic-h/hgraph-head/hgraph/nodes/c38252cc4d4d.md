private lemma germ_resHom {Y : Scheme.{u}} {V W : Y.Opens} (h : W ≤ V) (x : Y) (hx : x ∈ W)
    (t : Γ(Y, V)) :
    (Y.presheaf.germ W x hx).hom (Y.resHom h t) =
      (Y.presheaf.germ V x (h hx)).hom t :=
  Y.presheaf.germ_res_apply (homOfLE h) x hx t
