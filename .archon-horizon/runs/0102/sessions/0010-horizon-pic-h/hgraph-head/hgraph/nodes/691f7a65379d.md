theorem qc_at_arbitrary_parameter (n : ℕ)
    (b₁ : Module.Basis (Fin r₁) k ↥(Scheme.divisorSections k B ⊤))
    (b₂ : Module.Basis (Fin r₂) k ↥(Scheme.divisorSections k (A + B) ⊤))
    {J : Over (Spec (.of k))}
    (rep : (pic0TypeFunctor C).RepresentableBy J)
    (lam : pic0Subgroup C (divSchemeOver k A B n r₁ r₂ b₁ b₂))
    (hcl : ∀ y : J.left,
      ∃ q : overSpec k (Over.testPointField y) ⟶ divSchemeOver k A B n r₁ r₂ b₁ b₂,
        pic0Map C q lam = rep.homEquiv (Over.testPoint y)) :
    QuasiCompact J.hom :=
  quasiCompact_of_pic0_class_surjective rep lam hcl
