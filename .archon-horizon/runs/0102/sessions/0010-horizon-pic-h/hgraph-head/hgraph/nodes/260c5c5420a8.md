noncomputable def sectionIdeal (d : X.LocalEquations) (U : X.Opens) : Ideal Γ(X, U) where
  carrier := {s | ∀ (z : X) (hz : z ∈ U),
    (X.presheaf.germ U z hz).hom s ∈ d.stalkIdeal z}
  zero_mem' := by
    intro z hz
    rw [map_zero]
    exact Ideal.zero_mem _
  add_mem' := by
    intro s t hs ht z hz
    rw [map_add]
    exact Ideal.add_mem _ (hs z hz) (ht z hz)
  smul_mem' := by
    intro r s hs z hz
    rw [smul_eq_mul, map_mul]
    exact Ideal.mul_mem_left _ _ (hs z hz)
