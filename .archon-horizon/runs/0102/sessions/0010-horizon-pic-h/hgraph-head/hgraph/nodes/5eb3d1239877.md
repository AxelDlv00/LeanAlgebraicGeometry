noncomputable def jumpModule : ModuleCat.{u} K :=
  ModuleCat.of K
    (pointLattice K hx (coeffAt hx D) ⧸
      (pointLattice K hx (coeffAt hx D - 1)).submoduleOf (pointLattice K hx (coeffAt hx D)))
