theorem IsAffineOpen.exists_notMem_primeIdealOf_mul_mem_span_singleton_of_germ_mem_span
    {X : Scheme.{u}} {U : X.Opens} (hU : IsAffineOpen U) {z : X} (hz : z ∈ U)
    {b x : Γ(X, U)}
    (h : (X.presheaf.germ U z hz).hom x
      ∈ Ideal.span {(X.presheaf.germ U z hz).hom b}) :
    ∃ r ∉ (hU.primeIdealOf ⟨z, hz⟩).asIdeal, r * x ∈ Ideal.span {b} := by
  letI := X.presheaf.algebra_section_stalk (⟨z, hz⟩ : U)
  haveI := hU.isLocalization_stalk (⟨z, hz⟩ : U)
  set pz := hU.primeIdealOf (⟨z, hz⟩ : U) with hpz
  set g := (X.presheaf.germ U z hz).hom with hg
  rw [Ideal.mem_span_singleton] at h
  obtain ⟨y, hy⟩ := h
  -- write the stalk cofactor `y` as a fraction with denominator `t ∉ pz` (germ = algebraMap)
  obtain ⟨⟨a, t⟩, ht⟩ := IsLocalization.surj pz.asIdeal.primeCompl y
  have ht' : y * g (t : Γ(X, U)) = g a := ht
  -- clear denominators: `germ (t·x) = germ (a·b)`, so `t·x − a·b` dies in the stalk
  have hzero : g ((t : Γ(X, U)) * x - a * b) = 0 := by
    rw [map_sub, map_mul, map_mul, hy]
    linear_combination (g b) * ht'
  obtain ⟨m, hm⟩ := (IsLocalization.map_eq_zero_iff pz.asIdeal.primeCompl
    (X.presheaf.stalk z) _).mp hzero
  refine ⟨(m : Γ(X, U)) * (t : Γ(X, U)), ?_, ?_⟩
  · exact fun hc => (pz.isPrime.mem_or_mem hc).elim m.2 t.2
  · have hmt : (m : Γ(X, U)) * (t : Γ(X, U)) * x = (m : Γ(X, U)) * a * b := by
      linear_combination hm
    rw [hmt]
    exact Ideal.mem_span_singleton.mpr ⟨(m : Γ(X, U)) * a, by ring⟩
