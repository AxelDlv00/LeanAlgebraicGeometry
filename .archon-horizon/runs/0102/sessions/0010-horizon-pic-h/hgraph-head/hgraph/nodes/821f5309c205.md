theorem Scheme.surjective_selector_comp {X Y : Scheme.{u}} (f : X ⟶ Y) (sel : Y → Bool)
    (hsel : Function.Surjective sel) (hf : Function.Surjective f.base) :
    Function.Surjective (fun x ↦ sel (f.base x)) :=
  hsel.comp hf
