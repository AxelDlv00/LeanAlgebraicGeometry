lemma lift_diff_self {T : C} (G : C) [GrpObj G] (a : T ⟶ G) :
    lift a a ≫ diff G = toUnit T ≫ η[G] := by
  rw [diff, GrpObj.comp_div, lift_fst, lift_snd, div_self']
  rfl
