lemma actionMap_eq_lmul' (w : B ⊗[A] B) :
    actionMap A B B w = Algebra.TensorProduct.lmul' A (S := B) w := by
  induction w with
  | zero => simp
  | tmul b m => simp [smul_eq_mul]
  | add x y hx hy => simp [hx, hy]
