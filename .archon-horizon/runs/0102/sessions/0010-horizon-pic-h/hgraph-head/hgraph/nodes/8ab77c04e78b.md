    (lam : pic0Subgroup C DivO)
    (hcl : ∀ y : J.left, ∃ q : overSpec k (Over.testPointField y) ⟶ DivO,
      pic0Map C q lam = rep.homEquiv (Over.testPoint y)) :
    ∀ y : J.left, ∃ (T : Over (Spec (.of k))) (_ : Nonempty T.left)
