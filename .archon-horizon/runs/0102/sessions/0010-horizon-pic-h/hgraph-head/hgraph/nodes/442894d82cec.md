lemma relPicMk_eq_relPicMk_iff {T : Over (Spec (.of k))}
    {L L' : (C ⊗ T).left.CechPic} :
    relPicMk C T L = relPicMk C T L' ↔ L / L' ∈ picFromBase C T :=
  QuotientGroup.eq_iff_div_mem
