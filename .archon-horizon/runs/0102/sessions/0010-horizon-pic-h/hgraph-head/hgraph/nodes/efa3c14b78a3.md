noncomputable def sectionMul (A B : X.CurveDivisor) :
    ↥(divisorSections K A ⊤) ⊗[K] ↥(divisorSections K B ⊤) →ₗ[K]
      ↥(divisorSections K (A + B) ⊤) :=
  TensorProduct.lift (sectionMulBilin K A B)
