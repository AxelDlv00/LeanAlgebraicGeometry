def relPic (T : Over (Spec (.of k))) : Type u :=
  (C ⊗ T).left.CechPic ⧸ picFromBase C T

noncomputable instance (T : Over (Spec (.of k))) : CommGroup (relPic C T) :=
  QuotientGroup.Quotient.commGroup (picFromBase C T)
