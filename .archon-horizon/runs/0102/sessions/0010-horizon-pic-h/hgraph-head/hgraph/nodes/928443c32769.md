    A.Glued →ₗ[K]
      ∀ p : {p // p ∈ (Scheme.presentationDivisor K d.presentation).support},
        ((relCurve C K).presheaf.stalk p.1.1 ⧸ d.stalkIdeal p.1.1) :=
