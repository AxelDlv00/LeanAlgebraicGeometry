theorem picEtPullback_comp (g : D ⟶ E) (g' : E ⟶ F) (T : Over (Spec (.of k)))
    (s : picEt F T) :
    picEtPullback (g ≫ g') T s = picEtPullback g T (picEtPullback g' T s) :=
  picEt.ext fun U => PicEtAff.curveMap_comp g g' (s.1 U)
