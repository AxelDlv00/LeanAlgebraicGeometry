noncomputable def stalkStructureHom (f : X ⟶ Spec (CommRingCat.of k)) (x : X) :
    CommRingCat.of k ⟶ X.presheaf.stalk x :=
  (Scheme.ΓSpecIso (CommRingCat.of k)).inv ≫
    (Spec (CommRingCat.of k)).presheaf.germ ⊤ (f x) trivial ≫ f.stalkMap x
