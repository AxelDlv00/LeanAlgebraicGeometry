def abelDivTrans : divFunctor C π n ⟶ picDegLayerFunctor C (n : ℤ) where
  app T := ↾fun s => ⟨abelDiv C π n T.unop s, fun K _ _ t => degAt_abelDiv s t⟩
  naturality T T' g := by
    ext s
    refine Subtype.ext ?_
    exact (picEtMap_abelDiv g.unop s).symm
