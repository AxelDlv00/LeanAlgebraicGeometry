\chapter*{About This Blueprint}
\label{rigid-local-systems-multiplicative-eigenvalue-problem-about}

This project follows \emph{Rigid local systems and the multiplicative eigenvalue problem} by Prakash Belkale (2022). The
retrieved source is archived under
\href{../../../../../references/rigid-local-systems-multiplicative-eigenvalue-problem/}{\texttt{rigid-local-systems-multiplicative-eigenvalue-problem}} and is indexed in
the workspace reference manifest. The corresponding preprint is
\href{https://arxiv.org/abs/2005.12457}{arXiv:2005.12457}.

Fix multiplicative local monodromy data of rank $\ell$ and determinant one. The multiplicative eigenvalue problem produces a polytope whose distinguished faces are called $F$-vertices. The paper relates unitary rigid local systems to parabolic bundles, extremal rays, strange duality, and the KZ equations, translating rigidity into explicit inequalities.

This blueprint is an independently worded, source-faithful mathematical
outline of the paper's definitions, constructions, and principal results. It
should be read alongside the original source; the source anchor on each node
points to the complete extracted TeX. Statements are intentionally marked
\notready until a matching Lean declaration and proof have been added.

\section{Main theorem or conjecture}

\begin{theorem}[Rigid systems and polytope vertices]
\label{rigid-local-systems-multiplicative-eigenvalue-problem-main}
\source{rigid-local-systems-multiplicative-eigenvalue-problem}
\uses{rigid-local-systems-multiplicative-eigenvalue-problem-setup}
For fixed $n$, the assignment from multiplicative local monodromy data to its associated vertex is a bijection between rigid irreducible unitary local systems and the $F$-vertices of the multiplicative eigenvalue polytope.
\notready
\end{theorem}

\begin{theorem}[No finite-monodromy prime-rank examples]
\label{rigid-local-systems-multiplicative-eigenvalue-problem-secondary}
\source{rigid-local-systems-multiplicative-eigenvalue-problem}
\uses{rigid-local-systems-multiplicative-eigenvalue-problem-main}
If $n$ is prime, there is no rigid irreducible local system of rank greater than one with finite global monodromy and local monodromies of $n$-th-root-of-unity type.
\notready
\end{theorem}
