\chapter*{About This Blueprint}
\label{bounds-stalks-perverse-sheaves-characteristic-p-shende-about}

This project follows \emph{Bounds for the stalks of perverse sheaves in characteristic p and a conjecture of Shende and Tsimerman} by Will Sawin (2020). The
retrieved source is archived under
\href{../../../../../references/bounds-stalks-perverse-sheaves-characteristic-p-shende/}{\texttt{bounds-stalks-perverse-sheaves-characteristic-p-shende}} and is indexed in
the workspace reference manifest. The corresponding preprint is
\href{https://arxiv.org/abs/1907.04850}{arXiv:1907.04850}.

Let $X$ be smooth over a perfect field and let $K$ be a perverse sheaf of finite coefficients with the coefficient prime invertible on the base. The characteristic cycle of $K$ carries polar multiplicities at points of $X$. The paper compares equivalent descriptions of polar multiplicity and applies the resulting local estimate to intersection complexes on Jacobians and to equidistribution questions for vector bundles on the projective line.

This blueprint is an independently worded, source-faithful mathematical
outline of the paper's definitions, constructions, and principal results. It
should be read alongside the original source; the source anchor on each node
points to the complete extracted TeX. Statements are intentionally marked
\notready until a matching Lean declaration and proof have been added.

\section{Main theorem or conjecture}
\label{sec:bounds-stalks-perverse-sheaves-characteristic-p-shende-results}

\begin{theorem}[Stalk bound by polar multiplicity]
\label{bounds-stalks-perverse-sheaves-characteristic-p-shende-main}
\source{bounds-stalks-perverse-sheaves-characteristic-p-shende}
\uses{bounds-stalks-perverse-sheaves-characteristic-p-shende-setup}
For every point $x$ and integer $i$, the dimension of the stalk cohomology $\mathcal H^{-i}(K)_x$ is at most the $i$-th polar multiplicity of the characteristic cycle of $K$ at $x$.
\notready
\end{theorem}

\begin{theorem}[Betti bound for theta intersections]
\label{bounds-stalks-perverse-sheaves-characteristic-p-shende-secondary}
\source{bounds-stalks-perverse-sheaves-characteristic-p-shende}
\uses{bounds-stalks-perverse-sheaves-characteristic-p-shende-main}
For a genus-$g$ curve and the indicated intersections of theta divisors in its Jacobian, the total ℓ-adic Betti number is bounded by $28^g/16+4\cdot 8^g+2\cdot 4^g$.
\notready
\end{theorem}
