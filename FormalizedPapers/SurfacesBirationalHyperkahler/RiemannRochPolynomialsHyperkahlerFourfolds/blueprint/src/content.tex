\chapter*{About This Blueprint}
\label{riemann-roch-polynomials-hyperkahler-fourfolds-about}

This project follows \emph{Computing Riemann--Roch polynomials and classifying hyper-Kähler fourfolds} by Olivier Debarre; Daniel Huybrechts; Emanuele Macrì; Claire Voisin (2023). The
retrieved source is archived under
\href{../../../../../references/riemann-roch-polynomials-hyperkahler-fourfolds/}{\texttt{riemann-roch-polynomials-hyperkahler-fourfolds}} and is indexed in
the workspace reference manifest. The corresponding preprint is
\href{https://arxiv.org/abs/2201.08152}{arXiv:2201.08152}.

Let $X$ be a hyper-Kähler fourfold with Beauville--Bogomolov--Fujiki form and Fujiki constant. Nef isotropic classes and Lagrangian fibrations control its divisor cones. The proof computes the Huybrechts--Riemann--Roch polynomial, analyzes nef cones and divisorial contractions, and applies the SYZ conjecture in dimension four.

This blueprint is an independently worded, source-faithful mathematical
outline of the paper's definitions, constructions, and principal results. It
should be read alongside the original source; the source anchor on each node
points to the complete extracted TeX. Statements are intentionally marked
\notready until a matching Lean declaration and proof have been added.

\section{Main theorem or conjecture}
\label{sec:riemann-roch-polynomials-hyperkahler-fourfolds-results}

\begin{theorem}[Riemann--Roch polynomial]
\label{riemann-roch-polynomials-hyperkahler-fourfolds-main}
\source{riemann-roch-polynomials-hyperkahler-fourfolds}
\uses{riemann-roch-polynomials-hyperkahler-fourfolds-setup}
If classes $\ell,m\in H^2(X,\mathbb Z)$ satisfy $\int_X\ell^4=0$ and $\int_X\ell^2m^2=2$, then the Huybrechts--Riemann--Roch polynomial and the Fujiki, Hodge, and Chern numbers are those of the Hilbert square of a K3 surface.
\notready
\end{theorem}

\begin{theorem}[Classification of the fourfold]
\label{riemann-roch-polynomials-hyperkahler-fourfolds-secondary}
\source{riemann-roch-polynomials-hyperkahler-fourfolds}
\uses{riemann-roch-polynomials-hyperkahler-fourfolds-main}
Under the same hypotheses, $X$ is of $\mathrm{K3}^{[2]}$ deformation type and a suitable nef isotropic line bundle induces a Lagrangian fibration to $\mathbb P^2$.
\notready
\end{theorem}
