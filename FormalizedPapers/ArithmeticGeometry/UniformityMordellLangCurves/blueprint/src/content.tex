\chapter*{About This Blueprint}
\label{uniformity-mordell-lang-curves-about}

This project follows \emph{Uniformity in Mordell--Lang for curves} by Vesselin Dimitrov; Ziyang Gao; Philipp Habegger (2021). The
retrieved source is archived under
\href{../../../../../references/uniformity-mordell-lang-curves/}{\texttt{uniformity-mordell-lang-curves}} and is indexed in
the workspace reference manifest. The corresponding preprint is
\href{https://arxiv.org/abs/2001.10276}{arXiv:2001.10276}.

Let $C$ be a smooth curve of genus at least two over a number field, embedded in its Jacobian. The Betti map and the associated height functions measure non-degeneracy in families of abelian varieties. A Vojta--Mumford height inequality is combined with intersection theory, a counting argument for large points, and lower bounds for canonical heights of small points.

This blueprint is an independently worded, source-faithful mathematical
outline of the paper's definitions, constructions, and principal results. It
should be read alongside the original source; the source anchor on each node
points to the complete extracted TeX. Statements are intentionally marked
\notready until a matching Lean declaration and proof have been added.

\section{Main theorem or conjecture}
\label{sec:uniformity-mordell-lang-curves-results}

\begin{theorem}[Height inequality for non-degenerate subvarieties]
\label{uniformity-mordell-lang-curves-main}
\source{uniformity-mordell-lang-curves}
\uses{uniformity-mordell-lang-curves-setup}
A non-degenerate subvariety of an abelian scheme dominating its base satisfies a lower bound $\widehat h(P)\geq c_1h(\pi(P))-c_2$ on a dense open subset, for positive constants $c_1$ and $c_2$.
\notready
\end{theorem}

\begin{theorem}[Uniform bound for rational points]
\label{uniformity-mordell-lang-curves-secondary}
\source{uniformity-mordell-lang-curves}
\uses{uniformity-mordell-lang-curves-main}
For fixed genus $g\geq2$ and degree bound $d$, there is a constant $c(g,d)\geq1$ such that every genus-$g$ curve over a number field of degree at most $d$ satisfies $\#C(F)\leq c(g,d)^{1+\operatorname{rank}J(C)(F)}$.
\notready
\end{theorem}
